117 void fromJson(const Json &in) {
118 m_known.clear();
119 m_prepared.clear();
120 m_slotsUsed.clear();
121 if (!in.is_object()) {
122 return;
123 }
124 if (in.contains("known") && in.at("known").is_array()) {
125 for (const Json &id : in.at("known")) {
126 m_known.push_back(id.get<std::string>());
127 }
128 }
129 if (in.contains("prepared") && in.at("prepared").is_array()) {

131 m_prepared.push_back(id.get<std::string>());
132 }
133 }
134 if (in.contains("slots_used") && in.at("slots_used").is_object()) {
135 for (const auto &[level, count] : in.at("slots_used").items()) {
136 m_slotsUsed[std::stoi(level)] = count.get<int32_t>();
137 }
138 }
139 }
