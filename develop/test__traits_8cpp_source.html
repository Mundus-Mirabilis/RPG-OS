10TEST_CASE("bookkeeping: creature traits contribute check and stat modifiers") {
11 const std::string rulesetJson = R"json({
12 "schema_version": 1, "ruleset_id": "mini", "licence": "test",
13 "attributes": [{"id": "STR", "name": "Strength", "min": 1, "max": 30, "default": 10},
14 {"id": "WIS", "name": "Wisdom", "min": 1, "max": 30, "default": 10}],
15 "derived_stats": [{"id": "WIS_mod", "name": "Wisdom Mod", "formula": "floor((WIS - 10) / 2)"}],
16 "check_types": {
17 "mini_perception": {
18 "resolution": "threshold", "dice": "1d20", "comparison": "ge",
19 "threshold_source": "difficulty", "bonus_stats": ["WIS_mod"]
20 }
21 },
22 "data": {
23 "traits": [
24 {"id": "keen_smell", "name": "Keen Smell",
25 "check_modifiers": [{"scope": "all", "mode": "advantage"}]},
26 {"id": "mighty", "name": "Mighty",
27 "stat_modifiers": [{"stat": "STR", "type": "add", "value": 2}]}
28 ],
29 "creatures": [
30 {"id": "wolf", "name": "Wolf", "traits": ["keen_smell", "mighty"],
31 "attributes": {"STR": 12, "WIS": 12}}
32 ]
33 }
34 })json";
35 RulesetEngine engine;
36 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
37 auto wolf = engine.createCreature("wolf");
38 REQUIRE(wolf != nullptr);
39 CHECK(wolf->hasTrait("keen_smell"));
40 CHECK(wolf->hasTrait("mighty"));
41 CHECK_FALSE(wolf->hasTrait("nope"));
42
43 // The trait's stat modifier raises the effective stat.
44 CHECK(wolf->getEffectiveStat("STR") == 14);
45
46 // The trait's check modifier gives advantage: roll twice, keep the higher
47 // (17 + WIS_mod 1 = 18 >= DC 15 -> success).
48 CheckParams params;
49 params.difficulty = 15;
50 auto rng = script({3, 17});
51 const auto result = engine.executeCheck("mini_perception", *wolf, nullptr, params, rng);
52 CHECK(result.isSuccess);
53 CHECK(result.rawDiceRolls == std::vector<int>{17});
54
55 // Traits survive serialization.
56 rpg_os::Json saved;
57 wolf->toJson(saved);
58 DynamicEntity restored(engine.ruleset(), "wolf");
59 restored.fromJson(saved);
60 CHECK(restored.hasTrait("keen_smell"));
61 CHECK(restored.getEffectiveStat("STR") == 14);
62}

74
75 // Pack Tactics gives Advantage on attack rolls (wolf STR 14 -> mod +2):
76 // roll twice, keep 17; 17 + 2 = 19 >= AC 15.

79 CHECK(hit.isSuccess);
80 CHECK(hit.rawDiceRolls == std::vector<int>{17});
81
82 // Without the trait, the same roll misses (3 + 2 = 5 < 15).
83 wolf->removeTrait("pack_tactics");

86 CHECK_FALSE(miss.isSuccess);
