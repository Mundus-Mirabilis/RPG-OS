25TEST_CASE("Variance: pickVariant honours min/max/third selection") {
26 // Deterministic scripted RNG always returns 0 from rng(0, span).
27 auto rng = script({0, 0, 0, 0, 0});
28 CHECK(rpg_os::pickVariant(0, 30, Variance::Weakest, rng) == 0);
29 CHECK(rpg_os::pickVariant(0, 30, Variance::Strongest, rng) == 30);
30 CHECK(rpg_os::pickVariant(0, 30, Variance::Weak, rng) <= 10); // lower third
31 CHECK(rpg_os::pickVariant(0, 30, Variance::Average, rng) >= 10); // middle third
32 CHECK(rpg_os::pickVariant(0, 30, Variance::Average, rng) <= 20);
33 CHECK(rpg_os::pickVariant(0, 30, Variance::Strong, rng) >= 20); // upper third
34 // Random stays within the full range.
35 CHECK(rpg_os::pickVariant(5, 5, Variance::Random, rng) == 5); // degenerate
36}

38TEST_CASE("Variance: readVariantValue handles scalars, ranges, and dice") {
39 rpg_os::DefaultRandom rng;
40
41 // Plain integer: no range, same for every variance.
42 CHECK(rpg_os::readVariantValue(rpg_os::Json(7), Variance::Weakest, rng) == 7);
43 CHECK(rpg_os::readVariantValue(rpg_os::Json(7), Variance::Strongest, rng) == 7);
44
45 // Explicit min/max range.
46 const rpg_os::Json range = {{"min", 2}, {"max", 12}};
47 CHECK(rpg_os::readVariantValue(range, Variance::Weakest, rng) == 2);
48 CHECK(rpg_os::readVariantValue(range, Variance::Strongest, rng) == 12);
49 const int32_t weak = rpg_os::readVariantValue(range, Variance::Weak, rng);
50 CHECK(weak >= 2);
51 CHECK(weak <= 2 + 10 / 3);
52 const int32_t strong = rpg_os::readVariantValue(range, Variance::Strong, rng);
53 CHECK(strong >= 2 + 20 / 3);
54 CHECK(strong <= 12);
55 const int32_t random = rpg_os::readVariantValue(range, Variance::Random, rng);
56 CHECK(random >= 2);
57 CHECK(random <= 12);
58
59 // Dice expression: weakest/strongest map to the dice bounds, random rolls.
60 CHECK(rpg_os::readVariantValue(rpg_os::Json("2d6"), Variance::Weakest, rng) == 2);
61 CHECK(rpg_os::readVariantValue(rpg_os::Json("2d6"), Variance::Strongest, rng) == 12);
62 const int32_t rolled = rpg_os::readVariantValue(rpg_os::Json("2d6"), Variance::Random, rng);
63 CHECK(rolled >= 2);
64 CHECK(rolled <= 12);
65}

67TEST_CASE("Engine: createCreature with variance selects ranged hit points") {
68 // Reuse the shipped dnd5e ruleset (goblin warrior has "HitPoints_Max": "3d6",
69 // so its HP resource ranges over [3, 18]).
70 const std::string path = std::string(RPG_OS_SOURCE_DIR) + "/rulesets/dnd5e_srd.json";
71 std::ifstream file(path);
72 REQUIRE(file.good());
73 rpg_os::RulesetEngine engine;
74 std::ostringstream buffer;
75 buffer << file.rdbuf();
76 REQUIRE(engine.loadRulesetFromJson(buffer.str()));
77
78 auto weakest = engine.createCreature("goblin_warrior", Variance::Weakest);
79 auto strongest = engine.createCreature("goblin_warrior", Variance::Strongest);
80 REQUIRE(weakest != nullptr);
81 REQUIRE(strongest != nullptr);
82 CHECK(weakest->baseAttribute("STR") == 8); // scalar attributes unchanged
83 CHECK(strongest->baseAttribute("STR") == 8);
84 CHECK(weakest->getStat("HitPoints_Max") == 3);
85 CHECK(strongest->getStat("HitPoints_Max") == 18);
86 CHECK(weakest->resource("HP") == 3);
87 CHECK(strongest->resource("HP") == 18);
88
89 // Random stays within the dice bounds.
90 auto random = engine.createCreature("goblin_warrior", Variance::Random);
91 REQUIRE(random != nullptr);
92 CHECK(random->resource("HP") >= 3);
93 CHECK(random->resource("HP") <= 18);
94}
