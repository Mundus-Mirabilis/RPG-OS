33std::string readFile(const char *name) {
34 const std::string path = std::string(RPG_OS_SOURCE_DIR) + "/rulesets/" + name;
35 std::ifstream file(path);
36 REQUIRE(file.good());
37 std::ostringstream buffer;
38 buffer << file.rdbuf();

74TEST_CASE("generated tde5e: named skill checks match the universal engine (parity)") {
75 const rpg_os::Json ruleset = loadRuleset("tde5e_core.json");
76 const TdeCharacter geron = TdeCharacter::fromArchetype(ruleset, "geron");
77
78 auto rng = script({14, 12, 11});
79 const CheckResult specific = geron.checkClimbing(CheckParams{}, rng);
80
81 CHECK(specific.isSuccess);
82 CHECK(specific.remainingPool == 5);
83 CHECK(specific.qualityLevel == 2);
84
85 // The same scenario through the universal engine must give identical results.
86 rpg_os::RulesetEngine engine;
87 REQUIRE(engine.loadRulesetFromJson(readFile("tde5e_core.json")));
88 auto universal = engine.createEntity("geron");
89 REQUIRE(universal != nullptr);
90 auto rng2 = script({14, 12, 11});
91 const CheckResult universalResult =
92 engine.executeSkillCheck("climbing", *universal, CheckParams{}, rng2);
93 CHECK(universalResult.isSuccess == specific.isSuccess);
94 CHECK(universalResult.remainingPool == specific.remainingPool);
95 CHECK(universalResult.qualityLevel == specific.qualityLevel);
96 CHECK(universalResult.rawDiceRolls == specific.rawDiceRolls);
97}
