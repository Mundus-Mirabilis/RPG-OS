36std::string readFile(const char *name) {
37 const std::string path = std::string(RPG_OS_SOURCE_DIR) + "/rulesets/" + name;
38 std::ifstream file(path);
39 REQUIRE(file.good());
40 std::ostringstream buffer;
41 buffer << file.rdbuf();
42 return buffer.str();
43}

313TEST_CASE("generated characters store narrow stat types (no wasted int32)") {
314 // Attributes and skills fit in a single byte for the shipped rulesets, so
315 // the generated characters store them as uint8_t — the templates in
316 // core/checks.hpp and the StatProvider concept work with any integral
317 // storage type. Wide base stats (armor, hit-point maxima, level) and
318 // resource pools stay int32_t where they can genuinely grow.
319 static_assert(std::is_same_v<decltype(TdeCharacter::courage), uint8_t>,
320 "TDE attributes must be byte-sized");
321 static_assert(std::is_same_v<decltype(TdeCharacter::climbing), uint8_t>,
322 "TDE skills must be byte-sized");
323 static_assert(std::is_same_v<decltype(TdeCharacter::armorRating), int32_t>,
324 "armor rating stays wide");
325 static_assert(std::is_same_v<decltype(TdeCharacter::lifePoints), int32_t>,
326 "resource pools stay wide");
327 static_assert(std::is_same_v<decltype(DndCharacter::strength), uint8_t>,
328 "D&D attributes must be byte-sized");
329 static_assert(std::is_same_v<decltype(DndCharacter::hitPoints), int32_t>,
330 "D&D hit points stay wide");
331 static_assert(std::is_same_v<decltype(BrpCharacter::brawl), uint8_t>,
332 "BRP skills must be byte-sized");
333 static_assert(std::is_same_v<decltype(BrpCharacter::sanity), uint8_t>,
334 "BRP Sanity (0..100) fits a byte");
335 static_assert(std::is_same_v<decltype(TdeCharacter::money), rpg_os::Money>,
336 "wealth uses the shared Money type");
337 static_assert(std::is_same_v<decltype(TdeCharacter::advancement), rpg_os::Advancement>,
338 "advancement uses the shared Advancement type");
339 static_assert(std::is_same_v<decltype(TdeCharacter::inventory), rpg_os::Inventory>,
340 "inventory uses the shared Inventory type");
341 static_assert(std::is_same_v<decltype(TdeCharacter::equipment), rpg_os::Equipment>,
342 "gear uses the shared Equipment type");
343 static_assert(std::is_same_v<decltype(TdeCharacter::spellbook), rpg_os::Spellbook>,
344 "spells use the shared Spellbook type");
345
346 // The byte-sized storage keeps the hot stat block small; the sheet
347 // bookkeeping members (inventory/equipment/spellbook/money/advancement) and
348 // the living-sheet runtime state (conditions map, effect timeline,
349 // resistances, traits, afflictions) are all fixed-size handles over heap
350 // state, so the whole object stays compact.
351 CHECK(sizeof(TdeCharacter) <= 1024);
352}
