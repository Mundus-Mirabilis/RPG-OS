148 template <RandomNumberGenerator Rng>
149 void rollInitiative(Rng &rng, std::string_view statId = "Initiative") {
150 for (Combatant &combatant : m_combatants) {
151 DynamicEntity *entity = combatant.handle.resolve();
152 combatant.initiative = entity != nullptr ? entity->getStat(statId) + rng(1, 6) : 0;
153 }
154 std::stable_sort(
155 m_combatants.begin(), m_combatants.end(),
156 [](const Combatant &a, const Combatant &b) { return a.initiative > b.initiative; });
157 }

163 template <RandomNumberGenerator Rng>
164 void beginRound(Rng &rng, std::string_view statId = "Initiative") {
165 ++m_round;
166 rollInitiative(rng, statId);
167 m_nextIndex = 0;
168 m_currentIndex = m_combatants.size();
169 }

173 bool next() {
174 while (m_nextIndex < m_combatants.size()) {
175 const std::size_t index = m_nextIndex++;
176 if (m_combatants[index].handle.resolve() != nullptr) {
177 m_currentIndex = index;
178 return true;
179 }
180 }
181 m_currentIndex = m_combatants.size();
182 return false;
183 }
184
186 [[nodiscard]] EntityHandle current() const {
187 return m_currentIndex < m_combatants.size() ? m_combatants[m_currentIndex].handle
188 : EntityHandle{};
189 }

196 template <RandomNumberGenerator Rng> EntityHandle runCurrentTurn(Rng &rng) {
197 if (m_currentIndex >= m_combatants.size()) {
198 return EntityHandle{};
199 }
200 Combatant &combatant = m_combatants[m_currentIndex];
201 DynamicEntity *entity = combatant.handle.resolve();
202 if (entity == nullptr) {
203 return EntityHandle{};
204 }
205 resetBudget(combatant);
206 m_engine->runTurn(*entity, rng);
207 return combatant.handle;
208 }

213 [[nodiscard]] bool canSpend(EntityId id, ActionKind kind, int32_t amount = 1) const {
214 const Combatant *combatant = findCombatant(id);
215 if (combatant == nullptr) {
216 return false;
217 }
218 const DynamicEntity *entity = combatant->handle.resolve();
219 if (entity == nullptr || amount < 0) {
220 return false;
221 }
222 if (!m_engine->actionAllowed(*entity, actionName(kind))) {
223 return false;
224 }
225 return availableFor(*combatant, kind) >= amount;
226 }
227
231 bool spendAction(EntityId id, ActionKind kind, int32_t amount = 1) {
232 if (!canSpend(id, kind, amount)) {
233 return false;
234 }
235 Combatant *combatant = findCombatant(id);
236 spentFor(*combatant, kind) += amount;
237 return true;
238 }

241 [[nodiscard]] int32_t remaining(EntityId id, ActionKind kind) const {

243 return combatant == nullptr ? 0 : availableFor(*combatant, kind);

268 [[nodiscard]] static int32_t budgetTotal(const ActionBudget &budget, ActionKind kind) noexcept {
269 switch (kind) {
270 case ActionKind::Action:
271 return budget.actions;
272 case ActionKind::BonusAction:
273 return budget.bonusActions;
274 case ActionKind::Reaction:
275 return budget.reactions;
276 case ActionKind::Move:
277 return budget.movement;
278 }
279 return 0;
280 }

282 [[nodiscard]] static int32_t &spentFor(Combatant &combatant, ActionKind kind) noexcept {
283 switch (kind) {
284 case ActionKind::Action:
285 return combatant.spentActions;
286 case ActionKind::BonusAction:
287 return combatant.spentBonusActions;
288 case ActionKind::Reaction:
289 return combatant.spentReactions;
290 case ActionKind::Move:
291 return combatant.spentMovement;
292 }
293 return combatant.spentActions;
294 }

297 [[nodiscard]] int32_t availableFor(const Combatant &combatant, ActionKind kind) const noexcept {
298 const int32_t total = budgetTotal(m_budget, kind);
299 switch (kind) {
300 case ActionKind::Action:
301 return total - combatant.spentActions;
302 case ActionKind::BonusAction:
303 return total - combatant.spentBonusActions;
304 case ActionKind::Reaction:
305 return total - combatant.spentReactions;
306 case ActionKind::Move:
307 return total - combatant.spentMovement;
308 }
309 return 0;
310 }

312 static void resetBudget(Combatant &combatant) noexcept {
313 combatant.spentActions = 0;
314 combatant.spentBonusActions = 0;
315 combatant.spentReactions = 0;
316 combatant.spentMovement = 0;
317 }

319 [[nodiscard]] Combatant *findCombatant(EntityId id) {
320 for (Combatant &combatant : m_combatants) {
321 if (combatant.handle.entityId() == id) {
322 return &combatant;
323 }
324 }
325 return nullptr;
326 }
327 [[nodiscard]] const Combatant *findCombatant(EntityId id) const {
328 for (const Combatant &combatant : m_combatants) {
329 if (combatant.handle.entityId() == id) {
330 return &combatant;
331 }
332 }
333 return nullptr;
334 }
