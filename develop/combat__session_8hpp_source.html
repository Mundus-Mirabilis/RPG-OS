147 template <RandomNumberGenerator Rng>
148 void rollInitiative(Rng &rng, std::string_view statId = "Initiative") {
149 for (Combatant &combatant : m_combatants) {
150 DynamicEntity *entity = combatant.handle.resolve();
151 combatant.initiative = entity != nullptr ? entity->getStat(statId) + rng(1, 6) : 0;
152 }
153 std::stable_sort(
154 m_combatants.begin(), m_combatants.end(),
155 [](const Combatant &a, const Combatant &b) { return a.initiative > b.initiative; });
156 }

162 template <RandomNumberGenerator Rng>
163 void beginRound(Rng &rng, std::string_view statId = "Initiative") {
164 ++m_round;
165 rollInitiative(rng, statId);
166 m_nextIndex = 0;
167 m_currentIndex = m_combatants.size();
168 }
169
172 bool next() {
173 while (m_nextIndex < m_combatants.size()) {
174 const std::size_t index = m_nextIndex++;
175 if (m_combatants[index].handle.resolve() != nullptr) {
176 m_currentIndex = index;
177 return true;
178 }
179 }
180 m_currentIndex = m_combatants.size();
181 return false;
182 }
183
185 [[nodiscard]] EntityHandle current() const {
186 return m_currentIndex < m_combatants.size() ? m_combatants[m_currentIndex].handle
187 : EntityHandle{};
188 }

195 template <RandomNumberGenerator Rng> EntityHandle runCurrentTurn(Rng &rng) {
196 if (m_currentIndex >= m_combatants.size()) {
197 return EntityHandle{};
198 }
199 Combatant &combatant = m_combatants[m_currentIndex];
200 DynamicEntity *entity = combatant.handle.resolve();
201 if (entity == nullptr) {
202 return EntityHandle{};
203 }
204 resetBudget(combatant);
205 m_engine->runTurn(*entity, rng);
206 return combatant.handle;
207 }

212 [[nodiscard]] bool canSpend(EntityId id, ActionKind kind, int32_t amount = 1) const {
213 const Combatant *combatant = findCombatant(id);
214 if (combatant == nullptr) {
215 return false;
216 }
217 const DynamicEntity *entity = combatant->handle.resolve();
218 if (entity == nullptr || amount < 0) {
219 return false;
220 }
221 if (!m_engine->actionAllowed(*entity, actionName(kind))) {
222 return false;
223 }
224 return availableFor(*combatant, kind) >= amount;
225 }
226
230 bool spendAction(EntityId id, ActionKind kind, int32_t amount = 1) {
231 if (!canSpend(id, kind, amount)) {
232 return false;
233 }
234 Combatant *combatant = findCombatant(id);
235 spentFor(*combatant, kind) += amount;
236 return true;
237 }

240 [[nodiscard]] int32_t remaining(EntityId id, ActionKind kind) const {
241 const Combatant *combatant = findCombatant(id);
242 return combatant == nullptr ? 0 : availableFor(*combatant, kind);
243 }

267 [[nodiscard]] static int32_t budgetTotal(const ActionBudget &budget, ActionKind kind) noexcept {
268 switch (kind) {
269 case ActionKind::Action:
270 return budget.actions;
271 case ActionKind::BonusAction:
272 return budget.bonusActions;
273 case ActionKind::Reaction:
274 return budget.reactions;
275 case ActionKind::Move:
276 return budget.movement;
277 }
278 return 0;
279 }
280
281 [[nodiscard]] static int32_t &spentFor(Combatant &combatant, ActionKind kind) noexcept {
282 switch (kind) {
283 case ActionKind::Action:
284 return combatant.spentActions;
285 case ActionKind::BonusAction:
286 return combatant.spentBonusActions;
287 case ActionKind::Reaction:
288 return combatant.spentReactions;
289 case ActionKind::Move:
290 return combatant.spentMovement;
291 }
292 return combatant.spentActions;
293 }
294
296 [[nodiscard]] int32_t availableFor(const Combatant &combatant, ActionKind kind) const noexcept {
297 const int32_t total = budgetTotal(m_budget, kind);
298 switch (kind) {
299 case ActionKind::Action:
300 return total - combatant.spentActions;
301 case ActionKind::BonusAction:
302 return total - combatant.spentBonusActions;
303 case ActionKind::Reaction:
304 return total - combatant.spentReactions;
305 case ActionKind::Move:
306 return total - combatant.spentMovement;
307 }
308 return 0;
309 }

311 static void resetBudget(Combatant &combatant) noexcept {
312 combatant.spentActions = 0;
313 combatant.spentBonusActions = 0;
314 combatant.spentReactions = 0;
315 combatant.spentMovement = 0;
316 }

318 [[nodiscard]] Combatant *findCombatant(EntityId id) {
319 for (Combatant &combatant : m_combatants) {
320 if (combatant.handle.entityId() == id) {
321 return &combatant;
322 }
323 }
324 return nullptr;
325 }
326 [[nodiscard]] const Combatant *findCombatant(EntityId id) const {
327 for (const Combatant &combatant : m_combatants) {
328 if (combatant.handle.entityId() == id) {
329 return &combatant;
330 }
331 }
332 return nullptr;
333 }
