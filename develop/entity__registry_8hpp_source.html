113 [[nodiscard]] DynamicEntity *get(EntityId id) const {
114 const auto it = m_entities.find(id);
115 return it == m_entities.end() ? nullptr : it->second.get();

120 const auto it = m_entities.find(id);

122 return EntityHandle{};

124 return EntityHandle{id, it->second};
125 }

134 return m_entities.size();
135 }

146 m_entities.clear();

148
150 [[nodiscard]] std::vector<DynamicEntity *> all() const {
151 std::vector<DynamicEntity *> result;
152 result.reserve(m_entities.size());
