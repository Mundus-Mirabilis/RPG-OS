57void printHelp(std::ostream &os) {
58 os << "Usage: rpg_os_example_fight [options] [combatant_a] [combatant_b]\n"
59 << "\n"
60 << "Fight two entities to the end and announce the winner. Combatants are\n"
61 << "ruleset archetypes or bestiary entries; if none are given a default\n"
62 << "pair is used.\n"
63 << "\n"
64 << "Options:\n"
65 << " -h, --help show this help and exit\n"
66 << " --list list all combatants in the ruleset and exit\n"
67 << " --root <dir> project root containing rulesets/ (default: \".\")\n"
68 << " --ruleset <f> ruleset file (default: <root>/rulesets/tde5e_core.json)\n"
69 << " --seed <n> RNG seed for reproducible fights (default: fresh entropy)\n"
70 << " --rounds <n> max rounds before a fight counts as a draw (default: 1000)\n"
71 << " --batch <n> run n fights between the same pair (default: 1)\n"
72 << " --weapon <dice> weapon damage for archetypes (default: \"1d6+4\")\n"
73 << " --csv one tab-separated line per fight (for scripts)\n";

79bool parseArgs(int argc, char **argv, Options &opts, std::string &error) {
80 for (int i = 1; i < argc; ++i) {
81 const std::string arg(argv[i]);
82 const auto nextValue = [&](const std::string &flag) -> std::optional<std::string> {
83 if (i + 1 >= argc) {
84 error = "missing value for " + flag;
85 return std::nullopt;
86 }
87 return std::string(argv[++i]);
88 };
89 if (arg == "--root") {
90 const auto value = nextValue(arg);
91 if (!value) {
92 return false;
93 }
94 opts.root = *value;
95 opts.rootExplicit = true;
96 } else if (arg == "--ruleset") {
97 const auto value = nextValue(arg);
98 if (!value) {
99 return false;
100 }
101 opts.ruleset = *value;
102 } else if (arg == "--seed") {
103 const auto value = nextValue(arg);
104 if (!value) {
105 return false;
106 }
107 opts.seed = static_cast<uint32_t>(std::stoul(*value));
108 } else if (arg == "--rounds") {
109 const auto value = nextValue(arg);
110 if (!value) {
111 return false;
112 }
113 opts.maxRounds = std::stoi(*value);
114 } else if (arg == "--batch") {
115 const auto value = nextValue(arg);
116 if (!value) {
117 return false;
118 }
119 opts.batch = std::stoi(*value);
120 } else if (arg == "--weapon") {
121 const auto value = nextValue(arg);
122 if (!value) {
123 return false;
124 }
125 opts.weapon = *value;
126 } else if (arg == "--csv") {
127 opts.csv = true;
128 } else if (arg == "--list") {
129 opts.listOnly = true;
130 } else if (arg == "-h" || arg == "--help") {
131 printHelp(std::cout);
132 std::exit(0);
133 } else if (!arg.empty() && arg[0] == '-') {
134 error = "unknown option '" + arg + "'";
135 return false;
136 } else {
137 opts.combatants.push_back(arg);
138 }
139 }
140 return true;
141}

179 << '\t' << outcome.rounds << '\n';
180 } else if (outcome.winnerIndex == 1) {
181 os << b.id << '\t' << a.id << '\t' << outcome.remainingLp[1] << '\t' << outcome.remainingLp[0]
182 << '\t' << outcome.rounds << '\n';
183 } else {
184 os << "DRAW\t" << a.id << '\t' << b.id << '\t' << outcome.remainingLp[0] << '\t'
185 << outcome.remainingLp[1] << '\t' << outcome.rounds << '\n';

191void printHumanLine(std::ostream &os, const rpg_os::CombatantSpec &a,
192 const rpg_os::CombatantSpec &b, const rpg_os::FightOutcome &outcome) {
193 const char *plural = outcome.rounds == 1 ? "" : "s";
194 if (outcome.winnerIndex == 0) {
195 os << a.name << " defeats " << b.name << " in " << outcome.rounds << " round" << plural << " ("
196 << a.name << ' ' << outcome.remainingLp[0] << '/' << outcome.maxLp[0] << " LP, " << b.name
197 << ' ' << outcome.remainingLp[1] << '/' << outcome.maxLp[1] << " LP).\n";
198 } else if (outcome.winnerIndex == 1) {
199 os << b.name << " defeats " << a.name << " in " << outcome.rounds << " round" << plural << " ("
200 << b.name << ' ' << outcome.remainingLp[1] << '/' << outcome.maxLp[1] << " LP, " << a.name
201 << ' ' << outcome.remainingLp[0] << '/' << outcome.maxLp[0] << " LP).\n";
202 } else {
203 os << "Draw: " << a.name << " vs " << b.name << " after " << outcome.rounds << " rounds ("
204 << a.name << ' ' << outcome.remainingLp[0] << '/' << outcome.maxLp[0] << " LP, " << b.name
205 << ' ' << outcome.remainingLp[1] << '/' << outcome.maxLp[1] << " LP).\n";

211int main(int argc, char **argv) {
212 Options opts;
213 std::string parseError;
214 if (!parseArgs(argc, argv, opts, parseError)) {
215 std::cerr << "error: " << parseError << "\n\n";
216 printHelp(std::cerr);
217 return 2;
218 }
219
220 // Match the sibling examples' `[project-root]` convention: when the first
221 // positional argument is an existing directory and --root was not given
222 // explicitly, treat it as the project root instead of a combatant id.
223 if (!opts.rootExplicit && !opts.combatants.empty() &&
224 std::filesystem::is_directory(opts.combatants.front())) {
225 opts.root = opts.combatants.front();
226 opts.combatants.erase(opts.combatants.begin());
227 }
228
229 const std::string rulesetPath =
230 opts.ruleset.empty() ? opts.root + "/rulesets/tde5e_core.json" : opts.ruleset;
231
232 rpg_os::RulesetEngine engine;
233 if (!engine.loadRulesetFromFile(rulesetPath)) {
234 std::cerr << "error: failed to load '" << rulesetPath << "': " << engine.lastError() << '\n';
235 return 1;
236 }
237
238 if (opts.listOnly) {
239 printCombatantList(engine);
240 return 0;
241 }
242
243 const std::string checkType = rpg_os::resolveAttackCheckType(engine.ruleset());
244 if (checkType.empty()) {
245 std::cerr << "error: ruleset '" << rulesetPath << "' has no attack-vs-defence check type\n";
246 return 1;
247 }
248 const std::string hpPool = rpg_os::resolveHitPointPool(engine.ruleset());
249 if (hpPool.empty()) {
250 std::cerr << "error: ruleset '" << rulesetPath << "' has no hit-point pool\n";
251 return 1;
252 }
253
254 std::string idA;
255 std::string idB;
256 if (opts.combatants.size() == 2) {
257 idA = opts.combatants[0];
258 idB = opts.combatants[1];
259 } else if (opts.combatants.empty()) {
260 idA = "geron";
261 idB = "gotongi";
262 } else {
263 std::cerr << "error: expected exactly two combatant ids (got " << opts.combatants.size()
264 << ")\n";
265 return 2;
266 }
267
268 rpg_os::CombatantSpec specA;
269 rpg_os::CombatantSpec specB;
270 if (!rpg_os::makeCombatantSpec(engine, idA, specA, opts.weapon)) {
271 std::cerr << "error: unknown combatant '" << idA << "' (see --list)\n";
272 return 1;
273 }
274 if (!rpg_os::makeCombatantSpec(engine, idB, specB, opts.weapon)) {
275 std::cerr << "error: unknown combatant '" << idB << "' (see --list)\n";
276 return 1;
277 }
278
279 // Without --seed the engine is seeded from fresh entropy, so every run
280 // differs; with --seed the run is exactly reproducible.
281 rpg_os::DefaultRandom rng =
282 opts.seed ? rpg_os::DefaultRandom(*opts.seed) : rpg_os::DefaultRandom{};
283 for (int i = 0; i < opts.batch; ++i) {
284 const rpg_os::FightOutcome outcome =
285 rpg_os::runFight(engine, specA, specB, checkType, hpPool, opts.maxRounds, rng);
286 if (opts.csv) {
287 printCsvLine(std::cout, specA, specB, outcome);
288 } else {
289 if (i == 0) {
290 printCombatants(std::cout, specA, specB);
291 }
292 printHumanLine(std::cout, specA, specB, outcome);
293 }
294 }
295 return 0;
296}
