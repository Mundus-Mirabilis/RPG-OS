65void printHelp(std::ostream &os) {
66 os << "Usage: rpg_os_example_fight [options] [combatant_a] [combatant_b]\n"
67 << "\n"
68 << "Fight two entities to the end and announce the winner. Combatants are\n"
69 << "ruleset archetypes or bestiary entries; if none are given a default\n"
70 << "pair is used.\n"
71 << "\n"
72 << "Options:\n"
73 << " -h, --help show this help and exit\n"
74 << " --list list all combatants in the ruleset and exit\n"
75 << " --root <dir> project root containing rulesets/ (default: \".\")\n"
76 << " --ruleset <f> ruleset file (default: <root>/rulesets/tde5e_core.json)\n"
77 << " --seed <n> RNG seed for reproducible fights (default: fresh entropy)\n"
78 << " --rounds <n> max rounds before a fight counts as a draw (default: 1000)\n"
79 << " --batch <n> run n fights between the same pair (default: 1)\n"
80 << " --weapon <dice> weapon damage for archetypes (default: \"1d6+4\")\n"
81 << " --no-magic disable spellcasting: fight with weapons only\n"
82 << " --csv one tab-separated line per fight (for scripts)\n";

88bool parseArgs(int argc, char **argv, Options &opts, std::string &error) {
89 for (int i = 1; i < argc; ++i) {
90 const std::string arg(argv[i]);
91 const auto nextValue = [&](const std::string &flag) -> std::optional<std::string> {
92 if (i + 1 >= argc) {
93 error = "missing value for " + flag;
94 return std::nullopt;
95 }
96 return std::string(argv[++i]);
97 };
98 if (arg == "--root") {
99 const auto value = nextValue(arg);
100 if (!value) {
101 return false;
102 }
103 opts.root = *value;
104 opts.rootExplicit = true;
105 } else if (arg == "--ruleset") {
106 const auto value = nextValue(arg);
107 if (!value) {
108 return false;
109 }
110 opts.ruleset = *value;
111 } else if (arg == "--seed") {
112 const auto value = nextValue(arg);
113 if (!value) {
114 return false;
115 }
116 opts.seed = static_cast<uint32_t>(std::stoul(*value));
117 } else if (arg == "--rounds") {
118 const auto value = nextValue(arg);
119 if (!value) {
120 return false;
121 }
122 opts.maxRounds = std::stoi(*value);
123 } else if (arg == "--batch") {
124 const auto value = nextValue(arg);
125 if (!value) {
126 return false;
127 }
128 opts.batch = std::stoi(*value);
129 } else if (arg == "--weapon") {
130 const auto value = nextValue(arg);
131 if (!value) {
132 return false;
133 }
134 opts.weapon = *value;
135 } else if (arg == "--no-magic") {
136 opts.useMagic = false;
137 } else if (arg == "--csv") {
138 opts.csv = true;
139 } else if (arg == "--list") {
140 opts.listOnly = true;
141 } else if (arg == "-h" || arg == "--help") {
142 printHelp(std::cout);
143 std::exit(0);
144 } else if (!arg.empty() && arg[0] == '-') {
145 error = "unknown option '" + arg + "'";
146 return false;
147 } else {
148 opts.combatants.push_back(arg);
149 }
150 }
151 return true;
152}

204void printCsvLine(std::ostream &os, const rpg_os::CombatantSpec &a, const rpg_os::CombatantSpec &b,

208 << '\t' << outcome.rounds << '\n';
209 } else if (outcome.winnerIndex == 1) {
210 os << b.id << '\t' << a.id << '\t' << outcome.remainingLp[1] << '\t' << outcome.remainingLp[0]
211 << '\t' << outcome.rounds << '\n';
212 } else {
213 os << "DRAW\t" << a.id << '\t' << b.id << '\t' << outcome.remainingLp[0] << '\t'
214 << outcome.remainingLp[1] << '\t' << outcome.rounds << '\n';

220void printHumanLine(std::ostream &os, const rpg_os::CombatantSpec &a,
221 const rpg_os::CombatantSpec &b, const rpg_os::FightOutcome &outcome) {
222 const char *plural = outcome.rounds == 1 ? "" : "s";
223 if (outcome.winnerIndex == 0) {
224 os << a.name << " defeats " << b.name << " in " << outcome.rounds << " round" << plural << " ("
225 << a.name << ' ' << outcome.remainingLp[0] << '/' << outcome.maxLp[0] << " LP, " << b.name
226 << ' ' << outcome.remainingLp[1] << '/' << outcome.maxLp[1] << " LP).\n";
227 } else if (outcome.winnerIndex == 1) {
228 os << b.name << " defeats " << a.name << " in " << outcome.rounds << " round" << plural << " ("
229 << b.name << ' ' << outcome.remainingLp[1] << '/' << outcome.maxLp[1] << " LP, " << a.name
230 << ' ' << outcome.remainingLp[0] << '/' << outcome.maxLp[0] << " LP).\n";
231 } else {
232 os << "Draw: " << a.name << " vs " << b.name << " after " << outcome.rounds << " rounds ("
233 << a.name << ' ' << outcome.remainingLp[0] << '/' << outcome.maxLp[0] << " LP, " << b.name
234 << ' ' << outcome.remainingLp[1] << '/' << outcome.maxLp[1] << " LP).\n";

240int main(int argc, char **argv) {
241 Options opts;
242 std::string parseError;
243 if (!parseArgs(argc, argv, opts, parseError)) {
244 std::cerr << "error: " << parseError << "\n\n";
245 printHelp(std::cerr);
246 return 2;
247 }
248
249 // Match the sibling examples' `[project-root]` convention: when the first
250 // positional argument is an existing directory and --root was not given
251 // explicitly, treat it as the project root instead of a combatant id.
252 if (!opts.rootExplicit && !opts.combatants.empty() &&
253 std::filesystem::is_directory(opts.combatants.front())) {
254 opts.root = opts.combatants.front();
255 opts.combatants.erase(opts.combatants.begin());
256 }
257
258 const std::string rulesetPath =
259 opts.ruleset.empty() ? opts.root + "/rulesets/tde5e_core.json" : opts.ruleset;
260
261 rpg_os::RulesetEngine engine;
262 if (!engine.loadRulesetFromFile(rulesetPath)) {
263 std::cerr << "error: failed to load '" << rulesetPath << "': " << engine.lastError() << '\n';
264 return 1;
265 }
266
267 if (opts.listOnly) {
268 printCombatantList(engine);
269 return 0;
270 }
271
272 const std::string checkType = rpg_os::resolveAttackCheckType(engine.ruleset());
273 if (checkType.empty()) {
274 std::cerr << "error: ruleset '" << rulesetPath << "' has no attack-vs-defence check type\n";
275 return 1;
276 }
277 const std::string hpPool = rpg_os::resolveHitPointPool(engine.ruleset());
278 if (hpPool.empty()) {
279 std::cerr << "error: ruleset '" << rulesetPath << "' has no hit-point pool\n";
280 return 1;
281 }
282
283 std::string idA;
284 std::string idB;
285 if (opts.combatants.size() == 2) {
286 idA = opts.combatants[0];
287 idB = opts.combatants[1];
288 } else if (opts.combatants.empty()) {
289 idA = "geron";
290 idB = "gotongi";
291 } else {
292 std::cerr << "error: expected exactly two combatant ids (got " << opts.combatants.size()
293 << ")\n";
294 return 2;
295 }
296
297 rpg_os::CombatantSpec specA;
298 rpg_os::CombatantSpec specB;
299 if (!rpg_os::makeCombatantSpec(engine, idA, specA, opts.weapon)) {
300 std::cerr << "error: unknown combatant '" << idA << "' (see --list)\n";
301 return 1;
302 }
303 if (!rpg_os::makeCombatantSpec(engine, idB, specB, opts.weapon)) {
304 std::cerr << "error: unknown combatant '" << idB << "' (see --list)\n";
305 return 1;
306 }
307
308 // Without --seed the engine is seeded from fresh entropy, so every run
309 // differs; with --seed the run is exactly reproducible.
310 rpg_os::DefaultRandom rng =
311 opts.seed ? rpg_os::DefaultRandom(*opts.seed) : rpg_os::DefaultRandom{};
312 for (int i = 0; i < opts.batch; ++i) {
313 const rpg_os::FightOutcome outcome = rpg_os::runFight(engine, specA, specB, checkType, hpPool,
314 opts.maxRounds, rng, opts.useMagic);
315 if (opts.csv) {
316 printCsvLine(std::cout, specA, specB, outcome);
317 } else {
318 if (i == 0) {
319 printCombatants(std::cout, specA, specB);
320 }
321 printHumanLine(std::cout, specA, specB, outcome);
322 }
323 }
324 return 0;
325}
