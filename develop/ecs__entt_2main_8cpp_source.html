54int main(int argc, char **argv) {
55 const std::string root = argc > 1 ? argv[1] : ".";
56 rpg_os::RulesetEngine engine;
57 if (!engine.loadRulesetFromFile(root + "/rulesets/tde5e_core.json")) {
58 std::cerr << "failed to load tde5e_core.json: " << engine.lastError() << '\n';
59 return 1;
60 }
61
62 entt::registry registry;
63
64 // EnTT -> RPG OS: an EnTT entity carries a Sheet component that owns the
65 // live DynamicEntity; the id map links RPG OS ids back to EnTT entities so
66 // events can update components.
67 std::unordered_map<rpg_os::EntityId, entt::entity> idToEntt;
68
69 const auto addUnit = [&](const std::string &kind, const std::string &label, int32_t x,
70 int32_t y) {
71 const entt::entity enttEntity = registry.create();
72 registry.emplace<Tag>(enttEntity, label);
73 registry.emplace<Position>(enttEntity, x, y);
74
75 std::shared_ptr<rpg_os::DynamicEntity> entity =
76 kind == "character" ? engine.createEntity("geron") : engine.createCreature("gotongi");
77 registry.emplace<Sheet>(enttEntity, Sheet{entity});
78 registry.emplace<HpBar>(enttEntity, entity->resource("LP"));
79
80 const rpg_os::EntityId id = entity->entityId();
81 idToEntt[id] = enttEntity;
82 return id;
83 };
84
85 const rpg_os::EntityId heroId = addUnit("character", "hero", 0, 0);
86 const rpg_os::EntityId enemyId = addUnit("creature", "gotongi", 5, 0);
87
88 // RPG OS -> EnTT: an observer event updates the HpBar component of exactly
89 // the EnTT entity whose sheet changed — the "UI health bar" use case from
90 // the code review, without polling.
91 (void)engine.registerEventListener(
92 rpg_os::EventType::OnResourceChanged, [&](const rpg_os::EventData &data) {
93 if (data.getString("resource", "") != "LP") {
94 return;
95 }
96 const auto id = rpg_os::EntityId{static_cast<uint64_t>(data.getInt("instance_id"))};
97 const auto it = idToEntt.find(id);
98 if (it == idToEntt.end()) {
99 return;
100 }
101 registry.get<HpBar>(it->second).displayed = data.getInt("new_value");
102 });
103
104 // A system (EnTT view) prints every unit.
105 std::cout << "=== ECS pattern (EnTT) ===\n";
106 auto printUnits = [&]() {
107 registry.view<const Position, const Tag, const HpBar, const Sheet>().each(
108 [](const Position &pos, const Tag &tag, const HpBar &bar, const Sheet &) {
109 std::cout << " " << tag.label << " @" << pos.x << "," << pos.y << " HP "
110 << bar.displayed << '\n';
111 });
112 };
113 printUnits();
114
115 // A "system" resolves a rule through the engine: the hero is poisoned for 2
116 // rounds and the enemy takes a hit. Both flow through the event bus, so the
117 // HpBar components update themselves.
118 auto sheetOf = [&](rpg_os::EntityId id) -> rpg_os::DynamicEntity & {
119 return *registry.get<Sheet>(idToEntt[id]).entity;
120 };
121 engine.applyCondition(sheetOf(heroId), "pain", 1, 2);
122 rpg_os::DynamicEntity attacker(engine.ruleset(), "attacker");
123 (void)engine.applyDamage(attacker, sheetOf(enemyId), "LP", 8);
124
125 std::cout << "After the gotongi takes an 8-point hit:\n";
126 printUnits();
127 std::cout << "hero pain stacks: " << sheetOf(heroId).conditionStacks("pain") << "\n";
128
129 // Destroying an EnTT entity removes its Sheet (shared_ptr releases the
130 // sheet); the hero's sheet is still alive through its own component.
131 registry.destroy(idToEntt[enemyId]);
132 std::cout << "gotongi removed from the ECS registry; hero sheet still alive: "
133 << (sheetOf(heroId).resource("LP") > 0 ? "yes" : "no") << '\n';
134 return 0;
135}
