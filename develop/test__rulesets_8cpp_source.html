29TEST_CASE("dnd5e_srd: loads and resolves real SRD values") {
30 rpg_os::RulesetEngine engine;
31 REQUIRE(engine.loadRulesetFromFile(rulesetPath("dnd5e_srd.json")));
32 CHECK(engine.ruleset().attributes.size() == 9); // 6 abilities + prof + level + HP max
33 CHECK(engine.ruleset().skills.size() == 18);
34 CHECK(engine.ruleset().checkTypes.size() == 9);
35 CHECK(engine.ruleset().costTables.size() == 1);
36
37 auto fighter = engine.createEntity("fighter_lvl1");
38 REQUIRE(fighter != nullptr);
39 // Ability modifiers from the real SRD table: STR 16 -> +3, CON 14 -> +2.
40 CHECK(engine.calculateStat(*fighter, "STR_mod") == 3);
41 CHECK(engine.calculateStat(*fighter, "CON_mod") == 2);
42 // Base AC = 10 + DEX modifier (DEX 14 -> +2).
43 CHECK(engine.calculateStat(*fighter, "AC") == 12);
44 // Hit points as set on the archetype (fighter: 10 + CON mod = 12).
45 CHECK(fighter->resource("HP") == 12);
46 CHECK(fighter->baseAttribute("proficiency_bonus") == 2);
47
48 // Real XP -> level table (SRD Character Advancement).
49 const auto *xp = engine.ruleset().findCostTable("xp_to_level");
50 REQUIRE(xp != nullptr);
51 CHECK(xp->table.lookup(0) == 1);
52 CHECK(xp->table.lookup(2700) == 4);
53 CHECK(xp->table.lookup(6500) == 5);
54 CHECK(xp->table.lookup(355000) == 20);
55}

70TEST_CASE("tde5e_core: loads and resolves real TDE values") {
71 rpg_os::RulesetEngine engine;
72 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
73 CHECK(engine.ruleset().attributes.size() == 9); // 8 attributes + armor rating
74 CHECK(engine.ruleset().skills.size() > 40);
75 CHECK(engine.ruleset().checkTypes.size() == 3);
76
77 auto geron = engine.createEntity("geron");
78 REQUIRE(geron != nullptr);
79 // Life Points = 5 + 2 * CON (CON 13) = 31; current = max.
80 CHECK(engine.calculateStat(*geron, "LifePoints_Max") == 31);
81 CHECK(geron->resource("LP") == 31);
82 // Dodge = AGI / 2 = 13 / 2 = 6.
83 CHECK(engine.calculateStat(*geron, "Dodge") == 6);
84 // Attack = 6 + COU_Bonus; COU_Bonus = floor((12 - 8) / 3) = 1 -> 7.
85 CHECK(engine.calculateStat(*geron, "Attack") == 7);
86 // Parry = 3 + AGI_Bonus; AGI_Bonus = floor((13 - 8) / 3) = 1 -> 4.
87 CHECK(engine.calculateStat(*geron, "Parry") == 4);
88
89 // Real TDE skill check: climbing links COU/AGI/STR (12/13/13); skill rating 7.
90 auto rng = script({14, 12, 11}); // overshoots 2 + 0 + 0 = 2 -> 5 SP left.
91 const rpg_os::CheckResult result =
92 engine.executeSkillCheck("climbing", *geron, rpg_os::CheckParams{}, rng);
93 CHECK(result.isSuccess);
94 CHECK(result.remainingPool == 5);
95 CHECK(result.qualityLevel == 2);
96}

98TEST_CASE("tde5e_core: armor absorption and wound triggers (real TDE)") {
99 rpg_os::RulesetEngine engine;
100 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
101 auto geron = engine.createEntity("geron"); // Armor Rating 2
102 REQUIRE(geron != nullptr);
103 rpg_os::DynamicEntity orc(engine.ruleset(), "orc");
104
105 // 10 raw -> 10 - 2 (AR) = 8 applied; LP 31 -> 23.
106 CHECK(engine.applyDamage(orc, *geron, "LP", 10) == -8);
107 CHECK(geron->resource("LP") == 23);
108 CHECK_FALSE(geron->hasCondition("Wound"));
109
110 // 20 raw -> 18 applied (LP 23 -> 5); 18 > CON 13 -> Wound 1.
111 CHECK(engine.applyDamage(orc, *geron, "LP", 20) == -18);
112 CHECK(geron->resource("LP") == 5);
113 CHECK(geron->hasCondition("Wound"));
114 CHECK(geron->conditionStacks("Wound") == 1);
115}
