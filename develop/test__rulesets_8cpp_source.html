29TEST_CASE("dnd5e_srd: loads and resolves real SRD values") {
30 rpg_os::RulesetEngine engine;
31 REQUIRE(engine.loadRulesetFromFile(rulesetPath("dnd5e_srd.json")));
32 CHECK(engine.ruleset().attributes.size() == 9); // 6 abilities + prof + level + HP max
33 CHECK(engine.ruleset().skills.size() == 18);
34 CHECK(engine.ruleset().checkTypes.size() == 14);
35 CHECK(engine.ruleset().costTables.size() == 1);
36
37 // The SRD ruleset stores text descriptions (classes/species/backgrounds/feats)
38 // rather than archetype blocks, so build a fighter entity from a JSON record.
39 rpg_os::DynamicEntity fighter(engine.ruleset(), "fighter");
40 rpg_os::Json record = rpg_os::Json::parse(
41 R"({"attributes":{"STR":16,"DEX":14,"CON":14,"INT":10,"WIS":12,"CHA":8,"proficiency_bonus":2,"level":1,"HitPoints_Max":12}})");
42 fighter.loadFromArchetype(record);
43 // Ability modifiers from the real SRD table: STR 16 -> +3, CON 14 -> +2.
44 CHECK(engine.calculateStat(fighter, "STR_mod") == 3);
45 CHECK(engine.calculateStat(fighter, "CON_mod") == 2);
46 // Base AC = 10 + DEX modifier (DEX 14 -> +2).
47 CHECK(engine.calculateStat(fighter, "AC") == 12);
48 // Hit points from the record (fighter: 10 + CON mod = 12).
49 CHECK(fighter.resource("HP") == 12);
50 CHECK(fighter.baseAttribute("proficiency_bonus") == 2);
51
52 // Real XP -> level table (SRD Character Advancement).
53 const auto *xp = engine.ruleset().findCostTable("xp_to_level");
54 REQUIRE(xp != nullptr);
55 CHECK(xp->table.lookup(0) == 1);
56 CHECK(xp->table.lookup(2700) == 4);
57 CHECK(xp->table.lookup(6500) == 5);
58 CHECK(xp->table.lookup(355000) == 20);
59}

76TEST_CASE("tde5e_core: loads and resolves real TDE values") {
77 rpg_os::RulesetEngine engine;
78 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
79 CHECK(engine.ruleset().attributes.size() == 9); // 8 attributes + armor rating
80 CHECK(engine.ruleset().skills.size() > 40);
81 CHECK(engine.ruleset().checkTypes.size() == 3);
82
83 auto geron = engine.createEntity("geron");
84 REQUIRE(geron != nullptr);
85 // Life Points = 5 + 2 * CON (CON 13) = 31; current = max.
86 CHECK(engine.calculateStat(*geron, "LifePoints_Max") == 31);
87 CHECK(geron->resource("LP") == 31);
88 // Dodge = AGI / 2 = 13 / 2 = 6.5, rounded up = 7 (TDE rounds mathematically).
89 CHECK(engine.calculateStat(*geron, "Dodge") == 7);
90 // Attack = 6 + COU_Bonus; COU_Bonus = floor((12 - 8) / 3) = 1 -> 7.
91 CHECK(engine.calculateStat(*geron, "Attack") == 7);
92 // Parry = 3 + AGI_Bonus; AGI_Bonus = floor((13 - 8) / 3) = 1 -> 4.
93 CHECK(engine.calculateStat(*geron, "Parry") == 4);
94 // Spirit = round((COU + SGC + INT) / 6) = round(33 / 6) = 6 (TDE rounds up).
95 CHECK(engine.calculateStat(*geron, "Spirit") == 6);
96 // Toughness = round((CON + CON + STR) / 6) = round(39 / 6) = 7.
97 CHECK(engine.calculateStat(*geron, "Toughness") == 7);
98
99 // Real TDE skill check: climbing links COU/AGI/STR (12/13/13); skill rating 7.
100 auto rng = script({14, 12, 11}); // overshoots 2 + 0 + 0 = 2 -> 5 SP left.
101 const rpg_os::CheckResult result =
102 engine.executeSkillCheck("climbing", *geron, rpg_os::CheckParams{}, rng);
103 CHECK(result.isSuccess);
104 CHECK(result.remainingPool == 5);
105 CHECK(result.qualityLevel == 2);
106}
