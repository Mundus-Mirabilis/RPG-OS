52inline bool readJsonNumber(const Json &obj, std::string_view key, double &out) {
53 if (!obj.is_object()) {
54 return false;
55 }
56 const auto it = obj.find(std::string(key));
57 if (it == obj.end() || !it->is_number()) {
58 return false;
59 }
60 out = it->get<double>();
61 return true;
62}

203 template <RandomNumberGenerator Rng>
204 void loadFromArchetype(const Json &archetype, Variance variance, Rng &rng) {
205 if (archetype.contains("attributes")) {
206 for (const auto &[statId, value] : archetype.at("attributes").items()) {
207 m_stats[statId] = readVariantValue(value, variance, rng);
208 }
209 }
210 if (archetype.contains("skills")) {
211 for (const auto &[skillId, value] : archetype.at("skills").items()) {
212 m_stats[skillId] = readVariantValue(value, variance, rng);
213 }
214 }
215 refreshResources();
216 if (archetype.contains("resources")) {
217 for (const auto &[resourceId, value] : archetype.at("resources").items()) {
218 m_resources[resourceId].current = readVariantValue(value, variance, rng);
219 }
220 }
221 if (archetype.contains("conditions")) {
222 for (const auto &[conditionId, value] : archetype.at("conditions").items()) {
223 m_conditions[conditionId] = value.get<int32_t>();
224 }
225 }
226 }
227
228private:
233 void refreshResources() {
234 for (const ResourcePoolDef &def : m_ruleset->resourcePools) {
235 const int32_t maxValue = getStat(def.maxStat);
236 const auto it = m_resources.find(def.id);
237 if (it == m_resources.end()) {
238 m_resources.emplace(def.id, ResourcePool{maxValue, def.minValue, maxValue});
239 } else {
240 it->second.min = def.minValue;
241 it->second.max = maxValue;
242 }
243 }
244 }

249 [[nodiscard]] int32_t computeDerived(std::string_view statId) const {
250 const DerivedStatDef *def = m_ruleset->findDerivedStat(statId);
251 if (def == nullptr) {
252 return 0;
253 }
254 const Json emptyEnv = Json::object();
255 const Json emptyParams = Json::object();
256 const EntityContext context(*this, nullptr, emptyEnv, emptyParams);
257 return math::toStat(def->expression.evaluate(context));
258 }
259
260 const Ruleset *m_ruleset;

263 std::unordered_map<std::string, ResourcePool> m_resources;
264 std::unordered_map<std::string, int32_t> m_conditions;

267inline bool EntityContext::resolve(std::string_view path, double &out) const {
268 const std::size_t dot = path.find('.');
269 if (dot == std::string_view::npos) {
270 out = static_cast<double>(m_actor.getStat(path));
271 return true;
272 }
273 const std::string_view prefix = path.substr(0, dot);
274 const std::string_view rest = path.substr(dot + 1);
275 if (prefix == "actor") {
276 out = static_cast<double>(m_actor.getStat(rest));
277 return true;
278 }
279 if (prefix == "target") {
280 out = m_target != nullptr ? static_cast<double>(m_target->getStat(rest)) : 0.0;
281 return true;
282 }
283 if (prefix == "env") {
284 return readJsonNumber(m_env, rest, out);
285 }
286 if (prefix == "event" || prefix == "action") {
287 return readJsonNumber(m_params, rest, out);
288 }
289 return false;
290}
