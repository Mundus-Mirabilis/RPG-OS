66inline bool readJsonNumber(const Json &obj, std::string_view key, double &out) {
67 if (!obj.is_object()) {
68 return false;
69 }
70 const auto it = obj.find(std::string(key));
71 if (it == obj.end() || !it->is_number()) {
72 return false;
73 }
74 out = it->get<double>();
75 return true;

88inline Modifier parseModifierJson(const Json &mod) {
89 Modifier parsed;
90 const std::string type = mod.value("type", "add");
91 if (type == "override") {
92 parsed.type = ModifierType::Override;
93 parsed.value = mod.value("value", 0);
94 } else if (type == "multiply") {
95 parsed.type = ModifierType::Multiply;
96 parsed.factor = mod.value("factor", mod.value("value", 1.0));
97 } else if (type == "clamp") {
98 parsed.type = ModifierType::Clamp;
99 parsed.clampMin = mod.value("clamp_min", mod.value("min", 0));
100 parsed.clampMax = mod.value("clamp_max", mod.value("max", 0));
101 } else { // "add"
102 parsed.type = ModifierType::Add;
103 parsed.value = mod.value("value", 0);
104 }
105 return parsed;
106}

247 [[nodiscard]] int32_t modifyResource(std::string_view resourceId, int32_t delta) {
248 const auto it = m_resources.find(std::string(resourceId));
249 if (it == m_resources.end()) {
250 return 0;
251 }
252 const int32_t old = it->second.current;
253 const int32_t applied = it->second.modify(delta);
254 if (applied != 0) {
255 emitEvent(EventType::OnResourceChanged, Json{{"resource", std::string(resourceId)},

325 [[nodiscard]] std::vector<std::string> collectTokens(std::string_view field) const {
326 std::vector<std::string> out;
327 const auto addToken = [&out](const std::string &token) {
328 if (std::find(out.begin(), out.end(), token) == out.end()) {
329 out.push_back(token);
330 }
331 };
332 for (const auto &[conditionId, stacks] : m_conditions) {
333 if (stacks <= 0) {
334 continue;
335 }
336 const Json *record = findDataRecord("conditions", conditionId);
337 if (record == nullptr || !record->contains(field) || !record->at(field).is_array()) {
338 continue;
339 }
340 for (const Json &token : record->at(field)) {
341 addToken(token.get<std::string>());
342 }
343 }
344 for (const std::string &traitId : m_traits) {
345 const Json *record = findDataRecord("traits", traitId);
346 if (record == nullptr || !record->contains(field) || !record->at(field).is_array()) {
347 continue;
348 }
349 for (const Json &token : record->at(field)) {
350 addToken(token.get<std::string>());
351 }
352 }
353 return out;
354 }

377 [[nodiscard]] std::vector<std::string> capabilities() const {
378 std::vector<std::string> out = collectTokens("capabilities");
379 if (m_terrain.empty()) {
380 return out;
381 }
382 const auto addToken = [&out](const std::string &token) {
383 if (std::find(out.begin(), out.end(), token) == out.end()) {
384 out.push_back(token);
385 }
386 };
387 for (const auto &[slot, itemId] : m_equipment.slots()) {
388 const TerrainItemEffect effect = itemTerrainEffect(itemId);
389 for (const std::string &grant : effect.grants) {
390 addToken(grant);
391 }
392 }
393 return out;
394 }

552 void toJson(Json &out) const {
553 out = Json::object();
554 out["id"] = m_id;
555 out["entity_id"] = m_entityId.value;
556 out["stats"] = m_stats;
557 out["terrain"] = m_terrain;
558 Json resources = Json::object();
559 for (const auto &[id, pool] : m_resources) {
560 resources[id] = pool.current;
561 }
562 out["resources"] = resources;
563 out["conditions"] = m_conditions;
564 Json traits = Json::array();
565 for (const std::string &trait : m_traits) {
566 traits.push_back(trait);
567 }
568 out["traits"] = traits;
569 Json inventoryJson;
570 m_inventory.toJson(inventoryJson);
571 out["inventory"] = inventoryJson;
572 Json equipmentJson;
573 m_equipment.toJson(equipmentJson);
574 out["equipment"] = equipmentJson;
575 out["money"] = m_money.baseUnits();
576 out["temp_hp"] = m_tempHp;
577 Json spellbookJson;
578 m_spellbook.toJson(spellbookJson);
579 out["spellbook"] = spellbookJson;
580 Json advancementJson;
581 m_advancement.toJson(advancementJson);
582 out["advancement"] = advancementJson;
583 Json effectsJson;
584 m_effects.toJson(effectsJson);
585 out["effects"] = effectsJson;
586 Json afflictions = Json::array();
587 for (const AppliedAffliction &aff : m_afflictions) {
588 afflictions.push_back(
589 {{"section", aff.section}, {"id", aff.id}, {"conditions", aff.conditions}});
590 }
591 out["afflictions"] = afflictions;
592 Json resistances = Json::array();
593 for (const std::string &type : m_resistances) {
594 resistances.push_back(type);
595 }
596 out["resistances"] = resistances;
597 }

604 }

610 restoreTraits(in);

613 restoreEquipment(in);

616 restoreSpellbook(in);
617 restoreAdvancement(in);

619 restoreAfflictions(in);
620 restoreResistances(in);
621 restoreTerrain(in);
622 }

634 void refreshResources() {
635 for (const ResourcePoolDef &def : m_ruleset->resourcePools) {
636 const int32_t maxValue = getStat(def.maxStat);
637 const auto it = m_resources.find(def.id);
638 if (it == m_resources.end()) {
639 m_resources.emplace(def.id, ResourcePool{maxValue, def.minValue, maxValue});
640 } else {
641 it->second.min = def.minValue;
642 it->second.max = maxValue;
643 }
644 }

886 [[nodiscard]] int32_t resolveModifierNumber(const Json &mod, std::string_view key,
887 int32_t fallback, const EntityContext &ctx) const {
888 const auto it = mod.find(std::string(key));
889 if (it == mod.end()) {
890 return fallback;
891 }
892 const Json &value = *it;
893 if (value.is_number()) {
894 return value.get<int32_t>();
895 }
896 if (value.is_string()) {
897 const std::string text = value.get<std::string>();
898 auto cit = m_formulaCache.find(text);
899 if (cit == m_formulaCache.end()) {
900 try {
901 cit = m_formulaCache.emplace(text, rpg_os::Expression(text)).first;
902 } catch (const std::exception &) {
903 return fallback; // malformed formula: ignore, like other misuse
904 }
905 }
906 try {
907 return math::toStat(cit->second.evaluate(ctx));
908 } catch (const std::exception &) {
909 return fallback; // unresolvable identifier / division by zero
910 }
911 }
912 return fallback;
913 }

917 [[nodiscard]] double resolveModifierFactor(const Json &mod, std::string_view key, double fallback,
918 const EntityContext &ctx) const {
919 const auto it = mod.find(std::string(key));
920 if (it == mod.end()) {
921 return fallback;
922 }
923 const Json &value = *it;
924 if (value.is_number()) {
925 return value.get<double>();
926 }
927 if (value.is_string()) {
928 const std::string text = value.get<std::string>();
929 auto cit = m_formulaCache.find(text);
930 if (cit == m_formulaCache.end()) {
931 try {
932 cit = m_formulaCache.emplace(text, rpg_os::Expression(text)).first;
933 } catch (const std::exception &) {
934 return fallback;
935 }
936 }
937 try {
938 return cit->second.evaluate(ctx);
939 } catch (const std::exception &) {
940 return fallback;
941 }
942 }
943 return fallback;
944 }

958 void appendRecordModifiers(std::vector<Modifier> &result, const EntityContext &ctx,
959 const Json *record, std::string_view arrayField,
960 std::string_view statField, std::string_view statId,
961 int32_t stackScale = 1) const {
962 if (record == nullptr || !record->contains(arrayField) || !record->at(arrayField).is_array()) {
963 return;
964 }
965 const auto hasFormulaField = [](const Json &mod) {
966 for (const char *key : {"value", "factor", "clamp_min", "clamp_max", "min", "max"}) {
967 if (mod.contains(key) && mod.at(key).is_string()) {
968 return true;
969 }
970 }
971 return false;
972 };
973 for (const Json &mod : record->at(arrayField)) {
974 if (mod.value(statField, "") != statId) {
975 continue;
976 }
977 Modifier parsed;
978 if (!hasFormulaField(mod)) {
979 parsed = parseModifierJson(mod); // numeric fast path
980 } else {
981 const std::string type = mod.value("type", "add");
982 if (type == "override") {
983 parsed.type = ModifierType::Override;
984 parsed.value = resolveModifierNumber(mod, "value", 0, ctx);
985 } else if (type == "multiply") {
986 parsed.type = ModifierType::Multiply;
987 parsed.factor = mod.contains("factor")
988 ? resolveModifierFactor(mod, "factor", 1.0, ctx)
989 : static_cast<double>(resolveModifierNumber(mod, "value", 1, ctx));
990 } else if (type == "clamp") {
991 parsed.type = ModifierType::Clamp;
992 parsed.clampMin =
993 resolveModifierNumber(mod, mod.contains("clamp_min") ? "clamp_min" : "min", 0, ctx);
994 parsed.clampMax =
995 resolveModifierNumber(mod, mod.contains("clamp_max") ? "clamp_max" : "max", 0, ctx);
996 } else { // "add"
997 parsed.type = ModifierType::Add;
998 parsed.value = resolveModifierNumber(mod, "value", 0, ctx);
999 }
1000 }
1001 if (stackScale != 1 && parsed.type == ModifierType::Add) {
1002 parsed.value *= stackScale;
1003 }
1004 result.push_back(parsed);
1005 }
1006 }

1013 std::vector<Modifier> result;
1014 // One context for the whole sweep: formula-valued modifiers resolve against
1015 // this entity, and reusing the context avoids rebuilding it per record.
1016 const EntityContext ctx(*this, nullptr, Json::object(), Json::object());
1017 for (const auto &[slot, itemId] : m_equipment.slots()) {
1018 appendRecordModifiers(result, ctx, findDataRecord("items", itemId), "modifiers",

1023 continue;

1026 "stat_modifiers", "stat", statId, stacks);
1027 }
1028 for (const std::string &traitId : m_traits) {
1029 appendRecordModifiers(result, ctx, findDataRecord("traits", traitId), "stat_modifiers",
1030 "stat", statId);
1031 }
1032 return result;
1033 }

1117inline bool EntityContext::resolve(std::string_view path, double &out) const {
1118 const std::size_t dot = path.find('.');
1119 if (dot == std::string_view::npos) {
1120 out = static_cast<double>(m_actor.getStat(path));
1121 return true;
1122 }
1123 const std::string_view prefix = path.substr(0, dot);
1124 const std::string_view rest = path.substr(dot + 1);
1125 if (prefix == "actor") {
1126 out = static_cast<double>(m_actor.getStat(rest));
1127 return true;
1128 }
1129 if (prefix == "target") {
1130 out = m_target != nullptr ? static_cast<double>(m_target->getStat(rest)) : 0.0;
1131 return true;
1132 }
1133 if (prefix == "effective") {
1134 const DynamicEntity &subject = m_target != nullptr ? *m_target : m_actor;
1135 out = static_cast<double>(subject.getEffectiveStat(rest));
1136 return true;
1137 }
1138 if (prefix == "env") {
1139 return readJsonNumber(m_env, rest, out);
1140 }
1141 if (prefix == "event" || prefix == "action") {
1142 return readJsonNumber(m_params, rest, out);
1143 }
1144 return false;
1145}
