62inline bool readJsonNumber(const Json &obj, std::string_view key, double &out) {

84inline Modifier parseModifierJson(const Json &mod) {
85 Modifier parsed;
86 const std::string type = mod.value("type", "add");
87 if (type == "override") {
88 parsed.type = ModifierType::Override;
89 parsed.value = mod.value("value", 0);
90 } else if (type == "multiply") {
91 parsed.type = ModifierType::Multiply;
92 parsed.factor = mod.value("factor", mod.value("value", 1.0));
93 } else if (type == "clamp") {
94 parsed.type = ModifierType::Clamp;
95 parsed.clampMin = mod.value("clamp_min", mod.value("min", 0));
96 parsed.clampMax = mod.value("clamp_max", mod.value("max", 0));
97 } else { // "add"
98 parsed.type = ModifierType::Add;
99 parsed.value = mod.value("value", 0);
100 }
101 return parsed;
102}

268 [[nodiscard]] std::vector<std::string> collectTokens(std::string_view field) const {
269 std::vector<std::string> out;
270 const auto addToken = [&out](const std::string &token) {
271 if (std::find(out.begin(), out.end(), token) == out.end()) {
272 out.push_back(token);
273 }
274 };
275 for (const auto &[conditionId, stacks] : m_conditions) {
276 if (stacks <= 0) {
277 continue;
278 }
279 const Json *record = findDataRecord("conditions", conditionId);
280 if (record == nullptr || !record->contains(field) || !record->at(field).is_array()) {
281 continue;
282 }
283 for (const Json &token : record->at(field)) {
284 addToken(token.get<std::string>());
285 }
286 }
287 for (const std::string &traitId : m_traits) {
288 const Json *record = findDataRecord("traits", traitId);
289 if (record == nullptr || !record->contains(field) || !record->at(field).is_array()) {
290 continue;
291 }
292 for (const Json &token : record->at(field)) {
293 addToken(token.get<std::string>());
294 }
295 }
296 return out;

573 template <RandomNumberGenerator Rng>
574 void loadFromArchetype(const Json &archetype, Variance variance, Rng &rng) {
575 if (archetype.contains("attributes")) {
576 for (const auto &[statId, value] : archetype.at("attributes").items()) {
577 m_stats[statId] = readVariantValue(value, variance, rng);
578 }
579 }
580 if (archetype.contains("skills")) {
581 for (const auto &[skillId, value] : archetype.at("skills").items()) {
582 m_stats[skillId] = readVariantValue(value, variance, rng);
583 }
584 }
585 refreshResources();
586 if (archetype.contains("resources")) {
587 for (const auto &[resourceId, value] : archetype.at("resources").items()) {
588 m_resources[resourceId].current = readVariantValue(value, variance, rng);
589 }
590 }
591 if (archetype.contains("conditions")) {
592 for (const auto &[conditionId, value] : archetype.at("conditions").items()) {
593 m_conditions[conditionId] = value.get<int32_t>();
594 }
595 }
596 if (archetype.contains("traits") && archetype.at("traits").is_array()) {
597 for (const Json &trait : archetype.at("traits")) {
598 if (trait.is_string()) {
599 m_traits.insert(trait.get<std::string>());
600 }
601 }
602 }
603 // Bookkeeping fields an archetype (or bestiary entry) may declare. Each is
604 // optional: a ruleset without money / gear / spells simply omits them.
605 if (archetype.contains("wealth")) {
606 const Json &wealth = archetype.at("wealth");
607 if (wealth.is_object() && !m_ruleset->currencySystem.id.empty()) {
608 int64_t total = 0;
609 for (const auto &[denom, qty] : wealth.items()) {
610 total += m_ruleset->currencySystem.valueOf(denom, qty.get<int64_t>());
611 }
612 m_money = Money{total};
613 } else if (wealth.is_number_integer()) {
614 m_money = Money{wealth.get<int64_t>()};
615 }
616 }
617 if (archetype.contains("equipment")) {
618 const Json &equip = archetype.at("equipment");
619 if (equip.is_object()) {
620 for (const auto &[slot, item] : equip.items()) {
621 (void)m_equipment.equip(slot, item.get<std::string>());
622 }
623 } else if (equip.is_array()) {
624 for (const Json &entry : equip) {
625 (void)m_equipment.equip(entry.value("slot", ""), entry.value("item", ""));
626 }
627 }
628 }
629 if (archetype.contains("spells_known")) {
630 for (const Json &spellId : archetype.at("spells_known")) {
631 m_spellbook.learn(spellId.get<std::string>());
632 }
633 }
634 if (archetype.contains("xp")) {
635 m_advancement.gainXp(archetype.at("xp").get<int64_t>());
636 }
637 if (archetype.contains("level")) {
638 m_advancement.level = archetype.at("level").get<int32_t>();
639 }
640 }

647 [[nodiscard]] std::vector<Modifier> modifiersFor(std::string_view statId) const {
648 std::vector<Modifier> result;
649 for (const auto &[slot, itemId] : m_equipment.slots()) {
650 const Json *item = findDataRecord("items", itemId);
651 if (item == nullptr || !item->contains("modifiers") || !item->at("modifiers").is_array()) {
652 continue;
653 }
654 for (const Json &mod : item->at("modifiers")) {
655 if (mod.value("target_stat", "") == statId) {
656 result.push_back(parseModifierJson(mod));
657 }
658 }
659 }
660 for (const auto &[conditionId, stacks] : m_conditions) {
661 if (stacks <= 0) {
662 continue;
663 }
664 const Json *cond = findDataRecord("conditions", conditionId);
665 if (cond == nullptr || !cond->contains("stat_modifiers") ||
666 !cond->at("stat_modifiers").is_array()) {
667 continue;
668 }
669 for (const Json &mod : cond->at("stat_modifiers")) {
670 if (mod.value("stat", "") == statId) {
671 Modifier parsed = parseModifierJson(mod);
672 if (parsed.type == ModifierType::Add) {
673 parsed.value *= stacks;
674 }
675 result.push_back(parsed);
676 }
677 }
678 }
679 for (const std::string &traitId : m_traits) {
680 const Json *trait = findDataRecord("traits", traitId);
681 if (trait == nullptr || !trait->contains("stat_modifiers") ||
682 !trait->at("stat_modifiers").is_array()) {
683 continue;
684 }
685 for (const Json &mod : trait->at("stat_modifiers")) {
686 if (mod.value("stat", "") == statId) {
687 result.push_back(parseModifierJson(mod));
688 }
689 }
690 }
691 return result;
692 }

712 void refreshResources() {
713 for (const ResourcePoolDef &def : m_ruleset->resourcePools) {
714 const int32_t maxValue = getStat(def.maxStat);
715 const auto it = m_resources.find(def.id);
716 if (it == m_resources.end()) {
717 m_resources.emplace(def.id, ResourcePool{maxValue, def.minValue, maxValue});
718 } else {
719 it->second.min = def.minValue;
720 it->second.max = maxValue;
721 }

756inline bool EntityContext::resolve(std::string_view path, double &out) const {
757 const std::size_t dot = path.find('.');
758 if (dot == std::string_view::npos) {
759 out = static_cast<double>(m_actor.getStat(path));
760 return true;
761 }
762 const std::string_view prefix = path.substr(0, dot);
763 const std::string_view rest = path.substr(dot + 1);
764 if (prefix == "actor") {
765 out = static_cast<double>(m_actor.getStat(rest));
766 return true;
767 }
768 if (prefix == "target") {
769 out = m_target != nullptr ? static_cast<double>(m_target->getStat(rest)) : 0.0;
770 return true;
771 }
772 if (prefix == "effective") {
773 const DynamicEntity &subject = m_target != nullptr ? *m_target : m_actor;
774 out = static_cast<double>(subject.getEffectiveStat(rest));
775 return true;
776 }
777 if (prefix == "env") {
778 return readJsonNumber(m_env, rest, out);
779 }
780 if (prefix == "event" || prefix == "action") {
781 return readJsonNumber(m_params, rest, out);
782 }
783 return false;
784}
