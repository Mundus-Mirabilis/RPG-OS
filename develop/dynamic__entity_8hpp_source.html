65inline bool readJsonNumber(const Json &obj, std::string_view key, double &out) {
66 if (!obj.is_object()) {
67 return false;
68 }
69 const auto it = obj.find(std::string(key));
70 if (it == obj.end() || !it->is_number()) {
71 return false;
72 }
73 out = it->get<double>();
74 return true;
75}

87inline Modifier parseModifierJson(const Json &mod) {
88 Modifier parsed;
89 const std::string type = mod.value("type", "add");
90 if (type == "override") {
91 parsed.type = ModifierType::Override;
92 parsed.value = mod.value("value", 0);
93 } else if (type == "multiply") {
94 parsed.type = ModifierType::Multiply;
95 parsed.factor = mod.value("factor", mod.value("value", 1.0));
96 } else if (type == "clamp") {
97 parsed.type = ModifierType::Clamp;
98 parsed.clampMin = mod.value("clamp_min", mod.value("min", 0));
99 parsed.clampMax = mod.value("clamp_max", mod.value("max", 0));
100 } else { // "add"
101 parsed.type = ModifierType::Add;
102 parsed.value = mod.value("value", 0);
103 }
104 return parsed;
105}

324 [[nodiscard]] std::vector<std::string> collectTokens(std::string_view field) const {
325 std::vector<std::string> out;
326 const auto addToken = [&out](const std::string &token) {
327 if (std::find(out.begin(), out.end(), token) == out.end()) {
328 out.push_back(token);
329 }
330 };
331 for (const auto &[conditionId, stacks] : m_conditions) {
332 if (stacks <= 0) {
333 continue;
334 }
335 const Json *record = findDataRecord("conditions", conditionId);
336 if (record == nullptr || !record->contains(field) || !record->at(field).is_array()) {
337 continue;
338 }
339 for (const Json &token : record->at(field)) {
340 addToken(token.get<std::string>());
341 }
342 }
343 for (const std::string &traitId : m_traits) {
344 const Json *record = findDataRecord("traits", traitId);
345 if (record == nullptr || !record->contains(field) || !record->at(field).is_array()) {
346 continue;
347 }
348 for (const Json &token : record->at(field)) {
349 addToken(token.get<std::string>());
350 }
351 }
352 return out;
353 }

506 void toJson(Json &out) const {
507 out = Json::object();
508 out["id"] = m_id;
509 out["entity_id"] = m_entityId.value;
510 out["stats"] = m_stats;
511 Json resources = Json::object();
512 for (const auto &[id, pool] : m_resources) {
513 resources[id] = pool.current;
514 }
515 out["resources"] = resources;
516 out["conditions"] = m_conditions;
517 Json traits = Json::array();
518 for (const std::string &trait : m_traits) {
519 traits.push_back(trait);
520 }
521 out["traits"] = traits;
522 Json inventoryJson;
523 m_inventory.toJson(inventoryJson);
524 out["inventory"] = inventoryJson;
525 Json equipmentJson;
526 m_equipment.toJson(equipmentJson);
527 out["equipment"] = equipmentJson;
528 out["money"] = m_money.baseUnits();
529 out["temp_hp"] = m_tempHp;
530 Json spellbookJson;
531 m_spellbook.toJson(spellbookJson);
532 out["spellbook"] = spellbookJson;
533 Json advancementJson;
534 m_advancement.toJson(advancementJson);
535 out["advancement"] = advancementJson;
536 Json effectsJson;
537 m_effects.toJson(effectsJson);
538 out["effects"] = effectsJson;
539 Json afflictions = Json::array();
540 for (const AppliedAffliction &aff : m_afflictions) {
541 afflictions.push_back(
542 {{"section", aff.section}, {"id", aff.id}, {"conditions", aff.conditions}});
543 }
544 out["afflictions"] = afflictions;
545 Json resistances = Json::array();
546 for (const std::string &type : m_resistances) {
547 resistances.push_back(type);
548 }
549 out["resistances"] = resistances;
550 }

554 void fromJson(const Json &in) {
555 if (!in.is_object()) {
556 return;
557 }
558 if (in.contains("entity_id") && in.at("entity_id").is_number_unsigned()) {
559 m_entityId = EntityId{in.at("entity_id").get<uint64_t>()};

562 restoreConditions(in);
563 restoreTraits(in);
564 restoreResources(in);
565 restoreInventory(in);
566 restoreEquipment(in);
567 restoreMoney(in);

569 restoreSpellbook(in);
570 restoreAdvancement(in);
571 restoreEffects(in);
572 restoreAfflictions(in);
573 restoreResistances(in);

586 void refreshResources() {
587 for (const ResourcePoolDef &def : m_ruleset->resourcePools) {
588 const int32_t maxValue = getStat(def.maxStat);
589 const auto it = m_resources.find(def.id);
590 if (it == m_resources.end()) {
591 m_resources.emplace(def.id, ResourcePool{maxValue, def.minValue, maxValue});
592 } else {
593 it->second.min = def.minValue;
594 it->second.max = maxValue;
595 }
596 }

829 static void appendRecordModifiers(std::vector<Modifier> &result, const Json *record,
830 std::string_view arrayField, std::string_view statField,
831 std::string_view statId, int32_t stackScale = 1) {
832 if (record == nullptr || !record->contains(arrayField) || !record->at(arrayField).is_array()) {
833 return;
834 }
835 for (const Json &mod : record->at(arrayField)) {
836 if (mod.value(statField, "") != statId) {
837 continue;
838 }
839 Modifier parsed = parseModifierJson(mod);
840 if (stackScale != 1 && parsed.type == ModifierType::Add) {
841 parsed.value *= stackScale;
842 }
843 result.push_back(parsed);
844 }
845 }

851 [[nodiscard]] std::vector<Modifier> modifiersFor(std::string_view statId) const {
852 std::vector<Modifier> result;
853 for (const auto &[slot, itemId] : m_equipment.slots()) {
854 appendRecordModifiers(result, findDataRecord("items", itemId), "modifiers", "target_stat",
855 statId);
856 }
857 for (const auto &[conditionId, stacks] : m_conditions) {
858 if (stacks <= 0) {
859 continue;
860 }
861 appendRecordModifiers(result, findDataRecord("conditions", conditionId), "stat_modifiers",
862 "stat", statId, stacks);
863 }
864 for (const std::string &traitId : m_traits) {
865 appendRecordModifiers(result, findDataRecord("traits", traitId), "stat_modifiers", "stat",
866 statId);
867 }
868 return result;
869 }

880 [[nodiscard]] int32_t computeDerived(std::string_view statId) const {
881 const DerivedStatDef *def = m_ruleset->findDerivedStat(statId);
882 if (def == nullptr) {
883 return 0;
884 }
885 const Json emptyEnv = Json::object();
886 const Json emptyParams = Json::object();
887 const EntityContext context(*this, nullptr, emptyEnv, emptyParams);
888 return math::toStat(def->expression.evaluate(context));
889 }

920inline bool EntityContext::resolve(std::string_view path, double &out) const {
921 const std::size_t dot = path.find('.');
922 if (dot == std::string_view::npos) {
923 out = static_cast<double>(m_actor.getStat(path));
924 return true;
925 }
926 const std::string_view prefix = path.substr(0, dot);
927 const std::string_view rest = path.substr(dot + 1);
928 if (prefix == "actor") {
929 out = static_cast<double>(m_actor.getStat(rest));
930 return true;
931 }
932 if (prefix == "target") {
933 out = m_target != nullptr ? static_cast<double>(m_target->getStat(rest)) : 0.0;
934 return true;
935 }
936 if (prefix == "effective") {
937 const DynamicEntity &subject = m_target != nullptr ? *m_target : m_actor;
938 out = static_cast<double>(subject.getEffectiveStat(rest));
939 return true;
940 }
941 if (prefix == "env") {
942 return readJsonNumber(m_env, rest, out);
943 }
944 if (prefix == "event" || prefix == "action") {
945 return readJsonNumber(m_params, rest, out);
946 }
947 return false;
948}
