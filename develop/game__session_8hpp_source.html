285 [[nodiscard]] Json saveState() const {
286 Json out = Json::object();
287 Json worldJson;
288 m_world.toJson(worldJson);
289 out["world"] = std::move(worldJson);
290 Json sheets = Json::array();
291 for (const DynamicEntity *sheet : m_registry.all()) {
292 Json sheetJson;
293 sheet->toJson(sheetJson);
294 sheets.push_back(std::move(sheetJson));
295 }
296 out["sheets"] = sheets;
297 return out;
298 }
299
307 std::size_t loadState(const Json &in) {
308 if (!in.is_object()) {
309 return 0;
310 }
311 m_registry.clear();
312 if (in.contains("world") && in.at("world").is_object()) {
313 m_world.fromJson(in.at("world"));
314 }
315 if (!in.contains("sheets") || !in.at("sheets").is_array()) {
316 return 0;
317 }
318 std::size_t restored = 0;
319 for (const Json &sheetJson : in.at("sheets")) {
320 if (!sheetJson.is_object()) {
321 continue;
322 }
323 const std::string typeId = sheetJson.value("id", "");
324 if (typeId.empty()) {
325 continue;
326 }
327 std::shared_ptr<DynamicEntity> sheet = m_engine.createEntity(typeId);
328 if (sheet == nullptr) {
329 sheet = m_engine.createCreature(typeId);
330 }
331 if (sheet == nullptr) {
332 continue; // type not present in this ruleset
333 }
334 sheet->setEventsSuppressed(true);
335 sheet->fromJson(sheetJson);
336 sheet->setEventsSuppressed(false);
337 (void)m_registry.add(std::move(sheet));
338 ++restored;
339 }
340 return restored;
341 }
