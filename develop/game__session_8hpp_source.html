218 // ---- save / load ------------------------------------------------------
222 [[nodiscard]] Json saveState() const {
223 Json out = Json::object();
224 Json worldJson;
225 m_world.toJson(worldJson);
226 out["world"] = std::move(worldJson);
227 Json sheets = Json::array();
228 for (const DynamicEntity *sheet : m_registry.all()) {
229 Json sheetJson;
230 sheet->toJson(sheetJson);
231 sheets.push_back(std::move(sheetJson));
232 }
233 out["sheets"] = sheets;
234 return out;
235 }
236
244 std::size_t loadState(const Json &in) {
245 if (!in.is_object()) {
246 return 0;
247 }
248 m_registry.clear();
249 if (in.contains("world") && in.at("world").is_object()) {
250 m_world.fromJson(in.at("world"));
251 }
252 if (!in.contains("sheets") || !in.at("sheets").is_array()) {
253 return 0;
254 }
255 std::size_t restored = 0;
256 for (const Json &sheetJson : in.at("sheets")) {
257 if (!sheetJson.is_object()) {
258 continue;
259 }
260 const std::string typeId = sheetJson.value("id", "");
261 if (typeId.empty()) {
262 continue;
263 }
264 std::shared_ptr<DynamicEntity> sheet = m_engine.createEntity(typeId);
265 if (sheet == nullptr) {
266 sheet = m_engine.createCreature(typeId);
267 }
268 if (sheet == nullptr) {
269 continue; // type not present in this ruleset
270 }
271 sheet->setEventsSuppressed(true);
272 sheet->fromJson(sheetJson);
273 sheet->setEventsSuppressed(false);
274 (void)m_registry.add(std::move(sheet));
275 ++restored;
276 }
277 return restored;
278 }
