223 [[nodiscard]] Json saveState() const {
224 Json out = Json::object();
225 Json worldJson;
226 m_world.toJson(worldJson);
227 out["world"] = std::move(worldJson);
228 Json sheets = Json::array();
229 for (const DynamicEntity *sheet : m_registry.all()) {
230 Json sheetJson;
231 sheet->toJson(sheetJson);
232 sheets.push_back(std::move(sheetJson));
233 }
234 out["sheets"] = sheets;
235 return out;
236 }
237
245 std::size_t loadState(const Json &in) {
246 if (!in.is_object()) {
247 return 0;
248 }
249 m_registry.clear();
250 if (in.contains("world") && in.at("world").is_object()) {
251 m_world.fromJson(in.at("world"));
252 }
253 if (!in.contains("sheets") || !in.at("sheets").is_array()) {
254 return 0;
255 }
256 std::size_t restored = 0;
257 for (const Json &sheetJson : in.at("sheets")) {
258 if (!sheetJson.is_object()) {
259 continue;
260 }
261 const std::string typeId = sheetJson.value("id", "");
262 if (typeId.empty()) {
263 continue;
264 }
265 std::shared_ptr<DynamicEntity> sheet = m_engine.createEntity(typeId);
266 if (sheet == nullptr) {
267 sheet = m_engine.createCreature(typeId);
268 }
269 if (sheet == nullptr) {
270 continue; // type not present in this ruleset
271 }
272 sheet->setEventsSuppressed(true);
273 sheet->fromJson(sheetJson);
274 sheet->setEventsSuppressed(false);
275 (void)m_registry.add(std::move(sheet));
276 ++restored;
277 }
278 return restored;
279 }
