44rpg_os::Json loadRuleset(const std::string &root, std::string_view name) {

53int main(int argc, char **argv) {
54 // Optional leading positional argument = project root, matching the sibling
55 // examples so all three demos share the same invocation convention.
56 const std::string root = argc > 1 ? argv[1] : ".";
57
58 std::cout << "=== The Dark Eye 5e (specific mode) ===\n";
59 const rpg_os::Json tde = loadRuleset(root, "tde5e_core.json");
60 const auto geron = rpg_os::generated::tde5e::Character::fromArchetype(tde, "geron");
61 // Byte-sized members stream as characters; cast to int for display.
62 std::cout << "Geron: COU " << static_cast<int>(geron.courage) << ", Life Points "
63 << geron.lifePoints << '/' << geron.maxLifePoints() << ", Attack "
64 << geron.attackSwordsSr6() << ", Dodge " << geron.dodge() << '\n';
65
66 rpg_os::CheckParams params;
67 rpg_os::DefaultRandom rng;
68 // The generated check method instantiates the shared 3d20 pool algorithm
69 // with the skill's attributes baked in — same result as universal mode.
70 const rpg_os::CheckResult climb = geron.checkClimbing(params, rng);
71 std::cout << "Climbing check: " << (climb.isSuccess ? "success" : "failure") << " (SP left "
72 << climb.remainingPool << ", QL " << climb.qualityLevel << ")\n";
73
74 std::cout << "=== D&D 5e SRD (specific mode) ===\n";
75 const rpg_os::Json dnd = loadRuleset(root, "dnd5e_srd.json");
76 // The SRD ruleset stores text descriptions (classes/species/backgrounds/feats)
77 // rather than PC archetype blocks, so load a bestiary entry instead.
78 const auto fighter = rpg_os::generated::dnd5e::Character::fromCreature(dnd, "goblin_warrior");
79 std::cout << "Goblin: STR " << static_cast<int>(fighter.strength) << " (mod "
80 << fighter.strengthModifier() << "), AC " << fighter.armorClass() << ", HP "
81 << fighter.hitPoints << '\n';
82
83 rpg_os::CheckParams attackParams;
84 rpg_os::DefaultRandom attackRng;
85 // A generated check method taking the target as a StatProvider argument.
86 const rpg_os::CheckResult attack = fighter.dnd5eAttackMelee(fighter, attackParams, attackRng);
87 std::cout << "Melee attack vs AC " << fighter.armorClass() << ": "
88 << (attack.isSuccess ? "hit" : "miss") << " (roll "
89 << (attack.rawDiceRolls.empty() ? 0 : attack.rawDiceRolls.front()) << ")\n";
90 return 0;
