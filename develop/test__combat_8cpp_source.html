129TEST_CASE("combat: a detailed fight log records dice and stat changes") {
130 rpg_os::RulesetEngine engine;
131 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
132 rpg_os::CombatantSpec irrhalk;
133 rpg_os::CombatantSpec toad;
134 REQUIRE(rpg_os::makeCombatantSpec(engine, "irrhalk", irrhalk));
135 REQUIRE(rpg_os::makeCombatantSpec(engine, "toad", toad));
136
137 // Same script as the plain runFight test above: irrhalk wins initiative
138 // (3 vs 2), lands a critical (1) the toad fails to parry (20), and rolls
139 // maximum Claws damage (6, 6) + 4 = 16 — more than the toad's 2 LP.
140 auto rng = script({3, 2, 1, 20, 6, 6});
141 rpg_os::FightLog log;
142 const rpg_os::FightOutcome outcome =
143 rpg_os::runFight(engine, irrhalk, toad, "tde_attack", "LP", 100, rng, true, log);
144 CHECK(outcome.winnerIndex == 0);
145 CHECK(outcome.rounds == 1);
146
147 CHECK(log.winnerIndex == 0);
148 CHECK(log.names[0] == "Irrhalk");
149 CHECK(log.names[1] == "Toad (Kosh Toad)");
150 CHECK(log.maxLp[0] == 90);
151 CHECK(log.maxLp[1] == 2);
152 REQUIRE(log.rounds.size() == 1);
153
154 const rpg_os::FightRoundLog &round = log.rounds[0];
155 CHECK(round.round == 1);
156 CHECK(round.initRoll[0] == 3); // the raw initiative dice
157 CHECK(round.initRoll[1] == 2);
158 CHECK(round.goesFirst == 0); // irrhalk acts first
159 REQUIRE(round.actions.size() == 1); // the toad never gets to act
160
161 const rpg_os::FightActionLog &action = round.actions[0];
162 CHECK(action.actorIndex == 0);
163 CHECK(action.targetIndex == 1);
164 CHECK(action.kind == "attack");
165 CHECK(action.isHit);
166 CHECK(action.checkDice == std::vector<int32_t>({1, 20})); // attack, then parry
167 CHECK(action.damageDice == std::vector<int32_t>({6, 6}));
168 CHECK(action.damage == 16);
169 CHECK(action.hpBefore == 2); // the toad starts at its 2 LP
170 CHECK(action.targetHp == 0); // the toad is reduced to 0 LP
171}

191
192 CHECK(logged.winnerIndex == plain.winnerIndex);
193 CHECK(logged.rounds == plain.rounds);
194 CHECK(logged.remainingLp[0] == plain.remainingLp[0]);
195 CHECK(logged.remainingLp[1] == plain.remainingLp[1]);
196 REQUIRE(log.rounds.size() == static_cast<std::size_t>(logged.rounds));
197}
