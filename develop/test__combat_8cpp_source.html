78TEST_CASE("combat: a bestiary entry without attacks falls back to derived values") {
79 // The shipped Heshthot has attacks, so strip them from a loaded copy to
80 // exercise the fallback path (no natural attack defined).
81 const std::string path = rulesetPath("tde5e_core.json");
82 std::ifstream file(path);
83 REQUIRE(file.good());
84 rpg_os::Json ruleset = rpg_os::Json::parse(
85 std::string(std::istreambuf_iterator<char>(file), std::istreambuf_iterator<char>()));
86 for (rpg_os::Json &creature : ruleset["data"]["creatures"]) {
87 if (creature.value("id", "") == "heshthot") {
88 creature.erase("attacks");
89 }
90 }
91 rpg_os::RulesetEngine engine;
92 REQUIRE(engine.loadRulesetFromJson(ruleset.dump()));
93 rpg_os::CombatantSpec spec;
94 REQUIRE(rpg_os::makeCombatantSpec(engine, "heshthot", spec));
95 CHECK_FALSE(spec.isArchetype);
96 CHECK(spec.defenseValue == 7); // bestiary `dodge`
97 CHECK(spec.attackValue == 8); // 6 + COU_Bonus (COU 16)
98 CHECK(spec.damageExpression == "1d6"); // unarmed fallback
99}

101TEST_CASE("combat: bestiary armour rating is applied through the damage pipeline") {
102 rpg_os::RulesetEngine engine;
103 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
104 rpg_os::CombatantSpec heshthot;
105 REQUIRE(rpg_os::makeCombatantSpec(engine, "heshthot", heshthot)); // armor_rating 2
106 rpg_os::DefaultRandom rng(1);
107 auto fighter = rpg_os::createFighter(engine, heshthot, rng);
108 REQUIRE(fighter != nullptr);
109 const int32_t before = fighter->resource("LP");
110 CHECK(before == 35);
111 rpg_os::DynamicEntity orc(engine.ruleset(), "orc");
112 CHECK(engine.applyDamage(orc, *fighter, "LP", 10) == -8); // 10 - Armor_Rating 2
113 CHECK(fighter->resource("LP") == before - 8);
114}

116TEST_CASE("combat: the stronger beast defeats a helpless one to the end") {
117 rpg_os::RulesetEngine engine;
118 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
119 rpg_os::CombatantSpec irrhalk;
120 rpg_os::CombatantSpec toad;
121 REQUIRE(rpg_os::makeCombatantSpec(engine, "irrhalk", irrhalk));
122 REQUIRE(rpg_os::makeCombatantSpec(engine, "toad", toad));
123
124 // Scripted RNG: irrhalk wins initiative (3 vs 2), lands a critical hit (1)
125 // that is not parried (20) and rolls maximum damage on Claws (6, 6) -> 16,
126 // which is more than the toad's 2 life points.
127 auto rng = script({3, 2, 1, 20, 6, 6});
128 const rpg_os::FightOutcome outcome =
129 rpg_os::runFight(engine, irrhalk, toad, "tde_attack", "LP", 100, rng);
130 CHECK(outcome.winnerIndex == 0);
131 CHECK(outcome.rounds == 1);
132 CHECK(outcome.remainingLp[0] == 90); // Irrhalk untouched
133 CHECK(outcome.remainingLp[1] <= 0); // Toad at 0
134}
