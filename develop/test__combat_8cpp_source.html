91TEST_CASE("combat: the stronger beast defeats a helpless one to the end") {
92 rpg_os::RulesetEngine engine;
93 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
94 rpg_os::CombatantSpec irrhalk;
95 rpg_os::CombatantSpec toad;
96 REQUIRE(rpg_os::makeCombatantSpec(engine, "irrhalk", irrhalk));
97 REQUIRE(rpg_os::makeCombatantSpec(engine, "toad", toad));
98
99 // Scripted RNG: irrhalk wins initiative (3 vs 2), lands a critical hit (1)
100 // that is not parried (20) and rolls maximum damage on Claws (6, 6) -> 16,
101 // which is more than the toad's 2 life points.
102 auto rng = script({3, 2, 1, 20, 6, 6});
103 const rpg_os::FightOutcome outcome =
104 rpg_os::runFight(engine, irrhalk, toad, "tde_attack", "LP", 100, rng);
105 CHECK(outcome.winnerIndex == 0);
106 CHECK(outcome.rounds == 1);
107 CHECK(outcome.remainingLp[0] == 90); // Irrhalk untouched
108 CHECK(outcome.remainingLp[1] <= 0); // Toad at 0
109}
