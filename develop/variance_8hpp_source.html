65template <RandomNumberGenerator Rng>
66[[nodiscard]] constexpr int32_t pickVariant(int32_t min, int32_t max, Variance variance, Rng &rng) {
67 if (min >= max) {
68 return min;
69 }
70 const int32_t span = max - min;
71 switch (variance) {
72 case Variance::Weakest:
73 return min;
74 case Variance::Strongest:
75 return max;
76 case Variance::Weak: {
77 const int32_t hi = min + span / 3;
78 return min + rng(0, hi - min);
79 }
80 case Variance::Average: {
81 const int32_t lo = min + span / 3;
82 const int32_t hi = min + 2 * span / 3;
83 return lo + rng(0, hi - lo);
84 }
85 case Variance::Strong: {
86 const int32_t lo = min + 2 * span / 3;
87 return lo + rng(0, max - lo);
88 }
89 case Variance::Random:
90 default:
91 return min + rng(0, span);
92 }
93}

103[[nodiscard]] constexpr std::pair<int32_t, int32_t> diceBounds(const DiceExpression &expression) {
104 int32_t lo = expression.constant();
105 int32_t hi = expression.constant();
106 for (const DieSpec &die : expression.dice()) {
107 if (die.sign > 0) {
108 lo += die.count * 1;
109 hi += die.count * die.sides;
110 } else {
111 lo += die.count * die.sides * die.sign;
112 hi += die.count * 1 * die.sign;
113 }
114 }
115 return {lo, hi};
116}

131template <RandomNumberGenerator Rng>
132[[nodiscard]] constexpr int32_t readVariantValue(const Json &entry, Variance variance, Rng &rng) {
133 if (entry.is_number_integer()) {
134 return entry.get<int32_t>();
135 }
136 if (entry.is_string()) {
137 const DiceExpression expression(entry.get<std::string>());
138 if (variance == Variance::Random) {
139 return expression.rollSum(rng);
140 }
141 const auto [lo, hi] = diceBounds(expression);
142 return pickVariant(lo, hi, variance, rng);
143 }
144 if (entry.is_object() && entry.contains("min") && entry.contains("max")) {
145 const int32_t lo = entry.at("min").get<int32_t>();
146 const int32_t hi = entry.at("max").get<int32_t>();
147 return pickVariant(lo, hi, variance, rng);
148 }
149 // Non-numeric entries (e.g. a name) are treated as "no numeric value".
150 return 0;
151}
