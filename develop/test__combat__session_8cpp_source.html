65TEST_CASE("CombatSession: initiative orders combatants by stat + die") {
66 GameSession session = combatDemo();
67 CombatSession combat(session.engine());
68 const rpg_os::EntityHandle a = session.createCharacterHandle("geron");
69 const rpg_os::EntityHandle b = session.createCharacterHandle("geron");
70 const EntityId aId = a.entityId();
71 const EntityId bId = b.entityId();

73 (void)combat.addCombatant(b, 1);
74

110TEST_CASE("CombatSession: durations decay across rounds via runCurrentTurn") {
111 GameSession session = combatDemo();
112 CombatSession combat(session.engine());
113 const rpg_os::EntityHandle hero = session.createCharacterHandle("geron");
114 const rpg_os::EntityHandle goblin = session.createCharacterHandle("geron");
115 (void)combat.addCombatant(hero, 0);
116 (void)combat.addCombatant(goblin, 1);
117
118 // The hero is paralyzed for 2 rounds (2 ticks).
119 session.applyCondition(*hero.resolve(), "paralyzed", 1, 2);
120 CHECK(hero.resolve()->conditionStacks("paralyzed") == 1);
121
122 // Round 1: both act; the hero's condition ticks 2 -> 1.
123 auto rng = script({6, 1, 6, 1}); // initiative for round 1 and round 2
124 combat.beginRound(rng);
125 while (combat.next()) {
126 combat.runCurrentTurn(rng);
127 }
128 CHECK(hero.resolve()->conditionStacks("paralyzed") == 1);
129
130 // Round 2: the condition expires during the hero's turn.
131 combat.beginRound(rng);
132 while (combat.next()) {
133 combat.runCurrentTurn(rng);
134 }
135 CHECK_FALSE(hero.resolve()->hasCondition("paralyzed"));
136 CHECK(hero.resolve()->conditionStacks("paralyzed") == 0);
137 CHECK(combat.round() == 2);
138}
