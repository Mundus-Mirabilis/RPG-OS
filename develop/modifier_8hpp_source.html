78[[nodiscard]] inline int32_t applyModifierPipeline(int32_t base,
79 const std::vector<Modifier> &modifiers) {
80 int32_t value = base;
81 for (const Modifier &mod : modifiers) {
82 if (mod.type == ModifierType::Override) {
83 value = mod.value;
84 }
85 }
86 for (const Modifier &mod : modifiers) {
87 if (mod.type == ModifierType::Add) {
88 value += mod.value;
89 }
90 }
91 for (const Modifier &mod : modifiers) {
92 if (mod.type == ModifierType::Multiply) {
93 // D&D-style multiplicative modifiers round down, which is why the
94 // multiplication goes through math::floor before narrowing to a stat.
95 value = math::toStat(math::floor(static_cast<double>(value) * mod.factor));
96 }
97 }
98 for (const Modifier &mod : modifiers) {
99 if (mod.type == ModifierType::Clamp) {
100 value = math::clampInt(value, mod.clampMin, mod.clampMax);
101 }
102 }
103 return value;
104}
