22inline std::string trimWhitespace(std::string_view text) {
23 std::size_t begin = 0;
24 while (begin < text.size() && (text[begin] == ' ' || text[begin] == '\t' || text[begin] == '\r' ||
25 text[begin] == '\n')) {
26 ++begin;
27 }
28 std::size_t end = text.size();
29 while (end > begin && (text[end - 1] == ' ' || text[end - 1] == '\t' || text[end - 1] == '\r' ||
30 text[end - 1] == '\n')) {
31 --end;
32 }
33 return std::string(text.substr(begin, end - begin));
34}

38inline double parseFractionString(std::string_view text) {
39 std::string trimmed = trimWhitespace(text);
40 if (trimmed.empty()) {
41 return 0.0;
42 }
43 const std::size_t space = trimmed.find(' ');
44 std::string whole;
45 std::string frac;
46 if (space != std::string::npos) {
47 whole = trimWhitespace(trimmed.substr(0, space));
48 frac = trimWhitespace(trimmed.substr(space + 1));
49 } else {
50 frac = trimmed;
51 }
52 double result = whole.empty() ? 0.0 : std::strtod(whole.c_str(), nullptr);
53 const std::size_t slash = frac.find('/');
54 if (slash != std::string::npos) {
55 const double numerator = std::strtod(frac.substr(0, slash).c_str(), nullptr);
56 const double denominator = std::strtod(frac.substr(slash + 1).c_str(), nullptr);
57 if (denominator != 0.0) {
58 result += numerator / denominator;
59 }
60 } else if (!frac.empty()) {
61 result += std::strtod(frac.c_str(), nullptr);
62 }
63 return result;
64}

69inline double parseWeightValue(std::string_view text) {
70 std::string trimmed = trimWhitespace(text);
71 if (trimmed.empty() || trimmed == "-" || trimmed == "—" || trimmed == "Varies" ||
72 trimmed == "varies" || trimmed == "Slow" || trimmed == "Push") {
73 return -1.0;
74 }
75 std::size_t pos = 0;
76 while (pos < trimmed.size() &&
77 ((trimmed[pos] >= '0' && trimmed[pos] <= '9') || trimmed[pos] == '.' ||
78 trimmed[pos] == '/' || trimmed[pos] == ' ' || trimmed[pos] == ',')) {
79 ++pos;
80 }
81 if (pos == 0) {
82 return -1.0;
83 }
84 return parseFractionString(trimmed.substr(0, pos));
85}

335 static ItemInstance itemFromJson(const Json &in) {
336 ItemInstance item;
