39TEST_CASE("bookkeeping: a long rest restores resources and ends timed conditions (phase 5)") {
40 RulesetEngine tde;
41 REQUIRE(tde.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
42 auto geron = tde.createEntity("geron");

44 const int32_t maxLp = geron->resource("LP");
45 (void)geron->modifyResource("LP", -10);
