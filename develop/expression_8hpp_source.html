125 Node parse() {
126 skipWs();
127 Node node = parseOr();
128 skipWs();
129 if (pos != text.size()) {
130 throw std::invalid_argument("unexpected trailing characters in formula '" +
131 std::string(text) + "'");
132 }
133 return node;
134 }

136 void skipWs() {
137 while (pos < text.size() && (text[pos] == ' ' || text[pos] == '\t' || text[pos] == '\n')) {
138 ++pos;
139 }
140 }

142 bool consume(std::string_view token) {
143 if (text.substr(pos, token.size()) == token) {
144 pos += token.size();
145 return true;
146 }
147 return false;
148 }

150 [[nodiscard]] bool isDigit(char c) const {
151 return c >= '0' && c <= '9';
152 }
153 [[nodiscard]] bool isIdStart(char c) const {
154 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c == '_';
155 }
156 [[nodiscard]] bool isIdChar(char c) const {
157 return isIdStart(c) || isDigit(c) || c == '.';

199 double parseNumber() {
200 std::size_t start = pos;
201 while (pos < text.size() && isDigit(text[pos])) {
202 ++pos;
203 }
204 if (pos < text.size() && text[pos] == '.') {
205 ++pos;
206 while (pos < text.size() && isDigit(text[pos])) {
207 ++pos;
208 }
209 }
210 if (pos == start) {
211 throw std::invalid_argument("expected a number in formula '" + std::string(text) + "'");
212 }
213 const double value = std::stod(std::string(text.substr(start, pos - start)));
214 return value;
215 }
216
217 std::string parseIdentifier() {
218 const std::size_t start = pos;
219 while (pos < text.size() && isIdChar(text[pos])) {
220 ++pos;
221 }
222 return std::string(text.substr(start, pos - start));
223 }

225 // ---- grammar (precedence climbing) ----
226
227 Node parseOr() {
228 Node left = parseAnd();
229 while (true) {
230 skipWs();
231 if (consume("||")) {
232 Node right = parseAnd();
233 left = makeBinary("||", std::move(left), std::move(right));
234 } else {
235 return left;
236 }
237 }
238 }
239
240 Node parseAnd() {
241 Node left = parseEquality();
242 while (true) {
243 skipWs();
244 if (consume("&&")) {
245 Node right = parseEquality();
246 left = makeBinary("&&", std::move(left), std::move(right));
247 } else {
248 return left;
249 }
250 }
251 }
252
253 Node parseEquality() {
254 Node left = parseRelational();
255 while (true) {
256 skipWs();
257 std::string op;
258 if (consume("==")) {
259 op = "==";
260 } else if (consume("!=")) {
261 op = "!=";
262 } else {
263 return left;
264 }
265 Node right = parseRelational();
266 left = makeBinary(std::move(op), std::move(left), std::move(right));
267 }
268 }
269
270 Node parseRelational() {
271 Node left = parseAdditive();
272 while (true) {
273 skipWs();
274 std::string op;
275 if (consume("<=")) {
276 op = "<=";
277 } else if (consume(">=")) {
278 op = ">=";
279 } else if (consume("<")) {
280 op = "<";
281 } else if (consume(">")) {
282 op = ">";
283 } else {
284 return left;
285 }
286 Node right = parseAdditive();
287 left = makeBinary(std::move(op), std::move(left), std::move(right));
288 }
289 }
290
291 Node parseAdditive() {
292 Node left = parseMultiplicative();
293 while (true) {
294 skipWs();
295 std::string op;
296 if (consume("+")) {
297 op = "+";
298 } else if (consume("-")) {
299 op = "-";
300 } else {
301 return left;
302 }
303 Node right = parseMultiplicative();
304 left = makeBinary(std::move(op), std::move(left), std::move(right));
305 }
306 }
307
308 Node parseMultiplicative() {
309 Node left = parseUnary();
310 while (true) {
311 skipWs();
312 std::string op;
313 if (consume("*")) {
314 op = "*";
315 } else if (consume("/")) {
316 op = "/";
317 } else if (consume("%")) {
318 op = "%";
319 } else {
320 return left;
321 }
322 Node right = parseUnary();
323 left = makeBinary(std::move(op), std::move(left), std::move(right));
324 }
325 }
326
327 Node parseUnary() {
328 skipWs();
329 if (consume("!")) {
330 return makeUnary("!", parseUnary());
331 }
332 if (consume("-")) {
333 return makeUnary("-", parseUnary());
334 }
335 return parsePower();
336 }
337
338 Node parsePower() {
339 Node left = parsePrimary();
340 skipWs();
341 if (consume("^")) {
342 // Right-associative: 2 ^ 3 ^ 2 == 2 ^ (3 ^ 2).
343 Node right = parseUnary();
344 return makeBinary("^", std::move(left), std::move(right));
345 }
346 return left;
347 }
348
349 Node parsePrimary() {
350 skipWs();
351 if (pos >= text.size()) {
352 throw std::invalid_argument("unexpected end of formula '" + std::string(text) + "'");
353 }
354 if (text[pos] == '(') {
355 ++pos;
356 Node node = parseOr();
357 skipWs();
358 if (!consume(")")) {
359 throw std::invalid_argument("missing ')' in formula '" + std::string(text) + "'");
360 }
361 return node;
362 }
363 if (isDigit(text[pos]) || text[pos] == '.') {
364 return makeNumber(parseNumber());
365 }
366 if (isIdStart(text[pos])) {
367 const std::string id = parseIdentifier();
368 skipWs();
369 if (consume("(")) {
370 std::vector<Node> args;
371 skipWs();
372 if (!consume(")")) {
373 while (true) {
374 args.push_back(parseOr());
375 skipWs();
376 if (consume(")")) {
377 break;
378 }
379 if (!consume(",")) {
380 throw std::invalid_argument("expected ',' or ')' in formula '" + std::string(text) +
381 "'");
382 }
383 skipWs();
384 }
385 }
386 return makeCall(id, std::move(args));
387 }
388 return makeIdentifier(id);
389 }
390 throw std::invalid_argument("unexpected character '" + std::string(1, text[pos]) +
391 "' in formula '" + std::string(text) + "'");
392 }

416inline double Expression::evaluateNode(const Node &node, const EvalContext &context) {
417 switch (node.kind) {
418 case Node::Kind::Number:
419 return node.number;
420 case Node::Kind::Identifier: {
421 double value = 0.0;
422 if (!context.resolve(node.identifier, value)) {
423 throw std::runtime_error("unresolved identifier '" + node.identifier + "'");
424 }
425 return value;
426 }
427 case Node::Kind::Unary: {
428 const double operand = evaluateNode(node.children[0], context);
429 if (node.op == "!") {
430 return operand == 0.0 ? 1.0 : 0.0;
431 }
432 if (node.op == "-") {
433 return -operand;
434 }
435 throw std::runtime_error("unknown unary operator '" + node.op + "'");
436 }
437 case Node::Kind::Binary: {
438 const double lhs = evaluateNode(node.children[0], context);
439 const double rhs = evaluateNode(node.children[1], context);
440 const std::string &op = node.op;
441 if (op == "+") {
442 return lhs + rhs;
443 }
444 if (op == "-") {
445 return lhs - rhs;
446 }
447 if (op == "*") {
448 return lhs * rhs;
449 }
450 if (op == "/") {
451 if (rhs == 0.0) {
452 throw std::runtime_error("division by zero in formula");
453 }
454 return lhs / rhs;
455 }
456 if (op == "%") {
457 if (rhs == 0.0) {
458 throw std::runtime_error("modulo by zero in formula");
459 }
460 return std::fmod(lhs, rhs);
461 }
462 if (op == "^") {
463 return std::pow(lhs, rhs);
464 }
465 if (op == "==") {
466 return lhs == rhs ? 1.0 : 0.0;
467 }
468 if (op == "!=") {
469 return lhs != rhs ? 1.0 : 0.0;
470 }
471 if (op == "<") {
472 return lhs < rhs ? 1.0 : 0.0;
473 }
474 if (op == ">") {
475 return lhs > rhs ? 1.0 : 0.0;
476 }
477 if (op == "<=") {
478 return lhs <= rhs ? 1.0 : 0.0;
479 }
480 if (op == ">=") {
481 return lhs >= rhs ? 1.0 : 0.0;
482 }
483 if (op == "&&") {
484 return (lhs != 0.0 && rhs != 0.0) ? 1.0 : 0.0;
485 }
486 if (op == "||") {
487 return (lhs != 0.0 || rhs != 0.0) ? 1.0 : 0.0;
488 }
489 throw std::runtime_error("unknown binary operator '" + op + "'");
490 }
491 case Node::Kind::Call: {
492 std::vector<double> args;
493 args.reserve(node.children.size());
494 for (const Node &child : node.children) {
495 args.push_back(evaluateNode(child, context));
496 }
497 const std::string &fn = node.function;
498 if (fn == "min") {
499 if (args.size() != 2) {
500 throw std::runtime_error("min() expects 2 arguments");
501 }
502 return math::min(args[0], args[1]);
503 }
504 if (fn == "max") {
505 if (args.size() != 2) {
506 throw std::runtime_error("max() expects 2 arguments");
507 }
508 return math::max(args[0], args[1]);
509 }
510 if (fn == "floor") {
511 if (args.size() != 1) {
512 throw std::runtime_error("floor() expects 1 argument");
513 }
514 return math::floor(args[0]);
515 }
516 if (fn == "ceil") {
517 if (args.size() != 1) {
518 throw std::runtime_error("ceil() expects 1 argument");
519 }
520 return math::ceil(args[0]);
521 }
522 if (fn == "round") {
523 if (args.size() != 1) {
524 throw std::runtime_error("round() expects 1 argument");
525 }
526 return math::round(args[0]);
527 }
528 if (fn == "clamp") {
529 if (args.size() != 3) {
530 throw std::runtime_error("clamp() expects 3 arguments");
531 }
532 return math::clamp(args[0], args[1], args[2]);
533 }
534 throw std::runtime_error("unknown function '" + fn + "'");
535 }
536 }
537 return 0.0;
538}

540inline void Expression::collectIdentifiers(const Node &node, std::vector<std::string> &out) {
541 if (node.kind == Node::Kind::Identifier) {
