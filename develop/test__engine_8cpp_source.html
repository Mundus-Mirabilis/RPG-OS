129TEST_CASE("Engine: executeCheck resolves a DSA skill check (real TDE rules)") {
130 RulesetEngine engine = makeEngine();
131 auto geron = engine.createEntity("geron");
132 REQUIRE(geron != nullptr);
133
134 // Unmodified: rolls 14, 12, 11 vs COU 12 / AGI 13 / STR 11; overshoot 2 is
135 // paid from the 7 skill points -> 5 left, success, QL 2.
136 auto rng = script({14, 12, 11});
137 const rpg_os::CheckResult result =
138 engine.executeCheck("dsa4_talent", *geron, nullptr, CheckParams{}, rng);
139 CHECK(result.isSuccess);
140 CHECK(result.remainingPool == 5);
141 CHECK(result.qualityLevel == 2);
142
143 // A -4 penalty lowers the EAVs to 8 / 9 / 7; overshoot 6 + 3 + 4 = 13 > 7 -> fail.
144 CheckParams penalty;
145 penalty.difficulty = -4;
146 auto rngPenalty = script({14, 12, 11});
147 const rpg_os::CheckResult failed =
148 engine.executeCheck("dsa4_talent", *geron, nullptr, penalty, rngPenalty);
149 CHECK_FALSE(failed.isSuccess);
150 CHECK(failed.remainingPool == -6);
151}

153TEST_CASE("Engine: executeSkillCheck uses the skill's own linked attributes") {
154 RulesetEngine engine = makeEngine();
155 auto geron = engine.createEntity("geron");
156 REQUIRE(geron != nullptr);
157
158 // The climbing skill links COU/AGI/STR and its rating is the pool.
159 auto rng = script({14, 12, 11});
160 const rpg_os::CheckResult result =
161 engine.executeSkillCheck("climbing", *geron, CheckParams{}, rng);
162 CHECK(result.isSuccess);
163 CHECK(result.remainingPool == 5);
164 // executeSkillCheck is [[nodiscard]]; the explicit void cast is required so
165 // doctest's CHECK_THROWS_AS (which discards the expression) does not trip
166 // -Wunused-result under clang -Werror.

168 static_cast<void>(engine.executeSkillCheck("no_such_skill", *geron, CheckParams{}, rng)),
169 std::invalid_argument);
170}

182TEST_CASE("Engine: applyDamage runs armor absorption and wound triggers") {
183 RulesetEngine engine = makeEngine();
184 auto geron = engine.createEntity("geron");
185 REQUIRE(geron != nullptr);
186 DynamicEntity attacker(engine.ruleset(), "attacker");
187 attacker.setBaseAttribute("STR", 13);
188
189 // 10 raw -> 10 - 3 (AR) = 7 applied. VP 18 -> 11.
190 CHECK(engine.applyDamage(attacker, *geron, "VP", 10) == -7);
191 CHECK(geron->resource("VP") == 11);
192 CHECK_FALSE(geron->hasCondition("Wound"));
193
194 // 20 raw -> 20 - 3 = 17 > CN 12 -> Wound stacks = floor(17 / 12) = 1.
195 CHECK(engine.applyDamage(attacker, *geron, "VP", 20) == -11);
196 CHECK(geron->resource("VP") == 0);
197 CHECK(geron->hasCondition("Wound"));
198 CHECK(geron->conditionStacks("Wound") == 1);
199}

211TEST_CASE("Engine: user event listeners run after rule triggers") {
212 RulesetEngine engine = makeEngine();
213 auto geron = engine.createEntity("geron");
214 REQUIRE(geron != nullptr);
215 DynamicEntity attacker(engine.ruleset(), "attacker");
216
217 int seenApplied = -1;
218 const uint64_t id = engine.registerEventListener(
219 rpg_os::EventType::OnDamageTaken,
220 [&](const rpg_os::EventData &data) { seenApplied = data.getInt("applied_damage", -1); });
221 CHECK(id != 0);
222
223 engine.applyDamage(attacker, *geron, "VP", 10);
224 CHECK(seenApplied == 7); // post-armor damage
225 CHECK(engine.unregisterEventListener(id));
226}
