141TEST_CASE("Engine: executeCheck resolves a DSA skill check (real TDE rules)") {
142 RulesetEngine engine = makeEngine();
143 auto geron = engine.createEntity("geron");
144 REQUIRE(geron != nullptr);
145
146 // Unmodified: rolls 14, 12, 11 vs COU 12 / AGI 13 / STR 11; overshoot 2 is
147 // paid from the 7 skill points -> 5 left, success, QL 2.
148 auto rng = script({14, 12, 11});
149 const rpg_os::CheckResult result =
150 engine.executeCheck("dsa4_talent", *geron, nullptr, CheckParams{}, rng);
151 CHECK(result.isSuccess);
152 CHECK(result.remainingPool == 5);
153 CHECK(result.qualityLevel == 2);
154
155 // A -4 penalty lowers the EAVs to 8 / 9 / 7; overshoot 6 + 3 + 4 = 13 > 7 -> fail.
156 CheckParams penalty;
157 penalty.difficulty = -4;
158 auto rngPenalty = script({14, 12, 11});
159 const rpg_os::CheckResult failed =
160 engine.executeCheck("dsa4_talent", *geron, nullptr, penalty, rngPenalty);
161 CHECK_FALSE(failed.isSuccess);
162 CHECK(failed.remainingPool == -6);
163}

255 REQUIRE(geron != nullptr);
256
257 int resolved = 0;
258 int before = 0;
259 int after = 0;

261 uint64_t lastInstance = 0;

263 rpg_os::EventType::OnCheckResolved, [&](const rpg_os::EventData &data) {
264 ++resolved;
265 lastSuccess = data.getBool("is_success");
266 lastInstance = static_cast<uint64_t>(data.getInt("instance_id"));
267 });
268 (void)engine.registerEventListener(rpg_os::EventType::OnBeforeCheckRoll,
269 [&](const rpg_os::EventData &) { ++before; });
270 (void)engine.registerEventListener(rpg_os::EventType::OnAfterCheckRoll,
271 [&](const rpg_os::EventData &) { ++after; });
272
273 auto rng = script({14, 12, 11});
274 const rpg_os::CheckResult ok =
275 engine.executeCheck("dsa4_talent", *geron, nullptr, CheckParams{}, rng);
276 CHECK(ok.isSuccess);
277 CHECK(resolved == 1);
278 CHECK(lastSuccess);
279 CHECK(before == 1);
280 CHECK(after == 1);
281 CHECK(lastInstance == geron->entityId().toUint64());
282
283 CheckParams penalty;
284 penalty.difficulty = -4;

286 const rpg_os::CheckResult failed =
287 engine.executeCheck("dsa4_talent", *geron, nullptr, penalty, rngPenalty);
288 CHECK_FALSE(failed.isSuccess);
289 CHECK(resolved == 2);
290 CHECK_FALSE(lastSuccess);

343TEST_CASE("Engine: OnConditionChanged fires from every condition path") {
344 RulesetEngine engine = makeEngine();
345 auto geron = engine.createEntity("geron");
346 REQUIRE(geron != nullptr);
347
348 int applied = 0;
349 int removed = 0;
350 int expired = 0;
351 (void)engine.registerEventListener(rpg_os::EventType::OnConditionChanged,
352 [&](const rpg_os::EventData &data) {
353 const std::string action = data.getString("action", "");
354 if (action == "applied") {
355 ++applied;
356 } else if (action == "removed") {
357 ++removed;
358 } else if (action == "expired") {
359 ++expired;
360 }
361 });
362
363 // facade wrapper validates and fires "applied".
364 engine.applyCondition(*geron, "blinded", 1, 1);
365 CHECK(applied == 1);
366 CHECK(geron->hasCondition("blinded"));
367
368 // Duration expiry after a tick fires "expired".
369 engine.tickEffects(*geron);
370 CHECK(expired == 1);
371 CHECK_FALSE(geron->hasCondition("blinded"));
372
373 // The spell-effect path applies a condition through castSpell -> resolveEffects.
374 auto victim = engine.createEntity("geron");
375 REQUIRE(victim != nullptr);
376 auto rng = script({});
377 const rpg_os::RulesetEngine::SpellResult cast =
378 engine.castSpell("blinding_bolt", *geron, victim.get(), CheckParams{}, rng);
379 CHECK(cast.cast);
380 CHECK(victim->hasCondition("blinded"));
381 CHECK(applied == 2);
382
383 // Removal fires "removed".
384 engine.removeCondition(*victim, "blinded");
385 CHECK(removed == 1);
386 CHECK_FALSE(victim->hasCondition("blinded"));
387}
