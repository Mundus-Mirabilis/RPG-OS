25TEST_CASE("bookkeeping: stat bonuses tick, expire, and survive serialization (phase 0)") {
26 rpg_os::EffectTimeline timeline;
27 timeline.addBonus(rpg_os::StatBonus{"STR", 3, 2, "spell"});
28 timeline.addBonus(rpg_os::StatBonus{"PER", 5, 0, ""});
29 CHECK(timeline.bonusFor("STR") == 3);
30 CHECK(timeline.bonusFor("PER") == 5);
31 CHECK(timeline.bonusFor("DEX") == 0);
32
33 // A permanent bonus survives ticks; the timed one ticks down.
34 CHECK(timeline.tick() == 0); // STR 2 -> 1
35 CHECK(timeline.bonusFor("STR") == 3);
36 CHECK(timeline.bonusFor("PER") == 5);
37 CHECK(timeline.tick() == 1); // STR 1 -> 0: expires
38 CHECK(timeline.bonusFor("STR") == 0);
39 CHECK(timeline.bonusFor("PER") == 5);
40
41 // Serialization round-trip preserves the remaining bonuses.
42 rpg_os::Json saved;
43 timeline.toJson(saved);
44 rpg_os::EffectTimeline restored;
45 restored.fromJson(saved);
46 CHECK(restored.bonusFor("PER") == 5);
47 CHECK(restored.bonusFor("STR") == 0);
48}

171 target2.setBaseAttribute("AC", 15);
172 target2.setBaseAttribute("HitPoints_Max", 30);
173 target2.loadFromArchetype(rpg_os::Json::object());
174 auto rng3 = script({1, 1, 1, 1, 5}); // 4d4 rolled, attack 5 + 6 = 11 < 15 miss
