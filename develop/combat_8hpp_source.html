124[[nodiscard]] inline bool makeCombatantSpec(RulesetEngine &engine, std::string_view id,
125 CombatantSpec &out,
126 std::string_view weaponDamage = "1d6+4") {
127 const Ruleset &ruleset = engine.ruleset();
128 if (!ruleset.data.is_object()) {
129 return false;
130 }
131 DefaultRandom probeRng;
132
133 if (ruleset.data.contains("archetypes")) {
134 for (const Json &entry : ruleset.data.at("archetypes")) {
135 if (entry.value("id", "") != id) {
136 continue;
137 }
138 auto probe = engine.createEntityWith(id, Variance::Average, probeRng);
139 if (probe == nullptr) {
140 return false;
141 }
142 out.id = std::string(id);
143 out.name = entry.value("name", out.id);
144 out.isArchetype = true;
145 out.attackValue = probe->getStat("Attack");
146 out.defenseValue = probe->getStat("Parry");
147 out.armorRating = probe->baseAttribute("Armor_Rating");
148 out.damageExpression = std::string(weaponDamage);
149 // An archetype's `spells_known` were loaded into the probe's spellbook;
150 // copy them so the fight loop can cast magic for this combatant.
151 out.spellIds = probe->spellbook().known();
152 return true;
153 }
154 }
155
156 if (ruleset.data.contains("creatures")) {
157 for (const Json &entry : ruleset.data.at("creatures")) {
158 if (entry.value("id", "") != id) {
159 continue;
160 }
161 auto probe = engine.createCreatureWith(id, Variance::Average, probeRng);
162 if (probe == nullptr) {
163 return false;
164 }
165 out.id = std::string(id);
166 out.name = entry.value("name", out.id);
167 out.isArchetype = false;
168 out.defenseValue = entry.value("dodge", 0);
169 out.armorRating = entry.value("armor_rating", 0);
170 out.attackValue = 0;
171 out.damageExpression = std::string(weaponDamage);
172 if (entry.contains("attacks") && entry.at("attacks").is_array()) {
173 int32_t bestToHit = -1;
174 for (const Json &attack : entry.at("attacks")) {
175 const int32_t toHit = attack.value("to_hit", 0);
176 if (toHit > bestToHit) {
177 bestToHit = toHit;
178 out.attackValue = toHit;
179 out.damageExpression = attack.value("damage", "1d6");
180 }
181 }
182 }
183 if (out.attackValue <= 0) {
184 // No natural attack defined: fall back to the creature's derived
185 // attack value and an unarmed strike.
186 out.attackValue = probe->getStat("Attack");
187 out.damageExpression = "1d6";
188 }
189 return true;
190 }
191 }
192 return false;
193}

206template <RandomNumberGenerator Rng>
207[[nodiscard]] std::shared_ptr<DynamicEntity> createFighter(RulesetEngine &engine,
208 const CombatantSpec &spec, Rng &rng) {
209 std::shared_ptr<DynamicEntity> entity =
210 spec.isArchetype ? engine.createEntityWith(spec.id, Variance::Average, rng)
211 : engine.createCreatureWith(spec.id, Variance::Average, rng);
212 if (entity != nullptr && !spec.isArchetype) {
213 entity->setBaseAttribute("Attack", spec.attackValue);
214 entity->setBaseAttribute("Parry", spec.defenseValue);
215 entity->setBaseAttribute("Armor_Rating", spec.armorRating);
216 }
217 return entity;
218}

227[[nodiscard]] inline double spellAverageDamage(const Json &spell) {
228 double total = 0.0;
229 const auto addDice = [&](const Json &value) {
230 if (value.is_number()) {
231 total += value.get<double>();
232 } else if (value.is_string()) {
233 total += DiceExpression(value.get<std::string>()).expectedValue();
234 }
235 };
236 if (spell.contains("damage")) {
237 addDice(spell.at("damage"));
238 }
239 if (spell.contains("effects") && spell.at("effects").is_array()) {
240 for (const Json &effect : spell.at("effects")) {
241 if (effect.value("kind", "") == "damage" && effect.contains("dice")) {
242 addDice(effect.at("dice"));
243 }
244 }
245 }
246 return total;
247}

258[[nodiscard]] inline std::string pickSpell(const RulesetEngine &engine, const DynamicEntity &caster,
259 const std::vector<std::string> &spellIds) {
260 const std::string &resourceId = engine.ruleset().spellResource;
261 std::string best;
262 double bestDamage = 0.0;
263 for (const std::string &spellId : spellIds) {
264 const Json *spell = engine.findSpell(spellId);
265 if (spell == nullptr) {
266 continue;
267 }
268 int32_t cost = 0;
269 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
270 cost = spell->at("cost").get<int32_t>();
271 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
272 cost = spell->at("ae_cost").get<int32_t>();
273 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
274 cost = spell->at("level").get<int32_t>();
275 }
276 if (cost > 0 && !resourceId.empty() && caster.resource(resourceId) < cost) {
277 continue; // cannot afford
278 }
279 const double damage = spellAverageDamage(*spell);
280 if (damage > 0.0 && damage > bestDamage) {
281 bestDamage = damage;
282 best = spellId;
283 }
284 }
285 return best;
286}

297template <RandomNumberGenerator Rng>
298bool attackOnce(RulesetEngine &engine, DynamicEntity &attacker, const DiceExpression &damage,
299 DynamicEntity &defender, std::string_view checkTypeId,
300 std::string_view hpResourceId, Rng &rng) {
301 // The attacker's restrictions are enforced: a creature that cannot take
302 // actions (incapacitated, stunned, paralyzed, ...) cannot attack.
303 if (!engine.actionAllowed(attacker, "action")) {
304 return false;
305 }
306 // Resolve against effective stats so worn gear and active conditions (armour,
307 // encumbrance, wounds) influence the fight — with nothing equipped the
308 // effective value equals the raw value, so ungeared fights are unchanged.
309 const CheckResult hit =
310 engine.executeCheckEffective(checkTypeId, attacker, &defender, CheckParams{}, rng);
311 if (!hit.isSuccess) {
312 return false;
313 }
314 const int32_t raw = static_cast<int32_t>(damage.rollSum(rng));
315 engine.applyDamage(attacker, defender, hpResourceId, raw);
316 return defender.resource(hpResourceId) <= 0;
317}

330template <RandomNumberGenerator Rng>
331bool actOnce(RulesetEngine &engine, DynamicEntity &actor, const CombatantSpec &spec,
332 const DiceExpression &weaponDamage, DynamicEntity &defender,
333 std::string_view checkTypeId, std::string_view hpResourceId, bool useMagic, Rng &rng) {
334 if (useMagic && !spec.spellIds.empty()) {
335 const std::string spellId = pickSpell(engine, actor, spec.spellIds);
336 if (!spellId.empty()) {
337 const auto cast = engine.castSpell(spellId, actor, &defender, CheckParams{}, rng);
338 if (cast.cast) {
339 return defender.resource(hpResourceId) <= 0;
340 }
341 return false; // the casting attempt failed — the round is spent
342 }
343 }
344 return attackOnce(engine, actor, weaponDamage, defender, checkTypeId, hpResourceId, rng);
345}

366template <RandomNumberGenerator Rng>
367[[nodiscard]] FightOutcome runFight(RulesetEngine &engine, const CombatantSpec &a,
368 const CombatantSpec &b, std::string_view checkTypeId,
369 std::string_view hpResourceId, int maxRounds, Rng &rng,
370 bool useMagic = true) {
371 FightOutcome outcome;
372 auto ea = createFighter(engine, a, rng);
373 auto eb = createFighter(engine, b, rng);
374 if (ea == nullptr || eb == nullptr) {
375 return outcome;
376 }
377 outcome.maxLp[0] = ea->resource(hpResourceId);
378 outcome.maxLp[1] = eb->resource(hpResourceId);
379 const DiceExpression damageA(a.damageExpression);
380 const DiceExpression damageB(b.damageExpression);
381
382 int roundsPlayed = 0;
383 for (; roundsPlayed < maxRounds; ++roundsPlayed) {
384 const int initA = ea->getStat("Initiative") + rng(1, 6);
385 const int initB = eb->getStat("Initiative") + rng(1, 6);
386 const bool aFirst = initA != initB ? initA > initB : rng(0, 1) == 0;
387
388 DynamicEntity *first = aFirst ? ea.get() : eb.get();
389 DynamicEntity *second = aFirst ? eb.get() : ea.get();
390 const CombatantSpec &firstSpec = aFirst ? a : b;
391 const CombatantSpec &secondSpec = aFirst ? b : a;
392 const DiceExpression &firstDamage = aFirst ? damageA : damageB;
393 const DiceExpression &secondDamage = aFirst ? damageB : damageA;
394
395 if (detail::actOnce(engine, *first, firstSpec, firstDamage, *second, checkTypeId, hpResourceId,
396 useMagic, rng)) {
397 outcome.winnerIndex = aFirst ? 0 : 1;
398 break;
399 }
400 if (detail::actOnce(engine, *second, secondSpec, secondDamage, *first, checkTypeId,
401 hpResourceId, useMagic, rng)) {
402 outcome.winnerIndex = aFirst ? 1 : 0;
403 break;
404 }
405 }
406
407 outcome.rounds = outcome.winnerIndex == -1 ? maxRounds : roundsPlayed + 1;
408 outcome.remainingLp[0] = ea->resource(hpResourceId);
409 outcome.remainingLp[1] = eb->resource(hpResourceId);
410 return outcome;
411}
