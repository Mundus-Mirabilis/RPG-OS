74[[nodiscard]] inline std::string resolveAttackCheckType(const Ruleset &ruleset) {
75 for (const CheckTypeDef &def : ruleset.checkTypes) {

116[[nodiscard]] inline bool makeCombatantSpec(RulesetEngine &engine, std::string_view id,
117 CombatantSpec &out,
118 std::string_view weaponDamage = "1d6+4") {
119 const Ruleset &ruleset = engine.ruleset();
120 if (!ruleset.data.is_object()) {
121 return false;
122 }
123 DefaultRandom probeRng;
124
125 if (ruleset.data.contains("archetypes")) {
126 for (const Json &entry : ruleset.data.at("archetypes")) {
127 if (entry.value("id", "") != id) {
128 continue;
129 }
130 auto probe = engine.createEntityWith(id, Variance::Average, probeRng);
131 if (probe == nullptr) {
132 return false;
133 }
134 out.id = std::string(id);
135 out.name = entry.value("name", out.id);
136 out.isArchetype = true;
137 out.attackValue = probe->getStat("Attack");
138 out.defenseValue = probe->getStat("Parry");
139 out.armorRating = probe->baseAttribute("Armor_Rating");
140 out.damageExpression = std::string(weaponDamage);
141 return true;
142 }
143 }
144
145 if (ruleset.data.contains("creatures")) {
146 for (const Json &entry : ruleset.data.at("creatures")) {
147 if (entry.value("id", "") != id) {
148 continue;
149 }
150 auto probe = engine.createCreatureWith(id, Variance::Average, probeRng);
151 if (probe == nullptr) {
152 return false;
153 }
154 out.id = std::string(id);
155 out.name = entry.value("name", out.id);
156 out.isArchetype = false;
157 out.defenseValue = entry.value("dodge", 0);
158 out.armorRating = entry.value("armor_rating", 0);
159 out.attackValue = 0;
160 out.damageExpression = std::string(weaponDamage);
161 if (entry.contains("attacks") && entry.at("attacks").is_array()) {
162 int32_t bestToHit = -1;
163 for (const Json &attack : entry.at("attacks")) {
164 const int32_t toHit = attack.value("to_hit", 0);
165 if (toHit > bestToHit) {
166 bestToHit = toHit;
167 out.attackValue = toHit;
168 out.damageExpression = attack.value("damage", "1d6");
169 }
170 }
171 }
172 if (out.attackValue <= 0) {
173 // No natural attack defined: fall back to the creature's derived
174 // attack value and an unarmed strike.
175 out.attackValue = probe->getStat("Attack");
176 out.damageExpression = "1d6";
177 }
178 return true;
179 }
180 }
181 return false;
182}

195template <RandomNumberGenerator Rng>
196[[nodiscard]] std::shared_ptr<DynamicEntity> createFighter(RulesetEngine &engine,
197 const CombatantSpec &spec, Rng &rng) {
198 std::shared_ptr<DynamicEntity> entity =
199 spec.isArchetype ? engine.createEntityWith(spec.id, Variance::Average, rng)
200 : engine.createCreatureWith(spec.id, Variance::Average, rng);
201 if (entity != nullptr && !spec.isArchetype) {
202 entity->setBaseAttribute("Attack", spec.attackValue);
203 entity->setBaseAttribute("Parry", spec.defenseValue);
204 entity->setBaseAttribute("Armor_Rating", spec.armorRating);
205 }
206 return entity;
207}

220template <RandomNumberGenerator Rng>
221bool attackOnce(RulesetEngine &engine, DynamicEntity &attacker, const DiceExpression &damage,
222 DynamicEntity &defender, std::string_view checkTypeId,
223 std::string_view hpResourceId, Rng &rng) {
224 const CheckResult hit = engine.executeCheck(checkTypeId, attacker, &defender, CheckParams{}, rng);
225 if (!hit.isSuccess) {
226 return false;
227 }
228 const int32_t raw = static_cast<int32_t>(damage.rollSum(rng));
229 engine.applyDamage(attacker, defender, hpResourceId, raw);
230 return defender.resource(hpResourceId) <= 0;
231}

246template <RandomNumberGenerator Rng>
247[[nodiscard]] FightOutcome runFight(RulesetEngine &engine, const CombatantSpec &a,
248 const CombatantSpec &b, std::string_view checkTypeId,
249 std::string_view hpResourceId, int maxRounds, Rng &rng) {
250 FightOutcome outcome;
251 auto ea = createFighter(engine, a, rng);
252 auto eb = createFighter(engine, b, rng);
253 if (ea == nullptr || eb == nullptr) {
254 return outcome;
255 }
256 outcome.maxLp[0] = ea->resource(hpResourceId);
257 outcome.maxLp[1] = eb->resource(hpResourceId);
258 const DiceExpression damageA(a.damageExpression);
259 const DiceExpression damageB(b.damageExpression);
260
261 int roundsPlayed = 0;
262 for (; roundsPlayed < maxRounds; ++roundsPlayed) {
263 const int initA = ea->getStat("Initiative") + rng(1, 6);
264 const int initB = eb->getStat("Initiative") + rng(1, 6);
265 const bool aFirst = initA != initB ? initA > initB : rng(0, 1) == 0;
266
267 DynamicEntity *first = aFirst ? ea.get() : eb.get();
268 DynamicEntity *second = aFirst ? eb.get() : ea.get();
269 const DiceExpression &firstDamage = aFirst ? damageA : damageB;
270 const DiceExpression &secondDamage = aFirst ? damageB : damageA;
271
272 if (detail::attackOnce(engine, *first, firstDamage, *second, checkTypeId, hpResourceId, rng)) {
273 outcome.winnerIndex = aFirst ? 0 : 1;
274 break;
275 }
276 if (detail::attackOnce(engine, *second, secondDamage, *first, checkTypeId, hpResourceId, rng)) {
277 outcome.winnerIndex = aFirst ? 1 : 0;
278 break;
279 }
280 }
281
282 outcome.rounds = outcome.winnerIndex == -1 ? maxRounds : roundsPlayed + 1;
283 outcome.remainingLp[0] = ea->resource(hpResourceId);
284 outcome.remainingLp[1] = eb->resource(hpResourceId);
285 return outcome;
286}
