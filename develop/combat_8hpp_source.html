135[[nodiscard]] inline std::string resolveAttackCheckType(const Ruleset &ruleset) {
136 for (const CheckTypeDef &def : ruleset.checkTypes) {
137 if (def.id == "tde_attack") {
138 return def.id;
139 }
140 }
141 for (const CheckTypeDef &def : ruleset.checkTypes) {
142 if (def.recipe.resolution == Resolution::Opposed) {
143 return def.id;
144 }
145 }
146 // D&D-style attack-vs-AC: a threshold check rolled against a *target* stat
147 // (e.g. AC) whose bonus comes from the combatant's promoted `Attack` stat
148 // (the bestiary `to_hit`). Systems without an opposed parry roll (D&D)
149 // use this instead of the opposed resolution above.
150 for (const CheckTypeDef &def : ruleset.checkTypes) {
151 if (def.recipe.resolution == Resolution::Threshold &&
152 def.recipe.thresholdSource == ThresholdSource::TargetStat &&
153 std::ranges::contains(def.recipe.bonusStats, "Attack")) {
154 return def.id;
155 }
156 }
157 return {};
158}

181[[nodiscard]] inline bool fillArchetypeSpec(RulesetEngine &engine, const Json &entry,
182 std::string_view id, CombatantSpec &out,
183 std::string_view weaponDamage,
184 DefaultRandom &probeRng) {
185 auto probe = engine.createEntityWith(id, Variance::Average, probeRng);
186 if (probe == nullptr) {
187 return false;
188 }
189 out.id = std::string(id);
190 out.name = entry.value("name", out.id);
191 out.isArchetype = true;
192 out.attackValue = probe->getStat("Attack");
193 out.defenseValue = probe->getStat("Parry");
194 out.armorRating = probe->baseAttribute("Armor_Rating");
195 out.damageExpression = std::string(weaponDamage);
196 // An archetype's `spells_known` were loaded into the probe's spellbook;
197 // copy them so the fight loop can cast magic for this combatant.
198 out.spellIds = probe->spellbook().known();
199 return true;
200}

205[[nodiscard]] inline bool fillCreatureSpec(RulesetEngine &engine, const Json &entry,
206 std::string_view id, CombatantSpec &out,
207 std::string_view weaponDamage, DefaultRandom &probeRng) {
208 auto probe = engine.createCreatureWith(id, Variance::Average, probeRng);
209 if (probe == nullptr) {
210 return false;
211 }
212 out.id = std::string(id);
213 out.name = entry.value("name", out.id);
214 out.isArchetype = false;
215 out.defenseValue = entry.value("dodge", 0);
216 out.armorRating = entry.value("armor_rating", 0);
217 out.acValue = entry.value("ac", 0);
218 out.initiativeValue = entry.value("initiative", 0);
219 out.attackValue = 0;
220 out.damageExpression = std::string(weaponDamage);
221 if (entry.contains("attacks") && entry.at("attacks").is_array()) {
222 int32_t bestToHit = -1;
223 for (const Json &attack : entry.at("attacks")) {
224 const int32_t toHit = attack.value("to_hit", 0);
225 if (toHit > bestToHit) {
226 bestToHit = toHit;
227 out.attackValue = toHit;
228 out.damageExpression = attack.value("damage", "1d6");
229 }
230 }
231 }
232 if (out.attackValue <= 0) {
233 // No natural attack defined: fall back to the creature's derived
234 // attack value and an unarmed strike.
235 out.attackValue = probe->getStat("Attack");
236 out.damageExpression = "1d6";
237 }
238 return true;
239}

253[[nodiscard]] inline bool makeCombatantSpec(RulesetEngine &engine, std::string_view id,
254 CombatantSpec &out,
255 std::string_view weaponDamage = "1d6+4") {
256 const Ruleset &ruleset = engine.ruleset();
257 if (!ruleset.data.is_object()) {
258 return false;
259 }
260 DefaultRandom probeRng;
261
262 if (ruleset.data.contains("archetypes")) {
263 for (const Json &entry : ruleset.data.at("archetypes")) {
264 if (entry.value("id", "") != id) {
265 continue;
266 }
267 return detail::fillArchetypeSpec(engine, entry, id, out, weaponDamage, probeRng);
268 }
269 }
270
271 if (ruleset.data.contains("creatures")) {
272 for (const Json &entry : ruleset.data.at("creatures")) {
273 if (entry.value("id", "") != id) {
274 continue;
275 }
276 return detail::fillCreatureSpec(engine, entry, id, out, weaponDamage, probeRng);
277 }
278 }
279 return false;
280}

294[[nodiscard]] inline bool makeCombatantSpecFromEntity(RulesetEngine &engine,
295 const DynamicEntity &entity,
296 CombatantSpec &out,
297 std::string_view weaponDamage = "1d6+4") {
298 out.id = entity.id();
299 out.name = entity.id();
300 // Resolve a human-readable name when the sheet happens to be a known
301 // ruleset entry (archetype or bestiary); otherwise the id is the name.
302 const Json *record = engine.findDataRecord("archetypes", entity.id());
303 if (record == nullptr) {
304 record = engine.findDataRecord("creatures", entity.id());
305 }
306 if (record != nullptr) {
307 out.name = record->value("name", out.name);
308 }
309 out.isArchetype = true; // the sheet fully describes the character
310 out.attackValue = entity.getStat("Attack");
311 out.defenseValue = entity.getStat("Parry");
312 out.armorRating = entity.getStat("Armor_Rating");
313 out.acValue = entity.getStat("AC");
314 out.initiativeValue = entity.getStat("Initiative");
315 out.damageExpression = std::string(weaponDamage);
316 out.spellIds = entity.spellbook().known();
317 entity.toJson(out.sheet);
318 return true;
319}

332template <RandomNumberGenerator Rng>
333[[nodiscard]] std::shared_ptr<DynamicEntity> createFighter(RulesetEngine &engine,
334 const CombatantSpec &spec, Rng &rng) {
335 // A spec carrying an embedded sheet fights that sheet directly: ANY
336 // character — not just a named archetype or bestiary entry — can take part
337 // in a fight this way (e.g. a character entered into a form). The sheet is
338 // restored into a fresh entity per fight and its resource pools are
339 // (re)created so the fighter starts at its full derived hit points.
340 if (spec.sheet.is_object()) {
341 auto entity = std::make_shared<DynamicEntity>(engine.ruleset(), spec.id);
342 entity->fromJson(spec.sheet);
343 entity->refreshResources();
344 return entity;
345 }
346 std::shared_ptr<DynamicEntity> entity =
347 spec.isArchetype ? engine.createEntityWith(spec.id, Variance::Average, rng)
348 : engine.createCreatureWith(spec.id, Variance::Average, rng);
349 if (entity != nullptr && !spec.isArchetype) {
350 entity->setBaseAttribute("Attack", spec.attackValue);
351 entity->setBaseAttribute("Parry", spec.defenseValue);
352 entity->setBaseAttribute("Armor_Rating", spec.armorRating);
353 // D&D-style systems promote the bestiary's real AC and initiative into
354 // base attributes: the derived `AC = 10 + DEX_mod` does not match a
355 // creature's stat block, and `Initiative` has no derived stat at all.
356 // Skipping a zero value keeps rulesets without those fields (TDE, BRP)
357 // on their derived stats.
358 if (spec.acValue != 0) {
359 entity->setBaseAttribute("AC", spec.acValue);
360 }
361 if (spec.initiativeValue != 0) {
362 entity->setBaseAttribute("Initiative", spec.initiativeValue);
363 }
364 }
365 return entity;
366}

375[[nodiscard]] inline double spellAverageDamage(const Json &spell) {
376 double total = 0.0;
377 const auto addDice = [&](const Json &value) {
378 if (value.is_number()) {
379 total += value.get<double>();
380 } else if (value.is_string()) {
381 total += DiceExpression(value.get<std::string>()).expectedValue();
382 }
383 };
384 if (spell.contains("damage")) {
385 addDice(spell.at("damage"));
386 }
387 if (spell.contains("effects") && spell.at("effects").is_array()) {
388 for (const Json &effect : spell.at("effects")) {
389 if (effect.value("kind", "") == "damage" && effect.contains("dice")) {
390 addDice(effect.at("dice"));
391 }
392 }
393 }
394 return total;
395}

406[[nodiscard]] inline std::string pickSpell(const RulesetEngine &engine, const DynamicEntity &caster,
407 const std::vector<std::string> &spellIds) {
408 const std::string &resourceId = engine.ruleset().spellResource;
409 std::string best;
410 double bestDamage = 0.0;
411 for (const std::string &spellId : spellIds) {
412 const Json *spell = engine.findSpell(spellId);
413 if (spell == nullptr) {
414 continue;
415 }
416 int32_t cost = 0;
417 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
418 cost = spell->at("cost").get<int32_t>();
419 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
420 cost = spell->at("ae_cost").get<int32_t>();
421 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
422 cost = spell->at("level").get<int32_t>();
423 }
424 if (cost > 0 && !resourceId.empty() && caster.resource(resourceId) < cost) {
425 continue; // cannot afford
426 }
427 const double damage = spellAverageDamage(*spell);
428 if (damage > 0.0 && damage > bestDamage) {
429 bestDamage = damage;
430 best = spellId;
431 }
432 }
433 return best;
434}

445template <RandomNumberGenerator Rng>
446bool attackOnce(RulesetEngine &engine, DynamicEntity &attacker, const DiceExpression &damage,
447 DynamicEntity &defender, std::string_view checkTypeId,
448 std::string_view hpResourceId, Rng &rng, FightRoundLog *roundLog = nullptr,
449 int actorIndex = 0, int targetIndex = 1) {
450 // The attacker's restrictions are enforced: a creature that cannot take
451 // actions (incapacitated, stunned, paralyzed, ...) cannot attack.
452 if (!engine.actionAllowed(attacker, "action")) {
453 return false;
454 }
455 // Resolve against effective stats so worn gear and active conditions (armour,
456 // encumbrance, wounds) influence the fight — with nothing equipped the
457 // effective value equals the raw value, so ungeared fights are unchanged.
458 const CheckResult hit =
459 engine.executeCheckEffective(checkTypeId, attacker, &defender, CheckParams{}, rng);
460 FightActionLog action;
461 if (roundLog != nullptr) {
462 action.actorIndex = actorIndex;
463 action.targetIndex = targetIndex;
464 action.kind = "attack";
465 action.checkDice = hit.rawDiceRolls;
466 action.isHit = hit.isSuccess;
467 action.hpBefore = defender.resource(hpResourceId);
468 }
469 if (!hit.isSuccess) {
470 if (roundLog != nullptr) {
471 action.targetHp = defender.resource(hpResourceId);
472 roundLog->actions.push_back(std::move(action));
473 }
474 return false;
475 }
476 // Roll the damage explicitly so the individual dice can be logged. This is
477 // observation-only: roll() consumes exactly the RNG stream rollSum() would,
478 // so a logged fight and a plain fight stay byte-identical (pinned by test).
479 int32_t raw = 0;
480 if (roundLog != nullptr) {
481 const std::vector<int> rolled = damage.roll(rng);
482 action.damageDice.assign(rolled.begin(), rolled.end());
483 raw = damage.constant();
484 const auto &groups = damage.dice();
485 for (std::size_t g = 0, i = 0; g < groups.size(); ++g) {
486 for (int j = 0; j < groups[g].count; ++j, ++i) {
487 raw += groups[g].sign * rolled[i];
488 }
489 }
490 } else {
491 raw = static_cast<int32_t>(damage.rollSum(rng));
492 }
493 engine.applyDamage(attacker, defender, hpResourceId, raw);
494 if (roundLog != nullptr) {
495 action.damage = raw;
496 action.targetHp = defender.resource(hpResourceId);
497 roundLog->actions.push_back(std::move(action));
498 }
499 return defender.resource(hpResourceId) <= 0;
500}

513template <RandomNumberGenerator Rng>
514bool actOnce(RulesetEngine &engine, DynamicEntity &actor, const CombatantSpec &spec,
515 const DiceExpression &weaponDamage, DynamicEntity &defender,
516 std::string_view checkTypeId, std::string_view hpResourceId, bool useMagic, Rng &rng,
517 FightRoundLog *roundLog = nullptr, int actorIndex = 0, int targetIndex = 1) {
518 if (useMagic && !spec.spellIds.empty()) {
519 const std::string spellId = pickSpell(engine, actor, spec.spellIds);
520 if (!spellId.empty()) {
521 const int32_t hpBefore = defender.resource(hpResourceId);
522 const auto cast = engine.castSpell(spellId, actor, &defender, CheckParams{}, rng);
523 if (roundLog != nullptr) {
524 FightActionLog action;
525 action.actorIndex = actorIndex;
526 action.targetIndex = targetIndex;
527 action.kind = "cast";
528 action.spellId = spellId;
529 action.hpBefore = hpBefore;
530 action.checkDice = cast.check.rawDiceRolls;
531 action.isHit = cast.cast;
532 action.damage = cast.appliedDamage;
533 action.targetHp = defender.resource(hpResourceId);
534 action.resourceCost = cast.cost;
535 action.resourceId = cast.resourceId;
536 roundLog->actions.push_back(std::move(action));
537 }
538 if (cast.cast) {
539 return defender.resource(hpResourceId) <= 0;
540 }
541 return false; // the casting attempt failed — the round is spent
542 }
543 }
544 return attackOnce(engine, actor, weaponDamage, defender, checkTypeId, hpResourceId, rng, roundLog,
545 actorIndex, targetIndex);
546}

556template <RandomNumberGenerator Rng>
557[[nodiscard]] int resolveRound(RulesetEngine &engine, const CombatantSpec &a,
558 const CombatantSpec &b, DynamicEntity &ea, DynamicEntity &eb,
559 const DiceExpression &damageA, const DiceExpression &damageB,
560 std::string_view checkTypeId, std::string_view hpResourceId,
561 bool useMagic, Rng &rng, FightLog *log = nullptr,
562 int roundNumber = 0) {
563 const int initA = ea.getStat("Initiative") + rng(1, 6);
564 const int initB = eb.getStat("Initiative") + rng(1, 6);
565 const bool aFirst = initA != initB ? initA > initB : rng(0, 1) == 0;
566
567 // The round's transcript (built only when a log is requested — observation
568 // of the same dice, never an extra draw, so plain and logged fights match).
569 FightRoundLog roundLog;
570 if (log != nullptr) {
571 roundLog.round = roundNumber;
572 roundLog.initStat[0] = ea.getStat("Initiative");
573 roundLog.initStat[1] = eb.getStat("Initiative");
574 roundLog.initRoll[0] = initA - ea.getStat("Initiative");
575 roundLog.initRoll[1] = initB - eb.getStat("Initiative");
576 roundLog.initTotal[0] = initA;
577 roundLog.initTotal[1] = initB;
578 roundLog.goesFirst = aFirst ? 0 : 1;
579 }
580
581 DynamicEntity &first = aFirst ? ea : eb;
582 DynamicEntity &second = aFirst ? eb : ea;
583 const CombatantSpec &firstSpec = aFirst ? a : b;
584 const CombatantSpec &secondSpec = aFirst ? b : a;
585 const DiceExpression &firstDamage = aFirst ? damageA : damageB;
586 const DiceExpression &secondDamage = aFirst ? damageB : damageA;
587 const int firstIndex = aFirst ? 0 : 1;
588 const int secondIndex = aFirst ? 1 : 0;
589
590 if (actOnce(engine, first, firstSpec, firstDamage, second, checkTypeId, hpResourceId, useMagic,
591 rng, log != nullptr ? &roundLog : nullptr, firstIndex, secondIndex)) {
592 if (log != nullptr) {
593 log->rounds.push_back(std::move(roundLog));
594 }
595 return firstIndex;
596 }
597 if (actOnce(engine, second, secondSpec, secondDamage, first, checkTypeId, hpResourceId, useMagic,
598 rng, log != nullptr ? &roundLog : nullptr, secondIndex, firstIndex)) {
599 if (log != nullptr) {
600 log->rounds.push_back(std::move(roundLog));
601 }
602 return secondIndex;
603 }
604 if (log != nullptr) {
605 log->rounds.push_back(std::move(roundLog));
606 }
607 return -1;
608}

616template <RandomNumberGenerator Rng>
617[[nodiscard]] FightOutcome runFightImpl(RulesetEngine &engine, const CombatantSpec &a,
618 const CombatantSpec &b, std::string_view checkTypeId,
619 std::string_view hpResourceId, int maxRounds, Rng &rng,
620 bool useMagic, FightLog *log) {
621 FightOutcome outcome;
622 if (log != nullptr) {
623 log->names = {a.name, b.name};
624 }
625 auto ea = createFighter(engine, a, rng);
626 auto eb = createFighter(engine, b, rng);
627 if (ea == nullptr || eb == nullptr) {
628 return outcome;
629 }
630 outcome.maxLp[0] = ea->resource(hpResourceId);
631 outcome.maxLp[1] = eb->resource(hpResourceId);
632 if (log != nullptr) {
633 log->maxLp = outcome.maxLp;
634 }
635 const DiceExpression damageA(a.damageExpression);
636 const DiceExpression damageB(b.damageExpression);
637
638 int roundsPlayed = 0;
639 for (; roundsPlayed < maxRounds; ++roundsPlayed) {
640 const int winner = detail::resolveRound(engine, a, b, *ea, *eb, damageA, damageB, checkTypeId,
641 hpResourceId, useMagic, rng, log, roundsPlayed + 1);
642 if (winner >= 0) {
643 outcome.winnerIndex = winner;
644 break;
645 }
646 }
647
648 outcome.rounds = outcome.winnerIndex == -1 ? maxRounds : roundsPlayed + 1;
649 outcome.remainingLp[0] = ea->resource(hpResourceId);
650 outcome.remainingLp[1] = eb->resource(hpResourceId);
651 if (log != nullptr) {
652 log->winnerIndex = outcome.winnerIndex;
653 }
654 return outcome;
655}
