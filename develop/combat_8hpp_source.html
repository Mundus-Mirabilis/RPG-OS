135[[nodiscard]] inline std::string resolveAttackCheckType(const Ruleset &ruleset) {
136 for (const CheckTypeDef &def : ruleset.checkTypes) {
137 if (def.id == "tde_attack") {
138 return def.id;
139 }
140 }
141 for (const CheckTypeDef &def : ruleset.checkTypes) {
142 if (def.recipe.resolution == Resolution::Opposed) {
143 return def.id;
144 }
145 }
146 // D&D-style attack-vs-AC: a threshold check rolled against a *target* stat
147 // (e.g. AC) whose bonus comes from the combatant's promoted `Attack` stat
148 // (the bestiary `to_hit`). Systems without an opposed parry roll (D&D)
149 // use this instead of the opposed resolution above.
150 for (const CheckTypeDef &def : ruleset.checkTypes) {
151 if (def.recipe.resolution == Resolution::Threshold &&
152 def.recipe.thresholdSource == ThresholdSource::TargetStat &&
153 std::ranges::contains(def.recipe.bonusStats, "Attack")) {
154 return def.id;
155 }
156 }
157 return {};
158}

164[[nodiscard]] inline bool fillArchetypeSpec(RulesetEngine &engine, const Json &entry,
165 std::string_view id, CombatantSpec &out,
166 std::string_view weaponDamage,
167 DefaultRandom &probeRng) {
168 auto probe = engine.createEntityWith(id, Variance::Average, probeRng);
169 if (probe == nullptr) {
170 return false;
171 }
172 out.id = std::string(id);
173 out.name = entry.value("name", out.id);
174 out.isArchetype = true;
175 out.attackValue = probe->getStat("Attack");
176 out.defenseValue = probe->getStat("Parry");
177 out.armorRating = probe->baseAttribute("Armor_Rating");
178 out.damageExpression = std::string(weaponDamage);
179 // An archetype's `spells_known` were loaded into the probe's spellbook;
180 // copy them so the fight loop can cast magic for this combatant.
181 out.spellIds = probe->spellbook().known();
182 return true;
183}

188[[nodiscard]] inline bool fillCreatureSpec(RulesetEngine &engine, const Json &entry,
189 std::string_view id, CombatantSpec &out,
190 std::string_view weaponDamage, DefaultRandom &probeRng) {
191 auto probe = engine.createCreatureWith(id, Variance::Average, probeRng);
192 if (probe == nullptr) {
193 return false;
194 }
195 out.id = std::string(id);
196 out.name = entry.value("name", out.id);
197 out.isArchetype = false;
198 out.defenseValue = entry.value("dodge", 0);
199 out.armorRating = entry.value("armor_rating", 0);
200 out.acValue = entry.value("ac", 0);
201 out.initiativeValue = entry.value("initiative", 0);
202 out.attackValue = 0;
203 out.damageExpression = std::string(weaponDamage);
204 if (entry.contains("attacks") && entry.at("attacks").is_array()) {
205 int32_t bestToHit = -1;
206 for (const Json &attack : entry.at("attacks")) {
207 const int32_t toHit = attack.value("to_hit", 0);
208 if (toHit > bestToHit) {
209 bestToHit = toHit;
210 out.attackValue = toHit;
211 out.damageExpression = attack.value("damage", "1d6");
212 }
213 }
214 }
215 if (out.attackValue <= 0) {
216 // No natural attack defined: fall back to the creature's derived
217 // attack value and an unarmed strike.
218 out.attackValue = probe->getStat("Attack");
219 out.damageExpression = "1d6";
220 }
221 return true;
222}

241[[nodiscard]] inline bool makeCombatantSpec(RulesetEngine &engine, std::string_view id,
242 CombatantSpec &out,
243 std::string_view weaponDamage = kDefaultWeaponDamage) {
244 const Ruleset &ruleset = engine.ruleset();
245 if (!ruleset.data.is_object()) {
246 return false;
247 }
248 DefaultRandom probeRng;
249
250 if (ruleset.data.contains("archetypes")) {
251 for (const Json &entry : ruleset.data.at("archetypes")) {
252 if (entry.value("id", "") != id) {
253 continue;
254 }
255 return detail::fillArchetypeSpec(engine, entry, id, out, weaponDamage, probeRng);
256 }
257 }
258
259 if (ruleset.data.contains("creatures")) {
260 for (const Json &entry : ruleset.data.at("creatures")) {
261 if (entry.value("id", "") != id) {
262 continue;
263 }
264 return detail::fillCreatureSpec(engine, entry, id, out, weaponDamage, probeRng);
265 }
266 }
267 return false;
268}

282[[nodiscard]] inline bool
283makeCombatantSpecFromEntity(RulesetEngine &engine, const DynamicEntity &entity, CombatantSpec &out,
284 std::string_view weaponDamage = kDefaultWeaponDamage) {
285 out.id = entity.id();
286 out.name = entity.id();
287 // Resolve a human-readable name when the sheet happens to be a known
288 // ruleset entry (archetype or bestiary); otherwise the id is the name.
289 const Json *record = engine.findDataRecord("archetypes", entity.id());
290 if (record == nullptr) {
291 record = engine.findDataRecord("creatures", entity.id());
292 }
293 if (record != nullptr) {
294 out.name = record->value("name", out.name);
295 }
296 out.isArchetype = true; // the sheet fully describes the character
297 out.attackValue = entity.getStat("Attack");
298 out.defenseValue = entity.getStat("Parry");
299 out.armorRating = entity.getStat("Armor_Rating");
300 out.acValue = entity.getStat("AC");
301 out.initiativeValue = entity.getStat("Initiative");
302 out.damageExpression = std::string(weaponDamage);
303 out.spellIds = entity.spellbook().known();
304 entity.toJson(out.sheet);
305 return true;

319template <RandomNumberGenerator Rng>
320[[nodiscard]] std::shared_ptr<DynamicEntity> createFighter(RulesetEngine &engine,
321 const CombatantSpec &spec, Rng &rng) {
322 // A spec carrying an embedded sheet fights that sheet directly: ANY
323 // character — not just a named archetype or bestiary entry — can take part
324 // in a fight this way (e.g. a character entered into a form). The sheet is
325 // restored into a fresh entity per fight and its resource pools are
326 // (re)created so the fighter starts at its full derived hit points.
327 if (spec.sheet.is_object()) {
328 auto entity = std::make_shared<DynamicEntity>(engine.ruleset(), spec.id);
329 entity->fromJson(spec.sheet);
330 entity->refreshResources();
331 return entity;
332 }
333 std::shared_ptr<DynamicEntity> entity =
334 spec.isArchetype ? engine.createEntityWith(spec.id, Variance::Average, rng)
335 : engine.createCreatureWith(spec.id, Variance::Average, rng);
336 if (entity != nullptr && !spec.isArchetype) {
337 entity->setBaseAttribute("Attack", spec.attackValue);
338 entity->setBaseAttribute("Parry", spec.defenseValue);
339 entity->setBaseAttribute("Armor_Rating", spec.armorRating);
340 // D&D-style systems promote the bestiary's real AC and initiative into
341 // base attributes: the derived `AC = 10 + DEX_mod` does not match a
342 // creature's stat block, and `Initiative` has no derived stat at all.
343 // Skipping a zero value keeps rulesets without those fields (TDE, BRP)
344 // on their derived stats.
345 if (spec.acValue != 0) {
346 entity->setBaseAttribute("AC", spec.acValue);
347 }
348 if (spec.initiativeValue != 0) {
349 entity->setBaseAttribute("Initiative", spec.initiativeValue);
350 }
351 }
352 return entity;
353}

362[[nodiscard]] inline double spellAverageDamage(const Json &spell) {
363 double total = 0.0;
364 const auto addDice = [&](const Json &value) {
365 if (value.is_number()) {
366 total += value.get<double>();
367 } else if (value.is_string()) {
368 total += DiceExpression(value.get<std::string>()).expectedValue();
369 }
370 };
371 if (spell.contains("damage")) {
372 addDice(spell.at("damage"));
373 }
374 if (spell.contains("effects") && spell.at("effects").is_array()) {
375 for (const Json &effect : spell.at("effects")) {
376 if (effect.value("kind", "") == "damage" && effect.contains("dice")) {
377 addDice(effect.at("dice"));
378 }
379 }
380 }
381 return total;
382}

393[[nodiscard]] inline std::string pickSpell(const RulesetEngine &engine, const DynamicEntity &caster,
394 const std::vector<std::string> &spellIds) {
395 const std::string &resourceId = engine.ruleset().spellResource;
396 std::string best;
397 double bestDamage = 0.0;
398 for (const std::string &spellId : spellIds) {
399 const Json *spell = engine.findSpell(spellId);
400 if (spell == nullptr) {
401 continue;
402 }
403 const int32_t cost = rpg_os::spellCost(*spell);
404 if (cost > 0 && !resourceId.empty() && caster.resource(resourceId) < cost) {
405 continue; // cannot afford
406 }
407 const double damage = spellAverageDamage(*spell);
408 if (damage > 0.0 && damage > bestDamage) {
409 bestDamage = damage;
410 best = spellId;
411 }
412 }
413 return best;
414}

425template <RandomNumberGenerator Rng>
426bool attackOnce(RulesetEngine &engine, DynamicEntity &attacker, const DiceExpression &damage,
427 DynamicEntity &defender, std::string_view checkTypeId,
428 std::string_view hpResourceId, Rng &rng, FightRoundLog *roundLog = nullptr,
429 int actorIndex = 0, int targetIndex = 1) {
430 // The attacker's restrictions are enforced: a creature that cannot take
431 // actions (incapacitated, stunned, paralyzed, ...) cannot attack.
432 if (!engine.actionAllowed(attacker, "action")) {
433 return false;
434 }
435 // Resolve against effective stats so worn gear and active conditions (armour,
436 // encumbrance, wounds) influence the fight — with nothing equipped the
437 // effective value equals the raw value, so ungeared fights are unchanged.
438 const CheckResult hit =
439 engine.executeCheckEffective(checkTypeId, attacker, &defender, CheckParams{}, rng);
440 FightActionLog action;
441 if (roundLog != nullptr) {
442 action.actorIndex = actorIndex;
443 action.targetIndex = targetIndex;
444 action.kind = "attack";
445 action.checkDice = hit.rawDiceRolls;
446 action.isHit = hit.isSuccess;
447 action.hpBefore = defender.resource(hpResourceId);
448 }
449 if (!hit.isSuccess) {
450 if (roundLog != nullptr) {
451 action.targetHp = defender.resource(hpResourceId);
452 roundLog->actions.push_back(std::move(action));
453 }
454 return false;
455 }
456 // Roll the damage explicitly so the individual dice can be logged. This is
457 // observation-only: roll() consumes exactly the RNG stream rollSum() would,
458 // so a logged fight and a plain fight stay byte-identical (pinned by test).
459 int32_t raw = 0;
460 if (roundLog != nullptr) {
461 const std::vector<int> rolled = damage.roll(rng);
462 action.damageDice.assign(rolled.begin(), rolled.end());
463 raw = damage.constant();
464 const auto &groups = damage.dice();
465 for (std::size_t g = 0, i = 0; g < groups.size(); ++g) {
466 for (int j = 0; j < groups[g].count; ++j, ++i) {
467 raw += groups[g].sign * rolled[i];
468 }
469 }
470 } else {
471 raw = static_cast<int32_t>(damage.rollSum(rng));
472 }
473 engine.applyDamage(attacker, defender, hpResourceId, raw);
474 if (roundLog != nullptr) {
475 action.damage = raw;
476 action.targetHp = defender.resource(hpResourceId);
477 roundLog->actions.push_back(std::move(action));
478 }
479 return defender.resource(hpResourceId) <= 0;
480}

493template <RandomNumberGenerator Rng>
494bool actOnce(RulesetEngine &engine, DynamicEntity &actor, const CombatantSpec &spec,
495 const DiceExpression &weaponDamage, DynamicEntity &defender,
496 std::string_view checkTypeId, std::string_view hpResourceId, bool useMagic, Rng &rng,
497 FightRoundLog *roundLog = nullptr, int actorIndex = 0, int targetIndex = 1) {
498 if (useMagic && !spec.spellIds.empty()) {
499 const std::string spellId = pickSpell(engine, actor, spec.spellIds);
500 if (!spellId.empty()) {
501 const int32_t hpBefore = defender.resource(hpResourceId);
502 const auto cast = engine.castSpell(spellId, actor, &defender, CheckParams{}, rng);
503 if (roundLog != nullptr) {
504 FightActionLog action;
505 action.actorIndex = actorIndex;
506 action.targetIndex = targetIndex;
507 action.kind = "cast";
508 action.spellId = spellId;
509 action.hpBefore = hpBefore;
510 action.checkDice = cast.check.rawDiceRolls;
511 action.isHit = cast.cast;
512 action.damage = cast.appliedDamage;
513 action.targetHp = defender.resource(hpResourceId);
514 action.resourceCost = cast.cost;
515 action.resourceId = cast.resourceId;
516 roundLog->actions.push_back(std::move(action));
517 }
518 if (cast.cast) {
519 return defender.resource(hpResourceId) <= 0;
520 }
521 return false; // the casting attempt failed — the round is spent
522 }
523 }
524 return attackOnce(engine, actor, weaponDamage, defender, checkTypeId, hpResourceId, rng, roundLog,
525 actorIndex, targetIndex);
526}

536template <RandomNumberGenerator Rng>
537[[nodiscard]] int resolveRound(RulesetEngine &engine, const CombatantSpec &a,
538 const CombatantSpec &b, DynamicEntity &ea, DynamicEntity &eb,
539 const DiceExpression &damageA, const DiceExpression &damageB,
540 std::string_view checkTypeId, std::string_view hpResourceId,
541 bool useMagic, Rng &rng, FightLog *log = nullptr,
542 int roundNumber = 0) {
543 const int initA = ea.getStat("Initiative") + rng(1, 6);
544 const int initB = eb.getStat("Initiative") + rng(1, 6);
545 const bool aFirst = initA != initB ? initA > initB : rng(0, 1) == 0;
546
547 // The round's transcript (built only when a log is requested — observation
548 // of the same dice, never an extra draw, so plain and logged fights match).
549 FightRoundLog roundLog;
550 if (log != nullptr) {
551 roundLog.round = roundNumber;
552 roundLog.initStat[0] = ea.getStat("Initiative");
553 roundLog.initStat[1] = eb.getStat("Initiative");
554 roundLog.initRoll[0] = initA - ea.getStat("Initiative");
555 roundLog.initRoll[1] = initB - eb.getStat("Initiative");
556 roundLog.initTotal[0] = initA;
557 roundLog.initTotal[1] = initB;
558 roundLog.goesFirst = aFirst ? 0 : 1;
559 }
560
561 DynamicEntity &first = aFirst ? ea : eb;
562 DynamicEntity &second = aFirst ? eb : ea;
563 const CombatantSpec &firstSpec = aFirst ? a : b;
564 const CombatantSpec &secondSpec = aFirst ? b : a;
565 const DiceExpression &firstDamage = aFirst ? damageA : damageB;
566 const DiceExpression &secondDamage = aFirst ? damageB : damageA;
567 const int firstIndex = aFirst ? 0 : 1;
568 const int secondIndex = aFirst ? 1 : 0;
569
570 if (actOnce(engine, first, firstSpec, firstDamage, second, checkTypeId, hpResourceId, useMagic,
571 rng, log != nullptr ? &roundLog : nullptr, firstIndex, secondIndex)) {
572 if (log != nullptr) {
573 log->rounds.push_back(std::move(roundLog));
574 }
575 return firstIndex;
576 }
577 if (actOnce(engine, second, secondSpec, secondDamage, first, checkTypeId, hpResourceId, useMagic,
578 rng, log != nullptr ? &roundLog : nullptr, secondIndex, firstIndex)) {
579 if (log != nullptr) {
580 log->rounds.push_back(std::move(roundLog));
581 }
582 return secondIndex;
583 }
584 if (log != nullptr) {
585 log->rounds.push_back(std::move(roundLog));
586 }
587 return -1;
588}

596template <RandomNumberGenerator Rng>
597[[nodiscard]] FightOutcome runFightImpl(RulesetEngine &engine, const CombatantSpec &a,
598 const CombatantSpec &b, std::string_view checkTypeId,
599 std::string_view hpResourceId, int maxRounds, Rng &rng,
600 bool useMagic, FightLog *log) {
601 FightOutcome outcome;
602 if (log != nullptr) {
603 log->names = {a.name, b.name};
604 }
605 auto ea = createFighter(engine, a, rng);
606 auto eb = createFighter(engine, b, rng);
607 if (ea == nullptr || eb == nullptr) {
608 return outcome;
609 }
610 outcome.maxLp[0] = ea->resource(hpResourceId);
611 outcome.maxLp[1] = eb->resource(hpResourceId);
612 if (log != nullptr) {
613 log->maxLp = outcome.maxLp;
614 }
615 const DiceExpression damageA(a.damageExpression);
616 const DiceExpression damageB(b.damageExpression);
617
618 int roundsPlayed = 0;
619 for (; roundsPlayed < maxRounds; ++roundsPlayed) {
620 const int winner = detail::resolveRound(engine, a, b, *ea, *eb, damageA, damageB, checkTypeId,
621 hpResourceId, useMagic, rng, log, roundsPlayed + 1);
622 if (winner >= 0) {
623 outcome.winnerIndex = winner;
624 break;
625 }
626 }
627
628 outcome.rounds = outcome.winnerIndex == -1 ? maxRounds : roundsPlayed + 1;
629 outcome.remainingLp[0] = ea->resource(hpResourceId);
630 outcome.remainingLp[1] = eb->resource(hpResourceId);
631 if (log != nullptr) {
632 log->winnerIndex = outcome.winnerIndex;
633 }
634 return outcome;
635}
