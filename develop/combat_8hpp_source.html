93[[nodiscard]] inline std::string resolveAttackCheckType(const Ruleset &ruleset) {
94 for (const CheckTypeDef &def : ruleset.checkTypes) {
95 if (def.id == "tde_attack") {
96 return def.id;
97 }
98 }
99 for (const CheckTypeDef &def : ruleset.checkTypes) {
100 if (def.recipe.resolution == Resolution::Opposed) {
101 return def.id;
102 }
103 }
104 // D&D-style attack-vs-AC: a threshold check rolled against a *target* stat
105 // (e.g. AC) whose bonus comes from the combatant's promoted `Attack` stat
106 // (the bestiary `to_hit`). Systems without an opposed parry roll (D&D)
107 // use this instead of the opposed resolution above.
108 for (const CheckTypeDef &def : ruleset.checkTypes) {
109 if (def.recipe.resolution == Resolution::Threshold &&
110 def.recipe.thresholdSource == ThresholdSource::TargetStat &&
111 std::ranges::contains(def.recipe.bonusStats, "Attack")) {
112 return def.id;
113 }
114 }
115 return {};
116}

139[[nodiscard]] inline bool fillArchetypeSpec(RulesetEngine &engine, const Json &entry,
140 std::string_view id, CombatantSpec &out,
141 std::string_view weaponDamage,
142 DefaultRandom &probeRng) {
143 auto probe = engine.createEntityWith(id, Variance::Average, probeRng);
144 if (probe == nullptr) {
145 return false;
146 }
147 out.id = std::string(id);
148 out.name = entry.value("name", out.id);
149 out.isArchetype = true;
150 out.attackValue = probe->getStat("Attack");
151 out.defenseValue = probe->getStat("Parry");
152 out.armorRating = probe->baseAttribute("Armor_Rating");
153 out.damageExpression = std::string(weaponDamage);
154 // An archetype's `spells_known` were loaded into the probe's spellbook;
155 // copy them so the fight loop can cast magic for this combatant.
156 out.spellIds = probe->spellbook().known();
157 return true;
158}

163[[nodiscard]] inline bool fillCreatureSpec(RulesetEngine &engine, const Json &entry,
164 std::string_view id, CombatantSpec &out,
165 std::string_view weaponDamage, DefaultRandom &probeRng) {
166 auto probe = engine.createCreatureWith(id, Variance::Average, probeRng);
167 if (probe == nullptr) {
168 return false;
169 }
170 out.id = std::string(id);
171 out.name = entry.value("name", out.id);
172 out.isArchetype = false;
173 out.defenseValue = entry.value("dodge", 0);
174 out.armorRating = entry.value("armor_rating", 0);
175 out.acValue = entry.value("ac", 0);
176 out.initiativeValue = entry.value("initiative", 0);
177 out.attackValue = 0;
178 out.damageExpression = std::string(weaponDamage);
179 if (entry.contains("attacks") && entry.at("attacks").is_array()) {
180 int32_t bestToHit = -1;
181 for (const Json &attack : entry.at("attacks")) {
182 const int32_t toHit = attack.value("to_hit", 0);
183 if (toHit > bestToHit) {
184 bestToHit = toHit;
185 out.attackValue = toHit;
186 out.damageExpression = attack.value("damage", "1d6");
187 }
188 }
189 }
190 if (out.attackValue <= 0) {
191 // No natural attack defined: fall back to the creature's derived
192 // attack value and an unarmed strike.
193 out.attackValue = probe->getStat("Attack");
194 out.damageExpression = "1d6";
195 }
196 return true;
197}

211[[nodiscard]] inline bool makeCombatantSpec(RulesetEngine &engine, std::string_view id,
212 CombatantSpec &out,
213 std::string_view weaponDamage = "1d6+4") {
214 const Ruleset &ruleset = engine.ruleset();
215 if (!ruleset.data.is_object()) {
216 return false;
217 }
218 DefaultRandom probeRng;
219
220 if (ruleset.data.contains("archetypes")) {
221 for (const Json &entry : ruleset.data.at("archetypes")) {
222 if (entry.value("id", "") != id) {
223 continue;
224 }
225 return detail::fillArchetypeSpec(engine, entry, id, out, weaponDamage, probeRng);
226 }
227 }
228
229 if (ruleset.data.contains("creatures")) {
230 for (const Json &entry : ruleset.data.at("creatures")) {
231 if (entry.value("id", "") != id) {
232 continue;
233 }
234 return detail::fillCreatureSpec(engine, entry, id, out, weaponDamage, probeRng);
235 }
236 }
237 return false;
238}

252[[nodiscard]] inline bool makeCombatantSpecFromEntity(RulesetEngine &engine,
253 const DynamicEntity &entity,
254 CombatantSpec &out,
255 std::string_view weaponDamage = "1d6+4") {
256 out.id = entity.id();
257 out.name = entity.id();
258 // Resolve a human-readable name when the sheet happens to be a known
259 // ruleset entry (archetype or bestiary); otherwise the id is the name.
260 const Json *record = engine.findDataRecord("archetypes", entity.id());
261 if (record == nullptr) {
262 record = engine.findDataRecord("creatures", entity.id());
263 }
264 if (record != nullptr) {
265 out.name = record->value("name", out.name);
266 }
267 out.isArchetype = true; // the sheet fully describes the character
268 out.attackValue = entity.getStat("Attack");
269 out.defenseValue = entity.getStat("Parry");
270 out.armorRating = entity.getStat("Armor_Rating");
271 out.acValue = entity.getStat("AC");
272 out.initiativeValue = entity.getStat("Initiative");
273 out.damageExpression = std::string(weaponDamage);
274 out.spellIds = entity.spellbook().known();
275 entity.toJson(out.sheet);
276 return true;
277}

290template <RandomNumberGenerator Rng>
291[[nodiscard]] std::shared_ptr<DynamicEntity> createFighter(RulesetEngine &engine,
292 const CombatantSpec &spec, Rng &rng) {
293 // A spec carrying an embedded sheet fights that sheet directly: ANY
294 // character — not just a named archetype or bestiary entry — can take part
295 // in a fight this way (e.g. a character entered into a form). The sheet is
296 // restored into a fresh entity per fight and its resource pools are
297 // (re)created so the fighter starts at its full derived hit points.
298 if (spec.sheet.is_object()) {
299 auto entity = std::make_shared<DynamicEntity>(engine.ruleset(), spec.id);
300 entity->fromJson(spec.sheet);
301 entity->refreshResources();
302 return entity;
303 }
304 std::shared_ptr<DynamicEntity> entity =
305 spec.isArchetype ? engine.createEntityWith(spec.id, Variance::Average, rng)
306 : engine.createCreatureWith(spec.id, Variance::Average, rng);
307 if (entity != nullptr && !spec.isArchetype) {
308 entity->setBaseAttribute("Attack", spec.attackValue);
309 entity->setBaseAttribute("Parry", spec.defenseValue);
310 entity->setBaseAttribute("Armor_Rating", spec.armorRating);
311 // D&D-style systems promote the bestiary's real AC and initiative into
312 // base attributes: the derived `AC = 10 + DEX_mod` does not match a
313 // creature's stat block, and `Initiative` has no derived stat at all.
314 // Skipping a zero value keeps rulesets without those fields (TDE, BRP)
315 // on their derived stats.
316 if (spec.acValue != 0) {
317 entity->setBaseAttribute("AC", spec.acValue);
318 }
319 if (spec.initiativeValue != 0) {
320 entity->setBaseAttribute("Initiative", spec.initiativeValue);
321 }
322 }
323 return entity;
324}

333[[nodiscard]] inline double spellAverageDamage(const Json &spell) {
334 double total = 0.0;
335 const auto addDice = [&](const Json &value) {
336 if (value.is_number()) {
337 total += value.get<double>();
338 } else if (value.is_string()) {
339 total += DiceExpression(value.get<std::string>()).expectedValue();
340 }
341 };
342 if (spell.contains("damage")) {
343 addDice(spell.at("damage"));
344 }
345 if (spell.contains("effects") && spell.at("effects").is_array()) {
346 for (const Json &effect : spell.at("effects")) {
347 if (effect.value("kind", "") == "damage" && effect.contains("dice")) {
348 addDice(effect.at("dice"));
349 }
350 }
351 }
352 return total;
353}

364[[nodiscard]] inline std::string pickSpell(const RulesetEngine &engine, const DynamicEntity &caster,
365 const std::vector<std::string> &spellIds) {
366 const std::string &resourceId = engine.ruleset().spellResource;
367 std::string best;
368 double bestDamage = 0.0;
369 for (const std::string &spellId : spellIds) {
370 const Json *spell = engine.findSpell(spellId);
371 if (spell == nullptr) {
372 continue;
373 }
374 int32_t cost = 0;
375 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
376 cost = spell->at("cost").get<int32_t>();
377 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
378 cost = spell->at("ae_cost").get<int32_t>();
379 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
380 cost = spell->at("level").get<int32_t>();
381 }
382 if (cost > 0 && !resourceId.empty() && caster.resource(resourceId) < cost) {
383 continue; // cannot afford
384 }
385 const double damage = spellAverageDamage(*spell);
386 if (damage > 0.0 && damage > bestDamage) {
387 bestDamage = damage;
388 best = spellId;
389 }
390 }
391 return best;
392}

403template <RandomNumberGenerator Rng>
404bool attackOnce(RulesetEngine &engine, DynamicEntity &attacker, const DiceExpression &damage,
405 DynamicEntity &defender, std::string_view checkTypeId,
406 std::string_view hpResourceId, Rng &rng) {
407 // The attacker's restrictions are enforced: a creature that cannot take
408 // actions (incapacitated, stunned, paralyzed, ...) cannot attack.
409 if (!engine.actionAllowed(attacker, "action")) {
410 return false;
411 }
412 // Resolve against effective stats so worn gear and active conditions (armour,
413 // encumbrance, wounds) influence the fight — with nothing equipped the
414 // effective value equals the raw value, so ungeared fights are unchanged.
415 const CheckResult hit =
416 engine.executeCheckEffective(checkTypeId, attacker, &defender, CheckParams{}, rng);
417 if (!hit.isSuccess) {
418 return false;
419 }
420 const int32_t raw = static_cast<int32_t>(damage.rollSum(rng));
421 engine.applyDamage(attacker, defender, hpResourceId, raw);
422 return defender.resource(hpResourceId) <= 0;
423}

436template <RandomNumberGenerator Rng>
437bool actOnce(RulesetEngine &engine, DynamicEntity &actor, const CombatantSpec &spec,
438 const DiceExpression &weaponDamage, DynamicEntity &defender,
439 std::string_view checkTypeId, std::string_view hpResourceId, bool useMagic, Rng &rng) {
440 if (useMagic && !spec.spellIds.empty()) {
441 const std::string spellId = pickSpell(engine, actor, spec.spellIds);
442 if (!spellId.empty()) {
443 const auto cast = engine.castSpell(spellId, actor, &defender, CheckParams{}, rng);
444 if (cast.cast) {
445 return defender.resource(hpResourceId) <= 0;
446 }
447 return false; // the casting attempt failed — the round is spent
448 }
449 }
450 return attackOnce(engine, actor, weaponDamage, defender, checkTypeId, hpResourceId, rng);
451}

461template <RandomNumberGenerator Rng>
462[[nodiscard]] int resolveRound(RulesetEngine &engine, const CombatantSpec &a,
463 const CombatantSpec &b, DynamicEntity &ea, DynamicEntity &eb,
464 const DiceExpression &damageA, const DiceExpression &damageB,
465 std::string_view checkTypeId, std::string_view hpResourceId,
466 bool useMagic, Rng &rng) {
467 const int initA = ea.getStat("Initiative") + rng(1, 6);
468 const int initB = eb.getStat("Initiative") + rng(1, 6);
469 const bool aFirst = initA != initB ? initA > initB : rng(0, 1) == 0;
470
471 DynamicEntity &first = aFirst ? ea : eb;
472 DynamicEntity &second = aFirst ? eb : ea;
473 const CombatantSpec &firstSpec = aFirst ? a : b;
474 const CombatantSpec &secondSpec = aFirst ? b : a;
475 const DiceExpression &firstDamage = aFirst ? damageA : damageB;
476 const DiceExpression &secondDamage = aFirst ? damageB : damageA;
477
478 if (actOnce(engine, first, firstSpec, firstDamage, second, checkTypeId, hpResourceId, useMagic,
479 rng)) {
480 return aFirst ? 0 : 1;
481 }
482 if (actOnce(engine, second, secondSpec, secondDamage, first, checkTypeId, hpResourceId, useMagic,
483 rng)) {
484 return aFirst ? 1 : 0;
485 }
486 return -1;
487}

508template <RandomNumberGenerator Rng>
509[[nodiscard]] FightOutcome runFight(RulesetEngine &engine, const CombatantSpec &a,
510 const CombatantSpec &b, std::string_view checkTypeId,
511 std::string_view hpResourceId, int maxRounds, Rng &rng,
512 bool useMagic = true) {
513 FightOutcome outcome;
514 auto ea = createFighter(engine, a, rng);
515 auto eb = createFighter(engine, b, rng);
516 if (ea == nullptr || eb == nullptr) {
517 return outcome;
518 }
519 outcome.maxLp[0] = ea->resource(hpResourceId);
520 outcome.maxLp[1] = eb->resource(hpResourceId);
521 const DiceExpression damageA(a.damageExpression);
522 const DiceExpression damageB(b.damageExpression);
523
524 int roundsPlayed = 0;
525 for (; roundsPlayed < maxRounds; ++roundsPlayed) {
526 const int winner = detail::resolveRound(engine, a, b, *ea, *eb, damageA, damageB, checkTypeId,
527 hpResourceId, useMagic, rng);
528 if (winner >= 0) {
529 outcome.winnerIndex = winner;
530 break;
531 }
532 }
533
534 outcome.rounds = outcome.winnerIndex == -1 ? maxRounds : roundsPlayed + 1;
535 outcome.remainingLp[0] = ea->resource(hpResourceId);
536 outcome.remainingLp[1] = eb->resource(hpResourceId);
537 return outcome;
538}
