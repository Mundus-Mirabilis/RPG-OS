141[[nodiscard]] inline std::string resolveAttackCheckType(const Ruleset &ruleset) {
142 for (const CheckTypeDef &def : ruleset.checkTypes) {
143 if (def.id == "tde_attack") {
144 return def.id;
145 }
146 }
147 for (const CheckTypeDef &def : ruleset.checkTypes) {
148 if (def.recipe.resolution == Resolution::Opposed) {
149 return def.id;
150 }
151 }
152 // D&D-style attack-vs-AC: a threshold check rolled against a *target* stat
153 // (e.g. AC) whose bonus comes from the combatant's promoted `Attack` stat
154 // (the bestiary `to_hit`). Systems without an opposed parry roll (D&D)
155 // use this instead of the opposed resolution above.
156 for (const CheckTypeDef &def : ruleset.checkTypes) {
157 if (def.recipe.resolution == Resolution::Threshold &&
158 def.recipe.thresholdSource == ThresholdSource::TargetStat &&
159 std::ranges::contains(def.recipe.bonusStats, "Attack")) {
160 return def.id;
161 }
162 }
163 return {};
164}

170[[nodiscard]] inline bool fillArchetypeSpec(RulesetEngine &engine, const Json &entry,
171 std::string_view id, CombatantSpec &out,
172 std::string_view weaponDamage,
173 DefaultRandom &probeRng) {
174 auto probe = engine.createEntityWith(id, Variance::Average, probeRng);
175 if (probe == nullptr) {
176 return false;
177 }
178 out.id = std::string(id);
179 out.name = entry.value("name", out.id);
180 out.isArchetype = true;
181 out.attackValue = probe->getStat("Attack");
182 out.defenseValue = probe->getStat("Parry");
183 out.armorRating = probe->baseAttribute("Armor_Rating");
184 out.damageExpression = std::string(weaponDamage);
185 // An archetype's `spells_known` were loaded into the probe's spellbook;
186 // copy them so the fight loop can cast magic for this combatant.
187 out.spellIds = probe->spellbook().known();
188 return true;
189}

194[[nodiscard]] inline bool fillCreatureSpec(RulesetEngine &engine, const Json &entry,
195 std::string_view id, CombatantSpec &out,
196 std::string_view weaponDamage, DefaultRandom &probeRng) {
197 auto probe = engine.createCreatureWith(id, Variance::Average, probeRng);
198 if (probe == nullptr) {
199 return false;
200 }
201 out.id = std::string(id);
202 out.name = entry.value("name", out.id);
203 out.isArchetype = false;
204 out.defenseValue = entry.value("dodge", 0);
205 out.armorRating = entry.value("armor_rating", 0);
206 out.acValue = entry.value("ac", 0);
207 out.initiativeValue = entry.value("initiative", 0);
208 out.attackValue = 0;
209 out.damageExpression = std::string(weaponDamage);
210 if (entry.contains("attacks") && entry.at("attacks").is_array()) {
211 int32_t bestToHit = -1;
212 for (const Json &attack : entry.at("attacks")) {
213 const int32_t toHit = attack.value("to_hit", 0);
214 if (toHit > bestToHit) {
215 bestToHit = toHit;
216 out.attackValue = toHit;
217 out.damageExpression = attack.value("damage", "1d6");
218 }
219 }
220 }
221 if (out.attackValue <= 0) {
222 // No natural attack defined: fall back to the creature's derived
223 // attack value and an unarmed strike.
224 out.attackValue = probe->getStat("Attack");
225 out.damageExpression = "1d6";
226 }
227 // A bestiary entry may be a spellcaster too: the optional `spells` array
228 // lists the spell ids it can cast in a fight (e.g. a dragon's innate
229 // spellcasting). The fight loop then prefers casting just as it does for
230 // archetypes.
231 if (entry.contains("spells") && entry.at("spells").is_array()) {
232 for (const Json &spellId : entry.at("spells")) {
233 if (spellId.is_string()) {
234 out.spellIds.push_back(spellId.get<std::string>());
235 }
236 }
237 }
238 return true;
239}

258[[nodiscard]] inline bool makeCombatantSpec(RulesetEngine &engine, std::string_view id,
259 CombatantSpec &out,
260 std::string_view weaponDamage = kDefaultWeaponDamage) {
261 const Ruleset &ruleset = engine.ruleset();
262 if (!ruleset.data.is_object()) {
263 return false;
264 }
265 DefaultRandom probeRng;
266
267 if (ruleset.data.contains("archetypes")) {
268 for (const Json &entry : ruleset.data.at("archetypes")) {
269 if (entry.value("id", "") != id) {
270 continue;
271 }
272 return detail::fillArchetypeSpec(engine, entry, id, out, weaponDamage, probeRng);
273 }
274 }
275
276 if (ruleset.data.contains("creatures")) {
277 for (const Json &entry : ruleset.data.at("creatures")) {
278 if (entry.value("id", "") != id) {
279 continue;
280 }
281 return detail::fillCreatureSpec(engine, entry, id, out, weaponDamage, probeRng);
282 }
283 }
284 return false;
285}

299[[nodiscard]] inline bool
300makeCombatantSpecFromEntity(RulesetEngine &engine, const DynamicEntity &entity, CombatantSpec &out,
301 std::string_view weaponDamage = kDefaultWeaponDamage) {
302 out.id = entity.id();
303 out.name = entity.id();
304 // Resolve a human-readable name when the sheet happens to be a known
305 // ruleset entry (archetype or bestiary); otherwise the id is the name.
306 const Json *record = engine.findDataRecord("archetypes", entity.id());
307 if (record == nullptr) {
308 record = engine.findDataRecord("creatures", entity.id());
309 }
310 if (record != nullptr) {
311 out.name = record->value("name", out.name);
312 }
313 out.isArchetype = true; // the sheet fully describes the character
314 out.attackValue = entity.getStat("Attack");
315 out.defenseValue = entity.getStat("Parry");
316 out.armorRating = entity.getStat("Armor_Rating");
317 out.acValue = entity.getStat("AC");
318 out.initiativeValue = entity.getStat("Initiative");
319 out.damageExpression = std::string(weaponDamage);
320 out.spellIds = entity.spellbook().known();
321 entity.toJson(out.sheet);
322 return true;
323}

336template <RandomNumberGenerator Rng>
337[[nodiscard]] std::shared_ptr<DynamicEntity> createFighter(RulesetEngine &engine,
338 const CombatantSpec &spec, Rng &rng) {
339 // A spec carrying an embedded sheet fights that sheet directly: ANY
340 // character — not just a named archetype or bestiary entry — can take part
341 // in a fight this way (e.g. a character entered into a form). The sheet is
342 // restored into a fresh entity per fight and its resource pools are
343 // (re)created so the fighter starts at its full derived hit points.
344 if (spec.sheet.is_object()) {
345 auto entity = std::make_shared<DynamicEntity>(engine.ruleset(), spec.id);
346 entity->fromJson(spec.sheet);
347 entity->refreshResources();
348 return entity;
349 }
350 std::shared_ptr<DynamicEntity> entity =
351 spec.isArchetype ? engine.createEntityWith(spec.id, Variance::Average, rng)
352 : engine.createCreatureWith(spec.id, Variance::Average, rng);
353 if (entity != nullptr && !spec.isArchetype) {
354 entity->setBaseAttribute("Attack", spec.attackValue);
355 entity->setBaseAttribute("Parry", spec.defenseValue);
356 entity->setBaseAttribute("Armor_Rating", spec.armorRating);
357 // D&D-style systems promote the bestiary's real AC and initiative into
358 // base attributes: the derived `AC = 10 + DEX_mod` does not match a
359 // creature's stat block, and `Initiative` has no derived stat at all.
360 // Skipping a zero value keeps rulesets without those fields (TDE, BRP)
361 // on their derived stats.
362 if (spec.acValue != 0) {
363 entity->setBaseAttribute("AC", spec.acValue);
364 }
365 if (spec.initiativeValue != 0) {
366 entity->setBaseAttribute("Initiative", spec.initiativeValue);
367 }
368 }
369 return entity;
370}

383[[nodiscard]] inline double spellAverageDamage(const Json &spell) {
384 double total = 0.0;
385 const auto addDice = [&](const Json &value) {
386 if (value.is_number()) {
387 total += value.get<double>();
388 } else if (value.is_string()) {
389 total += DiceExpression(value.get<std::string>()).expectedValue();
390 }
391 };
392 if (spell.contains("effects") && spell.at("effects").is_array()) {
393 for (const Json &effect : spell.at("effects")) {
394 if (effect.value("kind", "") == "damage" && effect.contains("dice")) {
395 addDice(effect.at("dice"));
396 }
397 }
398 } else if (spell.contains("damage")) {
399 addDice(spell.at("damage"));
400 }
401 return total;
402}

413[[nodiscard]] inline std::string pickSpell(const RulesetEngine &engine, const DynamicEntity &caster,
414 const std::vector<std::string> &spellIds) {
415 const std::string &resourceId = engine.ruleset().spellResource;
416 std::string best;
417 double bestDamage = 0.0;
418 for (const std::string &spellId : spellIds) {
419 const Json *spell = engine.findSpell(spellId);
420 if (spell == nullptr) {
421 continue;
422 }
423 const int32_t cost = rpg_os::spellCost(*spell);
424 if (cost > 0 && !resourceId.empty() && caster.resource(resourceId) < cost) {
425 continue; // cannot afford
426 }
427 const double damage = spellAverageDamage(*spell);
428 if (damage > 0.0 && damage > bestDamage) {
429 bestDamage = damage;
430 best = spellId;
431 }
432 }
433 return best;
434}

445template <RandomNumberGenerator Rng>
446bool attackOnce(RulesetEngine &engine, DynamicEntity &attacker, const DiceExpression &damage,
447 DynamicEntity &defender, std::string_view checkTypeId,
448 std::string_view hpResourceId, Rng &rng, FightRoundLog *roundLog = nullptr,
449 int actorIndex = 0, int targetIndex = 1) {
450 // The attacker's restrictions are enforced: a creature that cannot take
451 // actions (incapacitated, stunned, paralyzed, ...) cannot attack.
452 if (!engine.actionAllowed(attacker, "action")) {
453 return false;
454 }
455 // Resolve against effective stats so worn gear and active conditions (armour,
456 // encumbrance, wounds) influence the fight — with nothing equipped the
457 // effective value equals the raw value, so ungeared fights are unchanged.
458 const CheckResult hit =
459 engine.executeCheckEffective(checkTypeId, attacker, &defender, CheckParams{}, rng);
460 FightActionLog action;
461 if (roundLog != nullptr) {
462 action.actorIndex = actorIndex;
463 action.targetIndex = targetIndex;
464 action.kind = "attack";
465 action.checkDice = hit.rawDiceRolls;
466 action.isHit = hit.isSuccess;
467 action.hpBefore = defender.resource(hpResourceId);
468 }
469 if (!hit.isSuccess) {
470 if (roundLog != nullptr) {
471 action.targetHp = defender.resource(hpResourceId);
472 roundLog->actions.push_back(std::move(action));
473 }
474 return false;
475 }
476 // Roll the damage explicitly so the individual dice can be logged. This is
477 // observation-only: roll() consumes exactly the RNG stream rollSum() would,
478 // so a logged fight and a plain fight stay byte-identical (pinned by test).
479 int32_t raw = 0;
480 if (roundLog != nullptr) {
481 const std::vector<int> rolled = damage.roll(rng);
482 action.damageDice.assign(rolled.begin(), rolled.end());
483 raw = damage.constant();
484 const auto &groups = damage.dice();
485 for (std::size_t g = 0, i = 0; g < groups.size(); ++g) {
486 for (int j = 0; j < groups[g].count; ++j, ++i) {
487 raw += groups[g].sign * rolled[i];
488 }
489 }
490 } else {
491 raw = static_cast<int32_t>(damage.rollSum(rng));
492 }
493 engine.applyDamage(attacker, defender, hpResourceId, raw);
494 if (roundLog != nullptr) {
495 action.damage = raw;
496 action.targetHp = defender.resource(hpResourceId);
497 roundLog->actions.push_back(std::move(action));
498 }
499 return defender.resource(hpResourceId) <= 0;
500}

513template <RandomNumberGenerator Rng>
514bool actOnce(RulesetEngine &engine, DynamicEntity &actor, const CombatantSpec &spec,
515 const DiceExpression &weaponDamage, DynamicEntity &defender,
516 std::string_view checkTypeId, std::string_view hpResourceId, bool useMagic, Rng &rng,
517 FightRoundLog *roundLog = nullptr, int actorIndex = 0, int targetIndex = 1) {
518 if (useMagic && !spec.spellIds.empty()) {
519 const std::string spellId = pickSpell(engine, actor, spec.spellIds);
520 if (!spellId.empty()) {
521 const int32_t hpBefore = defender.resource(hpResourceId);
522 // The spell resource is a pool that depletes during the fight (e.g. The
523 // Dark Eye's Astral Energy) — record it before the cast so the transcript
524 // can show it running down, and so the reader can see when the caster
525 // finally runs dry and falls back to its weapon.
526 const std::string &resourceId = engine.ruleset().spellResource;
527 const int32_t resourceBefore = resourceId.empty() ? 0 : actor.resource(resourceId);
528 const auto cast = engine.castSpell(spellId, actor, &defender, CheckParams{}, rng);
529 if (roundLog != nullptr) {
530 const Json *spell = engine.findSpell(spellId);
531 FightActionLog action;
532 action.actorIndex = actorIndex;
533 action.targetIndex = targetIndex;
534 action.kind = "cast";
535 action.spellId = spellId;
536 action.spellName = spell != nullptr ? spell->value("name", spellId) : spellId;
537 action.hpBefore = hpBefore;
538 action.checkDice = cast.check.rawDiceRolls;
539 action.isHit = cast.cast;
540 action.damageDice = cast.damageDice;
541 action.damage = cast.appliedDamage;
542 action.targetHp = defender.resource(hpResourceId);
543 action.resourceCost = cast.cost;
544 action.resourceId = cast.resourceId;
545 action.resourceBefore = resourceBefore;
546 action.resourceAfter = cast.resourceId.empty() ? 0 : actor.resource(cast.resourceId);
547 roundLog->actions.push_back(std::move(action));
548 }
549 if (cast.cast) {
550 return defender.resource(hpResourceId) <= 0;
551 }
552 return false; // the casting attempt failed — the round is spent
553 }
554 }
555 return attackOnce(engine, actor, weaponDamage, defender, checkTypeId, hpResourceId, rng, roundLog,
556 actorIndex, targetIndex);
557}

567template <RandomNumberGenerator Rng>
568[[nodiscard]] int resolveRound(RulesetEngine &engine, const CombatantSpec &a,
569 const CombatantSpec &b, DynamicEntity &ea, DynamicEntity &eb,
570 const DiceExpression &damageA, const DiceExpression &damageB,
571 std::string_view checkTypeId, std::string_view hpResourceId,
572 bool useMagic, Rng &rng, FightLog *log = nullptr,
573 int roundNumber = 0) {
574 const int initA = ea.getStat("Initiative") + rng(1, 6);
575 const int initB = eb.getStat("Initiative") + rng(1, 6);
576 const bool aFirst = initA != initB ? initA > initB : rng(0, 1) == 0;
577
578 // The round's transcript (built only when a log is requested — observation
579 // of the same dice, never an extra draw, so plain and logged fights match).
580 FightRoundLog roundLog;
581 if (log != nullptr) {
582 roundLog.round = roundNumber;
583 roundLog.initStat[0] = ea.getStat("Initiative");
584 roundLog.initStat[1] = eb.getStat("Initiative");
585 roundLog.initRoll[0] = initA - ea.getStat("Initiative");
586 roundLog.initRoll[1] = initB - eb.getStat("Initiative");
587 roundLog.initTotal[0] = initA;
588 roundLog.initTotal[1] = initB;
589 roundLog.goesFirst = aFirst ? 0 : 1;
590 }
591
592 DynamicEntity &first = aFirst ? ea : eb;
593 DynamicEntity &second = aFirst ? eb : ea;
594 const CombatantSpec &firstSpec = aFirst ? a : b;
595 const CombatantSpec &secondSpec = aFirst ? b : a;
596 const DiceExpression &firstDamage = aFirst ? damageA : damageB;
597 const DiceExpression &secondDamage = aFirst ? damageB : damageA;
598 const int firstIndex = aFirst ? 0 : 1;
599 const int secondIndex = aFirst ? 1 : 0;
600
601 if (actOnce(engine, first, firstSpec, firstDamage, second, checkTypeId, hpResourceId, useMagic,
602 rng, log != nullptr ? &roundLog : nullptr, firstIndex, secondIndex)) {
603 if (log != nullptr) {
604 log->rounds.push_back(std::move(roundLog));
605 }
606 return firstIndex;
607 }
608 if (actOnce(engine, second, secondSpec, secondDamage, first, checkTypeId, hpResourceId, useMagic,
609 rng, log != nullptr ? &roundLog : nullptr, secondIndex, firstIndex)) {
610 if (log != nullptr) {
611 log->rounds.push_back(std::move(roundLog));
612 }
613 return secondIndex;
614 }
615 if (log != nullptr) {
616 log->rounds.push_back(std::move(roundLog));
617 }
618 return -1;
619}

627template <RandomNumberGenerator Rng>
628[[nodiscard]] FightOutcome runFightImpl(RulesetEngine &engine, const CombatantSpec &a,
629 const CombatantSpec &b, std::string_view checkTypeId,
630 std::string_view hpResourceId, int maxRounds, Rng &rng,
631 bool useMagic, FightLog *log) {
632 FightOutcome outcome;
633 if (log != nullptr) {
634 log->names = {a.name, b.name};
635 // Each combatant's known spells, as human-readable names — so a fight
636 // transcript can show who is a spellcaster and what it can cast.
637 const auto spellNames = [&engine](const std::vector<std::string> &ids) {
638 std::vector<std::string> names;
639 names.reserve(ids.size());
640 for (const std::string &id : ids) {
641 const Json *spell = engine.findSpell(id);
642 names.push_back(spell != nullptr ? spell->value("name", id) : id);
643 }
644 return names;
645 };
646 log->spells = {spellNames(a.spellIds), spellNames(b.spellIds)};
647 }
648 auto ea = createFighter(engine, a, rng);
649 auto eb = createFighter(engine, b, rng);
650 if (ea == nullptr || eb == nullptr) {
651 return outcome;
652 }
653 outcome.maxLp[0] = ea->resource(hpResourceId);
654 outcome.maxLp[1] = eb->resource(hpResourceId);
655 if (log != nullptr) {
656 log->maxLp = outcome.maxLp;
657 // Each combatant's starting spell resource, so the transcript can show how
658 // much magic the casters began with and watch it run down during the fight.
659 const std::string &resourceId = engine.ruleset().spellResource;
660 log->resourceId = resourceId;
661 if (!resourceId.empty()) {
662 log->resourcePool = {ea->resource(resourceId), eb->resource(resourceId)};
663 }
664 }
665 const DiceExpression damageA(a.damageExpression);
666 const DiceExpression damageB(b.damageExpression);
667
668 int roundsPlayed = 0;
669 for (; roundsPlayed < maxRounds; ++roundsPlayed) {
670 const int winner = detail::resolveRound(engine, a, b, *ea, *eb, damageA, damageB, checkTypeId,
671 hpResourceId, useMagic, rng, log, roundsPlayed + 1);
672 if (winner >= 0) {
673 outcome.winnerIndex = winner;
674 break;
675 }
676 }
677
678 outcome.rounds = outcome.winnerIndex == -1 ? maxRounds : roundsPlayed + 1;
679 outcome.remainingLp[0] = ea->resource(hpResourceId);
680 outcome.remainingLp[1] = eb->resource(hpResourceId);
681 if (log != nullptr) {
682 log->winnerIndex = outcome.winnerIndex;
683 }
684 return outcome;
685}
