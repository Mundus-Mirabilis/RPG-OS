10TEST_CASE("bookkeeping: Money stores base units and converts (phase 0)") {

13 CHECK(twoGold.baseUnits() == 200);

15 CHECK(bag.baseUnits() == 253);

17 CHECK(bag.in(coins, "sp") == 25);

19 CHECK((bag - twoGold).baseUnits() == 53);

22 CHECK_FALSE(twoGold.baseUnits() > bag.baseUnits());

53TEST_CASE("bookkeeping: buy and pay transfer money and items (phase 1)") {
54 RulesetEngine tde;
55 REQUIRE(tde.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
56 auto geron = tde.createEntity("geron");
57 REQUIRE(geron != nullptr);
58 CHECK(geron->money().baseUnits() == 2500); // 25 ST starting wealth
59 geron->money() = Money{50000};
60
61 const auto total = tde.buy(*geron, nullptr, "dagger", 2);
62 REQUIRE(total.has_value());
63 CHECK(total->baseUnits() == 9000); // 2 * 45 ST
64 CHECK(geron->money().baseUnits() == 50000 - 9000);
65 CHECK(geron->inventory().count("dagger") == 2);
66
67 // insufficient funds fail without side effects
68 geron->money() = Money{10};
69 const auto poor = tde.buy(*geron, nullptr, "dagger", 1);
70 CHECK(poor.error() == BookkeepingError::NotEnoughMoney);
71 CHECK(geron->inventory().count("dagger") == 2);
72 CHECK(geron->money().baseUnits() == 10);
73
74 // pay transfers between two sheets
75 geron->money() = Money{1000};
76 auto other = tde.createEntity("geron");
77 REQUIRE(other != nullptr);
78 REQUIRE(tde.pay(*geron, other.get(), Money{400}).has_value());
79 CHECK(geron->money().baseUnits() == 600);
80 CHECK(other->money().baseUnits() == 2500 + 400);
81 CHECK(tde.pay(*geron, nullptr, Money{100000}).error() == BookkeepingError::NotEnoughMoney);
82}
