22TEST_CASE("bookkeeping: equip and unequip move items between slots (phase 2)") {
23 RulesetEngine tde;
24 REQUIRE(tde.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
25 auto sheet = tde.createEntity("geron");
26 REQUIRE(sheet != nullptr);
27 // geron starts with a longsword equipped
28 CHECK(sheet->equipment().isEquipped("weapon_hand"));
29 CHECK(sheet->equipment().itemIn("weapon_hand") == "longsword");
30
31 // cannot equip what you do not own
32 CHECK(tde.equip(*sheet, "weapon_hand", "dagger").error() == BookkeepingError::ItemNotOwned);
33
34 REQUIRE(tde.addItem(*sheet, "dagger", 1).has_value());
35 const auto equipped = tde.equip(*sheet, "weapon_hand", "dagger");
36 REQUIRE(equipped.has_value());
37 CHECK(equipped->previousItemId == "longsword"); // swap
38 CHECK(sheet->equipment().itemIn("weapon_hand") == "dagger");
39 CHECK(sheet->inventory().count("longsword") == 1); // old item returned
40 CHECK(sheet->inventory().count("dagger") == 0);
41
42 // unequip returns the item
43 REQUIRE(tde.unequip(*sheet, "weapon_hand").has_value());
44 CHECK_FALSE(sheet->equipment().isEquipped("weapon_hand"));
45 CHECK(sheet->inventory().count("dagger") == 1);
46
47 // slot / item mismatches fail
48 CHECK(tde.equip(*sheet, "weapon_hand", "leather_armor").error() ==
49 BookkeepingError::SlotMismatch);
50 CHECK(tde.equip(*sheet, "no_such_slot", "dagger").error() == BookkeepingError::SlotMismatch);
51 CHECK(tde.unequip(*sheet, "weapon_hand").error() == BookkeepingError::NotEquipped);
52}

70TEST_CASE("bookkeeping: effective armour reduces damage in the pipeline") {
71 RulesetEngine tde;
72 REQUIRE(tde.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
73 auto geron = tde.createEntity("geron");
74 REQUIRE(geron != nullptr);
75 rpg_os::DynamicEntity orc(tde.ruleset(), "orc");
76
77 // geron wears leather armour: raw Armor_Rating 2, effective 5.
78 CHECK(geron->baseAttribute("Armor_Rating") == 2);
79 CHECK(geron->getEffectiveStat("Armor_Rating") == 5);
80 const int32_t lpBefore = geron->resource("LP");
81 CHECK(tde.applyDamage(orc, *geron, "LP", 10) == -5); // 10 - effective 5
82
83 // without the armour only the raw rating applies
84 REQUIRE(tde.unequip(*geron, "body_armor").has_value());
85 const int32_t lpAfter = geron->resource("LP");
86 CHECK(tde.applyDamage(orc, *geron, "LP", 10) == -8); // 10 - raw 2
87 CHECK(geron->resource("LP") == lpBefore - 5 - 8);
88 CHECK(geron->resource("LP") == lpAfter - 8);
89}

91TEST_CASE("bookkeeping: an equipped item's formula modifier resolves against the wearer") {
92 // A gear modifier whose `value` is a *formula* (D&D medium armour caps the
93 // Dexterity contribution): AC = 10 + min(DEX_mod, 2) + 4.
94 const std::string rulesetJson = R"JSON({
95 "schema_version": 1, "ruleset_id": "mini", "licence": "test",
96 "attributes": [
97 {"id": "DEX", "name": "Dexterity", "min": 1, "max": 30, "default": 10}
98 ],
99 "derived_stats": [
100 {"id": "DEX_mod", "name": "Dexterity Modifier", "formula": "floor((DEX - 10) / 2)"},
101 {"id": "AC", "name": "Armor Class", "formula": "10 + DEX_mod"}
102 ],
103 "data": {
104 "items": [
105 {"id": "scale_mail", "name": "Scale Mail", "slot": "body_armor",
106 "modifiers": [{"target_stat": "AC", "type": "override",
107 "value": "10 + min(DEX_mod, 2) + 4"}]}
108 ]
109 }
110 })JSON";
111 RulesetEngine engine;
112 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
113
114 // DEX 18 -> DEX_mod +4, capped at +2 by the medium-armour formula: 16.
115 DynamicEntity hero(engine.ruleset(), "hero");
116 hero.setBaseAttribute("DEX", 18);
117 (void)hero.equipment().equip("body_armor", "scale_mail");
118 CHECK(hero.getEffectiveStat("AC") == 16);
119
120 // DEX 12 -> DEX_mod +1, no cap hit: 10 + 1 + 4 = 15.
121 DynamicEntity rogue(engine.ruleset(), "rogue");
122 rogue.setBaseAttribute("DEX", 12);
123 (void)rogue.equipment().equip("body_armor", "scale_mail");
124 CHECK(rogue.getEffectiveStat("AC") == 15);
125
126 // A plain numeric modifier keeps working alongside formula values.
127 const std::string numericJson = R"JSON({
128 "schema_version": 1, "ruleset_id": "mini2", "licence": "test",
129 "attributes": [
130 {"id": "DEX", "name": "Dexterity", "min": 1, "max": 30, "default": 10}
131 ],
132 "derived_stats": [
133 {"id": "DEX_mod", "name": "Dexterity Modifier", "formula": "floor((DEX - 10) / 2)"},
134 {"id": "AC", "name": "Armor Class", "formula": "10 + DEX_mod"}
135 ],
136 "data": {
137 "items": [
138 {"id": "shield", "name": "Shield", "slot": "shield",
139 "modifiers": [{"target_stat": "AC", "type": "add", "value": 2}]}
140 ]
141 }
142 })JSON";
143 RulesetEngine engine2;
144 REQUIRE(engine2.loadRulesetFromJson(numericJson));
145 DynamicEntity knight(engine2.ruleset(), "knight");
146 knight.setBaseAttribute("DEX", 14); // base AC 10 + 2 = 12
147 (void)knight.equipment().equip("shield", "shield");
148 CHECK(knight.getEffectiveStat("AC") == 14); // +2 from the shield
149}

170 skirmisher.setBaseAttribute("DEX", 18);
