81 void add(const ActiveEffect &effect) {
82 for (ActiveEffect &entry : m_effects) {
83 if (entry.conditionId == effect.conditionId && entry.source == effect.source) {
84 entry.stacks += effect.stacks;
85 if (effect.remaining > entry.remaining) {
86 entry.remaining = effect.remaining;
87 }
88 return;
89 }
90 }
91 m_effects.push_back(effect);
92 }

95 int32_t remove(std::string_view conditionId) {

98 if (it->conditionId == conditionId) {
99 it = m_effects.erase(it);
100 ++removed;
101 } else {
102 ++it;
103 }
104 }

226 int32_t tick() {
227 int32_t expired = 0;
228 for (auto it = m_effects.begin(); it != m_effects.end();) {
229 if (it->remaining > 0) {
230 --it->remaining;
231 if (it->remaining == 0) {
232 it = m_effects.erase(it);
233 ++expired;
234 continue;
235 }
236 }
237 ++it;
238 }
239 for (auto it = m_statBonuses.begin(); it != m_statBonuses.end();) {
240 if (it->remaining > 0) {
241 --it->remaining;
242 if (it->remaining == 0) {
243 it = m_statBonuses.erase(it);
244 ++expired;
245 continue;
246 }
247 }
248 ++it;
249 }
250 for (auto it = m_bonusDice.begin(); it != m_bonusDice.end();) {
251 if (it->remaining > 0) {
252 --it->remaining;
253 if (it->remaining == 0) {
254 it = m_bonusDice.erase(it);
255 ++expired;
256 continue;
257 }
258 }
259 ++it;
260 }
261 return expired;

268 int32_t expired = 0;
269 for (auto it = m_ongoing.begin(); it != m_ongoing.end();) {
270 if (it->phase == phase && it->remaining > 0) {
271 --it->remaining;

283 void toJson(Json &out) const {
284 out = Json::object();
285 Json conditions = Json::array();
286 for (const ActiveEffect &entry : m_effects) {
287 conditions.push_back({{"condition", entry.conditionId},
288 {"stacks", entry.stacks},
289 {"remaining", entry.remaining},
290 {"source", entry.source}});
291 }
292 out["conditions"] = conditions;
293 Json bonuses = Json::array();
294 for (const StatBonus &bonus : m_statBonuses) {
295 bonuses.push_back({{"stat", bonus.stat},
296 {"value", bonus.value},
297 {"remaining", bonus.remaining},
298 {"source", bonus.source}});
299 }
300 out["stat_bonuses"] = bonuses;
301 Json dice = Json::array();
302 for (const BonusDie &entry : m_bonusDice) {
303 dice.push_back({{"dice", entry.dice},
304 {"scope", entry.scope},
305 {"remaining", entry.remaining},
306 {"source", entry.source}});
307 }
308 out["bonus_dice"] = dice;
309 Json ongoing = Json::array();
310 for (const OngoingEffect &entry : m_ongoing) {
311 Json record = {{"effect", entry.effect},
312 {"phase", entry.phase},
313 {"remaining", entry.remaining},
314 {"source", entry.source}};
315 ongoing.push_back(record);
316 }
317 out["ongoing"] = ongoing;

320 void fromJson(const Json &in) {
321 m_effects.clear();
322 m_statBonuses.clear();
323 m_bonusDice.clear();
324 m_ongoing.clear();
325 if (!in.is_object()) {
326 return;
327 }
328 if (in.contains("conditions") && in.at("conditions").is_array()) {
329 for (const Json &entry : in.at("conditions")) {
330 ActiveEffect effect;
331 effect.conditionId = entry.value("condition", "");
332 effect.stacks = entry.value("stacks", 1);
333 effect.remaining = entry.value("remaining", 0);
334 effect.source = entry.value("source", "");
335 if (!effect.conditionId.empty()) {
336 m_effects.push_back(std::move(effect));
337 }
338 }
339 }
340 if (in.contains("stat_bonuses") && in.at("stat_bonuses").is_array()) {
341 for (const Json &entry : in.at("stat_bonuses")) {
342 StatBonus bonus;
343 bonus.stat = entry.value("stat", "");
344 bonus.value = entry.value("value", 0);
345 bonus.remaining = entry.value("remaining", 0);
346 bonus.source = entry.value("source", "");
347 if (!bonus.stat.empty()) {
348 m_statBonuses.push_back(std::move(bonus));
349 }
350 }
351 }
352 if (in.contains("bonus_dice") && in.at("bonus_dice").is_array()) {
353 for (const Json &entry : in.at("bonus_dice")) {
354 BonusDie die;
355 die.dice = entry.value("dice", "");
356 die.scope = entry.value("scope", "all");
357 die.remaining = entry.value("remaining", 0);
358 die.source = entry.value("source", "");
359 if (!die.dice.empty()) {
360 m_bonusDice.push_back(std::move(die));
361 }
362 }
363 }
364 if (in.contains("ongoing") && in.at("ongoing").is_array()) {
365 for (const Json &entry : in.at("ongoing")) {
366 OngoingEffect ongoing;
367 if (entry.contains("effect")) {
368 ongoing.effect = entry.at("effect");
369 }
370 ongoing.phase = entry.value("phase", "end_of_turn");
371 ongoing.remaining = entry.value("remaining", 0);
372 ongoing.source = entry.value("source", "");
373 if (ongoing.effect.is_object()) {
374 m_ongoing.push_back(std::move(ongoing));
375 }
376 }
377 }
