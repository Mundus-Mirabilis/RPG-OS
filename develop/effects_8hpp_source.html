66 void add(const ActiveEffect &effect) {
67 for (ActiveEffect &entry : m_effects) {
68 if (entry.conditionId == effect.conditionId && entry.source == effect.source) {
69 entry.stacks += effect.stacks;
70 if (effect.remaining > entry.remaining) {
71 entry.remaining = effect.remaining;
72 }
73 return;
74 }
75 }
76 m_effects.push_back(effect);
77 }

211 int32_t tick() {
212 int32_t expired = 0;
213 for (auto it = m_effects.begin(); it != m_effects.end();) {
214 if (it->remaining > 0) {
215 --it->remaining;
216 if (it->remaining == 0) {
217 it = m_effects.erase(it);
218 ++expired;
219 continue;
220 }
221 }
222 ++it;
223 }
224 for (auto it = m_statBonuses.begin(); it != m_statBonuses.end();) {
225 if (it->remaining > 0) {
226 --it->remaining;
227 if (it->remaining == 0) {
228 it = m_statBonuses.erase(it);
229 ++expired;
230 continue;
231 }
232 }
233 ++it;
234 }
235 for (auto it = m_bonusDice.begin(); it != m_bonusDice.end();) {
236 if (it->remaining > 0) {
237 --it->remaining;
238 if (it->remaining == 0) {
239 it = m_bonusDice.erase(it);
240 ++expired;
241 continue;
242 }
243 }
244 ++it;
245 }
246 return expired;

268 void toJson(Json &out) const {
269 out = Json::object();

272 conditions.push_back({{"condition", entry.conditionId},

280 bonuses.push_back({{"stat", bonus.stat},
281 {"value", bonus.value},
282 {"remaining", bonus.remaining},
283 {"source", bonus.source}});
284 }
285 out["stat_bonuses"] = bonuses;
286 Json dice = Json::array();
287 for (const BonusDie &entry : m_bonusDice) {
288 dice.push_back({{"dice", entry.dice},
289 {"scope", entry.scope},
290 {"remaining", entry.remaining},
291 {"source", entry.source}});
292 }
293 out["bonus_dice"] = dice;
294 Json ongoing = Json::array();
295 for (const OngoingEffect &entry : m_ongoing) {
296 Json record = {{"effect", entry.effect},
297 {"phase", entry.phase},
298 {"remaining", entry.remaining},

305 void fromJson(const Json &in) {
306 m_effects.clear();
307 m_statBonuses.clear();
308 m_bonusDice.clear();
309 m_ongoing.clear();
310 if (!in.is_object()) {
311 return;
312 }
313 if (in.contains("conditions") && in.at("conditions").is_array()) {
314 for (const Json &entry : in.at("conditions")) {
315 ActiveEffect effect;
316 effect.conditionId = entry.value("condition", "");
317 effect.stacks = entry.value("stacks", 1);
318 effect.remaining = entry.value("remaining", 0);
319 effect.source = entry.value("source", "");
320 if (!effect.conditionId.empty()) {
321 m_effects.push_back(std::move(effect));
322 }
323 }
324 }
325 if (in.contains("stat_bonuses") && in.at("stat_bonuses").is_array()) {
326 for (const Json &entry : in.at("stat_bonuses")) {
327 StatBonus bonus;
328 bonus.stat = entry.value("stat", "");
329 bonus.value = entry.value("value", 0);
330 bonus.remaining = entry.value("remaining", 0);
331 bonus.source = entry.value("source", "");
332 if (!bonus.stat.empty()) {
333 m_statBonuses.push_back(std::move(bonus));
334 }
335 }
336 }
337 if (in.contains("bonus_dice") && in.at("bonus_dice").is_array()) {
338 for (const Json &entry : in.at("bonus_dice")) {
339 BonusDie die;
340 die.dice = entry.value("dice", "");
341 die.scope = entry.value("scope", "all");
342 die.remaining = entry.value("remaining", 0);
343 die.source = entry.value("source", "");
344 if (!die.dice.empty()) {
345 m_bonusDice.push_back(std::move(die));
346 }
347 }
348 }
349 if (in.contains("ongoing") && in.at("ongoing").is_array()) {
350 for (const Json &entry : in.at("ongoing")) {
351 OngoingEffect ongoing;
352 if (entry.contains("effect")) {
353 ongoing.effect = entry.at("effect");
354 }
355 ongoing.phase = entry.value("phase", "end_of_turn");
356 ongoing.remaining = entry.value("remaining", 0);
357 ongoing.source = entry.value("source", "");
358 if (ongoing.effect.is_object()) {
359 m_ongoing.push_back(std::move(ongoing));
360 }
361 }
362 }
