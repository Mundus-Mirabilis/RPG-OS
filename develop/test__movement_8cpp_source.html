67TEST_CASE("movement: a ruleset without movement rules is unconstrained") {
68 const std::string rulesetJson = R"({
69 "schema_version": 1, "ruleset_id": "mini", "licence": "test",
70 "attributes": [{"id": "STR", "name": "Strength", "min": 1, "max": 30, "default": 10}],
71 "resource_pools": [{"id": "HP", "name": "Hit Points", "max_stat": "STR", "min": 0}]
72 })";
73 RulesetEngine engine;
74 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
75 const auto &ruleset = engine.ruleset();
76 DynamicEntity sheet(ruleset, "test");
77 sheet.setBaseAttribute("STR", 10);
78 sheet.refreshResources();
79
80 CHECK_FALSE(engine.hasMovement());
81 const auto status = engine.movementStatus(sheet);
82 REQUIRE(status.has_value());
83 REQUIRE(status->options.size() == 1);
84 CHECK(status->options[0].modeId == "walk");
85 CHECK(status->options[0].possible);
86 CHECK(status->options[0].speed == doctest::Approx(0.0)); // unconstrained
87 CHECK_FALSE(status->loadReducesSpeed);
88
89 // Any mode is possible, unconstrained, and free.
90 const auto swim = engine.movementSpeed(sheet, "swim");
91 REQUIRE(swim.has_value());
92 CHECK(swim->possible);
93 CHECK(swim->speed == doctest::Approx(0.0));
94 CHECK(swim->exhaustionPerUnit == doctest::Approx(0.0));
95 CHECK(*engine.movementCost(sheet, "swim", 100.0) == doctest::Approx(0.0));
96 CHECK(engine.canRegenerate(sheet));
97
98 // Moving does nothing but report an unconstrained outcome.
99 const auto outcome = engine.move(sheet, "walk", 10.0);
100 REQUIRE(outcome.has_value());
101 CHECK(outcome->possible);
102 CHECK(outcome->exhaustion == doctest::Approx(0.0));
103
104 // A long rest still restores fully.
105 (void)sheet.modifyResource("HP", -5);
106 engine.longRest(sheet);
107 CHECK(sheet.resource("HP") == 10);
108}

127 CHECK(walk->exhaustionPerUnit == doctest::Approx(0.0));
128 CHECK(walk->regeneration);
129

131 const auto swim = engine.movementSpeed(sheet, "swim", "land");
132 REQUIRE(swim.has_value());
133 CHECK_FALSE(swim->possible);
134 CHECK(swim->reason == "not possible in land");
135
136 // fly on land: 60.
137 const auto fly = engine.movementSpeed(sheet, "fly", "land");
138 REQUIRE(fly.has_value());
139 CHECK(fly->speed == doctest::Approx(60.0));
140
141 // walk in water needs breath_water, which the sheet lacks.
142 const auto wade = engine.movementSpeed(sheet, "walk", "water");
143 REQUIRE(wade.has_value());
144 CHECK_FALSE(wade->possible);
145 CHECK(wade->reason == "requires breath_water");
146
147 // swim in water: full 20.
148 const auto swimWater = engine.movementSpeed(sheet, "swim", "water");
149 REQUIRE(swimWater.has_value());
150 CHECK(swimWater->possible);
151 CHECK(swimWater->speed == doctest::Approx(20.0));
152
153 // unknown mode / terrain are typed errors
154 CHECK(engine.movementSpeed(sheet, "burrow", "land").error() == BookkeepingError::UnknownMode);
155 CHECK(engine.movementSpeed(sheet, "walk", "bog").error() == BookkeepingError::UnknownTerrain);
156
157 // movementStatus lists every mode for the terrain
158 const auto status = engine.movementStatus(sheet, "land");
159 REQUIRE(status.has_value());
160 CHECK(status->terrainId == "land");
161 CHECK(status->options.size() == 3);

164TEST_CASE("movement: an equipped item grants the terrain-required capability") {
165 RulesetEngine engine;
166 REQUIRE(engine.loadRulesetFromJson(movementRuleset()));
167 const auto &ruleset = engine.ruleset();
168 DynamicEntity sheet(ruleset, "test");
169 sheet.setBaseAttribute("STR", 10);
170 sheet.refreshResources();
171
172 REQUIRE(engine.addItem(sheet, "amulet", 1).has_value());
173
174 // On land the amulet does nothing special.
175 CHECK_FALSE(sheet.hasCapability("breath_water"));
176 const auto onLand = engine.movementSpeed(sheet, "walk", "water");
177 REQUIRE(onLand.has_value());
178 CHECK_FALSE(onLand->possible);
179
180 // Carrying (but not wearing) the amulet does not grant breath_water.
181 sheet.setTerrain("water");
182 CHECK_FALSE(sheet.hasCapability("breath_water"));
183 CHECK(sheet.itemTerrainStatus("amulet").grants.size() == 1);
184
185 // Once equipped, the amulet grants breath_water in water automatically.
186 REQUIRE(engine.equip(sheet, "neck", "amulet").has_value());
187 CHECK(sheet.hasCapability("breath_water"));
188
189 // ...which makes walking in water possible (at the terrain's half speed).
190 const auto wade = engine.movementSpeed(sheet, "walk", "water");
191 REQUIRE(wade.has_value());
192 CHECK(wade->possible);
193 CHECK(wade->speed == doctest::Approx(15.0)); // 30 x 0.5
194
195 // Leaving the water removes the grant.
196 sheet.setTerrain("land");
197 CHECK_FALSE(sheet.hasCapability("breath_water"));
198}

226TEST_CASE("movement: move drains the exhaustion pool and the load slows speed") {
227 RulesetEngine engine;
228 REQUIRE(engine.loadRulesetFromJson(movementRuleset()));
229 const auto &ruleset = engine.ruleset();
230 DynamicEntity sheet(ruleset, "test");
231 sheet.setBaseAttribute("STR", 10);
232 sheet.refreshResources();
233
234 CHECK(sheet.resource("STAM") == 10);
235
236 // Swim 3 units in water: cost 2/unit x 3 = 6 stamina.
237 const auto outcome = engine.move(sheet, "swim", 3.0, "water");
238 REQUIRE(outcome.has_value());
239 CHECK(outcome->possible);
240 CHECK(outcome->exhaustion == doctest::Approx(6.0));
241 CHECK(outcome->timeUnits == doctest::Approx(3.0 / 20.0));
242 CHECK(sheet.resource("STAM") == 4);
243
244 // An impossible move is refused before any resource is spent.
245 const auto wade = engine.move(sheet, "walk", 1.0, "water");
246 REQUIRE(wade.has_value());
247 CHECK_FALSE(wade->possible);
248 CHECK(sheet.resource("STAM") == 4);
249
250 // Over half capacity (10 lb carried / 150 capacity would be ratio 0.066,
251 // so load the sheet to > 150 lb): walk speed halves.
252 REQUIRE(engine.addItem(sheet, "amulet", 200).has_value()); // 200 lb
253 const auto loaded = engine.movementSpeed(sheet, "walk", "land");
254 REQUIRE(loaded.has_value());
255 CHECK(loaded->speed == doctest::Approx(15.0)); // 30 x 0.5 load factor
256 const auto status = engine.movementStatus(sheet, "land");
257 REQUIRE(status.has_value());
258 CHECK(status->loadReducesSpeed);
259}
