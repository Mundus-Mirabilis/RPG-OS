10TEST_CASE("bookkeeping: sheet serialization round-trips the whole state") {
11 RulesetEngine tde;
12 REQUIRE(tde.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
13 auto original = tde.createEntity("geron");
14 REQUIRE(original != nullptr);
15 REQUIRE(tde.addItem(*original, "dagger", 2).has_value());
16 tde.applyCondition(*original, "pain", 1, 3);
17 original->advancement().gainXp(1000);
18 original->spellbook().learn("balsam_salabunde");
19
20 rpg_os::Json saved;
21 original->toJson(saved);
22 DynamicEntity restored(tde.ruleset(), "geron");
23 restored.fromJson(saved);
24
25 CHECK(restored.getStat("COU") == 12);
26 CHECK(restored.money().baseUnits() == original->money().baseUnits());
27 CHECK(restored.inventory().count("dagger") == 2);
28 CHECK(restored.equipment().itemIn("weapon_hand") == "longsword");
29 CHECK(restored.hasCondition("pain"));
30 CHECK(restored.advancement().xp() == 1000);
31 CHECK(restored.spellbook().knows("balsam_salabunde"));
32 CHECK(restored.getEffectiveStat("Armor_Rating") == 5); // gear survives too
33}

59TEST_CASE("bookkeeping: advanceTime ticks durations and fires OnTimePassed") {
60 rpg_os::GameSession session;
61 REQUIRE(session.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
62 auto &geron = session.createCharacter("geron");
63
64 // A short-lived condition decays once per day.
65 session.applyCondition(geron, "pain", 1, 2);
66 CHECK(geron.conditionStacks("pain") == 1);
67
68 int timeEvents = 0;
69 int32_t lastDay = -1;
70 (void)session.onEvent(rpg_os::EventType::OnTimePassed, [&](const rpg_os::EventData &data) {
71 ++timeEvents;
72 lastDay = data.getInt("day");
73 });
74
75 const int32_t newDay = session.advanceTime(1);
76 CHECK(newDay == 1);
77 CHECK(session.world().day == 1);
78 CHECK(timeEvents == 1);
79 CHECK(lastDay == 1);
80
81 // Second day: the 2-tick pain condition expires and is stripped.
82 session.advanceTime(1);
83 CHECK(timeEvents == 2);
84 CHECK_FALSE(geron.hasCondition("pain"));
85 CHECK(geron.conditionStacks("pain") == 0);
86}

88TEST_CASE("bookkeeping: GameSession save/load round-trips living state") {
89 rpg_os::GameSession session;
90 REQUIRE(session.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
91 auto &geron = session.createCharacter("geron");
92 const rpg_os::EntityId geronId = geron.entityId();
93
94 // Put the character into a non-default living state: spent pool, a timed
95 // condition, and extra inventory.
96 CHECK(geron.modifyResource("LP", -5) == -5);
97 session.applyCondition(geron, "pain", 1, 3);
98 REQUIRE(session.engine().addItem(geron, "dagger", 2).has_value());
99 session.world().addToTreasury(Money{500});
100 const int32_t lpAfterDamage = geron.resource("LP");
101 const int32_t painStacks = geron.conditionStacks("pain");
102
103 const rpg_os::Json save = session.saveState();
104 CHECK(save.contains("sheets"));
105 CHECK(save.at("sheets").size() == 1);
106 CHECK(save.at("sheets")[0]["entity_id"] == geronId.toUint64());
107
108 // Loading over the same session replaces the sheet and restores living state
109 // under the original instance id.
110 CHECK(session.loadState(save) == 1);
111 CHECK(session.entityCount() == 1);
112 rpg_os::DynamicEntity *restored = session.findEntity(geronId);
113 REQUIRE(restored != nullptr);
114 CHECK(restored->resource("LP") == lpAfterDamage);
115 CHECK(restored->conditionStacks("pain") == painStacks);
116 CHECK(restored->inventory().count("dagger") == 2);
117 bool painHasDuration = false;
118 for (const auto &effect : restored->effects().effects()) {
119 if (effect.conditionId == "pain" && effect.remaining == 3) {
120 painHasDuration = true;
121 }
122 }
123 CHECK(painHasDuration);
124 CHECK(session.world().treasury.baseUnits() == 500);
125}

127TEST_CASE("bookkeeping: event triggers can grant items, money, and XP") {
128 const std::string rulesetJson = R"({
129 "schema_version": 1, "ruleset_id": "mini", "licence": "test",
130 "attributes": [{"id": "STR", "name": "Strength", "min": 1, "max": 30, "default": 10}],
131 "resource_pools": [{"id": "HP", "name": "Hit Points", "max_stat": "STR", "min": 0}],
132 "event_triggers": [
133 {"id": "loot", "trigger": "on_damage_taken", "actions": [
134 {"type": "gain_item", "item": "gold_coin", "amount": 1},
135 {"type": "gain_currency", "amount": 5},
136 {"type": "gain_xp", "amount": 10}
137 ]},
138 {"id": "fee", "trigger": "on_damage_calculated", "actions": [
139 {"type": "spend_currency", "amount": 2}
140 ]}
141 ],
142 "data": {
143 "items": [{"id": "gold_coin", "name": "Gold Coin"}]
144 }
145 })";
146 RulesetEngine engine;
147 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
148 const auto &ruleset = engine.ruleset();
149 DynamicEntity victim(ruleset, "v");
150 victim.setBaseAttribute("STR", 10);
151 victim.loadFromArchetype(rpg_os::Json::object()); // initialise the HP pool
152 DynamicEntity orc(ruleset, "orc");
153
154 const int32_t hpBefore = victim.resource("HP");
155 CHECK(engine.applyDamage(orc, victim, "HP", 10) == -10);
156 CHECK(victim.resource("HP") == hpBefore - 10);
157 CHECK(victim.inventory().count("gold_coin") == 1); // gained on hit
158 CHECK(victim.money().baseUnits() == 3); // +5 - 2 fee
159 CHECK(victim.advancement().xp() == 10); // gained on hit
160}
