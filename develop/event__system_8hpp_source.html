148class EventBus {
149public:
151 using Callback = std::function<void(const EventData &)>;
152
156 [[nodiscard]] uint64_t addListener(EventType type, Callback callback) {
157 const uint64_t id = m_nextId++;
158 m_listeners[type].push_back(Entry{id, std::move(callback)});
159 return id;
160 }
161
165 bool removeListener(uint64_t listenerId) {
166 for (auto &kv : m_listeners) {
167 auto &entries = kv.second;
168 for (auto it = entries.begin(); it != entries.end(); ++it) {
169 if (it->id == listenerId) {
170 entries.erase(it);
171 return true;
172 }
173 }
174 }
175 return false;
176 }
177
182 void dispatch(EventType type, const EventData &data) const {
183 const auto it = m_listeners.find(type);
184 if (it == m_listeners.end()) {
185 return;
186 }
187 for (const Entry &entry : it->second) {
188 entry.callback(data);
189 }
190 }
191
193 [[nodiscard]] std::size_t listenerCount(EventType type) const {
194 const auto it = m_listeners.find(type);
195 return it == m_listeners.end() ? 0 : it->second.size();
196 }
197
198private:
199 struct Entry {
200 uint64_t id;
201 Callback callback;
202 };
203 std::unordered_map<EventType, std::vector<Entry>> m_listeners;
204 uint64_t m_nextId{1};
205};
