124 [[nodiscard]] uint64_t addListener(EventType type, Callback callback) {
125 const uint64_t id = m_nextId++;
126 m_listeners[type].push_back(Entry{id, std::move(callback)});
127 return id;
128 }
129
133 bool removeListener(uint64_t listenerId) {
134 for (auto &kv : m_listeners) {
135 auto &entries = kv.second;
136 for (auto it = entries.begin(); it != entries.end(); ++it) {
137 if (it->id == listenerId) {
138 entries.erase(it);
139 return true;
140 }
141 }
142 }
143 return false;
144 }
145
150 void dispatch(EventType type, const EventData &data) const {
151 const auto it = m_listeners.find(type);
152 if (it == m_listeners.end()) {
153 return;
154 }
155 for (const Entry &entry : it->second) {
156 entry.callback(data);
157 }
158 }
159
161 [[nodiscard]] std::size_t listenerCount(EventType type) const {
162 const auto it = m_listeners.find(type);
163 return it == m_listeners.end() ? 0 : it->second.size();
164 }
165
166private:
167 struct Entry {
168 uint64_t id;
169 Callback callback;
170 };
171 std::unordered_map<EventType, std::vector<Entry>> m_listeners;
172 uint64_t m_nextId{1};
173};
