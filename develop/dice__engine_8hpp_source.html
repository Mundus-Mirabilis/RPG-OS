76[[nodiscard]] inline std::array<uint32_t, 8> entropyWords() noexcept {
77 std::array<uint32_t, 8> words{};
78 try {
79 std::random_device rd;
80 for (std::size_t i = 0; i < words.size(); ++i) {
81 words[i] = rd();
82 }
83 } catch (...) {
84 // No OS entropy source available: the other sources mixed in below still
85 // provide enough variety between processes / runs.
86 }
87 const std::uint64_t now = static_cast<std::uint64_t>(
88 std::chrono::high_resolution_clock::now().time_since_epoch().count());
89 const std::uintptr_t stack = reinterpret_cast<std::uintptr_t>(&now);
90 words[0] ^= static_cast<uint32_t>(now);
91 words[1] ^= static_cast<uint32_t>(now >> 32);
92 words[2] ^= static_cast<uint32_t>(stack);
93 words[3] ^= processId();
94 return words;
95}

108[[nodiscard]] inline uint32_t randomSeed() noexcept {
109 const std::array<uint32_t, 8> words = detail::entropyWords();
110 uint32_t result = 0;
111 for (const uint32_t word : words) {
112 result ^= word;
113 }
114 return result;
115}

216 template <RandomNumberGenerator Rng> [[nodiscard]] int rollSum(Rng &rng) const {
217 int sum = m_constant;
218 for (const DieSpec &die : m_dice) {
219 for (int i = 0; i < die.count; ++i) {

235 void parse(std::string_view expression) {
236 if (expression.empty()) {
237 throw std::invalid_argument("empty dice expression");
238 }
239 std::size_t i = 0;
240 while (i < expression.size()) {
241 if (expression[i] == ' ') {
242 ++i;
243 continue;
244 }
245 int sign = 1;
246 if (expression[i] == '+') {
247 ++i;
248 } else if (expression[i] == '-') {
249 sign = -1;
250 ++i;
251 }
252 int number = 0;
253 bool hasNumber = false;
254 while (i < expression.size() && expression[i] >= '0' && expression[i] <= '9') {
255 number = number * 10 + (expression[i] - '0');
256 hasNumber = true;
257 ++i;
258 }
259 if (i < expression.size() && (expression[i] == 'd' || expression[i] == 'D')) {
260 ++i; // consume 'd'
261 if (i >= expression.size()) {
262 throw std::invalid_argument("missing die sides in dice expression");
263 }
264 int sides = 0;
265 while (i < expression.size() && expression[i] >= '0' && expression[i] <= '9') {
266 sides = sides * 10 + (expression[i] - '0');
267 ++i;
268 }
269 if (sides <= 0) {
270 throw std::invalid_argument("die sides must be positive");
271 }
272 const int count = hasNumber ? number : 1; // "d20" == "1d20"
273 if (count <= 0) {
274 throw std::invalid_argument("die count must be positive");
275 }
276 m_dice.push_back(DieSpec{count, sides, sign});
277 } else {
278 if (!hasNumber) {
279 throw std::invalid_argument("unexpected character in dice expression");
280 }
281 m_constant += sign * number;
282 }
283 }
284 }
285
286 std::vector<DieSpec> m_dice;
287 int m_constant{0};
288};
