62[[nodiscard]] inline uint32_t randomSeed() noexcept {
63 try {
64 std::random_device rd;
65 return rd();
66 } catch (...) {
67 // No OS entropy device: the high-resolution clock still differs between
68 // runs, which is all an unseeded game run needs.
69 return static_cast<uint32_t>(
70 std::chrono::high_resolution_clock::now().time_since_epoch().count());
71 }
72}

276 constexpr void parse(std::string_view expression) {
277 if (expression.empty()) {
278 throw std::invalid_argument("empty dice expression");
279 }
280 std::size_t i = 0;
281 while (i < expression.size()) {
282 if (expression[i] == ' ') {
283 ++i;
284 continue;
285 }
286 int sign = 1;
287 if (expression[i] == '+') {
288 ++i;
289 } else if (expression[i] == '-') {
290 sign = -1;
291 ++i;
292 }
293 int number = 0;
294 bool hasNumber = false;
295 while (i < expression.size() && expression[i] >= '0' && expression[i] <= '9') {
296 number = number * 10 + (expression[i] - '0');
297 hasNumber = true;
298 ++i;
299 }
300 if (i < expression.size() && (expression[i] == 'd' || expression[i] == 'D')) {
301 ++i; // consume 'd'
302 if (i >= expression.size()) {
303 throw std::invalid_argument("missing die sides in dice expression");
304 }
305 int sides = 0;
306 while (i < expression.size() && expression[i] >= '0' && expression[i] <= '9') {
307 sides = sides * 10 + (expression[i] - '0');
308 ++i;
309 }
310 if (sides <= 0) {
311 throw std::invalid_argument("die sides must be positive");
312 }
313 const int count = hasNumber ? number : 1; // "d20" == "1d20"
314 if (count <= 0) {
315 throw std::invalid_argument("die count must be positive");
316 }
317 m_dice.push_back(DieSpec{count, sides, sign});
318 } else {
319 if (!hasNumber) {
320 throw std::invalid_argument("unexpected character in dice expression");
321 }
322 m_constant += sign * number;
323 }
324 }
325 }
326
327 std::vector<DieSpec> m_dice;
328 int m_constant{0};
329};
