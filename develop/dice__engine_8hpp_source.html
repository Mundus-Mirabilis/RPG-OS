62[[nodiscard]] inline uint32_t randomSeed() noexcept {
63 try {
64 std::random_device rd;
65 return rd();
66 } catch (...) {
67 // No OS entropy device: the high-resolution clock still differs between
68 // runs, which is all an unseeded game run needs.
69 return static_cast<uint32_t>(
70 std::chrono::high_resolution_clock::now().time_since_epoch().count());
71 }
72}

253 constexpr void parse(std::string_view expression) {
254 if (expression.empty()) {
255 throw std::invalid_argument("empty dice expression");
256 }
257 std::size_t i = 0;
258 while (i < expression.size()) {
259 if (expression[i] == ' ') {
260 ++i;
261 continue;
262 }
263 int sign = 1;
264 if (expression[i] == '+') {
265 ++i;
266 } else if (expression[i] == '-') {
267 sign = -1;
268 ++i;
269 }
270 int number = 0;
271 bool hasNumber = false;
272 while (i < expression.size() && expression[i] >= '0' && expression[i] <= '9') {
273 number = number * 10 + (expression[i] - '0');
274 hasNumber = true;
275 ++i;
276 }
277 if (i < expression.size() && (expression[i] == 'd' || expression[i] == 'D')) {
278 ++i; // consume 'd'
279 if (i >= expression.size()) {
280 throw std::invalid_argument("missing die sides in dice expression");
281 }
282 int sides = 0;
283 while (i < expression.size() && expression[i] >= '0' && expression[i] <= '9') {
284 sides = sides * 10 + (expression[i] - '0');
285 ++i;
286 }
287 if (sides <= 0) {
288 throw std::invalid_argument("die sides must be positive");
289 }
290 const int count = hasNumber ? number : 1; // "d20" == "1d20"
291 if (count <= 0) {
292 throw std::invalid_argument("die count must be positive");
293 }
294 m_dice.push_back(DieSpec{count, sides, sign});
295 } else {
296 if (!hasNumber) {
297 throw std::invalid_argument("unexpected character in dice expression");
298 }
299 m_constant += sign * number;
300 }
301 }
302 }
303
304 std::vector<DieSpec> m_dice;
305 int m_constant{0};
