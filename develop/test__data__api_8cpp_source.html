60TEST_CASE("data API: castSpell spends the ruleset's spell resource (TDE)") {
61 RulesetEngine engine;
62 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
63 auto geron = engine.createEntity("geron");
64 REQUIRE(geron != nullptr);
65 const int32_t aeBefore = geron->resource("AE");
66
67 // Balsam Salabunde costs 1 AE and is a SGC/SGC/INT spell check; rolls 10/10/10
68 // all pass (SGC 11, INT 10).
69 auto rng = script({10, 10, 10});
70 const auto result = engine.castSpell("balsam_salabunde", *geron, nullptr, CheckParams{}, rng);
71 CHECK(result.cast);
72 CHECK(result.cost == 1);
73 CHECK(result.resourceId == "AE");
74 CHECK(geron->resource("AE") == aeBefore - 1);
75 CHECK(result.check.isSuccess);
76 CHECK(result.check.rawDiceRolls == std::vector<int>{10, 10, 10});

113TEST_CASE("data API: applyAffliction rolls a save and applies conditions (TDE)") {
114 RulesetEngine engine;
115 REQUIRE(engine.loadRulesetFromFile(rulesetPath("tde5e_core.json")));
116 auto geron = engine.createEntity("geron");
117 REQUIRE(geron != nullptr);
118
119 // Toad Poison: Toughness save (geron Toughness = round(39/6) = 7). A roll of
120 // 5 succeeds -> the poison is resisted and nothing is applied.
121 auto rngResist = script({5});
122 const auto resisted =
123 engine.applyAffliction("poisons", "toad_poison", *geron, CheckParams{}, rngResist);
124 CHECK(resisted.saveRolled);
125 CHECK(resisted.resisted);
126 CHECK(resisted.effectsApplied == 0);
127 CHECK_FALSE(geron->hasCondition("confusion"));
128
129 // A failed save (roll 8 > 7) applies 1 level of Confusion.
130 auto rngHit = script({8});
131 const auto hit = engine.applyAffliction("poisons", "toad_poison", *geron, CheckParams{}, rngHit);
132 CHECK(hit.saveRolled);
133 CHECK_FALSE(hit.resisted);
134 CHECK(hit.effectsApplied == 1);
135 CHECK(geron->hasCondition("confusion"));
136 CHECK(geron->conditionStacks("confusion") == 1);
137
138 // Unknown afflictions fail fast.
139 CHECK_THROWS_AS(static_cast<void>(engine.applyAffliction("poisons", "no_such_poison", *geron,
140 CheckParams{}, rngHit)),
141 std::invalid_argument);
142}
