21// EvalContext backed by a simple path -> value map.
22struct MapEvalContext : rpg_os::EvalContext {
23 std::unordered_map<std::string, double> values;
24
25 bool resolve(std::string_view path, double &out) const override {
26 const auto it = values.find(std::string(path));
27 if (it == values.end()) {
28 return false;
29 }
30 out = it->second;
31 return true;
32 }
33};
