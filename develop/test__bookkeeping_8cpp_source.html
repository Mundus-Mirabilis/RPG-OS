59TEST_CASE("bookkeeping: Money stores base units and converts (phase 0)") {
60 const CurrencySystem coins = dndCoins();
61 const Money twoGold = Money::fromCoins(coins, {{"gp", 2}});
62 CHECK(twoGold.baseUnits() == 200);
63 const Money bag = Money::fromCoins(coins, {{"gp", 2}, {"sp", 5}, {"cp", 3}});
64 CHECK(bag.baseUnits() == 253);
65 CHECK(bag.in(coins, "gp") == 2);
66 CHECK(bag.in(coins, "sp") == 25);
67 CHECK((twoGold + bag).baseUnits() == 453);
68 CHECK((bag - twoGold).baseUnits() == 53);
69 CHECK((bag * 3).baseUnits() == 759);
70 CHECK(twoGold.baseUnits() < bag.baseUnits());
71 CHECK_FALSE(twoGold.baseUnits() > bag.baseUnits());
72 CHECK_THROWS_AS(static_cast<void>(Money::fromCoins(coins, {{"nope", 1}})), std::invalid_argument);
73 CHECK(coins.valueOf("gp", 3) == 300);
74 CHECK(coins.valueOf("nope", 1) == 0);
75}

497 "data": {
498 "items": [
