248 const rpg_os::Json effects = rpg_os::Json::parse(R"json([
249 {"kind": "damage", "dice": "4d4", "type": "Acid",

252 "effect": {"kind": "damage", "dice": "2d4", "type": "Acid"}}}

267 target2.setBaseAttribute("HitPoints_Max", 30);
268 target2.loadFromArchetype(rpg_os::Json::object());
269 auto rng3 = script({1, 1, 1, 1, 5}); // 4d4 rolled, attack 5 + 6 = 11 < 15 miss
270 const auto miss = dnd.resolveEffects(caster, target2, effects, CheckParams{}, rng3);

272 auto rng4 = script({});
273 dnd.processOngoing(target2, "end_of_turn", rng4);
274 CHECK(target2.resource("HP") == 30); // no recurring damage on a miss
