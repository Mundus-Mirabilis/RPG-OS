94struct MapEvalContext : rpg_os::EvalContext {
95 std::unordered_map<std::string, double> values;
96 bool resolve(std::string_view path, double &out) const override {
97 const auto it = values.find(std::string(path));
98 if (it == values.end()) {
99 return false;
100 }
101 out = it->second;
102 return true;
103 }
104};
