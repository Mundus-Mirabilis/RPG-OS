70TEST_CASE("AdditiveD20: hit when total meets the target's AC") {
71 MockStats fighter;
72 fighter.values = {{"STR_mod", 3}, {"proficiency_bonus", 2}};
73 MockStats orc;
74 orc.values = {{"AC", 15}};
75 const CheckConfig cfg = additiveConfig();
76 const CheckParams params;
77
78 auto r1 = script({10}); // 10 + 3 + 2 = 15 >= 15 -> hit, margin 0
79 const CheckResult hit = rpg_os::resolveCheck(fighter, orc, cfg, params, r1);
80 CHECK(hit.isSuccess);
81 CHECK(hit.marginOfSuccess == 0);
82 CHECK(hit.rawDiceRolls == std::vector<int>{10});
83
84 auto r2 = script({8}); // 8 + 3 + 2 = 13 < 15 -> miss, margin -2
85 const CheckResult miss = rpg_os::resolveCheck(fighter, orc, cfg, params, r2);
86 CHECK_FALSE(miss.isSuccess);
87 CHECK(miss.marginOfSuccess == -2);
88
89 auto r3 = script({11}); // 11 + 3 + 2 = 16 -> hit, margin 1
90 const CheckResult good = rpg_os::resolveCheck(fighter, orc, cfg, params, r3);
91 CHECK(good.isSuccess);
92 CHECK(good.marginOfSuccess == 1);
93}
