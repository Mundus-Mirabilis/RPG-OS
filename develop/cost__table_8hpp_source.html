111inline CostTable CostTable::fromEntries(Kind kind, double baseFactor, std::vector<Entry> entries) {
112 // A stable sort is not needed: duplicate keys are resolved by "last wins",
113 // which is unaffected by which of two equal-keyed entries comes first — the
114 // later-in-vector one simply ends up later in the sorted order too.
115 std::sort(entries.begin(), entries.end(),
116 [](const Entry &a, const Entry &b) { return a.key < b.key; });
117 CostTable table;
118 table.m_kind = kind;
119 table.m_baseFactor = baseFactor;
120 table.m_entries = std::move(entries);
121 return table;
122}

124inline int32_t CostTable::lookup(int32_t query) const noexcept {
125 if (m_entries.empty()) {
126 return 0;
127 }
128 // std::upper_bound finds the first entry with key > query; the answer is
129 // the entry just before it (or the first/last when the query falls outside
130 // the table, handled by the clamp below).
131 const auto it = std::upper_bound(m_entries.begin(), m_entries.end(), query,
132 [](int32_t q, const Entry &entry) { return q < entry.key; });
133 if (it == m_entries.begin()) {
134 return m_entries.front().value;
135 }
136 if (it == m_entries.end()) {
137 return m_entries.back().value;
138 }
139 return (it - 1)->value;
140}
