365inline CheckRecipe RulesetLoader::parseRecipe(const Json &item, std::string_view checkId) {
366 const auto requireField = [&](bool condition, const std::string &message) {
367 if (!condition) {
368 throw std::invalid_argument(error("check_type '" + std::string(checkId) + "' " + message));
369 }
370 };
371
372 CheckRecipe recipe;
373 const std::string resolution = item.value("resolution", "threshold");
374 if (resolution == "threshold") {
375 recipe.resolution = Resolution::Threshold;
376 } else if (resolution == "pool") {
377 recipe.resolution = Resolution::Pool;
378 } else if (resolution == "opposed") {
379 recipe.resolution = Resolution::Opposed;
380 } else if (resolution == "resistance") {
381 recipe.resolution = Resolution::Resistance;
382 } else {
383 requireField(false, "has unknown resolution '" + resolution + "'");
384 }
385 recipe.dice = DiceExpression(item.value("dice", "1d20"));
386
387 const std::string comparison = item.value("comparison", "ge");
388 if (comparison == "ge") {
389 recipe.comparison = Comparison::GreaterEqual;
390 } else if (comparison == "le") {
391 recipe.comparison = Comparison::LessEqual;
392 } else {
393 requireField(false, "has unknown comparison '" + comparison + "'");
394 }
395
396 const std::string source = item.value("threshold_source", "difficulty");
397 if (source == "difficulty") {
398 recipe.thresholdSource = ThresholdSource::Difficulty;
399 } else if (source == "actor_stat") {
400 recipe.thresholdSource = ThresholdSource::ActorStat;
401 } else if (source == "target_stat") {
402 recipe.thresholdSource = ThresholdSource::TargetStat;
403 } else {
404 requireField(false, "has unknown threshold_source '" + source + "'");
405 }
406 recipe.thresholdStat = item.value("threshold_stat", "");
407
408 if (item.contains("bonus_stats")) {
409 for (const Json &stat : item.at("bonus_stats")) {
410 recipe.bonusStats.push_back(stat.get<std::string>());
411 }
412 }
413 if (item.contains("pool_attributes")) {
414 recipe.numPoolAttributes = 0;
415 for (const Json &attr : item.at("pool_attributes")) {
416 requireField(recipe.numPoolAttributes < 3, "has more than 3 pool_attributes");
417 recipe.poolAttributes[recipe.numPoolAttributes++] = attr.get<std::string>();
418 }
419 }
420 recipe.poolStat = item.value("pool_stat", "");
421 recipe.attackStat = item.value("attack_stat", "");
422 recipe.parryStat = item.value("parry_stat", "");
423 recipe.compareLevels = item.value("compare_levels", false);
424
425 const std::string critical = item.value("critical_style", "none");
426 if (critical == "none") {
427 recipe.criticalStyle = CriticalStyle::None;
428 } else if (critical == "face") {
429 recipe.criticalStyle = CriticalStyle::Face;
430 } else if (critical == "double") {
431 recipe.criticalStyle = CriticalStyle::DoubleRoll;
432 } else if (critical == "percentile") {
433 recipe.criticalStyle = CriticalStyle::PercentileBand;
434 } else {
435 requireField(false, "has unknown critical_style '" + critical + "'");
436 }
437 recipe.criticalFace = item.value("critical_face", 0);
438 recipe.criticalConfirm = item.value("critical_confirm", false);
439
440 const std::string fumble = item.value("fumble_style", "none");
441 if (fumble == "none") {
442 recipe.fumbleStyle = CriticalStyle::None;
443 } else if (fumble == "face") {
444 recipe.fumbleStyle = CriticalStyle::Face;
445 } else if (fumble == "double") {
446 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
447 } else if (fumble == "percentile") {
448 recipe.fumbleStyle = CriticalStyle::PercentileBand;
449 } else {
450 requireField(false, "has unknown fumble_style '" + fumble + "'");
451 }
452 recipe.fumbleFace = item.value("fumble_face", 0);
453 recipe.fumbleConfirm = item.value("fumble_confirm", false);
454
455 const std::string grading = item.value("grading", "none");
456 if (grading == "none") {
457 recipe.grading = Grading::None;
458 } else if (grading == "percentile") {
459 recipe.grading = Grading::Percentile;
460 } else if (grading == "pool_quality") {
461 recipe.grading = Grading::PoolQuality;
462 } else {
463 requireField(false, "has unknown grading '" + grading + "'");
464 }
465
466 const std::string difficulty = item.value("difficulty_mode", "to_threshold");
467 if (difficulty == "to_threshold") {
468 recipe.difficultyMode = DifficultyMode::ToThreshold;
469 } else if (difficulty == "to_stat") {
470 recipe.difficultyMode = DifficultyMode::ToStat;
471 } else {
472 requireField(false, "has unknown difficulty_mode '" + difficulty + "'");
473 }
474
475 const std::string multiplier = item.value("difficulty_multiplier", "none");
476 if (multiplier == "none") {
477 recipe.difficultyMultiplier = DifficultyMultiplier::None;
478 } else if (multiplier == "double_halve") {
479 recipe.difficultyMultiplier = DifficultyMultiplier::DoubleHalve;
480 } else {
481 requireField(false, "has unknown difficulty_multiplier '" + multiplier + "'");
482 }
483 return recipe;
484}

486inline EventType RulesetLoader::parseTrigger(std::string_view trigger) {
487 if (trigger == "on_before_check_roll") {
488 return EventType::OnBeforeCheckRoll;
489 }
490 if (trigger == "on_after_check_roll") {
491 return EventType::OnAfterCheckRoll;
492 }
493 if (trigger == "on_damage_calculated") {
494 return EventType::OnDamageCalculated;
495 }
496 if (trigger == "on_damage_taken") {
497 return EventType::OnDamageTaken;
498 }
499 if (trigger == "on_turn_start") {
500 return EventType::OnTurnStart;
501 }
502 if (trigger == "on_turn_end") {
503 return EventType::OnTurnEnd;
504 }
505 throw std::invalid_argument(error("unknown event trigger '" + std::string(trigger) + "'"));
506}

508inline void RulesetLoader::parseAttributes(const Json &obj, Ruleset &out) {
509 if (!obj.contains("attributes")) {
510 return;
511 }
512 for (const Json &item : obj.at("attributes")) {
513 require(item.contains("id"), "attribute entry missing 'id'");
514 require(item.contains("name"),
515 "attribute '" + item.at("id").get<std::string>() + "' missing 'name'");
516 AttributeDef def;
517 def.id = item.at("id").get<std::string>();
518 def.name = item.at("name").get<std::string>();
519 def.minValue = item.value("min", 1);
520 def.maxValue = item.value("max", 30);
521 def.defaultValue = item.value("default", 10);
522 out.attributes.push_back(std::move(def));
523 }

591inline void RulesetLoader::parseCostTables(const Json &obj, Ruleset &out) {
592 if (!obj.contains("cost_tables")) {
593 return;
594 }
595 for (const auto &[key, item] : obj.at("cost_tables").items()) {
596 require(item.contains("type"), "cost_table '" + key + "' missing 'type'");
597 const std::string type = item.at("type").get<std::string>();
598 require(type == "threshold" || type == "multiplier",
599 "cost_table '" + key + "' has unknown type '" + type + "'");
600 CostTableDef def;
601 def.id = key;
602 const CostTable::Kind kind =
603 type == "threshold" ? CostTable::Kind::Threshold : CostTable::Kind::Multiplier;
604 const double baseFactor = item.value("base_factor", 1.0);
605 std::vector<CostTable::Entry> entries;
606 if (item.contains("thresholds")) {
607 for (const Json &entry : item.at("thresholds")) {
608 require(entry.contains("key") && entry.contains("value"),
609 "cost_table '" + key + "' entry missing 'key' or 'value'");
610 entries.push_back(
611 CostTable::Entry{entry.at("key").get<int32_t>(), entry.at("value").get<int32_t>()});
612 }
613 } else if (item.contains("values")) {
614 for (const auto &[idx, value] : item.at("values").items()) {
615 const int32_t keyInt = static_cast<int32_t>(std::stoi(idx));
616 entries.push_back(CostTable::Entry{keyInt, value.get<int32_t>()});
617 }
618 }
619 def.table = CostTable::fromEntries(kind, baseFactor, std::move(entries));
620 out.costTables.push_back(std::move(def));
621 }
622}

634inline void RulesetLoader::parseEventTriggers(const Json &obj, Ruleset &out) {
635 if (!obj.contains("event_triggers")) {
636 return;
637 }
638 for (const Json &item : obj.at("event_triggers")) {
639 require(item.contains("id"), "event_trigger entry missing 'id'");
640 require(item.contains("trigger"),
641 "event_trigger '" + item.at("id").get<std::string>() + "' missing 'trigger'");
642 EventTriggerDef def;
643 def.id = item.at("id").get<std::string>();
644 def.trigger = parseTrigger(item.at("trigger").get<std::string>());
645 def.conditionText = item.value("condition", "");
646 if (!def.conditionText.empty()) {
647 def.condition = Expression(def.conditionText);
648 }
649 for (const Json &action : item.value("actions", Json::array())) {
650 require(action.contains("type"), "event action missing 'type'");
651 EventActionDef act;
652 act.type = action.at("type").get<std::string>();
653 act.resource = action.value("resource", "");
654 act.amount = action.value("amount", 0);
655 act.condition = action.value("condition", "");
656 act.formulaText = action.value("formula", "");
657 if (!act.formulaText.empty()) {
658 act.formula = Expression(act.formulaText);
659 }
660 act.stacksText = action.value("stacks", "");
661 if (!act.stacksText.empty()) {
662 act.stacks = Expression(act.stacksText);
663 }
664 def.actions.push_back(std::move(act));
665 }
666 out.eventTriggers.push_back(std::move(def));
667 }

670inline Ruleset RulesetLoader::load(const Json &root) {
671 require(root.is_object(), "ruleset root must be a JSON object");
672 require(root.contains("schema_version"), "ruleset missing 'schema_version'");
673 require(root.contains("ruleset_id"), "ruleset missing 'ruleset_id'");
674 require(root.contains("licence"), "ruleset missing required 'licence'");
675 require(root.at("licence").is_string() && !root.at("licence").get<std::string>().empty(),
676 "ruleset 'licence' must be a non-empty string");
677
678 Ruleset out;
679 out.schemaVersion = root.at("schema_version").get<int32_t>();
680 out.id = root.at("ruleset_id").get<std::string>();
681 out.name = root.value("ruleset_name", out.id);
682 out.source = root.value("source", "");
683 out.licence = root.value("licence", "");
684 out.comment = root.value("comment", "");
685 out.ns = root.value("namespace", "rpg_os::generated::" + out.id);
686 out.spellResource = root.value("spell_resource", "");
687
688 parseAttributes(root, out);
689 parseDerivedStats(root, out);
690 parseResources(root, out);
691 parseSkills(root, out);
692 parseCheckTypes(root, out);
693 parseCostTables(root, out);
694 parseEquipmentSlots(root, out);
695 parseEventTriggers(root, out);
696 if (root.contains("data")) {
697 out.data = root.at("data");
698 }
699
700 validate(out);
701 return out;
702}

709inline void RulesetLoader::validate(const Ruleset &ruleset) {
710 std::unordered_set<std::string> seen;
711 // Attribute ids must be unique.
712 for (const AttributeDef &def : ruleset.attributes) {
713 require(seen.insert(def.id).second, "duplicate attribute id '" + def.id + "'");
714 }
715 for (const DerivedStatDef &def : ruleset.derivedStats) {
716 require(seen.insert(def.id).second, "duplicate derived stat id '" + def.id + "'");
717 }
718 for (const SkillDef &def : ruleset.skills) {
719 require(seen.insert(def.id).second, "duplicate skill id '" + def.id + "'");
720 }
721 std::unordered_set<std::string> checkIds;
722 for (const CheckTypeDef &def : ruleset.checkTypes) {
723 require(checkIds.insert(def.id).second, "duplicate check type id '" + def.id + "'");
724 }
725 std::unordered_set<std::string> tableIds;
726 for (const CostTableDef &def : ruleset.costTables) {
727 require(tableIds.insert(def.id).second, "duplicate cost table id '" + def.id + "'");
728 }
729
730 // Derived-stat formulas: bare identifiers must be known stats.
731 for (const DerivedStatDef &def : ruleset.derivedStats) {
732 for (const std::string &ident : def.expression.identifiers()) {
733 if (ident.find('.') != std::string::npos) {
734 continue; // dotted paths (actor.*, target.*, env.*, ...) are allowed
735 }
736 require(ruleset.hasStat(ident),
737 "derived stat '" + def.id + "' references unknown stat '" + ident + "'");
738 }
739 }
740
741 // Derived-stat graph must be acyclic.
742 {
743 std::unordered_map<std::string, std::vector<std::string>> graph;
744 for (const DerivedStatDef &def : ruleset.derivedStats) {
745 for (const std::string &ident : def.expression.identifiers()) {
746 if (ident.find('.') == std::string::npos && ruleset.findDerivedStat(ident) != nullptr) {
747 graph[def.id].push_back(ident);
748 }
749 }
750 }
751 enum class State : uint8_t { Unvisited, Visiting, Done };
752 std::unordered_map<std::string, State> state;
753 std::function<void(const std::string &)> visit = [&](const std::string &nodeId) {
754 state[nodeId] = State::Visiting;
755 for (const std::string &dep : graph[nodeId]) {
756 if (state[dep] == State::Visiting) {
757 throw std::invalid_argument(error("derived stat cycle involving '" + nodeId + "'"));
758 }
759 if (state[dep] == State::Unvisited) {
760 visit(dep);
761 }
762 }
763 state[nodeId] = State::Done;
764 };
765 for (const DerivedStatDef &def : ruleset.derivedStats) {
766 if (state[def.id] == State::Unvisited) {
767 visit(def.id);
768 }
769 }
770 }
771
772 // Resource max stats must exist.
773 for (const ResourcePoolDef &def : ruleset.resourcePools) {
774 require(ruleset.hasStat(def.maxStat),
775 "resource pool '" + def.id + "' references unknown max_stat '" + def.maxStat + "'");
776 }
777
778 // Skill attribute references must exist.
779 for (const SkillDef &def : ruleset.skills) {
780 for (const std::string &attr : def.attributes) {
781 require(ruleset.findAttribute(attr) != nullptr,
782 "skill '" + def.id + "' references unknown attribute '" + attr + "'");
783 }
784 }
785
786 // Check type recipe references must exist.
787 for (const CheckTypeDef &def : ruleset.checkTypes) {
788 const CheckRecipe &recipe = def.recipe;
789 for (const std::string &stat : recipe.bonusStats) {
790 require(ruleset.hasStat(stat),
791 "check type '" + def.id + "' references unknown bonus stat '" + stat + "'");
792 }
793 if (recipe.thresholdSource == ThresholdSource::TargetStat ||
794 recipe.thresholdSource == ThresholdSource::ActorStat) {
795 require(ruleset.hasStat(recipe.thresholdStat), "check type '" + def.id +
796 "' references unknown threshold stat '" +
797 recipe.thresholdStat + "'");
798 }
799 if (!recipe.poolStat.empty()) {
800 require(ruleset.hasStat(recipe.poolStat),
801 "check type '" + def.id + "' references unknown pool stat '" + recipe.poolStat + "'");
802 }
803 if (!recipe.attackStat.empty()) {
804 require(ruleset.hasStat(recipe.attackStat), "check type '" + def.id +
805 "' references unknown attack stat '" +
806 recipe.attackStat + "'");
807 }
808 if (!recipe.parryStat.empty()) {
809 require(ruleset.hasStat(recipe.parryStat), "check type '" + def.id +
810 "' references unknown parry stat '" +
811 recipe.parryStat + "'");
812 }
813 for (std::size_t i = 0; i < recipe.numPoolAttributes; ++i) {
814 require(ruleset.hasStat(recipe.poolAttributes[i]),
815 "check type '" + def.id + "' references unknown pool attribute '" +
816 recipe.poolAttributes[i] + "'");
817 }
818 // A pool check needs as many dice as attributes.
819 if (recipe.resolution == Resolution::Pool) {
820 require(recipe.dice.dieCount() == recipe.numPoolAttributes,
821 "check type '" + def.id + "' pool resolution dice count must match pool_attributes");
822 }
823 }
824
825 // Event triggers: action types must be known (conditions and formulas were
826 // already parsed by Expression at load time).
827 for (const EventTriggerDef &def : ruleset.eventTriggers) {
828 for (const EventActionDef &action : def.actions) {
829 require(action.type == "modify_event_damage" || action.type == "consume_resource" ||
830 action.type == "apply_condition",
831 "event action in '" + def.id + "' has unknown type '" + action.type + "'");
832 }
833 }
834}
