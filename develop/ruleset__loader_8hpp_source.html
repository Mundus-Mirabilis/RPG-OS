369[[nodiscard]] inline const Json *findDataRecord(const Ruleset &ruleset, std::string_view section,
370 std::string_view id) {
371 if (!ruleset.data.is_object()) {
372 return nullptr;
373 }
374 const auto it = ruleset.data.find(std::string(section));
375 if (it == ruleset.data.end() || !it->is_array()) {
376 return nullptr;
377 }
378 for (const Json &record : *it) {
379 if (record.value("id", "") == id) {
380 return &record;
381 }
382 }
383 return nullptr;

485inline CheckRecipe RulesetLoader::parseRecipe(const Json &item, std::string_view checkId) {
486 const auto requireField = [&](bool condition, const std::string &message) {
487 if (!condition) {
488 throw std::invalid_argument(error("check_type '" + std::string(checkId) + "' " + message));
489 }
490 };
491
492 CheckRecipe recipe;
493 const std::string resolution = item.value("resolution", "threshold");
494 if (resolution == "threshold") {
495 recipe.resolution = Resolution::Threshold;
496 } else if (resolution == "pool") {
497 recipe.resolution = Resolution::Pool;
498 } else if (resolution == "opposed") {
499 recipe.resolution = Resolution::Opposed;
500 } else if (resolution == "resistance") {
501 recipe.resolution = Resolution::Resistance;
502 } else {
503 requireField(false, "has unknown resolution '" + resolution + "'");
504 }
505 recipe.dice = DiceExpression(item.value("dice", "1d20"));
506
507 const std::string comparison = item.value("comparison", "ge");
508 if (comparison == "ge") {
509 recipe.comparison = Comparison::GreaterEqual;
510 } else if (comparison == "le") {
511 recipe.comparison = Comparison::LessEqual;
512 } else {
513 requireField(false, "has unknown comparison '" + comparison + "'");
514 }
515
516 const std::string source = item.value("threshold_source", "difficulty");
517 if (source == "difficulty") {
518 recipe.thresholdSource = ThresholdSource::Difficulty;
519 } else if (source == "actor_stat") {
520 recipe.thresholdSource = ThresholdSource::ActorStat;
521 } else if (source == "target_stat") {
522 recipe.thresholdSource = ThresholdSource::TargetStat;
523 } else {
524 requireField(false, "has unknown threshold_source '" + source + "'");
525 }
526 recipe.thresholdStat = item.value("threshold_stat", "");
527
528 if (item.contains("bonus_stats")) {
529 for (const Json &stat : item.at("bonus_stats")) {
530 recipe.bonusStats.push_back(stat.get<std::string>());
531 }
532 }
533 if (item.contains("pool_attributes")) {
534 recipe.numPoolAttributes = 0;
535 for (const Json &attr : item.at("pool_attributes")) {
536 requireField(recipe.numPoolAttributes < 3, "has more than 3 pool_attributes");
537 recipe.poolAttributes[recipe.numPoolAttributes++] = attr.get<std::string>();
538 }
539 }
540 recipe.poolStat = item.value("pool_stat", "");
541 recipe.attackStat = item.value("attack_stat", "");
542 recipe.parryStat = item.value("parry_stat", "");
543 recipe.compareLevels = item.value("compare_levels", false);
544
545 const std::string critical = item.value("critical_style", "none");
546 if (critical == "none") {
547 recipe.criticalStyle = CriticalStyle::None;
548 } else if (critical == "face") {
549 recipe.criticalStyle = CriticalStyle::Face;
550 } else if (critical == "double") {
551 recipe.criticalStyle = CriticalStyle::DoubleRoll;
552 } else if (critical == "percentile") {
553 recipe.criticalStyle = CriticalStyle::PercentileBand;
554 } else {
555 requireField(false, "has unknown critical_style '" + critical + "'");
556 }
557 recipe.criticalFace = item.value("critical_face", 0);
558 recipe.criticalConfirm = item.value("critical_confirm", false);
559
560 const std::string fumble = item.value("fumble_style", "none");
561 if (fumble == "none") {
562 recipe.fumbleStyle = CriticalStyle::None;
563 } else if (fumble == "face") {
564 recipe.fumbleStyle = CriticalStyle::Face;
565 } else if (fumble == "double") {
566 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
567 } else if (fumble == "percentile") {
568 recipe.fumbleStyle = CriticalStyle::PercentileBand;
569 } else {
570 requireField(false, "has unknown fumble_style '" + fumble + "'");
571 }
572 recipe.fumbleFace = item.value("fumble_face", 0);
573 recipe.fumbleConfirm = item.value("fumble_confirm", false);
574
575 const std::string grading = item.value("grading", "none");
576 if (grading == "none") {
577 recipe.grading = Grading::None;
578 } else if (grading == "percentile") {
579 recipe.grading = Grading::Percentile;
580 } else if (grading == "pool_quality") {
581 recipe.grading = Grading::PoolQuality;
582 } else {
583 requireField(false, "has unknown grading '" + grading + "'");
584 }
585
586 const std::string difficulty = item.value("difficulty_mode", "to_threshold");
587 if (difficulty == "to_threshold") {
588 recipe.difficultyMode = DifficultyMode::ToThreshold;
589 } else if (difficulty == "to_stat") {
590 recipe.difficultyMode = DifficultyMode::ToStat;
591 } else {
592 requireField(false, "has unknown difficulty_mode '" + difficulty + "'");
593 }
594
595 const std::string multiplier = item.value("difficulty_multiplier", "none");
596 if (multiplier == "none") {
597 recipe.difficultyMultiplier = DifficultyMultiplier::None;
598 } else if (multiplier == "double_halve") {
599 recipe.difficultyMultiplier = DifficultyMultiplier::DoubleHalve;
600 } else {
601 requireField(false, "has unknown difficulty_multiplier '" + multiplier + "'");
602 }
603 return recipe;
604}

606inline EventType RulesetLoader::parseTrigger(std::string_view trigger) {
607 if (trigger == "on_before_check_roll") {
608 return EventType::OnBeforeCheckRoll;
609 }
610 if (trigger == "on_after_check_roll") {
611 return EventType::OnAfterCheckRoll;
612 }
613 if (trigger == "on_damage_calculated") {
614 return EventType::OnDamageCalculated;
615 }
616 if (trigger == "on_damage_taken") {
617 return EventType::OnDamageTaken;
618 }
619 if (trigger == "on_turn_start") {
620 return EventType::OnTurnStart;
621 }
622 if (trigger == "on_turn_end") {
623 return EventType::OnTurnEnd;
624 }
625 throw std::invalid_argument(error("unknown event trigger '" + std::string(trigger) + "'"));
626}

628inline void RulesetLoader::parseAttributes(const Json &obj, Ruleset &out) {
629 if (!obj.contains("attributes")) {
630 return;
631 }
632 for (const Json &item : obj.at("attributes")) {
633 require(item.contains("id"), "attribute entry missing 'id'");
634 require(item.contains("name"),
635 "attribute '" + item.at("id").get<std::string>() + "' missing 'name'");
636 AttributeDef def;
637 def.id = item.at("id").get<std::string>();
638 def.name = item.at("name").get<std::string>();
639 def.minValue = item.value("min", 1);
640 def.maxValue = item.value("max", 30);
641 def.defaultValue = item.value("default", 10);
642 out.attributes.push_back(std::move(def));
643 }
644}

711inline void RulesetLoader::parseCostTables(const Json &obj, Ruleset &out) {
712 if (!obj.contains("cost_tables")) {
713 return;
714 }
715 for (const auto &[key, item] : obj.at("cost_tables").items()) {
716 require(item.contains("type"), "cost_table '" + key + "' missing 'type'");
717 const std::string type = item.at("type").get<std::string>();
718 require(type == "threshold" || type == "multiplier",
719 "cost_table '" + key + "' has unknown type '" + type + "'");
720 CostTableDef def;
721 def.id = key;
722 const CostTable::Kind kind =
723 type == "threshold" ? CostTable::Kind::Threshold : CostTable::Kind::Multiplier;
724 const double baseFactor = item.value("base_factor", 1.0);
725 std::vector<CostTable::Entry> entries;
726 if (item.contains("thresholds")) {
727 for (const Json &entry : item.at("thresholds")) {
728 require(entry.contains("key") && entry.contains("value"),
729 "cost_table '" + key + "' entry missing 'key' or 'value'");
730 entries.push_back(
731 CostTable::Entry{entry.at("key").get<int32_t>(), entry.at("value").get<int32_t>()});
732 }
733 } else if (item.contains("values")) {
734 for (const auto &[idx, value] : item.at("values").items()) {
735 const int32_t keyInt = static_cast<int32_t>(std::stoi(idx));
736 entries.push_back(CostTable::Entry{keyInt, value.get<int32_t>()});
737 }
738 }
739 def.table = CostTable::fromEntries(kind, baseFactor, std::move(entries));
740 out.costTables.push_back(std::move(def));
741 }
742}

754inline void RulesetLoader::parseEventTriggers(const Json &obj, Ruleset &out) {
755 if (!obj.contains("event_triggers")) {
756 return;
757 }
758 for (const Json &item : obj.at("event_triggers")) {
759 require(item.contains("id"), "event_trigger entry missing 'id'");
760 require(item.contains("trigger"),
761 "event_trigger '" + item.at("id").get<std::string>() + "' missing 'trigger'");
762 EventTriggerDef def;
763 def.id = item.at("id").get<std::string>();
764 def.trigger = parseTrigger(item.at("trigger").get<std::string>());
765 def.conditionText = item.value("condition", "");
766 if (!def.conditionText.empty()) {
767 def.condition = Expression(def.conditionText);
768 }
769 for (const Json &action : item.value("actions", Json::array())) {
770 require(action.contains("type"), "event action missing 'type'");
771 EventActionDef act;
772 act.type = action.at("type").get<std::string>();
773 act.resource = action.value("resource", "");
774 act.amount = action.value("amount", 0);
775 act.item = action.value("item", "");
776 act.condition = action.value("condition", "");
777 act.formulaText = action.value("formula", "");
778 if (!act.formulaText.empty()) {
779 act.formula = Expression(act.formulaText);
780 }
781 act.stacksText = action.value("stacks", "");
782 if (!act.stacksText.empty()) {
783 act.stacks = Expression(act.stacksText);
784 }
785 def.actions.push_back(std::move(act));
786 }
787 out.eventTriggers.push_back(std::move(def));
788 }

791inline void RulesetLoader::parseCurrencies(const Json &obj, Ruleset &out) {
792 if (!obj.contains("currencies")) {
793 return;
794 }
795 const Json &cur = obj.at("currencies");
796 require(cur.is_object(), "'currencies' must be an object");
797 out.currencySystem.id = cur.value("id", "coins");
798 out.currencySystem.name = cur.value("name", out.currencySystem.id);
799 out.currencySystem.baseUnit = cur.value("base_unit", "");
800 require(!out.currencySystem.baseUnit.empty(), "'currencies' missing 'base_unit'");
801 require(cur.contains("denominations") && cur.at("denominations").is_array(),
802 "'currencies' missing 'denominations' array");
803 for (const Json &denom : cur.at("denominations")) {
804 require(denom.contains("id"), "currency denomination missing 'id'");
805 Denomination d;
806 d.id = denom.at("id").get<std::string>();
807 d.name = denom.value("name", d.id);
808 d.symbol = denom.value("symbol", d.id);
809 d.perBase = denom.value("per_base", 1);
810 out.currencySystem.denominations.push_back(std::move(d));
811 }
812 // The base unit must be one of the declared denominations.
813 require(out.currencySystem.find(out.currencySystem.baseUnit) != nullptr,
814 "currency base_unit '" + out.currencySystem.baseUnit + "' is not a denomination");
815}

840inline void RulesetLoader::parseSpellcasting(const Json &obj, Ruleset &out) {
841 if (!obj.contains("spellcasting")) {
842 return;
843 }
844 const Json &sc = obj.at("spellcasting");
845 require(sc.is_object(), "'spellcasting' must be an object");
846 const std::string style = sc.value("style", "pool");

901inline void RulesetLoader::validate(const Ruleset &ruleset) {
902 std::unordered_set<std::string> seen;
903 // Attribute ids must be unique.
904 for (const AttributeDef &def : ruleset.attributes) {
905 require(seen.insert(def.id).second, "duplicate attribute id '" + def.id + "'");
906 }
907 for (const DerivedStatDef &def : ruleset.derivedStats) {
908 require(seen.insert(def.id).second, "duplicate derived stat id '" + def.id + "'");
909 }
910 for (const SkillDef &def : ruleset.skills) {
911 require(seen.insert(def.id).second, "duplicate skill id '" + def.id + "'");
912 }
913 std::unordered_set<std::string> checkIds;
914 for (const CheckTypeDef &def : ruleset.checkTypes) {
915 require(checkIds.insert(def.id).second, "duplicate check type id '" + def.id + "'");
916 }
917 std::unordered_set<std::string> tableIds;
918 for (const CostTableDef &def : ruleset.costTables) {
919 require(tableIds.insert(def.id).second, "duplicate cost table id '" + def.id + "'");
920 }
921
922 // Derived-stat formulas: bare identifiers must be known stats.
923 for (const DerivedStatDef &def : ruleset.derivedStats) {
924 for (const std::string &ident : def.expression.identifiers()) {
925 if (ident.find('.') != std::string::npos) {
926 continue; // dotted paths (actor.*, target.*, env.*, ...) are allowed
927 }
928 require(ruleset.hasStat(ident),
929 "derived stat '" + def.id + "' references unknown stat '" + ident + "'");
930 }
931 }
932
933 // Derived-stat graph must be acyclic.
934 {
935 std::unordered_map<std::string, std::vector<std::string>> graph;
936 for (const DerivedStatDef &def : ruleset.derivedStats) {
937 for (const std::string &ident : def.expression.identifiers()) {
938 if (ident.find('.') == std::string::npos && ruleset.findDerivedStat(ident) != nullptr) {
939 graph[def.id].push_back(ident);
940 }
941 }
942 }
943 enum class State : uint8_t { Unvisited, Visiting, Done };
944 std::unordered_map<std::string, State> state;
945 std::function<void(const std::string &)> visit = [&](const std::string &nodeId) {
946 state[nodeId] = State::Visiting;
947 for (const std::string &dep : graph[nodeId]) {
948 if (state[dep] == State::Visiting) {
949 throw std::invalid_argument(error("derived stat cycle involving '" + nodeId + "'"));
950 }
951 if (state[dep] == State::Unvisited) {
952 visit(dep);
953 }
954 }
955 state[nodeId] = State::Done;
956 };
957 for (const DerivedStatDef &def : ruleset.derivedStats) {
958 if (state[def.id] == State::Unvisited) {
959 visit(def.id);
960 }
961 }
962 }
963
964 // Resource max stats must exist.
965 for (const ResourcePoolDef &def : ruleset.resourcePools) {
966 require(ruleset.hasStat(def.maxStat),
967 "resource pool '" + def.id + "' references unknown max_stat '" + def.maxStat + "'");
968 }
969
970 // Encumbrance capacity formula: bare identifiers must be known stats.
971 if (!ruleset.encumbrance.capacityText.empty()) {
972 for (const std::string &ident : ruleset.encumbrance.capacity.identifiers()) {
973 if (ident.find('.') == std::string::npos) {
974 require(ruleset.hasStat(ident),
975 "encumbrance capacity references unknown stat '" + ident + "'");
976 }
977 }
978 }
979
980 // Skill attribute references must exist.
981 for (const SkillDef &def : ruleset.skills) {
982 for (const std::string &attr : def.attributes) {
983 require(ruleset.findAttribute(attr) != nullptr,
984 "skill '" + def.id + "' references unknown attribute '" + attr + "'");
985 }
986 }
987
988 // Check type recipe references must exist. The combat simulator promotes a
989 // bestiary entry's attack/defence fields into the stats "Attack", "Parry",
990 // "Armor_Rating", "AC", and "Initiative" (see combat.hpp), so a check type
991 // may reference those names even when the ruleset has no such stat of its
992 // own — that is how a D&D-style ruleset declares its attack-vs-AC recipe.
993 const auto knownStat = [&ruleset](const std::string &stat) {
994 return ruleset.hasStat(stat) || stat == "Attack" || stat == "Parry" || stat == "Armor_Rating" ||
995 stat == "AC" || stat == "Initiative";
996 };
997 for (const CheckTypeDef &def : ruleset.checkTypes) {
998 const CheckRecipe &recipe = def.recipe;
999 for (const std::string &stat : recipe.bonusStats) {
1000 require(knownStat(stat),
1001 "check type '" + def.id + "' references unknown bonus stat '" + stat + "'");
1002 }
1003 if (recipe.thresholdSource == ThresholdSource::TargetStat ||
1004 recipe.thresholdSource == ThresholdSource::ActorStat) {
1005 require(knownStat(recipe.thresholdStat), "check type '" + def.id +
1006 "' references unknown threshold stat '" +
1007 recipe.thresholdStat + "'");
1008 }
1009 if (!recipe.poolStat.empty()) {
1010 require(knownStat(recipe.poolStat),
1011 "check type '" + def.id + "' references unknown pool stat '" + recipe.poolStat + "'");
1012 }
1013 if (!recipe.attackStat.empty()) {
1014 require(knownStat(recipe.attackStat), "check type '" + def.id +
1015 "' references unknown attack stat '" +
1016 recipe.attackStat + "'");
1017 }
1018 if (!recipe.parryStat.empty()) {
1019 require(knownStat(recipe.parryStat), "check type '" + def.id +
1020 "' references unknown parry stat '" +
1021 recipe.parryStat + "'");
1022 }
1023 for (std::size_t i = 0; i < recipe.numPoolAttributes; ++i) {
1024 require(knownStat(recipe.poolAttributes[i]), "check type '" + def.id +
1025 "' references unknown pool attribute '" +
1026 recipe.poolAttributes[i] + "'");
1027 }
1028 // A pool check needs as many dice as attributes.
1029 if (recipe.resolution == Resolution::Pool) {
1030 require(recipe.dice.dieCount() == recipe.numPoolAttributes,
1031 "check type '" + def.id + "' pool resolution dice count must match pool_attributes");
1032 }
1033 }
1034
1035 // Event triggers: action types must be known (conditions and formulas were
1036 // already parsed by Expression at load time).
1037 for (const EventTriggerDef &def : ruleset.eventTriggers) {
1038 for (const EventActionDef &action : def.actions) {
1039 require(action.type == "modify_event_damage" || action.type == "consume_resource" ||
1040 action.type == "apply_condition" || action.type == "gain_item" ||
1041 action.type == "remove_item" || action.type == "gain_currency" ||
1042 action.type == "spend_currency" || action.type == "gain_xp",
1043 "event action in '" + def.id + "' has unknown type '" + action.type + "'");
1044 }
1045 }
1046}
