445inline void RulesetLoader::parseCheckTypes(const Json &obj, Ruleset &out) {
446 if (!obj.contains("check_types")) {
447 return;
448 }
449 for (const auto &[key, item] : obj.at("check_types").items()) {
450 require(item.contains("kind"), "check_type '" + key + "' missing 'kind'");
451 CheckTypeDef def;
452 def.id = key;
453 def.config.kind = parseKind(item.at("kind").get<std::string>());
454 def.config.diceExpression = item.value("dice_expression", "1d20");
455 def.config.useConfirmationRoll = item.value("confirmation_roll", false);
456 def.config.targetStat = item.value("target_stat", "");
457 def.config.poolStat = item.value("pool_stat", "");
458 def.config.attackStat = item.value("attack_stat", "");
459 def.config.parryStat = item.value("parry_stat", "");
460 if (item.contains("bonus_stats")) {
461 for (const Json &stat : item.at("bonus_stats")) {
462 def.config.bonusStats.push_back(stat.get<std::string>());
463 }
464 }
465 if (item.contains("attributes")) {
466 def.config.numAttributes = 0;
467 for (const Json &attr : item.at("attributes")) {
468 require(def.config.numAttributes < 3,
469 "check_type '" + key + "' has more than 3 attributes");
470 def.config.attributes[def.config.numAttributes++] = attr.get<std::string>();
471 }
472 }
473 out.checkTypes.push_back(std::move(def));
474 }

477inline void RulesetLoader::parseCostTables(const Json &obj, Ruleset &out) {
478 if (!obj.contains("cost_tables")) {
479 return;
480 }
481 for (const auto &[key, item] : obj.at("cost_tables").items()) {
482 require(item.contains("type"), "cost_table '" + key + "' missing 'type'");
483 const std::string type = item.at("type").get<std::string>();
484 require(type == "threshold" || type == "multiplier",
485 "cost_table '" + key + "' has unknown type '" + type + "'");
486 CostTableDef def;
487 def.id = key;
488 const CostTable::Kind kind =
489 type == "threshold" ? CostTable::Kind::Threshold : CostTable::Kind::Multiplier;
490 const double baseFactor = item.value("base_factor", 1.0);
491 std::vector<CostTable::Entry> entries;
492 if (item.contains("thresholds")) {
493 for (const Json &entry : item.at("thresholds")) {
494 require(entry.contains("key") && entry.contains("value"),
495 "cost_table '" + key + "' entry missing 'key' or 'value'");
496 entries.push_back(
497 CostTable::Entry{entry.at("key").get<int32_t>(), entry.at("value").get<int32_t>()});
498 }
499 } else if (item.contains("values")) {
500 for (const auto &[idx, value] : item.at("values").items()) {
501 const int32_t keyInt = static_cast<int32_t>(std::stoi(idx));
502 entries.push_back(CostTable::Entry{keyInt, value.get<int32_t>()});
503 }
504 }
505 def.table = CostTable::fromEntries(kind, baseFactor, std::move(entries));
506 out.costTables.push_back(std::move(def));
507 }
508}

520inline void RulesetLoader::parseEventTriggers(const Json &obj, Ruleset &out) {
521 if (!obj.contains("event_triggers")) {
522 return;
523 }
524 for (const Json &item : obj.at("event_triggers")) {
525 require(item.contains("id"), "event_trigger entry missing 'id'");
526 require(item.contains("trigger"),
527 "event_trigger '" + item.at("id").get<std::string>() + "' missing 'trigger'");
528 EventTriggerDef def;
529 def.id = item.at("id").get<std::string>();
530 def.trigger = parseTrigger(item.at("trigger").get<std::string>());
531 def.conditionText = item.value("condition", "");
532 if (!def.conditionText.empty()) {
533 def.condition = Expression(def.conditionText);
534 }
535 for (const Json &action : item.value("actions", Json::array())) {
536 require(action.contains("type"), "event action missing 'type'");
537 EventActionDef act;
538 act.type = action.at("type").get<std::string>();
539 act.resource = action.value("resource", "");
540 act.amount = action.value("amount", 0);
541 act.condition = action.value("condition", "");
542 act.formulaText = action.value("formula", "");
543 if (!act.formulaText.empty()) {
544 act.formula = Expression(act.formulaText);
545 }
546 act.stacksText = action.value("stacks", "");
547 if (!act.stacksText.empty()) {
548 act.stacks = Expression(act.stacksText);
549 }
550 def.actions.push_back(std::move(act));
551 }
552 out.eventTriggers.push_back(std::move(def));
553 }

556inline Ruleset RulesetLoader::load(const Json &root) {
557 require(root.is_object(), "ruleset root must be a JSON object");
558 require(root.contains("schema_version"), "ruleset missing 'schema_version'");
559 require(root.contains("ruleset_id"), "ruleset missing 'ruleset_id'");
560 require(root.contains("licence"), "ruleset missing required 'licence'");
561 require(root.at("licence").is_string() && !root.at("licence").get<std::string>().empty(),
562 "ruleset 'licence' must be a non-empty string");
563
564 Ruleset out;
565 out.schemaVersion = root.at("schema_version").get<int32_t>();
566 out.id = root.at("ruleset_id").get<std::string>();
567 out.name = root.value("ruleset_name", out.id);
568 out.source = root.value("source", "");
569 out.licence = root.value("licence", "");
570 out.comment = root.value("comment", "");
571 out.ns = root.value("namespace", "rpg_os::generated::" + out.id);
572
573 parseAttributes(root, out);
574 parseDerivedStats(root, out);
575 parseResources(root, out);
576 parseSkills(root, out);
577 parseCheckTypes(root, out);
578 parseCostTables(root, out);
579 parseEquipmentSlots(root, out);
580 parseEventTriggers(root, out);
581 if (root.contains("data")) {
582 out.data = root.at("data");
583 }
584
585 validate(out);
586 return out;
587}

594inline void RulesetLoader::validate(const Ruleset &ruleset) {
595 std::unordered_set<std::string> seen;
596 // Attribute ids must be unique.
597 for (const AttributeDef &def : ruleset.attributes) {
598 require(seen.insert(def.id).second, "duplicate attribute id '" + def.id + "'");
599 }
600 for (const DerivedStatDef &def : ruleset.derivedStats) {
601 require(seen.insert(def.id).second, "duplicate derived stat id '" + def.id + "'");
602 }
603 for (const SkillDef &def : ruleset.skills) {
604 require(seen.insert(def.id).second, "duplicate skill id '" + def.id + "'");
605 }
606 std::unordered_set<std::string> checkIds;
607 for (const CheckTypeDef &def : ruleset.checkTypes) {
608 require(checkIds.insert(def.id).second, "duplicate check type id '" + def.id + "'");
609 }
610 std::unordered_set<std::string> tableIds;
611 for (const CostTableDef &def : ruleset.costTables) {
612 require(tableIds.insert(def.id).second, "duplicate cost table id '" + def.id + "'");
613 }
614
615 // Derived-stat formulas: bare identifiers must be known stats.
616 for (const DerivedStatDef &def : ruleset.derivedStats) {
617 for (const std::string &ident : def.expression.identifiers()) {
618 if (ident.find('.') != std::string::npos) {
619 continue; // dotted paths (actor.*, target.*, env.*, ...) are allowed
620 }
621 require(ruleset.hasStat(ident),
622 "derived stat '" + def.id + "' references unknown stat '" + ident + "'");
623 }
624 }
625
626 // Derived-stat graph must be acyclic.
627 {
628 std::unordered_map<std::string, std::vector<std::string>> graph;
629 for (const DerivedStatDef &def : ruleset.derivedStats) {
630 for (const std::string &ident : def.expression.identifiers()) {
631 if (ident.find('.') == std::string::npos && ruleset.findDerivedStat(ident) != nullptr) {
632 graph[def.id].push_back(ident);
633 }
634 }
635 }
636 enum class State : uint8_t { Unvisited, Visiting, Done };
637 std::unordered_map<std::string, State> state;
638 std::function<void(const std::string &)> visit = [&](const std::string &nodeId) {
639 state[nodeId] = State::Visiting;
640 for (const std::string &dep : graph[nodeId]) {
641 if (state[dep] == State::Visiting) {
642 throw std::invalid_argument(error("derived stat cycle involving '" + nodeId + "'"));
643 }
644 if (state[dep] == State::Unvisited) {
645 visit(dep);
646 }
647 }
648 state[nodeId] = State::Done;
649 };
650 for (const DerivedStatDef &def : ruleset.derivedStats) {
651 if (state[def.id] == State::Unvisited) {
652 visit(def.id);
653 }
654 }
655 }
656
657 // Resource max stats must exist.
658 for (const ResourcePoolDef &def : ruleset.resourcePools) {
659 require(ruleset.hasStat(def.maxStat),
660 "resource pool '" + def.id + "' references unknown max_stat '" + def.maxStat + "'");
661 }
662
663 // Skill attribute references must exist.
664 for (const SkillDef &def : ruleset.skills) {
665 for (const std::string &attr : def.attributes) {
666 require(ruleset.findAttribute(attr) != nullptr,
667 "skill '" + def.id + "' references unknown attribute '" + attr + "'");
668 }
669 }
670
671 // Check type references must exist.
672 for (const CheckTypeDef &def : ruleset.checkTypes) {
673 const CheckConfig &cfg = def.config;
674 for (const std::string &stat : cfg.bonusStats) {
675 require(ruleset.hasStat(stat),
676 "check type '" + def.id + "' references unknown bonus stat '" + stat + "'");
677 }
678 if (!cfg.targetStat.empty()) {
679 require(ruleset.hasStat(cfg.targetStat), "check type '" + def.id +
680 "' references unknown target stat '" +
681 cfg.targetStat + "'");
682 }
683 if (!cfg.poolStat.empty()) {
684 require(ruleset.hasStat(cfg.poolStat),
685 "check type '" + def.id + "' references unknown pool stat '" + cfg.poolStat + "'");
686 }
687 if (!cfg.attackStat.empty()) {
688 require(ruleset.hasStat(cfg.attackStat), "check type '" + def.id +
689 "' references unknown attack stat '" +
690 cfg.attackStat + "'");
691 }
692 if (!cfg.parryStat.empty()) {
693 require(ruleset.hasStat(cfg.parryStat),
694 "check type '" + def.id + "' references unknown parry stat '" + cfg.parryStat + "'");
695 }
696 for (std::size_t i = 0; i < cfg.numAttributes; ++i) {
697 require(ruleset.hasStat(cfg.attributes[i]), "check type '" + def.id +
698 "' references unknown attribute '" +
699 cfg.attributes[i] + "'");
700 }
701 }
702
703 // Event triggers: action types must be known (conditions and formulas were
704 // already parsed by Expression at load time).
705 for (const EventTriggerDef &def : ruleset.eventTriggers) {
706 for (const EventActionDef &action : def.actions) {
707 require(action.type == "modify_event_damage" || action.type == "consume_resource" ||
708 action.type == "apply_condition",
709 "event action in '" + def.id + "' has unknown type '" + action.type + "'");
710 }
711 }
