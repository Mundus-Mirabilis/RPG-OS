47TEST_CASE("bookkeeping: containers hold items (bag-in-bags)") {
48 const std::string rulesetJson = R"({
49 "schema_version": 1, "ruleset_id": "mini", "licence": "test",
50 "attributes": [{"id": "STR", "name": "Strength", "min": 1, "max": 30, "default": 10}],
51 "encumbrance": {"weight_unit": "lb", "default_weight": 0.0, "capacity": "15 * STR", "levels": []},
52 "data": {
53 "items": [
54 {"id": "backpack", "name": "Backpack", "weight": "5 lb."},
55 {"id": "club", "name": "Club", "weight": "2 lb."}
56 ]
57 }
58 })";
59 RulesetEngine engine;
60 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
61 const auto &ruleset = engine.ruleset();
62 DynamicEntity sheet(ruleset, "test");
63 sheet.setBaseAttribute("STR", 10);
64
65 REQUIRE(engine.addItem(sheet, "backpack", 1).has_value());
66 REQUIRE(engine.addItem(sheet, "club", 2).has_value());
67 // flat count does not see what is inside the container...
68 CHECK(sheet.inventory().count("club") == 2);
69 CHECK(sheet.inventory().totalCount("club") == 2);
70 CHECK(engine.carriedWeight(sheet).value == doctest::Approx(9.0)); // 5 + 2*2
71
72 // ...until we put a club into the backpack.
73 REQUIRE(engine.addItemToContainer(sheet, "backpack", "club", 1).has_value());
74 CHECK(sheet.inventory().countIn("backpack", "club") == 1);
75 CHECK(sheet.inventory().totalCount("club") == 3);
76 CHECK(engine.carriedWeight(sheet).value == doctest::Approx(11.0)); // +2 in the pack
77 REQUIRE(sheet.inventory().contentsOf("backpack") != nullptr);
78 CHECK(sheet.inventory().contentsOf("backpack")->size() == 1);
79
80 // a missing container is a distinct error
81 CHECK(engine.addItemToContainer(sheet, "nope", "club", 1).error() ==
82 BookkeepingError::UnknownContainer);
83
84 // removing from the container returns the count to flat-only
85 REQUIRE(engine.removeItemFromContainer(sheet, "backpack", "club", 1).has_value());
86 CHECK(sheet.inventory().countIn("backpack", "club") == 0);
87 CHECK(sheet.inventory().totalCount("club") == 2);
88 CHECK(engine.carriedWeight(sheet).value == doctest::Approx(9.0));
89
90 // serialization round-trips nested contents
91 REQUIRE(engine.addItemToContainer(sheet, "backpack", "club", 1).has_value());
92 rpg_os::Json saved;
93 sheet.toJson(saved);
94 DynamicEntity restored(ruleset, "test");
95 restored.fromJson(saved);
96 CHECK(restored.inventory().countIn("backpack", "club") == 1);
97 CHECK(restored.inventory().totalCount("club") == 3);
98}

143 CHECK_FALSE(status.overSize);

155 },
156 "data": {
157 "items": [
158 {"id": "heavy_box", "name": "Heavy Box", "weight": "20 lb.", "size": 3},
159 {"id": "light_box", "name": "Light Box", "weight": "1 lb.", "size": 1}
160 ]
161 }
162 })";
163 RulesetEngine engine;
164 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
165 const auto &ruleset = engine.ruleset();
166 DynamicEntity sheet(ruleset, "test");
167 sheet.setBaseAttribute("STR", 10);
168
169 // 7 heavy boxes: weight 140 lb <= 150, but size 21 > 10 -> size overflow.
170 auto ok = engine.canCarry(sheet, "heavy_box", 7);
171 REQUIRE(ok.has_value());
172 CHECK_FALSE(*ok);
173 // 3 heavy boxes fit on every axis.
174 ok = engine.canCarry(sheet, "heavy_box", 3);
175 CHECK(*ok);
176 // 5 light boxes = exactly the item limit (still allowed).
177 ok = engine.canCarry(sheet, "light_box", 5);
178 CHECK(*ok);
179 // 6 light boxes -> item-count overflow.
180 ok = engine.canCarry(sheet, "light_box", 6);
181 CHECK_FALSE(*ok);
182 // unknown item -> UnknownItem
183 CHECK(engine.canCarry(sheet, "nope", 1).error() == BookkeepingError::UnknownItem);
184}

216TEST_CASE("bookkeeping: containers have their own capacity (weight/size/count)") {
217 const std::string rulesetJson = R"({
218 "schema_version": 1, "ruleset_id": "mini", "licence": "test",
219 "attributes": [{"id": "STR", "name": "Strength", "min": 1, "max": 30, "default": 10}],
220 "encumbrance": {
221 "weight_unit": "lb", "default_weight": 1.0, "capacity": "15 * STR",
222 "size_unit": "slot", "default_item_size": 1.0, "levels": []
223 },
224 "data": {
225 "items": [
226 {"id": "backpack", "name": "Backpack", "weight": "5 lb.", "size": 1,
227 "capacity": {"weight": 30, "size": 6, "items": 5}},
228 {"id": "club", "name": "Club", "weight": "2 lb.", "size": 1}
229 ]
230 }
231 })";
232 RulesetEngine engine;
233 REQUIRE(engine.loadRulesetFromJson(rulesetJson));
234 const auto &ruleset = engine.ruleset();
235 DynamicEntity sheet(ruleset, "test");
236 sheet.setBaseAttribute("STR", 10);
237
238 const auto cap = engine.containerCapacity("backpack");
239 REQUIRE(cap.has_value());
240 CHECK(cap->weight == doctest::Approx(30.0));
241 CHECK(cap->size == doctest::Approx(6.0));
242 CHECK(cap->items == 5);
243 CHECK(cap->enabled);
244 CHECK(engine.containerCapacity("nope").error() == BookkeepingError::UnknownItem);
245 // a non-container item declares no capacity
246 const auto clubCap = engine.containerCapacity("club");
247 REQUIRE(clubCap.has_value());
248 CHECK_FALSE(clubCap->enabled);
249
250 REQUIRE(engine.addItem(sheet, "backpack", 1).has_value());
251 // 16 clubs = 32 lb > 30 weight -> reject
252 CHECK_FALSE(*engine.canAddToContainer(sheet, "backpack", "club", 16));
253 // 7 clubs = size 7 > 6 -> reject
254 CHECK_FALSE(*engine.canAddToContainer(sheet, "backpack", "club", 7));
255 // 6 clubs = 12 lb, size 6, items 6 > 5 -> reject by count
256 CHECK_FALSE(*engine.canAddToContainer(sheet, "backpack", "club", 6));
257 // 5 clubs fit exactly on every axis
258 CHECK(*engine.canAddToContainer(sheet, "backpack", "club", 5));
259 REQUIRE(engine.addItemToContainer(sheet, "backpack", "club", 5).has_value());
260 // the container is now full: one more club cannot fit
261 CHECK_FALSE(*engine.canAddToContainer(sheet, "backpack", "club", 1));
262 // ...and adding it is refused with OverCapacity
263 CHECK(engine.addItemToContainer(sheet, "backpack", "club", 1).error() ==
264 BookkeepingError::OverCapacity);
265 CHECK(sheet.inventory().countIn("backpack", "club") == 5);
266 // containerLoad reports what is inside
267 const auto cl = engine.containerLoad(sheet, "backpack");
268 CHECK(cl.items == 5);
269 CHECK(cl.weight == doctest::Approx(10.0));
270 CHECK(cl.size == doctest::Approx(5.0));
271}
