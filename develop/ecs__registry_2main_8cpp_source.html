80int main(int argc, char **argv) {
81 const std::string root = argc > 1 ? argv[1] : ".";
82 rpg_os::GameSession session;
83 if (!session.loadRulesetFromFile(root + "/rulesets/tde5e_core.json")) {
84 std::cerr << "failed to load tde5e_core.json: " << session.lastError() << '\n';
85 return 1;
86 }
87
88 // 1) Entities: two living sheets, addressed by stable handles. The handles
89 // stay valid even after the entities are removed from the session.
90 const rpg_os::EntityHandle hero = session.createCharacterHandle("geron");
91 const rpg_os::EntityHandle enemy = session.createCreatureHandle("gotongi");
92 const rpg_os::EntityId heroId = hero.entityId();
93 const rpg_os::EntityId enemyId = enemy.entityId();
94
95 // 2) Components: plain application data, keyed by entity id.
96 std::unordered_map<rpg_os::EntityId, Position> positions;
97 std::unordered_map<rpg_os::EntityId, Tag> tags;
98 std::unordered_map<rpg_os::EntityId, HpBar> bars;
99 positions[heroId] = {0, 0};
100 positions[enemyId] = {5, 0};
101 tags[heroId] = {"hero"};
102 tags[enemyId] = {"gotongi"};
103 renderHpSystem(session.entities(), bars);
104
105 std::cout << "=== ECS pattern (dependency-free registry) ===\n";
106 printEntities(session.entities(), positions, tags, bars);
107
108 // 3) An observer event: when the hero's hit points change, update only that
109 // entity's HpBar component — no per-frame polling of the sheet.
110 (void)session.onEvent(rpg_os::EventType::OnResourceChanged, [&](const rpg_os::EventData &data) {
111 if (data.getString("resource", "") != "LP") {
112 return;
113 }
114 const auto id = rpg_os::EntityId{static_cast<uint64_t>(data.getInt("instance_id"))};
115 const auto it = bars.find(id);
116 if (it != bars.end()) {
117 it->second.displayed = data.getInt("new_value");
118 }
119 });
120
121 // 4) A system applies a rule: the hero is poisoned for 2 rounds, and a
122 // CombatSession drives the turns so the duration decays and the expiry
123 // fires OnConditionChanged. The engine owns the bookkeeping; the
124 // application owns the presentation.
125 session.applyCondition(*hero.resolve(), "pain", 1, 2);
126 rpg_os::CombatSession combat(session.engine());
127 (void)combat.addCombatant(hero, 0);
128 (void)combat.addCombatant(enemy, 1);
129 rpg_os::DefaultRandom rng;
130 combat.beginRound(rng);
131 while (combat.next()) {
132 combat.runCurrentTurn(rng);
133 }
134
135 // The hero takes a hit (through the engine's damage pipeline), which flows
136 // through OnResourceChanged and updates the HpBar component automatically.
137 rpg_os::DynamicEntity attacker(session.engine().ruleset(), "attacker");
138 (void)session.engine().applyDamage(attacker, *hero.resolve(), "LP", 6);
139 std::cout << "After a 6-point hit and one round (pain: "
140 << hero.resolve()->conditionStacks("pain") << " stack left):\n";
141 printEntities(session.entities(), positions, tags, bars);
142
143 // 5) The handle layer: removing the enemy makes its handle expire safely.
144 std::cout << "Enemy alive: " << (enemy ? "yes" : "no") << " -> ";
145 (void)session.removeEntity(enemyId);
146 std::cout << (enemy ? "still yes" : "no, handle expired") << '\n';
147 return 0;
148}
