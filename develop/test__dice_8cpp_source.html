123TEST_CASE("DefaultRandom: unseeded instances draw from real entropy") {
124 // Two independent default-seeded engines must not share a stream: with a
125 // genuinely random seed a 32-bit collision is essentially impossible, so a
126 // difference within a few draws is the expected (and stable) outcome.
127 rpg_os::DefaultRandom a;
128 rpg_os::DefaultRandom b;
129 bool anyDifference = false;
130 for (int i = 0; i < 16; ++i) {
131 if (a(1, 1000) != b(1, 1000)) {
132 anyDifference = true;
133 break;
134 }
135 }
136 CHECK(anyDifference);
137}
