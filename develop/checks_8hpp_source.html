154template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
155[[nodiscard]] CheckResult
156resolveAdditiveD20(const Actor &actor, const Target &target, std::string_view diceExpression,
157 std::span<const std::string_view> bonusStats, std::string_view targetStat,
158 const CheckParams &params, Rng &rng) {
159 CheckResult result;
160 result.rawDiceRolls = DiceExpression(diceExpression).roll(rng);
161 int total = params.situationalModifier;
162 for (const std::string_view stat : bonusStats) {
163 total += actor.getStat(stat);
164 }
165 for (const int die : result.rawDiceRolls) {
166 total += die;
167 }
168 const int targetValue = targetStat.empty() ? params.difficulty : target.getStat(targetStat);
169 result.marginOfSuccess = static_cast<int32_t>(total - targetValue);
170
171 const bool natural20 = !result.rawDiceRolls.empty() && result.rawDiceRolls.front() == 20;
172 const bool natural1 = !result.rawDiceRolls.empty() && result.rawDiceRolls.front() == 1;
173 result.isCriticalSuccess = natural20;
174 result.isCriticalFailure = natural1;
175 if (natural20) {
176 result.isSuccess = true;
177 } else if (natural1) {
178 result.isSuccess = false;
179 } else {
180 result.isSuccess = total >= targetValue;
181 }
182 return result;
183}

193template <StatProvider Actor, RandomNumberGenerator Rng>
194[[nodiscard]] CheckResult resolveRollUnderD20(const Actor &actor, std::string_view attribute,
195 bool useConfirmationRoll, const CheckParams &params,
196 Rng &rng) {
197 CheckResult result;
198 const int32_t eav = actor.getStat(attribute) + params.difficulty;
199 if (eav < 1) {
200 // Cannot roll below 1, so success is impossible.
201 result.isSuccess = false;
202 return result;
203 }
204 const int roll = detail::rollDie(20, rng);
205 result.rawDiceRolls.push_back(roll);
206
207 if (useConfirmationRoll && (roll == 1 || roll == 20)) {
208 const int confirm = detail::rollDie(20, rng);
209 result.rawDiceRolls.push_back(confirm);
210 if (roll == 1) {
211 // A successful confirmation is a critical success; otherwise a regular
212 // success on the original check.
213 result.isSuccess = true;
214 result.isCriticalSuccess = confirm <= eav;
215 } else {
216 // A failed confirmation (or another 20) is a botch; otherwise a regular
217 // failure on the original check.
218 result.isSuccess = false;
219 result.isCriticalFailure = confirm > eav || confirm == 20;
220 }
221 } else {
222 result.isSuccess = roll <= eav;
223 result.isCriticalSuccess = roll == 1;
224 result.isCriticalFailure = roll == 20;
225 }
226 result.marginOfSuccess = static_cast<int32_t>(eav - roll);
227 return result;
228}

245template <StatProvider Actor, RandomNumberGenerator Rng>
246[[nodiscard]] CheckResult resolveTripleRollUnderPool(const Actor &actor, std::string_view attr1,
247 std::string_view attr2, std::string_view attr3,
248 std::string_view poolStat,
249 const CheckParams &params, Rng &rng) {
250 CheckResult result;
251 result.rawDiceRolls.reserve(3);
252
253 const std::array<std::string_view, 3> attributes{attr1, attr2, attr3};
254 int32_t eav[3]{};
255 for (std::size_t i = 0; i < 3; ++i) {
256 eav[i] = actor.getStat(attributes[i]) + params.difficulty;
257 if (eav[i] < 1) {
258 // Cannot roll below 1: this part of the check can never pass.
259 result.isSuccess = false;
260 return result;
261 }
262 }
263
264 int32_t dice[3]{};
265 int32_t totalMargin = 0;
266 for (std::size_t i = 0; i < 3; ++i) {
267 dice[i] = detail::rollDie(20, rng);
268 result.rawDiceRolls.push_back(dice[i]);
269 totalMargin += dice[i] > eav[i] ? (dice[i] - eav[i]) : 0;
270 }
271
272 const int32_t pool = actor.getStat(poolStat) + params.situationalModifier;
273 const int32_t finalPool = pool - totalMargin;
274 result.remainingPool = finalPool;
275 result.marginOfSuccess = finalPool;
276
277 const bool doubleOne = (dice[0] == 1 && dice[1] == 1) || (dice[0] == 1 && dice[2] == 1) ||
278 (dice[1] == 1 && dice[2] == 1);
279 const bool doubleTwenty = (dice[0] == 20 && dice[1] == 20) || (dice[0] == 20 && dice[2] == 20) ||
280 (dice[1] == 20 && dice[2] == 20);
281 result.isCriticalSuccess = doubleOne;
282 result.isCriticalFailure = doubleTwenty;
283 result.isSuccess = finalPool >= 0;
284 if (doubleOne) {
285 result.isSuccess = true;
286 } else if (doubleTwenty) {
287 result.isSuccess = false;
288 }
289 result.qualityLevel = result.isSuccess ? qualityLevelFromRemaining(finalPool) : 0;
290 return result;
291}

303template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
304[[nodiscard]] CheckResult
305resolveAttackVsDefense(const Actor &actor, const Target &target, std::string_view attackStat,
306 std::string_view parryStat, const CheckParams &params, Rng &rng) {
307 CheckResult result;
308 const int32_t attackValue = actor.getStat(attackStat) + params.difficulty;
309 const int32_t parryValue = target.getStat(parryStat);
310
311 const int attackRoll = detail::rollDie(20, rng);
312 result.rawDiceRolls.push_back(attackRoll);
313 result.isCriticalSuccess = attackRoll == 1;
314 result.isCriticalFailure = attackRoll == 20;
315 const bool attackHits =
316 attackRoll == 1 ? true : (attackRoll == 20 ? false : attackRoll <= attackValue);
317 result.marginOfSuccess = static_cast<int32_t>(attackValue - attackRoll);
318
319 if (!attackHits) {
320 result.isSuccess = false;
321 return result;
322 }
323
324 const int parryRoll = detail::rollDie(20, rng);
325 result.rawDiceRolls.push_back(parryRoll);
326 const bool parrySucceeds =
327 parryRoll == 1 ? true : (parryRoll == 20 ? false : parryRoll <= parryValue);
328 result.isSuccess = !parrySucceeds;
329 return result;
330}

343template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
344[[nodiscard]] CheckResult resolveCheck(const Actor &actor, const Target &target,
345 const CheckConfig &config, const CheckParams &params,
346 Rng &rng) {
347 switch (config.kind) {
348 case CheckKind::AdditiveD20: {
349 std::vector<std::string_view> views;
350 views.reserve(config.bonusStats.size());
351 for (const std::string &stat : config.bonusStats) {
352 views.push_back(stat);
353 }
354 return resolveAdditiveD20(actor, target, config.diceExpression, views, config.targetStat,
355 params, rng);
356 }
357 case CheckKind::RollUnderD20:
358 return resolveRollUnderD20(actor, config.attributes[0], config.useConfirmationRoll, params,
359 rng);
360 case CheckKind::TripleRollUnderPool:
361 return resolveTripleRollUnderPool(actor, config.attributes[0], config.attributes[1],
362 config.attributes[2], config.poolStat, params, rng);
363 case CheckKind::AttackVsDefense:
364 return resolveAttackVsDefense(actor, target, config.attackStat, config.parryStat, params, rng);
365 }
366 return {};
367}
