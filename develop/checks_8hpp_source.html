173[[nodiscard]] inline SuccessLevel successLevelFor(int32_t stat, int roll) noexcept {
174 const int rawFumbleMin = 95 + (stat + 19) / 20; // 95 + ceil(stat/20)
175 const int fumbleMin = rawFumbleMin < 100 ? rawFumbleMin : 100;
176 if (roll >= fumbleMin) {
177 return SuccessLevel::Fumble;
178 }
179 if (roll <= (stat + 19) / 20) { // ceil(stat/20)
180 return SuccessLevel::Critical;
181 }
182 if (roll <= (stat + 4) / 5) { // ceil(stat/5)
183 return SuccessLevel::Special;
184 }
185 return roll <= stat ? SuccessLevel::Success : SuccessLevel::Failure;
186}

205template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
206[[nodiscard]] CheckResult
207resolveAdditiveD20(const Actor &actor, const Target &target, std::string_view diceExpression,
208 std::span<const std::string_view> bonusStats, std::string_view targetStat,
209 const CheckParams &params, Rng &rng) {
210 CheckResult result;
211 result.rawDiceRolls = DiceExpression(diceExpression).roll(rng);
212 int total = params.situationalModifier;
213 for (const std::string_view stat : bonusStats) {
214 total += actor.getStat(stat);
215 }
216 for (const int die : result.rawDiceRolls) {
217 total += die;
218 }
219 const int targetValue = targetStat.empty() ? params.difficulty : target.getStat(targetStat);
220 result.marginOfSuccess = static_cast<int32_t>(total - targetValue);
221
222 const bool natural20 = !result.rawDiceRolls.empty() && result.rawDiceRolls.front() == 20;
223 const bool natural1 = !result.rawDiceRolls.empty() && result.rawDiceRolls.front() == 1;
224 result.isCriticalSuccess = natural20;
225 result.isCriticalFailure = natural1;
226 if (natural20) {
227 result.isSuccess = true;
228 } else if (natural1) {
229 result.isSuccess = false;
230 } else {
231 result.isSuccess = total >= targetValue;
232 }
233 return result;
234}

244template <StatProvider Actor, RandomNumberGenerator Rng>
245[[nodiscard]] CheckResult resolveRollUnderD20(const Actor &actor, std::string_view attribute,
246 bool useConfirmationRoll, const CheckParams &params,
247 Rng &rng) {
248 CheckResult result;
249 const int32_t eav = actor.getStat(attribute) + params.difficulty;
250 if (eav < 1) {
251 // Cannot roll below 1, so success is impossible.
252 result.isSuccess = false;
253 return result;
254 }
255 const int roll = detail::rollDie(20, rng);
256 result.rawDiceRolls.push_back(roll);
257
258 if (useConfirmationRoll && (roll == 1 || roll == 20)) {
259 const int confirm = detail::rollDie(20, rng);
260 result.rawDiceRolls.push_back(confirm);
261 if (roll == 1) {
262 // A successful confirmation is a critical success; otherwise a regular
263 // success on the original check.
264 result.isSuccess = true;
265 result.isCriticalSuccess = confirm <= eav;
266 } else {
267 // A failed confirmation (or another 20) is a botch; otherwise a regular
268 // failure on the original check.
269 result.isSuccess = false;
270 result.isCriticalFailure = confirm > eav || confirm == 20;
271 }
272 } else {
273 result.isSuccess = roll <= eav;
274 result.isCriticalSuccess = roll == 1;
275 result.isCriticalFailure = roll == 20;
276 }
277 result.marginOfSuccess = static_cast<int32_t>(eav - roll);
278 return result;
279}

296template <StatProvider Actor, RandomNumberGenerator Rng>
297[[nodiscard]] CheckResult resolveTripleRollUnderPool(const Actor &actor, std::string_view attr1,
298 std::string_view attr2, std::string_view attr3,
299 std::string_view poolStat,
300 const CheckParams &params, Rng &rng) {
301 CheckResult result;
302 result.rawDiceRolls.reserve(3);
303
304 const std::array<std::string_view, 3> attributes{attr1, attr2, attr3};
305 int32_t eav[3]{};
306 for (std::size_t i = 0; i < 3; ++i) {
307 eav[i] = actor.getStat(attributes[i]) + params.difficulty;
308 if (eav[i] < 1) {
309 // Cannot roll below 1: this part of the check can never pass.
310 result.isSuccess = false;
311 return result;
312 }
313 }
314
315 int32_t dice[3]{};
316 int32_t totalMargin = 0;
317 for (std::size_t i = 0; i < 3; ++i) {
318 dice[i] = detail::rollDie(20, rng);
319 result.rawDiceRolls.push_back(dice[i]);
320 totalMargin += dice[i] > eav[i] ? (dice[i] - eav[i]) : 0;
321 }
322
323 const int32_t pool = actor.getStat(poolStat) + params.situationalModifier;
324 const int32_t finalPool = pool - totalMargin;
325 result.remainingPool = finalPool;
326 result.marginOfSuccess = finalPool;
327
328 const bool doubleOne = (dice[0] == 1 && dice[1] == 1) || (dice[0] == 1 && dice[2] == 1) ||
329 (dice[1] == 1 && dice[2] == 1);
330 const bool doubleTwenty = (dice[0] == 20 && dice[1] == 20) || (dice[0] == 20 && dice[2] == 20) ||
331 (dice[1] == 20 && dice[2] == 20);
332 result.isCriticalSuccess = doubleOne;
333 result.isCriticalFailure = doubleTwenty;
334 result.isSuccess = finalPool >= 0;
335 if (doubleOne) {
336 result.isSuccess = true;
337 } else if (doubleTwenty) {
338 result.isSuccess = false;
339 }
340 result.qualityLevel = result.isSuccess ? qualityLevelFromRemaining(finalPool) : 0;
341 return result;
342}

354template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
355[[nodiscard]] CheckResult
356resolveAttackVsDefense(const Actor &actor, const Target &target, std::string_view attackStat,
357 std::string_view parryStat, const CheckParams &params, Rng &rng) {
358 CheckResult result;
359 const int32_t attackValue = actor.getStat(attackStat) + params.difficulty;
360 const int32_t parryValue = target.getStat(parryStat);
361
362 const int attackRoll = detail::rollDie(20, rng);
363 result.rawDiceRolls.push_back(attackRoll);
364 result.isCriticalSuccess = attackRoll == 1;
365 result.isCriticalFailure = attackRoll == 20;
366 const bool attackHits =
367 attackRoll == 1 ? true : (attackRoll == 20 ? false : attackRoll <= attackValue);
368 result.marginOfSuccess = static_cast<int32_t>(attackValue - attackRoll);
369
370 if (!attackHits) {
371 result.isSuccess = false;
372 return result;
373 }
374
375 const int parryRoll = detail::rollDie(20, rng);
376 result.rawDiceRolls.push_back(parryRoll);
377 const bool parrySucceeds =
378 parryRoll == 1 ? true : (parryRoll == 20 ? false : parryRoll <= parryValue);
379 result.isSuccess = !parrySucceeds;
380 return result;
381}

398template <StatProvider Actor, RandomNumberGenerator Rng>
399[[nodiscard]] CheckResult resolveRollUnderD100(const Actor &actor, std::string_view attribute,
400 const CheckParams &params, Rng &rng) {
401 CheckResult result;
402 int32_t stat = actor.getStat(attribute) + params.difficulty;
403 if (params.difficultyMode == DifficultyMode::Easy) {
404 stat *= 2;
405 } else if (params.difficultyMode == DifficultyMode::Difficult) {
406 stat /= 2;
407 }
408 const int roll = detail::rollDie(100, rng);
409 result.rawDiceRolls.push_back(roll);
410 result.marginOfSuccess = static_cast<int32_t>(stat - roll);
411 result.successLevel = successLevelFor(stat, roll);
412 result.isCriticalSuccess = result.successLevel == SuccessLevel::Critical;
413 result.isCriticalFailure = result.successLevel == SuccessLevel::Fumble;
414 result.isSuccess = result.successLevel >= SuccessLevel::Success;
415 return result;
416}

432template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
433[[nodiscard]] CheckResult
434resolveOpposedRollUnderD100(const Actor &actor, const Target &target, std::string_view attackStat,
435 std::string_view parryStat, const CheckParams &params, Rng &rng) {
436 const int32_t attackValue = actor.getStat(attackStat) + params.difficulty;
437 const int32_t parryValue = target.getStat(parryStat);
438
439 CheckResult result;
440 const int attackRoll = detail::rollDie(100, rng);
441 const int parryRoll = detail::rollDie(100, rng);
442 result.rawDiceRolls = {attackRoll, parryRoll};
443 result.marginOfSuccess = static_cast<int32_t>(attackValue - attackRoll);
444 const SuccessLevel attackLevel = successLevelFor(attackValue, attackRoll);
445 const SuccessLevel parryLevel = successLevelFor(parryValue, parryRoll);
446 result.successLevel = attackLevel;
447 result.isCriticalSuccess = attackLevel == SuccessLevel::Critical;
448 result.isCriticalFailure = attackLevel == SuccessLevel::Fumble;
449 // Attack and Defense Matrix: a hit needs a strictly higher success level
450 // (and at least a plain success); equal levels mean the defender parries.
451 result.isSuccess = attackLevel >= SuccessLevel::Success && attackLevel > parryLevel;
452 return result;
453}

466template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
467[[nodiscard]] CheckResult
468resolveResistanceRoll(const Actor &actor, const Target &target, std::string_view activeStat,
469 std::string_view passiveStat, const CheckParams &params, Rng &rng) {
470 CheckResult result;
471 const int32_t active = actor.getStat(activeStat) + params.difficulty;
472 const int32_t passive = target.getStat(passiveStat);
473 const int32_t chance = 50 + 5 * (active - passive);
474 if (chance >= 100) {
475 result.isSuccess = true; // automatic success (off the top of the table)
476 result.successLevel = SuccessLevel::Success;
477 return result;
478 }
479 if (chance <= 0) {
480 result.isSuccess = false; // automatic failure (off the bottom of the table)
481 result.successLevel = SuccessLevel::Failure;
482 return result;
483 }
484 const int roll = detail::rollDie(100, rng);
485 result.rawDiceRolls.push_back(roll);
486 result.marginOfSuccess = static_cast<int32_t>(chance - roll);
487 result.isSuccess = roll <= chance;
488 result.successLevel = result.isSuccess ? SuccessLevel::Success : SuccessLevel::Failure;
489 return result;
490}

503template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
504[[nodiscard]] CheckResult resolveCheck(const Actor &actor, const Target &target,
505 const CheckConfig &config, const CheckParams &params,
506 Rng &rng) {
507 switch (config.kind) {
508 case CheckKind::AdditiveD20: {
509 std::vector<std::string_view> views;
510 views.reserve(config.bonusStats.size());
511 for (const std::string &stat : config.bonusStats) {
512 views.push_back(stat);
513 }
514 return resolveAdditiveD20(actor, target, config.diceExpression, views, config.targetStat,
515 params, rng);
516 }
517 case CheckKind::RollUnderD20:
518 return resolveRollUnderD20(actor, config.attributes[0], config.useConfirmationRoll, params,
519 rng);
520 case CheckKind::TripleRollUnderPool:
521 return resolveTripleRollUnderPool(actor, config.attributes[0], config.attributes[1],
522 config.attributes[2], config.poolStat, params, rng);
523 case CheckKind::AttackVsDefense:
524 return resolveAttackVsDefense(actor, target, config.attackStat, config.parryStat, params, rng);
525 case CheckKind::RollUnderD100:
526 return resolveRollUnderD100(actor, config.attributes[0], params, rng);
527 case CheckKind::OpposedRollUnderD100:
528 return resolveOpposedRollUnderD100(actor, target, config.attackStat, config.parryStat, params,
529 rng);
530 case CheckKind::ResistanceRoll:
531 return resolveResistanceRoll(actor, target, config.attackStat, config.parryStat, params, rng);
532 }
533 return {};
534}
