248[[nodiscard]] constexpr SuccessLevel successLevelFor(int32_t stat, int roll) noexcept {
249 const int rawFumbleMin = 95 + (stat + 19) / 20; // 95 + ceil(stat/20)
250 const int fumbleMin = rawFumbleMin < 100 ? rawFumbleMin : 100;
251 if (roll >= fumbleMin) {
252 return SuccessLevel::Fumble;
253 }
254 if (roll <= (stat + 19) / 20) { // ceil(stat/20)
255 return SuccessLevel::Critical;
256 }
257 if (roll <= (stat + 4) / 5) { // ceil(stat/5)
258 return SuccessLevel::Special;
259 }
260 return roll <= stat ? SuccessLevel::Success : SuccessLevel::Failure;
261}

293template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
294[[nodiscard]] constexpr CheckResult resolveThresholdCheck(const Actor &actor, const Target &target,
295 const CheckRecipe &recipe,
296 const CheckParams &params, Rng &rng) {
297 CheckResult result;
298 const DiceExpression &dice = recipe.dice;
299
300 // Derive the reference value the roll is compared against.
301 int32_t reference = 0;
302 if (recipe.thresholdSource == ThresholdSource::TargetStat) {
303 reference = static_cast<int32_t>(target.getStat(recipe.thresholdStat));
304 } else if (recipe.thresholdSource == ThresholdSource::ActorStat) {
305 reference = static_cast<int32_t>(actor.getStat(recipe.thresholdStat));
306 }
307 if (recipe.difficultyMode == DifficultyMode::ToStat) {
308 reference += params.difficulty;
309 } else if (recipe.difficultyMode == DifficultyMode::ToThreshold &&
310 recipe.thresholdSource == ThresholdSource::Difficulty) {
311 reference = params.difficulty;
312 }
313 if (recipe.difficultyMultiplier == DifficultyMultiplier::DoubleHalve) {
314 reference = detail::scaleByDifficulty(reference, params.difficultyScale);
315 }
316 // Roll-under checks fail outright when the reference is below the minimum
317 // rollable value (1): no die can possibly meet it, so no RNG is consumed.
318 if (recipe.comparison == Comparison::LessEqual && reference < 1) {
319 result.isSuccess = false;
320 return result;
321 }
322
323 result.rawDiceRolls = dice.roll(rng);
324 int diceTotal = 0;
325 for (const int die : result.rawDiceRolls) {
326 diceTotal += die;
327 }
328
329 // Advantage / disadvantage: roll the whole expression a second time and
330 // keep the better (advantage) or worse (disadvantage) total, replacing the
331 // recorded dice so critical detection and callers see the kept roll.
332 if (params.advantage != AdvantageMode::None) {
333 const std::vector<int> second = dice.roll(rng);
334 int secondTotal = 0;
335 for (const int die : second) {
336 secondTotal += die;
337 }
338 const bool keepSecond = params.advantage == AdvantageMode::Advantage ? secondTotal > diceTotal
339 : secondTotal < diceTotal;
340 if (keepSecond) {
341 result.rawDiceRolls = second;
342 diceTotal = secondTotal;
343 }
344 }
345
346 // Additive checks sum bonus stats, bonus dice, and the situational modifier
347 // into the rolled total; roll-under checks compare the bare roll.
348 int total = diceTotal;
349 if (recipe.comparison == Comparison::GreaterEqual) {
350 for (const std::string &stat : recipe.bonusStats) {
351 total += static_cast<int32_t>(actor.getStat(stat));
352 }
353 for (const DiceExpression &bonus : params.bonusDice) {
354 const std::vector<int> rolls = bonus.roll(rng);
355 result.rawDiceRolls.insert(result.rawDiceRolls.end(), rolls.begin(), rolls.end());
356 for (const int die : rolls) {
357 total += die;
358 }
359 }
360 total += params.situationalModifier;
361 }
362
363 const int roll = diceTotal;
364 result.marginOfSuccess = static_cast<int32_t>(
365 recipe.comparison == Comparison::GreaterEqual ? total - reference : reference - roll);
366
367 if (recipe.criticalStyle == CriticalStyle::Face && roll == recipe.criticalFace) {
368 result.isSuccess = true;
369 if (recipe.criticalConfirm) {
370 const int confirm = dice.rollSum(rng);
371 result.rawDiceRolls.push_back(confirm);
372 result.isCriticalSuccess = confirm <= reference;
373 } else {
374 result.isCriticalSuccess = true;
375 }
376 } else if (recipe.fumbleStyle == CriticalStyle::Face && roll == recipe.fumbleFace) {
377 result.isSuccess = false;
378 if (recipe.fumbleConfirm) {
379 const int confirm = dice.rollSum(rng);
380 result.rawDiceRolls.push_back(confirm);
381 result.isCriticalFailure = confirm > reference || confirm == recipe.fumbleFace;
382 } else {
383 result.isCriticalFailure = true;
384 }
385 } else if (recipe.grading == Grading::Percentile) {
386 result.successLevel = successLevelFor(reference, roll);
387 result.isCriticalSuccess = result.successLevel == SuccessLevel::Critical;
388 result.isCriticalFailure = result.successLevel == SuccessLevel::Fumble;
389 result.isSuccess = result.successLevel >= SuccessLevel::Success;
390 } else {
391 result.isSuccess =
392 recipe.comparison == Comparison::GreaterEqual ? total >= reference : roll <= reference;
393 }
394 return result;
395}

408template <StatProvider Actor, RandomNumberGenerator Rng>
409[[nodiscard]] constexpr CheckResult resolvePoolCheck(const Actor &actor, const CheckRecipe &recipe,
410 const CheckParams &params, Rng &rng) {
411 CheckResult result;
412 const DiceExpression &dice = recipe.dice;
413
414 // Every effective attribute must be at least 1 before any die is rolled:
415 // otherwise that part of the check can never pass, and no RNG is consumed.
416 const std::size_t n = recipe.numPoolAttributes;
417 int32_t eav[3]{};
418 for (std::size_t i = 0; i < n; ++i) {
419 eav[i] = static_cast<int32_t>(actor.getStat(recipe.poolAttributes[i])) + params.difficulty;
420 if (eav[i] < 1) {
421 result.isSuccess = false;
422 return result;
423 }
424 }
425
426 result.rawDiceRolls = dice.roll(rng);
427 int32_t totalOvershoot = 0;
428 for (std::size_t i = 0; i < n; ++i) {
429 totalOvershoot += result.rawDiceRolls[i] > eav[i] ? (result.rawDiceRolls[i] - eav[i]) : 0;
430 }
431
432 const int32_t pool =
433 static_cast<int32_t>(actor.getStat(recipe.poolStat)) + params.situationalModifier;
434 const int32_t finalPool = pool - totalOvershoot;
435 result.remainingPool = finalPool;
436 result.marginOfSuccess = finalPool;
437
438 bool doubleLow = false;
439 bool doubleHigh = false;
440 for (std::size_t i = 0; i < n; ++i) {
441 for (std::size_t j = i + 1; j < n; ++j) {
442 doubleLow = doubleLow || (result.rawDiceRolls[i] == 1 && result.rawDiceRolls[j] == 1);
443 doubleHigh = doubleHigh || (result.rawDiceRolls[i] == 20 && result.rawDiceRolls[j] == 20);
444 }
445 }
446 result.isCriticalSuccess = doubleLow;
447 result.isCriticalFailure = doubleHigh;
448 result.isSuccess = finalPool >= 0;
449 if (doubleLow) {
450 result.isSuccess = true;
451 } else if (doubleHigh) {
452 result.isSuccess = false;
453 }
454 if (recipe.grading == Grading::PoolQuality) {
455 result.qualityLevel = result.isSuccess ? qualityLevelFromRemaining(finalPool) : 0;
456 }
457 return result;
458}

471template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
472[[nodiscard]] constexpr CheckResult resolveOpposedCheck(const Actor &actor, const Target &target,
473 const CheckRecipe &recipe,
474 const CheckParams &params, Rng &rng) {
475 CheckResult result;
476 const DiceExpression &dice = recipe.dice;
477 const int32_t attackValue =
478 static_cast<int32_t>(actor.getStat(recipe.attackStat)) + params.difficulty;
479 const int32_t parryValue = static_cast<int32_t>(target.getStat(recipe.parryStat));
480
481 if (recipe.compareLevels) {
482 // Both rolls are needed up front: the two success levels are compared
483 // directly, so the defender's roll must exist even when the attack is a
484 // critical. Rolling both keeps the RNG stream aligned with the rules.
485 const int attackRoll = dice.rollSum(rng);
486 const int parryRoll = dice.rollSum(rng);
487 result.rawDiceRolls = {attackRoll, parryRoll};
488 result.marginOfSuccess = static_cast<int32_t>(attackValue - attackRoll);
489 const SuccessLevel attackLevel = successLevelFor(attackValue, attackRoll);
490 const SuccessLevel parryLevel = successLevelFor(parryValue, parryRoll);
491 result.successLevel = attackLevel;
492 result.isCriticalSuccess = attackLevel == SuccessLevel::Critical;
493 result.isCriticalFailure = attackLevel == SuccessLevel::Fumble;
494 // A hit needs a strictly higher success level (and at least a plain
495 // success); equal levels mean the defender parries.
496 result.isSuccess = attackLevel >= SuccessLevel::Success && attackLevel > parryLevel;
497 return result;
498 }
499
500 // Staged roll-under contest: the parry only happens after a successful
501 // attack, so the defense roll must not consume RNG (or affect the result)
502 // when the attack already missed.
503 const int attackRoll = dice.rollSum(rng);
504 result.rawDiceRolls.push_back(attackRoll);
505 result.isCriticalSuccess = attackRoll == recipe.criticalFace;
506 result.isCriticalFailure = attackRoll == recipe.fumbleFace;
507 const bool attackHits =
508 attackRoll == recipe.criticalFace
509 ? true
510 : (attackRoll == recipe.fumbleFace ? false : attackRoll <= attackValue);
511 result.marginOfSuccess = static_cast<int32_t>(attackValue - attackRoll);
512 if (!attackHits) {
513 result.isSuccess = false;
514 return result;
515 }
516 const int parryRoll = dice.rollSum(rng);
517 result.rawDiceRolls.push_back(parryRoll);
518 const bool parrySucceeds =
519 parryRoll == recipe.criticalFace
520 ? true
521 : (parryRoll == recipe.fumbleFace ? false : parryRoll <= parryValue);
522 result.isSuccess = !parrySucceeds;
523 return result;
524}

533template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
534[[nodiscard]] constexpr CheckResult resolveResistanceCheck(const Actor &actor, const Target &target,
535 const CheckRecipe &recipe,
536 const CheckParams &params, Rng &rng) {
537 CheckResult result;
538 const int32_t active = static_cast<int32_t>(actor.getStat(recipe.attackStat)) + params.difficulty;
539 const int32_t passive = static_cast<int32_t>(target.getStat(recipe.parryStat));
540 const int32_t chance = 50 + 5 * (active - passive);
541 if (chance >= 100) {
542 result.isSuccess = true; // automatic success (off the top of the table)
543 result.successLevel = SuccessLevel::Success;
544 return result;
545 }
546 if (chance <= 0) {
547 result.isSuccess = false; // automatic failure (off the bottom of the table)
548 result.successLevel = SuccessLevel::Failure;
549 return result;
550 }
551 const DiceExpression &dice = recipe.dice;
552 const int roll = dice.rollSum(rng);
553 result.rawDiceRolls.push_back(roll);
554 result.marginOfSuccess = static_cast<int32_t>(chance - roll);
555 result.isSuccess = roll <= chance;
556 result.successLevel = result.isSuccess ? SuccessLevel::Success : SuccessLevel::Failure;
557 return result;
558}

572template <StatProvider Actor, StatProvider Target, RandomNumberGenerator Rng>
573[[nodiscard]] constexpr CheckResult resolveCheck(const Actor &actor, const Target &target,
574 const CheckRecipe &recipe,
575 const CheckParams &params, Rng &rng) {
576 // An automatic failure ("you automatically fail Strength and Dexterity
577 // saving throws") fails the check outright without consuming any RNG.
578 if (params.autoFail) {
579 CheckResult failed;
580 failed.isSuccess = false;
581 return failed;
582 }
583 switch (recipe.resolution) {
584 case Resolution::Threshold:
585 return resolveThresholdCheck(actor, target, recipe, params, rng);
586 case Resolution::Pool:
587 return resolvePoolCheck(actor, recipe, params, rng);
588 case Resolution::Opposed:
589 return resolveOpposedCheck(actor, target, recipe, params, rng);
590 case Resolution::Resistance:
591 return resolveResistanceCheck(actor, target, recipe, params, rng);
592 }
593 return {};
594}
