62TEST_CASE("check_resolver: resolves a named threshold check against a difficulty") {
63 const Ruleset ruleset = RulesetLoader::load(rpg_os::Json::parse(kRuleset));
64 MockStats actor;
65 actor.values["STR"] = 14;
66 const MockStats noTarget; // a solo threshold check has no target
67 CheckParams params;
68 params.difficulty = 20;
69
70 // 10 + 14 = 24 >= 20 -> success with margin 4.
71 ScriptedRng okRng = script({10});
72 const rpg_os::CheckResult ok =
73 CheckResolver::resolve(ruleset, actor, noTarget, "hit", params, okRng);
74 CHECK(ok.isSuccess);
75 CHECK(ok.marginOfSuccess == 4);
76 CHECK(ok.rawDiceRolls.size() == 1);
77
78 // 2 + 14 = 16 < 20 -> failure with margin -4.
79 ScriptedRng failRng = script({2});
80 const rpg_os::CheckResult fail =
81 CheckResolver::resolve(ruleset, actor, noTarget, "hit", params, failRng);
82 CHECK_FALSE(fail.isSuccess);
83 CHECK(fail.marginOfSuccess == -4);
84}

86TEST_CASE("check_resolver: resolves a named opposed check with a target") {
87 const Ruleset ruleset = RulesetLoader::load(rpg_os::Json::parse(kRuleset));
88 MockStats actor;
89 actor.values["STR"] = 14;
90 MockStats target;
91 target.values["DEF"] = 10;
92 CheckParams params;
93
94 // Staged contest: attack 8 <= 14 hits, parry 12 > 10 fails -> the blow lands.
95 ScriptedRng hitRng = script({8, 12});
96 const rpg_os::CheckResult hit =
97 CheckResolver::resolve(ruleset, actor, target, "fight", params, hitRng);
98 CHECK(hit.isSuccess);
99 CHECK(hit.rawDiceRolls.size() == 2);
100
101 // Attack 8 hits but parry 5 <= 10 succeeds -> the blow is parried.
102 ScriptedRng parriedRng = script({8, 5});
103 const rpg_os::CheckResult parried =
104 CheckResolver::resolve(ruleset, actor, target, "fight", params, parriedRng);
105 CHECK_FALSE(parried.isSuccess);
106}
