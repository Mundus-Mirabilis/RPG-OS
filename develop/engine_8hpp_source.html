74 bool loadRulesetFromFile(std::string_view path) {
75 const std::string pathStr(path);
76 std::ifstream file(pathStr);
77 if (!file) {
78 m_lastError = "cannot open file '" + pathStr + "'";
79 m_loaded = false;
80 return false;
81 }
82 // Read via rdbuf() rather than the istreambuf_iterator range idiom: the
83 // iterator path trips a -Wnull-dereference false positive in libstdc++ 13
84 // under -Werror, and streaming the streambuf is the idiomatic whole-file
85 // read anyway.
86 std::ostringstream buffer;
87 buffer << file.rdbuf();
88 return loadRulesetFromJson(buffer.str());

257 recipe.poolAttributes = {skill->attributes[0], skill->attributes[1], skill->attributes[2]};
258 recipe.poolStat = std::string(skillId);
259 recipe.criticalStyle = CriticalStyle::DoubleRoll;
260 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
261 recipe.grading = Grading::PoolQuality;
262 recipe.difficultyMode = DifficultyMode::ToStat;
263 return resolveCheck(actor, NullStatProvider{}, recipe, params, rng);

286 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
287 int32_t rawDamage, const Json &env = {}) {
288 EventData data;
289 data.payload = {{"damage", rawDamage},
290 {"raw_damage", rawDamage},
291 {"attacker_id", actor.id()},
292 {"target_id", target.id()}};
293 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
294 int32_t finalDamage = data.getInt("damage", rawDamage);
295 if (finalDamage < 0) {
296 finalDamage = 0;
297 }
298 // Temporary Hit Points are a buffer: they absorb damage before the real
299 // hit-point pool ("Lose Temporary Hit Points First").
300 if (target.temporaryHitPoints() > 0 && finalDamage > 0) {
301 const int32_t absorbed = std::min(target.temporaryHitPoints(), finalDamage);
302 target.addTemporaryHitPoints(-absorbed);
303 finalDamage -= absorbed;
304 data.payload["temporary_hp_absorbed"] = absorbed;
305 }
306 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
307 data.payload["applied_damage"] = -applied;
308 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
309 return applied;
310 }

393 template <RandomNumberGenerator Rng>
394 [[nodiscard]] SpellResult castSpell(std::string_view spellId, DynamicEntity &actor,
395 DynamicEntity *target, std::string_view resourceId,
396 const CheckParams &params, Rng &rng) {
397 SpellResult result;
398 const Json *spell = findSpell(spellId);
399 if (spell == nullptr) {
400 throw std::invalid_argument("unknown spell '" + std::string(spellId) + "'");
401 }
402 // Cost: `cost` or `ae_cost`, else `level` (one point per level).
403 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
404 result.cost = spell->at("cost").get<int32_t>();
405 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
406 result.cost = spell->at("ae_cost").get<int32_t>();
407 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
408 result.cost = spell->at("level").get<int32_t>();
409 }
410 result.resourceId = std::string(resourceId);
411 if (result.cost > 0 && !resourceId.empty()) {
412 if (actor.resource(resourceId) < result.cost) {
413 result.cast = false; // cannot afford the spell
414 return result;
415 }
416 (void)actor.modifyResource(resourceId, -result.cost);
417 }
418
419 // Casting check (named check type, or an "A/B/C" attribute list).
420 if (spell->contains("check") && spell->at("check").is_string()) {
421 const std::string checkText = spell->at("check").get<std::string>();
422 if (m_ruleset.findCheckType(checkText) != nullptr) {
423 const NullStatProvider noTarget;
424 result.check =
425 target != nullptr
426 ? CheckResolver::resolve(m_ruleset, actor, *target, checkText, params, rng)
427 : CheckResolver::resolve(m_ruleset, actor, noTarget, checkText, params, rng);
428 } else {
429 result.check = resolveSpellCheck(actor, checkText, params, rng);
430 }
431 result.cast = result.check.isSuccess;
432 } else {
433 result.cast = true;
434 }
435
436 // Structured effects take precedence over the bare `damage` field: they
437 // are the extracted, machine-readable form of the spell's prose (saves,
438 // half-on-save damage, conditions, healing, resistances). A spell with no
439 // `effects` falls back to the simple damage field.
440 if (target != nullptr && spell->contains("effects") && spell->at("effects").is_array()) {
441 const EffectsResult effects =
442 resolveEffects(actor, *target, spell->at("effects"), params, rng);
443 result.appliedDamage = effects.damageDealt;
444 } else if (target != nullptr && spell->contains("damage")) {
445 // `applyDamage` reports the (negative) pool delta, so negate it into the
446 // positive "damage dealt" the caller expects.
447 const int32_t damage = readVariantValue(spell->at("damage"), Variance::Random, rng);
448 const std::string hitPool = resolveHitPointPoolId();
449 if (!hitPool.empty()) {
450 result.appliedDamage = -applyDamage(actor, *target, hitPool, damage);
451 }
452 }
453 return result;
454 }

474 template <RandomNumberGenerator Rng>
475 [[nodiscard]] AfflictionResult applyAffliction(std::string_view section, std::string_view id,
476 DynamicEntity &victim, const CheckParams &params,
477 Rng &rng) {
478 AfflictionResult result;
479 const Json *affliction = findDataRecord(section, id);
480 if (affliction == nullptr) {
481 throw std::invalid_argument("unknown " + std::string(section) + " '" + std::string(id) + "'");
482 }
483 if (affliction->contains("save") && affliction->at("save").is_string()) {
484 const std::string save = affliction->at("save").get<std::string>();
485 if (m_ruleset.hasStat(save)) {
486 result.saveRolled = true;
487 CheckRecipe recipe;
488 recipe.resolution = Resolution::Threshold;
489 recipe.dice = "1d20"_dice;
490 recipe.comparison = Comparison::LessEqual;
491 recipe.thresholdSource = ThresholdSource::ActorStat;
492 recipe.thresholdStat = save;
493 recipe.difficultyMode = DifficultyMode::ToStat;
494 const CheckResult saveResult =
495 resolveCheck(victim, NullStatProvider{}, recipe, params, rng);
496 if (saveResult.isSuccess) {
497 result.resisted = true;
498 return result;
499 }
500 }
501 }
502 if (affliction->contains("effects") && affliction->at("effects").is_array()) {
503 std::vector<std::string> appliedConditions;
504 for (const Json &effect : affliction->at("effects")) {
505 const std::string stat = effect.value("stat", "");
506 if (stat.empty()) {
507 continue;
508 }
509 int32_t amount = 1;
510 if (effect.contains("amount")) {
511 amount = readVariantValue(effect.at("amount"), Variance::Random, rng);
512 }
513 if (m_ruleset.isResourcePool(stat)) {
514 (void)victim.modifyResource(stat, -amount);
515 ++result.effectsApplied;
516 } else if (effect.contains("condition") && effect.at("condition").is_string()) {
517 const std::string conditionId = effect.at("condition").get<std::string>();
518 victim.addCondition(conditionId, amount);
519 appliedConditions.push_back(conditionId);
520 ++result.effectsApplied;
521 } else if (m_ruleset.isCondition(stat)) {
522 victim.addCondition(stat, amount);
523 appliedConditions.push_back(stat);
524 ++result.effectsApplied;
525 } else if (m_ruleset.findAttribute(stat) != nullptr) {
526 victim.setBaseAttribute(stat, victim.baseAttribute(stat) - amount);
527 ++result.effectsApplied;
528 }
529 }
530 // Remember what was applied so an application can cure it later.
531 if (result.effectsApplied > 0) {
532 victim.addAffliction(section, id, appliedConditions);
533 EventData data;
534 data.payload = {{"section", std::string(section)},
535 {"affliction", std::string(id)},
536 {"victim_id", victim.id()},
537 {"effects_applied", result.effectsApplied}};
538 fireEvent(EventType::OnAfflictionApplied, data, victim, nullptr, Json{});
539 }
540 }
541 return result;
542 }

547 DefaultRandom rng;

569 [[nodiscard]] std::expected<Money, BookkeepingError> itemPrice(std::string_view itemId) const {
570 if (!m_loaded) {
571 return std::unexpected(BookkeepingError::NoRuleset);
572 }
573 const Json *item = findItem(itemId);
574 if (item == nullptr) {
575 return std::unexpected(BookkeepingError::UnknownItem);
576 }
577 if (!m_ruleset.hasCurrency()) {
578 return std::unexpected(BookkeepingError::UnknownCurrency);
579 }
580 const Json *price = nullptr;
581 if (item->contains("cost")) {
582 price = &item->at("cost");
583 } else if (item->contains("price")) {
584 price = &item->at("price");
585 } else if (item->contains("value")) {
586 price = &item->at("value");
587 }
588 if (price == nullptr) {
589 return Money{0};
590 }
591 if (price->is_number_integer()) {
592 return Money{price->get<int64_t>()};
593 }
594 if (price->is_string()) {
595 return parsePriceString(price->get<std::string>());
596 }
597 return Money{0};
598 }

697 [[nodiscard]] std::expected<EquipResult, BookkeepingError>
698 equip(DynamicEntity &sheet, std::string_view slotId, std::string_view itemId) {
699 if (!m_loaded) {
700 return std::unexpected(BookkeepingError::NoRuleset);
701 }
702 bool slotOk = false;
703 for (const std::string &slot : m_ruleset.equipmentSlots) {
704 if (slot == slotId) {
705 slotOk = true;
706 break;
707 }
708 }
709 if (!slotOk) {
710 return std::unexpected(BookkeepingError::SlotMismatch);
711 }
712 const Json *item = findItem(itemId);
713 if (item == nullptr) {
714 return std::unexpected(BookkeepingError::UnknownItem);
715 }
716 const std::string itemSlot = item->value("slot", "");
717 if (!itemSlot.empty() && itemSlot != slotId) {
718 return std::unexpected(BookkeepingError::SlotMismatch);
719 }
720 if (sheet.inventory().count(itemId) <= 0) {
721 return std::unexpected(BookkeepingError::ItemNotOwned);
722 }
723 EquipResult result;
724 result.slotId = std::string(slotId);
725 result.itemId = std::string(itemId);
726 result.quantityBefore = sheet.inventory().count(itemId);
727 if (sheet.equipment().isEquipped(slotId)) {
728 result.previousItemId = std::string(sheet.equipment().itemIn(slotId));
729 (void)sheet.equipment().unequip(slotId);
730 if (!result.previousItemId.empty()) {
731 sheet.inventory().add(ItemInstance{result.previousItemId, 1, {}});
732 }
733 }
734 (void)sheet.equipment().equip(slotId, itemId);
735 (void)sheet.inventory().remove(itemId, 1);
736 EventData data;
737 data.payload = {
738 {"slot", std::string(slotId)}, {"item", std::string(itemId)}, {"owner_id", sheet.id()}};
739 fireEvent(EventType::OnEquipChanged, data, sheet, nullptr, Json{});
740 return result;
741 }

765 [[nodiscard]] Weight carriedWeight(const DynamicEntity &sheet) const {
766 const EncumbranceConfig &cfg = m_ruleset.encumbrance;
767 Weight total{0.0, cfg.weightUnit};
768 const auto addWeight = [&](std::string_view itemId, int32_t quantity) {
769 double unit = cfg.defaultItemWeight;
770 const Json *item = findItem(itemId);
771 if (item != nullptr && item->contains("weight")) {
772 const Json &weightField = item->at("weight");
773 if (weightField.is_number()) {
774 unit = weightField.get<double>();
775 } else if (weightField.is_string()) {
776 const double parsed = parseWeightValue(weightField.get<std::string>());
777 if (parsed >= 0.0) {
778 unit = parsed;
779 }
780 }
781 }
782 total.value += unit * quantity;
783 };
784 sheet.inventory().visitStacks(
785 [&](const ItemInstance &stack) { addWeight(stack.itemId, stack.quantity); });
786 for (const auto &[slot, itemId] : sheet.equipment().slots()) {
787 addWeight(itemId, 1);
788 }
789 return total;
790 }

1084 template <RandomNumberGenerator Rng>
1085 [[nodiscard]] EffectsResult resolveEffects(DynamicEntity &source, DynamicEntity &target,
1086 const Json &effects, const CheckParams &params,
1087 Rng &rng) {
1088 EffectsResult result;
1089 if (!effects.is_array()) {
1090 return result;
1091 }
1092 const std::string hitPool = resolveHitPointPoolId();
1093 for (const Json &effect : effects) {
1094 const std::string kind = effect.value("kind", "");
1095 if (kind == "damage") {
1096 if (!effect.contains("dice")) {
1097 continue;
1098 }
1099 int32_t final =
1100 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1101 if (effect.contains("save")) {
1102 if (resolveSave(source, target, effect.at("save"), params, rng)) {
1103 ++result.savesPassed;
1104 final = effect.at("save").value("on_success", "none") == "half" ? final / 2 : 0;
1105 }
1106 } else if (effect.contains("attack")) {
1107 const Json &attack = effect.at("attack");
1108 if (!resolveAttack(source, target, attack, params, rng)) {
1109 final = attack.value("on_miss", "none") == "half" ? final / 2 : 0;
1110 }
1111 }
1112 if (final > 0 && !hitPool.empty()) {
1113 // applyDamage reports the negative pool delta; negate for "dealt".
1114 result.damageDealt += -applyDamage(source, target, hitPool, final);
1115 }
1116 } else if (kind == "condition") {
1117 const std::string conditionId = effect.value("condition", "");
1118 if (conditionId.empty()) {
1119 continue;
1120 }
1121 bool applied = true;
1122 if (effect.contains("save") &&
1123 resolveSave(source, target, effect.at("save"), params, rng)) {
1124 ++result.savesPassed;
1125 applied = false;
1126 } else if (effect.contains("attack") &&
1127 !resolveAttack(source, target, effect.at("attack"), params, rng)) {
1128 applied = false;
1129 }
1130 if (!applied) {
1131 continue;
1132 }
1133 const int32_t stacks = effect.value("stacks", 1);
1134 target.addCondition(conditionId, stacks);
1135 const int32_t duration = effect.value("duration", 0);
1136 if (duration > 0) {
1137 target.effects().add(ActiveEffect{conditionId, stacks, duration, "effect"});
1138 }
1139 ++result.conditionsApplied;
1140 } else if (kind == "heal") {
1141 if (!effect.contains("dice") || hitPool.empty()) {
1142 continue;
1143 }
1144 int32_t amount =
1145 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1146 if (effect.contains("add")) {
1147 const Json &add = effect.at("add");
1148 if (add.is_number()) {
1149 amount += add.get<int32_t>();
1150 } else if (add.is_string()) {
1151 const Json emptyEnv = Json::object();
1152 const EntityContext context(source, &target, emptyEnv, Json{});
1153 amount += math::toStat(Expression(add.get<std::string>()).evaluate(context));
1154 }
1155 }
1156 result.healingDone += target.modifyResource(hitPool, amount);
1157 } else if (kind == "temp_hp") {
1158 if (!effect.contains("dice")) {
1159 continue;
1160 }
1161 int32_t amount =
1162 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1163 if (effect.contains("add")) {
1164 const Json &add = effect.at("add");
1165 if (add.is_number()) {
1166 amount += add.get<int32_t>();
1167 } else if (add.is_string()) {
1168 const Json emptyEnv = Json::object();
1169 const EntityContext context(source, &target, emptyEnv, Json{});
1170 amount += math::toStat(Expression(add.get<std::string>()).evaluate(context));
1171 }
1172 }
1173 target.addTemporaryHitPoints(amount);
1174 } else if (kind == "resist") {
1175 if (effect.contains("types")) {
1176 for (const Json &type : effect.at("types")) {
1177 target.addResistance(type.get<std::string>());
1178 }
1179 }
1180 }
1181 }
1182 return result;
1183 }

1375 [[nodiscard]] std::expected<Money, BookkeepingError>
1376 parsePriceString(std::string_view text) const {
1377 std::string trimmed = trimWhitespace(text);
1378 if (trimmed.empty() || trimmed == "-" || trimmed == "—" || trimmed == "Varies" ||
1379 trimmed == "varies") {
1380 return Money{0};
1381 }
1382 std::size_t pos = 0;
1383 bool sawDigit = false;
1384 while (pos < trimmed.size()) {
1385 const char c = trimmed[pos];
1386 if (c == ',' || c == '.' || c == ' ' || (c >= '0' && c <= '9')) {
1387 if (c >= '0' && c <= '9') {
1388 sawDigit = true;
1389 }
1390 ++pos;
1391 continue;
1392 }
1393 break;
1394 }
1395 if (!sawDigit) {
1396 return Money{0};
1397 }
1398 std::string number = trimmed.substr(0, pos);
1399 std::string symbol = trimWhitespace(trimmed.substr(pos));
1400 number.erase(std::remove(number.begin(), number.end(), ','), number.end());
1401 number.erase(std::remove(number.begin(), number.end(), ' '), number.end());
1402 const int64_t amount = static_cast<int64_t>(std::llround(std::strtod(number.c_str(), nullptr)));
1403 if (symbol.empty()) {
1404 return Money{amount}; // a bare number is base units
1405 }
1406 for (const Denomination &d : m_ruleset.currencySystem.denominations) {
1407 if (d.symbol == symbol || d.id == symbol) {
1408 return Money{amount * d.perBase};
1409 }
1410 }
1411 return std::unexpected(BookkeepingError::UnknownDenomination);
1412 }

1507 template <RandomNumberGenerator Rng>
1508 [[nodiscard]] CheckResult resolveSpellCheck(const DynamicEntity &actor,
1509 std::string_view checkText, const CheckParams &params,
1510 Rng &rng) const {
1511 CheckRecipe recipe;
1512 recipe.resolution = Resolution::Pool;
1513 recipe.dice = "3d20"_dice;
1514 recipe.criticalStyle = CriticalStyle::DoubleRoll;
1515 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
1516 recipe.grading = Grading::PoolQuality;
1517 recipe.difficultyMode = DifficultyMode::ToStat;
1518
1519 const std::size_t paren = checkText.find('(');
1520 const std::string_view head =
1521 paren == std::string_view::npos ? checkText : checkText.substr(0, paren);
1522 std::size_t begin = 0;
1523 while (begin < head.size()) {
1524 while (begin < head.size() && (head[begin] == ' ' || head[begin] == '/')) {
1525 ++begin;
1526 }
1527 if (begin >= head.size()) {
1528 break;
1529 }
1530 std::size_t end = begin;
1531 while (end < head.size() && head[end] != '/' && head[end] != ' ') {
1532 ++end;
1533 }
