74 bool loadRulesetFromFile(std::string_view path) {
75 const std::string pathStr(path);
76 std::ifstream file(pathStr);
77 if (!file) {
78 m_lastError = "cannot open file '" + pathStr + "'";
79 m_loaded = false;
80 return false;
81 }
82 // Read via rdbuf() rather than the istreambuf_iterator range idiom: the
83 // iterator path trips a -Wnull-dereference false positive in libstdc++ 13
84 // under -Werror, and streaming the streambuf is the idiomatic whole-file
85 // read anyway.
86 std::ostringstream buffer;
87 buffer << file.rdbuf();
88 return loadRulesetFromJson(buffer.str());
89 }

200 [[nodiscard]] static bool scopeMatches(std::string_view pattern,
201 std::string_view scope) noexcept {
202 if (pattern == "all" || pattern == scope) {
203 return true;
204 }
205 if (!pattern.empty() && pattern.back() == '*') {
206 return scope.starts_with(pattern.substr(0, pattern.size() - 1));
207 }
208 std::size_t begin = 0;
209 while (begin <= scope.size()) {
210 const std::size_t end = scope.find('_', begin);
211 const std::string_view segment =
212 scope.substr(begin, end == std::string_view::npos ? std::string_view::npos : end - begin);
213 if (segment == pattern) {
214 return true;
215 }
216 if (end == std::string_view::npos) {
217 break;
218 }
219 begin = end + 1;
220 }
221 return false;
222 }

239 [[nodiscard]] CheckParams withConditionModifiers(const DynamicEntity &actor,
240 const DynamicEntity *target,
241 std::string_view scope,
242 CheckParams params) const {
243 int32_t bonus = 0;
244 int advantage = 0;
245 int disadvantage = 0;
246 bool autoFail = false;
247 std::vector<std::string> bonusDice;
248 accumulateCheckModifiers(actor, scope, "actor", bonus, advantage, disadvantage, autoFail,
249 bonusDice);
250 if (target != nullptr) {
251 accumulateCheckModifiers(*target, scope, "target", bonus, advantage, disadvantage, autoFail,
252 bonusDice);
253 }
254 // Temporary bonus dice from effects (the `bonus_die` effect kind) apply to
255 // the carrier's own checks of the matching scope.
256 for (const BonusDie &die : actor.effects().bonusDice()) {
257 if (scopeMatches(die.scope, scope)) {
258 bonusDice.push_back(die.dice);
259 }
260 }
261 params.situationalModifier += bonus;
262 if (advantage > 0 && disadvantage > 0) {
263 params.advantage = AdvantageMode::None;
264 } else if (advantage > 0) {
265 params.advantage = AdvantageMode::Advantage;
266 } else if (disadvantage > 0) {
267 params.advantage = AdvantageMode::Disadvantage;
268 }
269 params.autoFail = autoFail;
270 for (const std::string &dice : bonusDice) {
271 params.bonusDice.push_back(DiceExpression(dice));
272 }
273 return params;
274 }

280 void accumulateCheckModifiers(const DynamicEntity &entity, std::string_view scope,
281 std::string_view wantedSide, int32_t &bonus, int &advantage,
282 int &disadvantage, bool &autoFail,
283 std::vector<std::string> &bonusDice) const {
284 for (const auto &[conditionId, stacks] : entity.conditions()) {
285 const Json *condition = findCondition(conditionId);
286 if (condition == nullptr || !condition->contains("check_modifiers")) {
287 continue;
288 }
289 const Json &modifiers = condition->at("check_modifiers");
290 if (!modifiers.is_array()) {
291 continue;
292 }
293 for (const Json &mod : modifiers) {
294 accumulateModifierEntry(mod, scope, wantedSide, stacks, bonus, advantage, disadvantage,
295 autoFail, bonusDice);
296 }
297 }
298 for (const std::string &traitId : entity.traits()) {
299 const Json *trait = findDataRecord("traits", traitId);
300 if (trait == nullptr || !trait->contains("check_modifiers")) {
301 continue;
302 }
303 const Json &modifiers = trait->at("check_modifiers");
304 if (!modifiers.is_array()) {
305 continue;
306 }
307 for (const Json &mod : modifiers) {
308 accumulateModifierEntry(mod, scope, wantedSide, 1, bonus, advantage, disadvantage, autoFail,
309 bonusDice);
310 }
311 }
312 }
313
317 void accumulateModifierEntry(const Json &mod, std::string_view scope, std::string_view wantedSide,
318 int32_t stacks, int32_t &bonus, int &advantage, int &disadvantage,
319 bool &autoFail, std::vector<std::string> &bonusDice) const {
320 const std::string side = mod.value("side", "actor");
321 if (side != wantedSide) {
322 return;
323 }
324 if (!scopeMatches(mod.value("scope", "all"), scope)) {
325 return;
326 }
327 const std::string mode = mod.value("mode", "none");
328 if (mode == "advantage") {
329 ++advantage;
330 } else if (mode == "disadvantage") {
331 ++disadvantage;
332 } else if (mode == "auto_fail") {
333 autoFail = true;
334 }
335 if (mod.contains("bonus")) {
336 int32_t modBonus = mod.at("bonus").get<int32_t>();
337 if (mod.value("per_stack", false)) {
338 modBonus *= stacks;
339 }
340 bonus += modBonus;
341 }
342 if (mod.contains("bonus_dice")) {
343 bonusDice.push_back(mod.at("bonus_dice").get<std::string>());
344 }
345 }

418 template <RandomNumberGenerator Rng>
419 [[nodiscard]] CheckResult executeSkillCheck(std::string_view skillId, const DynamicEntity &actor,
420 const CheckParams &params, Rng &rng) const {
421 const SkillDef *skill = m_ruleset.findSkill(skillId);
422 if (skill == nullptr) {
423 throw std::invalid_argument("unknown skill '" + std::string(skillId) + "'");
424 }
425 if (skill->attributes.size() != 3) {
426 throw std::invalid_argument("skill '" + std::string(skillId) +
427 "' is not a three-attribute skill check");
428 }
429 CheckRecipe recipe;
430 recipe.resolution = Resolution::Pool;
431 recipe.dice = "3d20"_dice;
432 recipe.numPoolAttributes = 3;
433 recipe.poolAttributes = {skill->attributes[0], skill->attributes[1], skill->attributes[2]};
434 recipe.poolStat = std::string(skillId);
435 recipe.criticalStyle = CriticalStyle::DoubleRoll;
436 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
437 recipe.grading = Grading::PoolQuality;
438 recipe.difficultyMode = DifficultyMode::ToStat;
439 const CheckParams adjusted = withConditionModifiers(actor, nullptr, "skill", params);
440 return resolveCheck(actor, NullStatProvider{}, recipe, adjusted, rng);
441 }

463 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
464 int32_t rawDamage, const Json &env = {}) {
465 EventData data;
466 data.payload = {{"damage", rawDamage},
467 {"raw_damage", rawDamage},
468 {"attacker_id", actor.id()},
469 {"target_id", target.id()}};
470 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
471 int32_t finalDamage = data.getInt("damage", rawDamage);
472 if (finalDamage < 0) {
473 finalDamage = 0;
474 }
475 // Temporary Hit Points are a buffer: they absorb damage before the real
476 // hit-point pool ("Lose Temporary Hit Points First").
477 if (target.temporaryHitPoints() > 0 && finalDamage > 0) {
478 const int32_t absorbed = std::min(target.temporaryHitPoints(), finalDamage);
479 target.addTemporaryHitPoints(-absorbed);
480 finalDamage -= absorbed;
481 data.payload["temporary_hp_absorbed"] = absorbed;
482 }
483 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
484 data.payload["applied_damage"] = -applied;
485 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
486 return applied;
487 }

593 template <RandomNumberGenerator Rng>
594 [[nodiscard]] SpellResult
595 castSpell(std::string_view spellId, DynamicEntity &actor, DynamicEntity *target,
596 std::string_view resourceId, const CheckParams &params, Rng &rng,
597 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
598 SpellResult result;
599 const Json *spell = findSpell(spellId);
600 if (spell == nullptr) {
601 throw std::invalid_argument("unknown spell '" + std::string(spellId) + "'");
602 }
603 // The actor's restrictions are enforced: a creature that cannot take
604 // actions (incapacitated, stunned, ...) or cannot cast is refused before
605 // any resource is spent, with the reason recorded on the result.
606 if (!actionAllowed(actor, "action") || !actionAllowed(actor, "cast")) {
607 result.cast = false;
608 result.denied = actor.hasRestriction("no_cast") ? "no_cast" : "no_action";
609 return result;
610 }
611 // Cost: `cost` or `ae_cost`, else `level` (one point per level).
612 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
613 result.cost = spell->at("cost").get<int32_t>();
614 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
615 result.cost = spell->at("ae_cost").get<int32_t>();
616 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
617 result.cost = spell->at("level").get<int32_t>();
618 }
619 result.resourceId = std::string(resourceId);
620 if (result.cost > 0 && !resourceId.empty()) {
621 if (actor.resource(resourceId) < result.cost) {
622 result.cast = false; // cannot afford the spell
623 return result;
624 }
625 (void)actor.modifyResource(resourceId, -result.cost);
626 }
627
628 // Casting check (named check type, or an "A/B/C" attribute list).
629 if (spell->contains("check") && spell->at("check").is_string()) {
630 const std::string checkText = spell->at("check").get<std::string>();
631 if (m_ruleset.findCheckType(checkText) != nullptr) {
632 const NullStatProvider noTarget;
633 result.check =
634 target != nullptr
635 ? CheckResolver::resolve(m_ruleset, actor, *target, checkText, params, rng)
636 : CheckResolver::resolve(m_ruleset, actor, noTarget, checkText, params, rng);
637 } else {
638 result.check = resolveSpellCheck(actor, checkText, params, rng);
639 }
640 result.cast = result.check.isSuccess;
641 } else {
642 result.cast = true;
643 }
644
645 // Structured effects take precedence over the bare `damage` field: they
646 // are the extracted, machine-readable form of the spell's prose (saves,
647 // half-on-save damage, conditions, healing, resistances). A spell with no
648 // `effects` falls back to the simple damage field.
649 if (target != nullptr && spell->contains("effects") && spell->at("effects").is_array()) {
650 // The casting check's quality level (The Dark Eye's QL) is threaded into
651 // the effect formulas so QL-scaled spells ("2D6 + QLx2") resolve from
652 // data alone.
653 const int32_t ql = result.check.qualityLevel;
654 const EffectsResult effects =
655 resolveEffects(actor, *target, spell->at("effects"), params, rng, ql, selections);
656 result.appliedDamage = effects.damageDealt;
657 result.choicesRequired = effects.choicesRequired;
658 } else if (target != nullptr && spell->contains("damage")) {
659 // `applyDamage` reports the (negative) pool delta, so negate it into the
660 // positive "damage dealt" the caller expects.
661 const int32_t damage = readVariantValue(spell->at("damage"), Variance::Random, rng);
662 const std::string hitPool = resolveHitPointPoolId();
663 if (!hitPool.empty()) {
664 result.appliedDamage = -applyDamage(actor, *target, hitPool, damage);
665 }
666 }
667 return result;
668 }

695 template <RandomNumberGenerator Rng>
696 [[nodiscard]] AfflictionResult applyAffliction(std::string_view section, std::string_view id,
697 DynamicEntity &victim, const CheckParams &params,
698 Rng &rng) {
699 AfflictionResult result;
700 const Json *affliction = findDataRecord(section, id);
701 if (affliction == nullptr) {
702 throw std::invalid_argument("unknown " + std::string(section) + " '" + std::string(id) + "'");
703 }
704 if (affliction->contains("save") && affliction->at("save").is_string()) {
705 const std::string save = affliction->at("save").get<std::string>();
706 if (m_ruleset.hasStat(save)) {
707 result.saveRolled = true;
708 CheckRecipe recipe;
709 recipe.resolution = Resolution::Threshold;
710 recipe.dice = "1d20"_dice;
711 recipe.comparison = Comparison::LessEqual;
712 recipe.thresholdSource = ThresholdSource::ActorStat;
713 recipe.thresholdStat = save;
714 recipe.difficultyMode = DifficultyMode::ToStat;
715 const CheckResult saveResult =
716 resolveCheck(victim, NullStatProvider{}, recipe, params, rng);
717 if (saveResult.isSuccess) {
718 result.resisted = true;
719 return result;
720 }
721 }
722 }
723 if (affliction->contains("effects") && affliction->at("effects").is_array()) {
724 std::vector<std::string> appliedConditions;
725 for (const Json &effect : affliction->at("effects")) {
726 const std::string stat = effect.value("stat", "");
727 if (stat.empty()) {
728 continue;
729 }
730 int32_t amount = 1;
731 if (effect.contains("amount")) {
732 amount = readVariantValue(effect.at("amount"), Variance::Random, rng);
733 }
734 if (m_ruleset.isResourcePool(stat)) {
735 (void)victim.modifyResource(stat, -amount);
736 ++result.effectsApplied;
737 } else if (effect.contains("condition") && effect.at("condition").is_string()) {
738 const std::string conditionId = effect.at("condition").get<std::string>();
739 victim.addCondition(conditionId, amount);
740 appliedConditions.push_back(conditionId);
741 ++result.effectsApplied;
742 } else if (m_ruleset.isCondition(stat)) {
743 victim.addCondition(stat, amount);
744 appliedConditions.push_back(stat);
745 ++result.effectsApplied;
746 } else if (m_ruleset.findAttribute(stat) != nullptr) {
747 victim.setBaseAttribute(stat, victim.baseAttribute(stat) - amount);
748 ++result.effectsApplied;
749 }
750 }
751 // Remember what was applied so an application can cure it later.
752 if (result.effectsApplied > 0) {
753 victim.addAffliction(section, id, appliedConditions);
754 EventData data;
755 data.payload = {{"section", std::string(section)},
756 {"affliction", std::string(id)},
757 {"victim_id", victim.id()},
758 {"effects_applied", result.effectsApplied}};
759 fireEvent(EventType::OnAfflictionApplied, data, victim, nullptr, Json{});
760 }
761 }
762 return result;
763 }

790 [[nodiscard]] std::expected<Money, BookkeepingError> itemPrice(std::string_view itemId) const {
791 if (!m_loaded) {
792 return std::unexpected(BookkeepingError::NoRuleset);
793 }
794 const Json *item = findItem(itemId);
795 if (item == nullptr) {
796 return std::unexpected(BookkeepingError::UnknownItem);
797 }
798 if (!m_ruleset.hasCurrency()) {
799 return std::unexpected(BookkeepingError::UnknownCurrency);
800 }
801 const Json *price = nullptr;
802 if (item->contains("cost")) {
803 price = &item->at("cost");
804 } else if (item->contains("price")) {
805 price = &item->at("price");
806 } else if (item->contains("value")) {
807 price = &item->at("value");
808 }
809 if (price == nullptr) {
810 return Money{0};
811 }
812 if (price->is_number_integer()) {
813 return Money{price->get<int64_t>()};
814 }
815 if (price->is_string()) {

1346 struct EffectsResult {
1347 int32_t damageDealt{0};
1348 int32_t conditionsApplied{0};
1349 int32_t healingDone{0};
1350 int32_t savesPassed{0};
1351 int32_t statBonusesApplied{0};
1352 std::vector<std::string> choicesRequired;
1353 };
1354
1393 template <RandomNumberGenerator Rng>
1394 [[nodiscard]] EffectsResult
1395 resolveEffects(DynamicEntity &source, DynamicEntity &target, const Json &effects,
1396 const CheckParams &params, Rng &rng, int32_t qualityLevel = 0,
1397 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
1398 EffectsResult result;
1399 if (!effects.is_array()) {
1400 return result;
1401 }
1402 const std::string hitPool = resolveHitPointPoolId();
1403 for (const Json &effect : effects) {
1404 const std::string kind = effect.value("kind", "");
1405 bool applied = true; // whether the base effect actually took effect
1406 if (kind == "damage") {
1407 if (!effect.contains("dice")) {
1408 continue;
1409 }
1410 int32_t final =
1411 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1412 if (effect.contains("add")) {
1413 final += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1414 }
1415 if (effect.contains("save")) {
1416 if (resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
1417 ++result.savesPassed;
1418 if (effect.at("save").value("on_success", "none") == "half") {
1419 final /= 2;
1420 } else {
1421 final = 0;
1422 applied = false;
1423 }
1424 }
1425 } else if (effect.contains("attack")) {
1426 const Json &attack = effect.at("attack");
1427 if (!resolveAttack(source, target, attack, params, rng)) {
1428 if (attack.value("on_miss", "none") == "half") {
1429 final /= 2;
1430 } else {
1431 final = 0;
1432 applied = false;
1433 }
1434 }
1435 }
1436 if (final > 0 && !hitPool.empty()) {
1437 // applyDamage reports the negative pool delta; negate for "dealt".
1438 result.damageDealt += -applyDamage(source, target, hitPool, final);
1439 }
1440 } else if (kind == "condition") {
1441 const std::string conditionId = effect.value("condition", "");
1442 if (conditionId.empty()) {
1443 continue;
1444 }
1445 if (effect.contains("save") &&
1446 resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
1447 ++result.savesPassed;
1448 applied = false;
1449 } else if (effect.contains("attack") &&
1450 !resolveAttack(source, target, effect.at("attack"), params, rng)) {
1451 applied = false;
1452 }
1453 if (!applied) {
1454 continue;
1455 }
1456 // `stacks` may be a plain number or a formula over the source's stats
1457 // and the casting check's quality level (`env.ql`) — the latter lets
1458 // QL-scaled conditions ("QL 3: 2 levels of Pain") resolve from data.
1459 int32_t stacks = 1;
1460 if (effect.contains("stacks")) {
1461 const Json &stacksJson = effect.at("stacks");
1462 if (stacksJson.is_number()) {
1463 stacks = stacksJson.get<int32_t>();
1464 } else if (stacksJson.is_string()) {
1465 const Json env = {{"ql", qualityLevel}};
1466 const EntityContext context(source, &target, env, Json{});
1467 stacks = math::toStat(Expression(stacksJson.get<std::string>()).evaluate(context));
1468 }
1469 }
1470 target.addCondition(conditionId, stacks);
1471 const int32_t duration = effect.value("duration", 0);
1472 if (duration > 0) {
1473 target.effects().add(ActiveEffect{conditionId, stacks, duration, "effect"});
1474 }
1475 ++result.conditionsApplied;
1476 } else if (kind == "heal") {
1477 if (!effect.contains("dice") || hitPool.empty()) {
1478 continue;
1479 }
1480 int32_t amount =
1481 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1482 if (effect.contains("add")) {
1483 amount += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1484 }
1485 result.healingDone += target.modifyResource(hitPool, amount);
1486 } else if (kind == "temp_hp") {
1487 if (!effect.contains("dice")) {
1488 continue;
1489 }
1490 int32_t amount =
1491 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1492 if (effect.contains("add")) {
1493 amount += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1494 }
1495 target.addTemporaryHitPoints(amount);
1496 } else if (kind == "resist") {
1497 if (effect.contains("types")) {
1498 for (const Json &type : effect.at("types")) {
1499 target.addResistance(type.get<std::string>());
1500 }
1501 }
1502 } else if (kind == "stat_bonus") {
1503 const std::string stat = effect.value("stat", "");
1504 if (stat.empty()) {
1505 continue;
1506 }
1507 const int32_t value =
1508 effect.contains("add")
1509 ? evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel)
1510 : 0;
1511 const int32_t duration = effect.value("duration", 0);
1512 target.effects().addBonus(StatBonus{stat, value, duration, "effect"});
1513 ++result.statBonusesApplied;
1514 } else if (kind == "bonus_die") {
1515 const std::string dice = effect.value("dice", "");
1516 if (dice.empty()) {
1517 continue;
1518 }
1519 const std::string scope = effect.value("scope", "all");
1520 const int32_t duration = effect.value("duration", 0);
1521 target.effects().addBonusDie(BonusDie{dice, scope, duration, "effect"});
1522 } else if (kind == "options") {
1523 // A caller-side choice: the engine never picks for the caller. Without
1524 // a selection a required group is reported in choicesRequired (nothing
1525 // applied); an optional group is skipped silently; with a selection the
1526 // chosen option resolves (including its own save/attack/ongoing logic).
1527 const std::string groupId = effect.value("id", "");
1528 const Json &options = effect.at("options");
1529 int32_t chosen = -1;
1530 if (!groupId.empty() && selections != nullptr) {
1531 const auto it = selections->find(groupId);
1532 if (it != selections->end()) {
1533 chosen = it->second;
1534 }
1535 }
1536 if (chosen < 0) {
1537 if (!effect.value("optional", false)) {
1538 result.choicesRequired.push_back(groupId);
1539 }
1540 continue;
1541 }
1542 if (!options.is_array() || chosen >= static_cast<int32_t>(options.size())) {
1543 continue;
1544 }
1545 Json batch = Json::array();
1546 batch.push_back(options.at(static_cast<std::size_t>(chosen)));
1547 const EffectsResult sub =
1548 resolveEffects(source, target, batch, params, rng, qualityLevel, selections);
1549 result.damageDealt += sub.damageDealt;
1550 result.conditionsApplied += sub.conditionsApplied;
1551 result.healingDone += sub.healingDone;
1552 result.savesPassed += sub.savesPassed;
1553 result.statBonusesApplied += sub.statBonusesApplied;
1554 result.choicesRequired.insert(result.choicesRequired.end(), sub.choicesRequired.begin(),

1655 }
1656 }

1812 parsePriceString(std::string_view text) const {
1813 std::string trimmed = trimWhitespace(text);
1814 if (trimmed.empty() || trimmed == "-" || trimmed == "—" || trimmed == "Varies" ||
1815 trimmed == "varies") {
1816 return Money{0};
1817 }
1818 std::size_t pos = 0;
1819 bool sawDigit = false;
1820 while (pos < trimmed.size()) {
1821 const char c = trimmed[pos];

1823 if (c >= '0' && c <= '9') {
1824 sawDigit = true;
1825 }
1826 ++pos;
1827 continue;
1828 }
1829 break;
1830 }

1832 return Money{0};
1833 }
1834 std::string number = trimmed.substr(0, pos);
1835 std::string symbol = trimWhitespace(trimmed.substr(pos));
1836 number.erase(std::remove(number.begin(), number.end(), ','), number.end());
1837 number.erase(std::remove(number.begin(), number.end(), ' '), number.end());

1843 if (d.symbol == symbol || d.id == symbol) {

1945 std::string_view checkText, const CheckParams &params,
1946 Rng &rng) const {
1947 CheckRecipe recipe;
1948 recipe.resolution = Resolution::Pool;
1949 recipe.dice = "3d20"_dice;
1950 recipe.criticalStyle = CriticalStyle::DoubleRoll;
1951 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
1952 recipe.grading = Grading::PoolQuality;
1953 recipe.difficultyMode = DifficultyMode::ToStat;
1954
1955 const std::size_t paren = checkText.find('(');
1956 const std::string_view head =
1957 paren == std::string_view::npos ? checkText : checkText.substr(0, paren);
1958 std::size_t begin = 0;
1959 while (begin < head.size()) {
1960 while (begin < head.size() && (head[begin] == ' ' || head[begin] == '/')) {
1961 ++begin;
1962 }
1963 if (begin >= head.size()) {
1964 break;
1965 }
1966 std::size_t end = begin;
1967 while (end < head.size() && head[end] != '/' && head[end] != ' ') {
1968 ++end;
1969 }
1970 if (recipe.numPoolAttributes < 3) {
1971 recipe.poolAttributes[recipe.numPoolAttributes++] =
1972 std::string(head.substr(begin, end - begin));
1973 }
1974 begin = end;
1975 }
1976 return resolveCheck(actor, NullStatProvider{}, recipe, params, rng);
1977 }
