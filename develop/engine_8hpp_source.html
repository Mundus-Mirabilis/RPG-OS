68 bool loadRulesetFromFile(std::string_view path) {
69 const std::string pathStr(path);
70 std::ifstream file(pathStr);
71 if (!file) {
72 m_lastError = "cannot open file '" + pathStr + "'";
73 m_loaded = false;
74 return false;
75 }
76 // Read via rdbuf() rather than the istreambuf_iterator range idiom: the
77 // iterator path trips a -Wnull-dereference false positive in libstdc++ 13
78 // under -Werror, and streaming the streambuf is the idiomatic whole-file
79 // read anyway.
80 std::ostringstream buffer;
81 buffer << file.rdbuf();
82 return loadRulesetFromJson(buffer.str());
83 }

120 Variance variance, Rng &rng) {
121 if (!m_loaded || !m_ruleset.data.is_object() || !m_ruleset.data.contains("archetypes")) {
122 return nullptr;
123 }
124 for (const Json &archetype : m_ruleset.data.at("archetypes")) {
125 if (archetype.value("id", "") == archetypeId) {
126 auto entity = std::make_shared<DynamicEntity>(m_ruleset, std::string(archetypeId));

246 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
247 int32_t rawDamage, const Json &env = {}) {
248 EventData data;
249 data.payload = {{"damage", rawDamage},
250 {"raw_damage", rawDamage},
251 {"attacker_id", actor.id()},
252 {"target_id", target.id()}};
253 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
254 int32_t finalDamage = data.getInt("damage", rawDamage);
255 if (finalDamage < 0) {
256 finalDamage = 0;
257 }
258 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
259 data.payload["applied_damage"] = -applied;
260 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
261 return applied;
262 }

298 void fireEvent(EventType type, EventData &data, DynamicEntity &actor, DynamicEntity *target,
299 const Json &env) {
300 runRuleTriggers(type, data, actor, target, env);
301 m_eventBus.dispatch(type, data);
302 }
303
307 void runRuleTriggers(EventType type, EventData &data, DynamicEntity &actor, DynamicEntity *target,
308 const Json &env) {
309 for (const EventTriggerDef &trigger : m_ruleset.eventTriggers) {
310 if (trigger.trigger != type) {
311 continue;
312 }
313 if (!trigger.conditionText.empty()) {
314 const EntityContext context(actor, target, env, data.payload);
315 if (trigger.condition.evaluate(context) == 0.0) {
316 continue;
317 }
318 }
319 for (const EventActionDef &action : trigger.actions) {
320 executeAction(action, data, actor, target, env);
321 }
322 }
323 }
324
333 void executeAction(const EventActionDef &action, EventData &data, DynamicEntity &actor,
334 DynamicEntity *target, const Json &env) {
335 const EntityContext context(actor, target, env, data.payload);
336 DynamicEntity &subject = target != nullptr ? *target : actor;
337 if (action.type == "modify_event_damage") {
338 const double value = action.formula.evaluate(context);
339 data.payload["damage"] = value;
340 } else if (action.type == "consume_resource") {
341 (void)subject.modifyResource(action.resource, -action.amount);
342 } else if (action.type == "apply_condition") {
343 const int32_t stacks =
344 action.stacksText.empty() ? 1 : math::toStat(action.stacks.evaluate(context));
345 subject.addCondition(action.condition, stacks);
346 }
347 }
