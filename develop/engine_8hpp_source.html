68 bool loadRulesetFromFile(std::string_view path) {
69 const std::string pathStr(path);
70 std::ifstream file(pathStr);
71 if (!file) {
72 m_lastError = "cannot open file '" + pathStr + "'";
73 m_loaded = false;
74 return false;
75 }
76 // Read via rdbuf() rather than the istreambuf_iterator range idiom: the
77 // iterator path trips a -Wnull-dereference false positive in libstdc++ 13
78 // under -Werror, and streaming the streambuf is the idiomatic whole-file
79 // read anyway.
80 std::ostringstream buffer;
81 buffer << file.rdbuf();
82 return loadRulesetFromJson(buffer.str());
83 }

213 [[nodiscard]] CheckResult executeSkillCheck(std::string_view skillId, const DynamicEntity &actor,
214 const CheckParams &params, Rng &rng) const {
215 const SkillDef *skill = m_ruleset.findSkill(skillId);
216 if (skill == nullptr) {

225 recipe.dice = "3d20"_dice;
226 recipe.numPoolAttributes = 3;
227 recipe.poolAttributes = {skill->attributes[0], skill->attributes[1], skill->attributes[2]};
228 recipe.poolStat = std::string(skillId);
229 recipe.criticalStyle = CriticalStyle::DoubleRoll;
230 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
231 recipe.grading = Grading::PoolQuality;
232 recipe.difficultyMode = DifficultyMode::ToStat;
233 return resolveCheck(actor, NullStatProvider{}, recipe, params, rng);

256 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
257 int32_t rawDamage, const Json &env = {}) {
258 EventData data;
259 data.payload = {{"damage", rawDamage},
260 {"raw_damage", rawDamage},
261 {"attacker_id", actor.id()},
262 {"target_id", target.id()}};
263 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
264 int32_t finalDamage = data.getInt("damage", rawDamage);
265 if (finalDamage < 0) {
266 finalDamage = 0;
267 }
268 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
269 data.payload["applied_damage"] = -applied;
270 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
271 return applied;
272 }

355 template <RandomNumberGenerator Rng>
356 [[nodiscard]] SpellResult castSpell(std::string_view spellId, DynamicEntity &actor,
357 DynamicEntity *target, std::string_view resourceId,
358 const CheckParams &params, Rng &rng) {
359 SpellResult result;
360 const Json *spell = findSpell(spellId);
361 if (spell == nullptr) {
362 throw std::invalid_argument("unknown spell '" + std::string(spellId) + "'");
363 }
364 // Cost: `cost` or `ae_cost`, else `level` (one point per level).
365 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
366 result.cost = spell->at("cost").get<int32_t>();
367 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
368 result.cost = spell->at("ae_cost").get<int32_t>();
369 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
370 result.cost = spell->at("level").get<int32_t>();
371 }
372 result.resourceId = std::string(resourceId);
373 if (result.cost > 0 && !resourceId.empty()) {
374 if (actor.resource(resourceId) < result.cost) {
375 result.cast = false; // cannot afford the spell
376 return result;
377 }
378 (void)actor.modifyResource(resourceId, -result.cost);
379 }
380
381 // Casting check (named check type, or an "A/B/C" attribute list).
382 if (spell->contains("check") && spell->at("check").is_string()) {
383 const std::string checkText = spell->at("check").get<std::string>();
384 if (m_ruleset.findCheckType(checkText) != nullptr) {
385 const NullStatProvider noTarget;
386 result.check =
387 target != nullptr
388 ? CheckResolver::resolve(m_ruleset, actor, *target, checkText, params, rng)
389 : CheckResolver::resolve(m_ruleset, actor, noTarget, checkText, params, rng);
390 } else {
391 result.check = resolveSpellCheck(actor, checkText, params, rng);
392 }
393 result.cast = result.check.isSuccess;
394 } else {
395 result.cast = true;
396 }
397
398 // Damage: roll the spell's damage and apply it to the target's hit points.
399 // `applyDamage` reports the (negative) pool delta, so negate it into the
400 // positive "damage dealt" the caller expects.
401 if (target != nullptr && spell->contains("damage")) {
402 const int32_t damage = readVariantValue(spell->at("damage"), Variance::Random, rng);
403 const std::string hitPool = resolveHitPointPoolId();
404 if (!hitPool.empty()) {
405 result.appliedDamage = -applyDamage(actor, *target, hitPool, damage);
406 }
407 }
408 return result;
409 }

429 template <RandomNumberGenerator Rng>
430 [[nodiscard]] AfflictionResult applyAffliction(std::string_view section, std::string_view id,
431 DynamicEntity &victim, const CheckParams &params,
432 Rng &rng) {
433 AfflictionResult result;
434 const Json *affliction = findDataRecord(section, id);
435 if (affliction == nullptr) {
436 throw std::invalid_argument("unknown " + std::string(section) + " '" + std::string(id) + "'");
437 }
438 if (affliction->contains("save") && affliction->at("save").is_string()) {
439 const std::string save = affliction->at("save").get<std::string>();
440 if (m_ruleset.hasStat(save)) {
441 result.saveRolled = true;
442 CheckRecipe recipe;
443 recipe.resolution = Resolution::Threshold;
444 recipe.dice = "1d20"_dice;
445 recipe.comparison = Comparison::LessEqual;
446 recipe.thresholdSource = ThresholdSource::ActorStat;
447 recipe.thresholdStat = save;
448 recipe.difficultyMode = DifficultyMode::ToStat;
449 const CheckResult saveResult =
450 resolveCheck(victim, NullStatProvider{}, recipe, params, rng);
451 if (saveResult.isSuccess) {
452 result.resisted = true;
453 return result;
454 }
455 }
456 }
457 if (affliction->contains("effects") && affliction->at("effects").is_array()) {
458 for (const Json &effect : affliction->at("effects")) {
459 const std::string stat = effect.value("stat", "");
460 if (stat.empty()) {
461 continue;
462 }
463 int32_t amount = 1;
464 if (effect.contains("amount")) {
465 amount = readVariantValue(effect.at("amount"), Variance::Random, rng);
466 }
467 if (m_ruleset.isResourcePool(stat)) {
468 (void)victim.modifyResource(stat, -amount);
469 ++result.effectsApplied;
470 } else if (effect.contains("condition") && effect.at("condition").is_string()) {
471 victim.addCondition(effect.at("condition").get<std::string>(), amount);
472 ++result.effectsApplied;
473 } else if (m_ruleset.isCondition(stat)) {
474 victim.addCondition(stat, amount);
475 ++result.effectsApplied;
476 } else if (m_ruleset.findAttribute(stat) != nullptr) {
477 victim.setBaseAttribute(stat, victim.baseAttribute(stat) - amount);
478 ++result.effectsApplied;
479 }
480 }
481 }
482 return result;
483 }

526 void fireEvent(EventType type, EventData &data, DynamicEntity &actor, DynamicEntity *target,

528 runRuleTriggers(type, data, actor, target, env);
529 m_eventBus.dispatch(type, data);

531
535 void runRuleTriggers(EventType type, EventData &data, DynamicEntity &actor, DynamicEntity *target,
536 const Json &env) {
537 for (const EventTriggerDef &trigger : m_ruleset.eventTriggers) {
538 if (trigger.trigger != type) {
539 continue;
540 }
541 if (!trigger.conditionText.empty()) {
542 const EntityContext context(actor, target, env, data.payload);
543 if (trigger.condition.evaluate(context) == 0.0) {
544 continue;
545 }
546 }
547 for (const EventActionDef &action : trigger.actions) {
548 executeAction(action, data, actor, target, env);
549 }
550 }
551 }
552
561 void executeAction(const EventActionDef &action, EventData &data, DynamicEntity &actor,
562 DynamicEntity *target, const Json &env) {
563 const EntityContext context(actor, target, env, data.payload);
564 DynamicEntity &subject = target != nullptr ? *target : actor;
565 if (action.type == "modify_event_damage") {
566 const double value = action.formula.evaluate(context);
567 data.payload["damage"] = value;
568 } else if (action.type == "consume_resource") {
569 (void)subject.modifyResource(action.resource, -action.amount);
570 } else if (action.type == "apply_condition") {
571 const int32_t stacks =
572 action.stacksText.empty() ? 1 : math::toStat(action.stacks.evaluate(context));
573 subject.addCondition(action.condition, stacks);
574 }
575 }

597 template <RandomNumberGenerator Rng>
598 [[nodiscard]] CheckResult resolveSpellCheck(const DynamicEntity &actor,
599 std::string_view checkText, const CheckParams &params,
600 Rng &rng) const {
601 CheckRecipe recipe;
602 recipe.resolution = Resolution::Pool;
603 recipe.dice = "3d20"_dice;
604 recipe.criticalStyle = CriticalStyle::DoubleRoll;
605 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
606 recipe.grading = Grading::PoolQuality;
607 recipe.difficultyMode = DifficultyMode::ToStat;
608
609 const std::size_t paren = checkText.find('(');
610 const std::string_view head =
611 paren == std::string_view::npos ? checkText : checkText.substr(0, paren);
612 std::size_t begin = 0;
613 while (begin < head.size()) {
614 while (begin < head.size() && (head[begin] == ' ' || head[begin] == '/')) {
615 ++begin;
616 }
617 if (begin >= head.size()) {
618 break;
619 }
620 std::size_t end = begin;
621 while (end < head.size() && head[end] != '/' && head[end] != ' ') {
622 ++end;
623 }
624 if (recipe.numPoolAttributes < 3) {
625 recipe.poolAttributes[recipe.numPoolAttributes++] =
626 std::string(head.substr(begin, end - begin));
627 }
628 begin = end;
629 }
630 return resolveCheck(actor, NullStatProvider{}, recipe, params, rng);
631 }
