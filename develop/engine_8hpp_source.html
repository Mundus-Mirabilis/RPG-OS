74 bool loadRulesetFromFile(std::string_view path) {
75 const std::string pathStr(path);
76 std::ifstream file(pathStr);
77 if (!file) {
78 m_lastError = "cannot open file '" + pathStr + "'";
79 m_loaded = false;
80 return false;
81 }
82 // Read via rdbuf() rather than the istreambuf_iterator range idiom: the
83 // iterator path trips a -Wnull-dereference false positive in libstdc++ 13
84 // under -Werror, and streaming the streambuf is the idiomatic whole-file
85 // read anyway.
86 std::ostringstream buffer;
87 buffer << file.rdbuf();
88 return loadRulesetFromJson(buffer.str());

202 [[nodiscard]] static bool scopeMatches(std::string_view pattern,
203 std::string_view scope) noexcept {
204 if (pattern == "all" || pattern == scope) {
205 return true;
206 }
207 if (!pattern.empty() && pattern.back() == '*') {
208 return scope.starts_with(pattern.substr(0, pattern.size() - 1));
209 }
210 std::size_t begin = 0;
211 while (begin <= scope.size()) {
212 const std::size_t end = scope.find('_', begin);
213 const std::string_view segment =
214 scope.substr(begin, end == std::string_view::npos ? std::string_view::npos : end - begin);
215 if (segment == pattern) {
216 return true;
217 }
218 if (end == std::string_view::npos) {
219 break;
220 }
221 begin = end + 1;
222 }
223 return false;
224 }

241 [[nodiscard]] CheckParams withConditionModifiers(const DynamicEntity &actor,
242 const DynamicEntity *target,
243 std::string_view scope,
244 CheckParams params) const {
245 int32_t bonus = 0;
246 int advantage = 0;
247 int disadvantage = 0;
248 bool autoFail = false;
249 std::vector<std::string> bonusDice;
250 accumulateCheckModifiers(actor, scope, "actor", bonus, advantage, disadvantage, autoFail,
251 bonusDice);
252 if (target != nullptr) {
253 accumulateCheckModifiers(*target, scope, "target", bonus, advantage, disadvantage, autoFail,
254 bonusDice);
255 }
256 // Temporary bonus dice from effects (the `bonus_die` effect kind) apply to
257 // the carrier's own checks of the matching scope.
258 for (const BonusDie &die : actor.effects().bonusDice()) {
259 if (scopeMatches(die.scope, scope)) {
260 bonusDice.push_back(die.dice);
261 }
262 }
263 params.situationalModifier += bonus;
264 if (advantage > 0 && disadvantage > 0) {
265 params.advantage = AdvantageMode::None;
266 } else if (advantage > 0) {
267 params.advantage = AdvantageMode::Advantage;
268 } else if (disadvantage > 0) {
269 params.advantage = AdvantageMode::Disadvantage;
270 }
271 params.autoFail = autoFail;
272 for (const std::string &dice : bonusDice) {
273 params.bonusDice.push_back(DiceExpression(dice));
274 }
275 return params;
276 }

282 void accumulateCheckModifiers(const DynamicEntity &entity, std::string_view scope,
283 std::string_view wantedSide, int32_t &bonus, int &advantage,
284 int &disadvantage, bool &autoFail,
285 std::vector<std::string> &bonusDice) const {
286 for (const auto &[conditionId, stacks] : entity.conditions()) {
287 const Json *condition = findCondition(conditionId);
288 if (condition == nullptr || !condition->contains("check_modifiers")) {
289 continue;
290 }
291 const Json &modifiers = condition->at("check_modifiers");
292 if (!modifiers.is_array()) {
293 continue;
294 }
295 for (const Json &mod : modifiers) {
296 accumulateModifierEntry(mod, scope, wantedSide, stacks, bonus, advantage, disadvantage,
297 autoFail, bonusDice);
298 }
299 }
300 for (const std::string &traitId : entity.traits()) {
301 const Json *trait = findDataRecord("traits", traitId);
302 if (trait == nullptr || !trait->contains("check_modifiers")) {
303 continue;
304 }
305 const Json &modifiers = trait->at("check_modifiers");
306 if (!modifiers.is_array()) {
307 continue;
308 }
309 for (const Json &mod : modifiers) {
310 accumulateModifierEntry(mod, scope, wantedSide, 1, bonus, advantage, disadvantage, autoFail,
311 bonusDice);
312 }
313 }
314 }
315
319 void accumulateModifierEntry(const Json &mod, std::string_view scope, std::string_view wantedSide,
320 int32_t stacks, int32_t &bonus, int &advantage, int &disadvantage,
321 bool &autoFail, std::vector<std::string> &bonusDice) const {
322 const std::string side = mod.value("side", "actor");
323 if (side != wantedSide) {
324 return;
325 }
326 if (!scopeMatches(mod.value("scope", "all"), scope)) {
327 return;
328 }
329 const std::string mode = mod.value("mode", "none");
330 if (mode == "advantage") {
331 ++advantage;
332 } else if (mode == "disadvantage") {
333 ++disadvantage;
334 } else if (mode == "auto_fail") {
335 autoFail = true;
336 }
337 if (mod.contains("bonus")) {
338 int32_t modBonus = mod.at("bonus").get<int32_t>();
339 if (mod.value("per_stack", false)) {
340 modBonus *= stacks;
341 }
342 bonus += modBonus;
343 }
344 if (mod.contains("bonus_dice")) {
345 bonusDice.push_back(mod.at("bonus_dice").get<std::string>());
346 }
347 }

431 template <RandomNumberGenerator Rng>
432 [[nodiscard]] CheckResult executeSkillCheck(std::string_view skillId, const DynamicEntity &actor,
433 const CheckParams &params, Rng &rng) const {
434 const SkillDef *skill = m_ruleset.findSkill(skillId);
435 if (skill == nullptr) {
436 throw std::invalid_argument("unknown skill '" + std::string(skillId) + "'");
437 }
438 if (skill->attributes.size() != 3) {
439 throw std::invalid_argument("skill '" + std::string(skillId) +
440 "' is not a three-attribute skill check");
441 }
442 CheckRecipe recipe;
443 recipe.resolution = Resolution::Pool;
444 recipe.dice = "3d20"_dice;
445 recipe.numPoolAttributes = 3;
446 recipe.poolAttributes = {skill->attributes[0], skill->attributes[1], skill->attributes[2]};
447 recipe.poolStat = std::string(skillId);
448 recipe.criticalStyle = CriticalStyle::DoubleRoll;
449 recipe.fumbleStyle = CriticalStyle::DoubleRoll;
450 recipe.grading = Grading::PoolQuality;
451 recipe.difficultyMode = DifficultyMode::ToStat;
452 const CheckParams adjusted = withConditionModifiers(actor, nullptr, "skill", params);
453 announceCheckStart(skillId, actor);
454 const CheckResult result = resolveCheck(actor, NullStatProvider{}, recipe, adjusted, rng);
455 announceCheckResult(skillId, actor, result);
456 return result;
457 }

479 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
480 int32_t rawDamage, const Json &env = {}) {
481 EventData data;
482 data.payload = {{"damage", rawDamage}, {"raw_damage", rawDamage},
483 {"attacker_id", actor.id()}, {"attacker_instance_id", actor.entityId().value},
484 {"target_id", target.id()}, {"target_instance_id", target.entityId().value}};
485 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
486 int32_t finalDamage = data.getInt("damage", rawDamage);
487 if (finalDamage < 0) {
488 finalDamage = 0;
489 }
490 // Temporary Hit Points are a buffer: they absorb damage before the real
491 // hit-point pool ("Lose Temporary Hit Points First").
492 if (target.temporaryHitPoints() > 0 && finalDamage > 0) {
493 const int32_t absorbed = std::min(target.temporaryHitPoints(), finalDamage);
494 target.addTemporaryHitPoints(-absorbed);
495 finalDamage -= absorbed;
496 data.payload["temporary_hp_absorbed"] = absorbed;
497 }
498 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
499 data.payload["applied_damage"] = -applied;
500 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
501 return applied;
502 }

513 [[nodiscard]] const Json *findDataRecord(std::string_view section, std::string_view id) const {

519 return nullptr;

608 template <RandomNumberGenerator Rng>
609 [[nodiscard]] SpellResult
610 castSpell(std::string_view spellId, DynamicEntity &actor, DynamicEntity *target,
611 std::string_view resourceId, const CheckParams &params, Rng &rng,
612 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
613 SpellResult result;
614 const Json *spell = findSpell(spellId);
615 if (spell == nullptr) {
616 throw std::invalid_argument("unknown spell '" + std::string(spellId) + "'");
617 }
618 // The actor's restrictions are enforced: a creature that cannot take
619 // actions (incapacitated, stunned, ...) or cannot cast is refused before
620 // any resource is spent, with the reason recorded on the result.
621 if (!actionAllowed(actor, "action") || !actionAllowed(actor, "cast")) {
622 result.cast = false;
623 result.denied = actor.hasRestriction("no_cast") ? "no_cast" : "no_action";
624 return result;
625 }
626 // Cost: `cost` or `ae_cost`, else `level` (one point per level).
627 if (spell->contains("cost") && spell->at("cost").is_number_integer()) {
628 result.cost = spell->at("cost").get<int32_t>();
629 } else if (spell->contains("ae_cost") && spell->at("ae_cost").is_number_integer()) {
630 result.cost = spell->at("ae_cost").get<int32_t>();
631 } else if (spell->contains("level") && spell->at("level").is_number_integer()) {
632 result.cost = spell->at("level").get<int32_t>();
633 }
634 result.resourceId = std::string(resourceId);
635 if (result.cost > 0 && !resourceId.empty()) {
636 if (actor.resource(resourceId) < result.cost) {
637 result.cast = false; // cannot afford the spell
638 return result;
639 }
640 (void)actor.modifyResource(resourceId, -result.cost);
641 }
642
643 // Casting check (named check type, or an "A/B/C" attribute list).
644 if (spell->contains("check") && spell->at("check").is_string()) {
645 const std::string checkText = spell->at("check").get<std::string>();
646 if (m_ruleset.findCheckType(checkText) != nullptr) {
647 const NullStatProvider noTarget;
648 result.check =
649 target != nullptr
650 ? CheckResolver::resolve(m_ruleset, actor, *target, checkText, params, rng)
651 : CheckResolver::resolve(m_ruleset, actor, noTarget, checkText, params, rng);
652 } else {
653 result.check = resolveSpellCheck(actor, checkText, params, rng);
654 }
655 result.cast = result.check.isSuccess;
656 } else {
657 result.cast = true;
658 }
659
660 // Structured effects take precedence over the bare `damage` field: they
661 // are the extracted, machine-readable form of the spell's prose (saves,
662 // half-on-save damage, conditions, healing, resistances). A spell with no
663 // `effects` falls back to the simple damage field.
664 if (target != nullptr && spell->contains("effects") && spell->at("effects").is_array()) {
665 // The casting check's quality level (The Dark Eye's QL) is threaded into
666 // the effect formulas so QL-scaled spells ("2D6 + QLx2") resolve from
667 // data alone.
668 const int32_t ql = result.check.qualityLevel;
669 const EffectsResult effects =
670 resolveEffects(actor, *target, spell->at("effects"), params, rng, ql, selections);
671 result.appliedDamage = effects.damageDealt;
672 result.choicesRequired = effects.choicesRequired;
673 } else if (target != nullptr && spell->contains("damage")) {
674 // `applyDamage` reports the (negative) pool delta, so negate it into the
675 // positive "damage dealt" the caller expects.
676 const int32_t damage = readVariantValue(spell->at("damage"), Variance::Random, rng);
677 const std::string hitPool = resolveHitPointPoolId();
678 if (!hitPool.empty()) {
679 result.appliedDamage = -applyDamage(actor, *target, hitPool, damage);
680 }
681 }
682
683 // Announce the cast attempt (cast = true when it went off). Refused casts
684 // (not allowed to act / not enough resource) return before reaching here
685 // and are not announced; a failed casting check still announces with
686 // cast = false so UI can narrate the fizzle.
687 EventData data;
688 data.payload = {{"spell", std::string(spellId)},
689 {"actor_id", actor.id()},
690 {"instance_id", actor.entityId().value},
691 {"cast", result.cast}};
692 fireEvent(EventType::OnSpellCast, data, actor, target, Json{});
693 return result;
694 }

721 template <RandomNumberGenerator Rng>
722 [[nodiscard]] AfflictionResult applyAffliction(std::string_view section, std::string_view id,
723 DynamicEntity &victim, const CheckParams &params,
724 Rng &rng) {
725 AfflictionResult result;
726 const Json *affliction = findDataRecord(section, id);
727 if (affliction == nullptr) {
728 throw std::invalid_argument("unknown " + std::string(section) + " '" + std::string(id) + "'");
729 }
730 if (affliction->contains("save") && affliction->at("save").is_string()) {
731 const std::string save = affliction->at("save").get<std::string>();
732 if (m_ruleset.hasStat(save)) {
733 result.saveRolled = true;
734 CheckRecipe recipe;
735 recipe.resolution = Resolution::Threshold;
736 recipe.dice = "1d20"_dice;
737 recipe.comparison = Comparison::LessEqual;
738 recipe.thresholdSource = ThresholdSource::ActorStat;
739 recipe.thresholdStat = save;
740 recipe.difficultyMode = DifficultyMode::ToStat;
741 const CheckResult saveResult =
742 resolveCheck(victim, NullStatProvider{}, recipe, params, rng);
743 if (saveResult.isSuccess) {
744 result.resisted = true;
745 return result;
746 }
747 }
748 }
749 if (affliction->contains("effects") && affliction->at("effects").is_array()) {
750 std::vector<std::string> appliedConditions;
751 for (const Json &effect : affliction->at("effects")) {
752 const std::string stat = effect.value("stat", "");
753 if (stat.empty()) {
754 continue;
755 }
756 int32_t amount = 1;
757 if (effect.contains("amount")) {
758 amount = readVariantValue(effect.at("amount"), Variance::Random, rng);
759 }
760 if (m_ruleset.isResourcePool(stat)) {
761 (void)victim.modifyResource(stat, -amount);
762 ++result.effectsApplied;
763 } else if (effect.contains("condition") && effect.at("condition").is_string()) {
764 const std::string conditionId = effect.at("condition").get<std::string>();
765 victim.addCondition(conditionId, amount);
766 appliedConditions.push_back(conditionId);
767 ++result.effectsApplied;
768 } else if (m_ruleset.isCondition(stat)) {
769 victim.addCondition(stat, amount);
770 appliedConditions.push_back(stat);
771 ++result.effectsApplied;
772 } else if (m_ruleset.findAttribute(stat) != nullptr) {
773 victim.setBaseAttribute(stat, victim.baseAttribute(stat) - amount);
774 ++result.effectsApplied;
775 }
776 }
777 // Remember what was applied so an application can cure it later.
778 if (result.effectsApplied > 0) {
779 victim.addAffliction(section, id, appliedConditions);
780 EventData data;
781 data.payload = {{"section", std::string(section)},
782 {"affliction", std::string(id)},
783 {"victim_id", victim.id()},
784 {"effects_applied", result.effectsApplied}};
785 fireEvent(EventType::OnAfflictionApplied, data, victim, nullptr, Json{});
786 }
787 }
788 return result;
789 }

1446 template <RandomNumberGenerator Rng>
1447 [[nodiscard]] EffectsResult
1448 resolveEffects(DynamicEntity &source, DynamicEntity &target, const Json &effects,
1449 const CheckParams &params, Rng &rng, int32_t qualityLevel = 0,
1450 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
1451 EffectsResult result;
1452 if (!effects.is_array()) {
1453 return result;
1454 }
1455 const std::string hitPool = resolveHitPointPoolId();
1456 for (const Json &effect : effects) {
1457 const std::string kind = effect.value("kind", "");
1458 bool applied = true; // whether the base effect actually took effect
1459 if (kind == "damage") {
1460 if (!effect.contains("dice")) {
1461 continue;
1462 }
1463 int32_t final =
1464 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1465 if (effect.contains("add")) {
1466 final += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1467 }
1468 if (effect.contains("save")) {
1469 if (resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
1470 ++result.savesPassed;
1471 if (effect.at("save").value("on_success", "none") == "half") {
1472 final /= 2;
1473 } else {
1474 final = 0;
1475 applied = false;
1476 }
1477 }
1478 } else if (effect.contains("attack")) {
1479 const Json &attack = effect.at("attack");
1480 if (!resolveAttack(source, target, attack, params, rng)) {
1481 if (attack.value("on_miss", "none") == "half") {
1482 final /= 2;
1483 } else {
1484 final = 0;
1485 applied = false;
1486 }
1487 }
1488 }
1489 if (final > 0 && !hitPool.empty()) {
1490 // applyDamage reports the negative pool delta; negate for "dealt".
1491 result.damageDealt += -applyDamage(source, target, hitPool, final);
1492 }
1493 } else if (kind == "condition") {
1494 const std::string conditionId = effect.value("condition", "");
1495 if (conditionId.empty()) {
1496 continue;
1497 }
1498 if (effect.contains("save") &&
1499 resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
1500 ++result.savesPassed;
1501 applied = false;
1502 } else if (effect.contains("attack") &&
1503 !resolveAttack(source, target, effect.at("attack"), params, rng)) {
1504 applied = false;
1505 }
1506 if (!applied) {
1507 continue;
1508 }
1509 // `stacks` may be a plain number or a formula over the source's stats
1510 // and the casting check's quality level (`env.ql`) — the latter lets
1511 // QL-scaled conditions ("QL 3: 2 levels of Pain") resolve from data.
1512 int32_t stacks = 1;
1513 if (effect.contains("stacks")) {
1514 const Json &stacksJson = effect.at("stacks");
1515 if (stacksJson.is_number()) {
1516 stacks = stacksJson.get<int32_t>();
1517 } else if (stacksJson.is_string()) {
1518 const Json env = {{"ql", qualityLevel}};
1519 const EntityContext context(source, &target, env, Json{});
1520 stacks = math::toStat(Expression(stacksJson.get<std::string>()).evaluate(context));
1521 }
1522 }
1523 target.addCondition(conditionId, stacks);
1524 fireConditionChanged(target, conditionId, stacks, "applied");
1525 const int32_t duration = effect.value("duration", 0);
1526 if (duration > 0) {
1527 target.effects().add(ActiveEffect{conditionId, stacks, duration, "effect"});
1528 }
1529 ++result.conditionsApplied;
1530 } else if (kind == "heal") {
1531 if (!effect.contains("dice") || hitPool.empty()) {
1532 continue;
1533 }
1534 int32_t amount =
1535 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1536 if (effect.contains("add")) {
1537 amount += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1538 }
1539 result.healingDone += target.modifyResource(hitPool, amount);
1540 } else if (kind == "temp_hp") {
1541 if (!effect.contains("dice")) {
1542 continue;
1543 }
1544 int32_t amount =
1545 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1546 if (effect.contains("add")) {
1547 amount += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1548 }
1549 target.addTemporaryHitPoints(amount);
1550 } else if (kind == "resist") {
1551 if (effect.contains("types")) {
1552 for (const Json &type : effect.at("types")) {
1553 target.addResistance(type.get<std::string>());
1554 }
1555 }
1556 } else if (kind == "stat_bonus") {
1557 const std::string stat = effect.value("stat", "");
1558 if (stat.empty()) {
1559 continue;
1560 }
1561 const int32_t value =
1562 effect.contains("add")
1563 ? evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel)
1564 : 0;
1565 const int32_t duration = effect.value("duration", 0);
1566 target.effects().addBonus(StatBonus{stat, value, duration, "effect"});
1567 ++result.statBonusesApplied;
1568 } else if (kind == "bonus_die") {
1569 const std::string dice = effect.value("dice", "");
1570 if (dice.empty()) {
1571 continue;
1572 }
1573 const std::string scope = effect.value("scope", "all");
1574 const int32_t duration = effect.value("duration", 0);
1575 target.effects().addBonusDie(BonusDie{dice, scope, duration, "effect"});
1576 } else if (kind == "options") {
1577 // A caller-side choice: the engine never picks for the caller. Without
1578 // a selection a required group is reported in choicesRequired (nothing
1579 // applied); an optional group is skipped silently; with a selection the
1580 // chosen option resolves (including its own save/attack/ongoing logic).
1581 const std::string groupId = effect.value("id", "");
1582 const Json &options = effect.at("options");
1583 int32_t chosen = -1;
1584 if (!groupId.empty() && selections != nullptr) {
1585 const auto it = selections->find(groupId);
1586 if (it != selections->end()) {
1587 chosen = it->second;
1588 }
1589 }
1590 if (chosen < 0) {
1591 if (!effect.value("optional", false)) {
1592 result.choicesRequired.push_back(groupId);
1593 }
1594 continue;
1595 }
1596 if (!options.is_array() || chosen >= static_cast<int32_t>(options.size())) {
1597 continue;
1598 }
1599 Json batch = Json::array();
1600 batch.push_back(options.at(static_cast<std::size_t>(chosen)));
1601 const EffectsResult sub =
1602 resolveEffects(source, target, batch, params, rng, qualityLevel, selections);
1603 result.damageDealt += sub.damageDealt;
1604 result.conditionsApplied += sub.conditionsApplied;
1605 result.healingDone += sub.healingDone;
1606 result.savesPassed += sub.savesPassed;
1607 result.statBonusesApplied += sub.statBonusesApplied;
1608 result.choicesRequired.insert(result.choicesRequired.end(), sub.choicesRequired.begin(),
1609 sub.choicesRequired.end());
1610 }
1611
1612 // Recurring (ongoing) effects: after the base effect resolves and
1613 // *applies* (the attack hit / the save failed / it is unconditional),
1614 // remember what to re-apply at the declared phase of the target's turn.
1615 // The stored record is the `ongoing.effect` when given (the recurring
1616 // part may differ from the initial one — an arrow deals 4d4 now and 2d4
1617 // at the end of its next turn), else the base effect with `ongoing`
1618 // stripped, so a re-resolution never re-registers.
1619 if (applied && effect.contains("ongoing") && effect.at("ongoing").is_object()) {
1620 const Json &ongoingJson = effect.at("ongoing");
1621 OngoingEffect ongoing;
1622 if (ongoingJson.contains("effect")) {
1623 ongoing.effect = ongoingJson.at("effect");
1624 } else {
1625 ongoing.effect = effect;
1626 ongoing.effect.erase("ongoing");
1627 }
1628 ongoing.phase = ongoingJson.value("at", "end_of_turn");
1629 ongoing.remaining = ongoingJson.value("duration", 1);
1630 ongoing.source = "effect";
1631 target.effects().addOngoing(ongoing);
1632 }
1633 }
1634 return result;
1635 }

1677 const Json &save, const CheckParams &params, Rng &rng,

1690 }

1693 recipe.resolution = Resolution::Threshold;
1694 recipe.dice = DiceExpression(save.value("dice", "1d20"));

1697 // dc is a modifier (the caster's QL applies a penalty, dc = "-env.ql").
1698 recipe.comparison = Comparison::LessEqual;
1699 recipe.thresholdSource = ThresholdSource::ActorStat;
1700 recipe.thresholdStat = stat;
1701 recipe.difficultyMode = DifficultyMode::ToStat;

1704 recipe.comparison = Comparison::GreaterEqual;

1712 adjusted.difficulty = dc;

1894 trimmed == "varies") {

1907 }
