216 [[nodiscard]] static bool scopeMatches(std::string_view pattern,
217 std::string_view scope) noexcept {
218 if (pattern == "all" || pattern == scope) {
219 return true;
220 }
221 if (!pattern.empty() && pattern.back() == '*') {
222 return scope.starts_with(pattern.substr(0, pattern.size() - 1));
223 }
224 std::size_t begin = 0;
225 while (begin <= scope.size()) {
226 const std::size_t end = scope.find('_', begin);
227 const std::string_view segment =
228 scope.substr(begin, end == std::string_view::npos ? std::string_view::npos : end - begin);
229 if (segment == pattern) {
230 return true;
231 }
232 if (end == std::string_view::npos) {
233 break;
234 }
235 begin = end + 1;
236 }
237 return false;
238 }

255 [[nodiscard]] CheckParams withConditionModifiers(const DynamicEntity &actor,
256 const DynamicEntity *target,
257 std::string_view scope,
258 CheckParams params) const {
259 int32_t bonus = 0;
260 int advantage = 0;
261 int disadvantage = 0;
262 bool autoFail = false;
263 std::vector<std::string> bonusDice;
264 accumulateCheckModifiers(actor, scope, "actor", bonus, advantage, disadvantage, autoFail,
265 bonusDice);
266 if (target != nullptr) {
267 accumulateCheckModifiers(*target, scope, "target", bonus, advantage, disadvantage, autoFail,
268 bonusDice);
269 }
270 // Temporary bonus dice from effects (the `bonus_die` effect kind) apply to
271 // the carrier's own checks of the matching scope.
272 for (const BonusDie &die : actor.effects().bonusDice()) {
273 if (scopeMatches(die.scope, scope)) {
274 bonusDice.push_back(die.dice);
275 }
276 }
277 params.situationalModifier += bonus;
278 if (advantage > 0 && disadvantage > 0) {
279 params.advantage = AdvantageMode::None;
280 } else if (advantage > 0) {
281 params.advantage = AdvantageMode::Advantage;
282 } else if (disadvantage > 0) {
283 params.advantage = AdvantageMode::Disadvantage;
284 }
285 params.autoFail = autoFail;
286 for (const std::string &dice : bonusDice) {
287 params.bonusDice.push_back(DiceExpression(dice));
288 }
289 return params;
290 }

296 void accumulateCheckModifiers(const DynamicEntity &entity, std::string_view scope,
297 std::string_view wantedSide, int32_t &bonus, int &advantage,
298 int &disadvantage, bool &autoFail,
299 std::vector<std::string> &bonusDice) const {
300 for (const auto &[conditionId, stacks] : entity.conditions()) {
301 const Json *condition = findCondition(conditionId);
302 if (condition == nullptr || !condition->contains("check_modifiers")) {
303 continue;
304 }
305 const Json &modifiers = condition->at("check_modifiers");
306 if (!modifiers.is_array()) {
307 continue;
308 }
309 for (const Json &mod : modifiers) {
310 accumulateModifierEntry(mod, scope, wantedSide, stacks, bonus, advantage, disadvantage,
311 autoFail, bonusDice);
312 }
313 }
314 for (const std::string &traitId : entity.traits()) {
315 const Json *trait = findDataRecord("traits", traitId);
316 if (trait == nullptr || !trait->contains("check_modifiers")) {
317 continue;
318 }
319 const Json &modifiers = trait->at("check_modifiers");
320 if (!modifiers.is_array()) {
321 continue;
322 }
323 for (const Json &mod : modifiers) {
324 accumulateModifierEntry(mod, scope, wantedSide, 1, bonus, advantage, disadvantage, autoFail,
325 bonusDice);
326 }
327 }
328 }

333 void accumulateModifierEntry(const Json &mod, std::string_view scope, std::string_view wantedSide,
334 int32_t stacks, int32_t &bonus, int &advantage, int &disadvantage,
335 bool &autoFail, std::vector<std::string> &bonusDice) const {
336 const std::string side = mod.value("side", "actor");
337 if (side != wantedSide) {
338 return;
339 }
340 if (!scopeMatches(mod.value("scope", "all"), scope)) {
341 return;
342 }
343 const std::string mode = mod.value("mode", "none");
344 if (mode == "advantage") {
345 ++advantage;
346 } else if (mode == "disadvantage") {
347 ++disadvantage;
348 } else if (mode == "auto_fail") {
349 autoFail = true;
350 }
351 if (mod.contains("bonus")) {
352 int32_t modBonus = mod.at("bonus").get<int32_t>();
353 if (mod.value("per_stack", false)) {
354 modBonus *= stacks;
355 }
356 bonus += modBonus;
357 }
358 if (mod.contains("bonus_dice")) {
359 bonusDice.push_back(mod.at("bonus_dice").get<std::string>());
360 }
361 }

446 [[nodiscard]] CheckResult executeSkillCheck(std::string_view skillId, const DynamicEntity &actor,
447 const CheckParams &params, Rng &rng) const {
448 const SkillDef *skill = m_ruleset.findSkill(skillId);
449 if (skill == nullptr) {
450 throw std::invalid_argument("unknown skill '" + std::string(skillId) + "'");
451 }
452 if (skill->attributes.size() != 3) {
453 throw std::invalid_argument("skill '" + std::string(skillId) +
454 "' is not a three-attribute skill check");
455 }
456 CheckRecipe recipe = makePoolRecipe();
457 recipe.numPoolAttributes = 3;
458 recipe.poolAttributes = {skill->attributes[0], skill->attributes[1], skill->attributes[2]};
459 recipe.poolStat = std::string(skillId);
460 const CheckParams adjusted = withConditionModifiers(actor, nullptr, "skill", params);
461 announceCheckStart(skillId, actor);
462 const CheckResult result = resolveCheck(actor, NullStatProvider{}, recipe, adjusted, rng);
463 announceCheckResult(skillId, actor, result);
464 return result;
465 }

487 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
488 int32_t rawDamage, const Json &env = {}) {
489 EventData data;
490 data.payload = {{"damage", rawDamage}, {"raw_damage", rawDamage},
491 {"attacker_id", actor.id()}, {"attacker_instance_id", actor.entityId().value},
492 {"target_id", target.id()}, {"target_instance_id", target.entityId().value}};
493 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
494 int32_t finalDamage = data.getInt("damage", rawDamage);
495 if (finalDamage < 0) {
496 finalDamage = 0;
497 }
498 // Temporary Hit Points are a buffer: they absorb damage before the real
499 // hit-point pool ("Lose Temporary Hit Points First").
500 if (target.temporaryHitPoints() > 0 && finalDamage > 0) {
501 const int32_t absorbed = std::min(target.temporaryHitPoints(), finalDamage);
502 target.addTemporaryHitPoints(-absorbed);
503 finalDamage -= absorbed;
504 data.payload["temporary_hp_absorbed"] = absorbed;
505 }
506 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
507 data.payload["applied_damage"] = -applied;
508 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
509 return applied;
510 }

595 template <RandomNumberGenerator Rng>
596 void applySpellEffects(DynamicEntity &actor, DynamicEntity *target, const Json *spell,
597 SpellResult &result, const CheckParams &params, Rng &rng,
598 int32_t qualityLevel,
599 const std::unordered_map<std::string, int32_t> *selections) {
600 if (!result.cast || target == nullptr) {
601 return;
602 }
603 if (spell->contains("effects") && spell->at("effects").is_array()) {
604 const EffectsResult effects = resolveEffects(actor, *target, spell->at("effects"), params,
605 rng, qualityLevel, selections);
606 result.appliedDamage = effects.damageDealt;
607 result.choicesRequired = effects.choicesRequired;
608 result.damageDice = effects.damageDice;
609 } else if (spell->contains("damage")) {
610 // `applyDamage` reports the (negative) pool delta, so negate it into the
611 // positive "damage dealt" the caller expects. The dice are recorded the
612 // same observation-only way as the effects path.
613 const Json &damageJson = spell->at("damage");
614 int32_t damage = 0;
615 if (damageJson.is_string()) {
616 const DiceExpression diceExpr(damageJson.get<std::string>());
617 const std::vector<int> rolled = diceExpr.roll(rng);
618 damage = diceExpr.constant();
619 const auto &groups = diceExpr.dice();
620 for (std::size_t g = 0, i = 0; g < groups.size(); ++g) {
621 for (int j = 0; j < groups[g].count; ++j, ++i) {
622 damage += groups[g].sign * rolled[i];
623 result.damageDice.push_back(rolled[i]);
624 }
625 }
626 } else {
627 damage = readVariantValue(damageJson, Variance::Random, rng);
628 }
629 const std::string hitPool = resolveHitPointPoolId();
630 if (!hitPool.empty()) {
631 result.appliedDamage = -applyDamage(actor, *target, hitPool, damage);
632 }
633 }
634 }

652 template <RandomNumberGenerator Rng>
653 [[nodiscard]] SpellResult
654 castSpell(std::string_view spellId, DynamicEntity &actor, DynamicEntity *target,
655 std::string_view resourceId, const CheckParams &params, Rng &rng,
656 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
657 SpellResult result;
658 const Json *spell = findSpell(spellId);
659 if (spell == nullptr) {
660 throw std::invalid_argument("unknown spell '" + std::string(spellId) + "'");
661 }
662 // The actor's restrictions are enforced: a creature that cannot take
663 // actions (incapacitated, stunned, ...) or cannot cast is refused before
664 // any resource is spent, with the reason recorded on the result.
665 if (!actionAllowed(actor, "action") || !actionAllowed(actor, "cast")) {
666 result.cast = false;
667 result.denied = actor.hasRestriction("no_cast") ? "no_cast" : "no_action";
668 return result;
669 }
670 // Cost: `cost` or `ae_cost`, else `level` (one point per level).
671 result.cost = rpg_os::spellCost(*spell);
672 result.resourceId = std::string(resourceId);
673 if (result.cost > 0 && !resourceId.empty()) {
674 if (actor.resource(resourceId) < result.cost) {
675 result.cast = false; // cannot afford the spell
676 return result;
677 }
678 (void)actor.modifyResource(resourceId, -result.cost);
679 }
680
681 // Casting check (named check type, or an "A/B/C" attribute list).
682 if (spell->contains("check") && spell->at("check").is_string()) {
683 const std::string checkText = spell->at("check").get<std::string>();
684 if (m_ruleset.findCheckType(checkText) != nullptr) {
685 const NullStatProvider noTarget;
686 result.check =
687 target != nullptr
688 ? CheckResolver::resolve(m_ruleset, actor, *target, checkText, params, rng)
689 : CheckResolver::resolve(m_ruleset, actor, noTarget, checkText, params, rng);
690 } else {
691 result.check = resolveSpellCheck(actor, checkText, params, rng);
692 }
693 result.cast = result.check.isSuccess;
694 } else {
695 result.cast = true;
696 }
697
698 // Structured effects take precedence over the bare `damage` field: they
699 // are the extracted, machine-readable form of the spell's prose (saves,
700 // half-on-save damage, conditions, healing, resistances). A spell with no
701 // `effects` falls back to the simple damage field. Both paths require the
702 // casting check to have passed: a failed check fizzles — the action and
703 // the resource are spent, but the spell has no effect (and no dice are
704 // rolled for its damage), so the combat log can trust that a fizzle
705 // changes no stat.
706 if (result.cast) {
707 // The casting check's quality level (The Dark Eye's QL) is threaded into
708 // the effect formulas so QL-scaled spells ("2D6 + QLx2") resolve from
709 // data alone.
710 applySpellEffects(actor, target, spell, result, params, rng, result.check.qualityLevel,
711 selections);
712 }
713
714 // Announce the cast attempt (cast = true when it went off). Refused casts
715 // (not allowed to act / not enough resource) return before reaching here
716 // and are not announced; a failed casting check still announces with
717 // cast = false so UI can narrate the fizzle.
718 EventData data;
719 data.payload = {{"spell", std::string(spellId)},
720 {"actor_id", actor.id()},
721 {"instance_id", actor.entityId().value},
722 {"cast", result.cast}};
723 fireEvent(EventType::OnSpellCast, data, actor, target, Json{});
724 return result;
725 }

752 template <RandomNumberGenerator Rng>
753 [[nodiscard]] AfflictionResult applyAffliction(std::string_view section, std::string_view id,
754 DynamicEntity &victim, const CheckParams &params,
755 Rng &rng) {
756 AfflictionResult result;
757 const Json *affliction = findDataRecord(section, id);
758 if (affliction == nullptr) {
759 throw std::invalid_argument("unknown " + std::string(section) + " '" + std::string(id) + "'");
760 }
761 if (affliction->contains("save") && affliction->at("save").is_string()) {
762 const std::string save = affliction->at("save").get<std::string>();
763 if (m_ruleset.hasStat(save)) {
764 result.saveRolled = true;
765 CheckRecipe recipe;
766 recipe.resolution = Resolution::Threshold;
767 recipe.dice = "1d20"_dice;
768 recipe.comparison = Comparison::LessEqual;
769 recipe.thresholdSource = ThresholdSource::ActorStat;
770 recipe.thresholdStat = save;
771 recipe.difficultyMode = DifficultyMode::ToStat;
772 const CheckResult saveResult =
773 resolveCheck(victim, NullStatProvider{}, recipe, params, rng);
774 if (saveResult.isSuccess) {
775 result.resisted = true;
776 return result;
777 }
778 }
779 }
780 if (affliction->contains("effects") && affliction->at("effects").is_array()) {
781 std::vector<std::string> appliedConditions;
782 for (const Json &effect : affliction->at("effects")) {
783 const std::string stat = effect.value("stat", "");
784 if (stat.empty()) {
785 continue;
786 }
787 int32_t amount = 1;
788 if (effect.contains("amount")) {
789 amount = readVariantValue(effect.at("amount"), Variance::Random, rng);
790 }
791 if (m_ruleset.isResourcePool(stat)) {
792 (void)victim.modifyResource(stat, -amount);
793 ++result.effectsApplied;
794 } else if (effect.contains("condition") && effect.at("condition").is_string()) {
795 const std::string conditionId = effect.at("condition").get<std::string>();
796 victim.addCondition(conditionId, amount);
797 appliedConditions.push_back(conditionId);
798 ++result.effectsApplied;
799 } else if (m_ruleset.isCondition(stat)) {
800 victim.addCondition(stat, amount);
801 appliedConditions.push_back(stat);
802 ++result.effectsApplied;
803 } else if (m_ruleset.findAttribute(stat) != nullptr) {
804 victim.setBaseAttribute(stat, victim.baseAttribute(stat) - amount);
805 ++result.effectsApplied;
806 }
807 }
808 // Remember what was applied so an application can cure it later.
809 if (result.effectsApplied > 0) {
810 victim.addAffliction(section, id, appliedConditions);
811 EventData data;
812 data.payload = {{"section", std::string(section)},
813 {"affliction", std::string(id)},
814 {"victim_id", victim.id()},

1901 template <RandomNumberGenerator Rng>
1902 [[nodiscard]] EffectsResult
1903 resolveEffects(DynamicEntity &source, DynamicEntity &target, const Json &effects,
1904 const CheckParams &params, Rng &rng, int32_t qualityLevel = 0,
1905 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
1906 EffectsResult result;
1907 if (!effects.is_array()) {
1908 return result;
1909 }
1910 const std::string hitPool = resolveHitPointPoolId();
1911 for (const Json &effect : effects) {
1912 const std::string kind = effect.value("kind", "");
1913 bool applied = true; // whether the base effect actually took effect
1914 if (kind == "damage") {
1915 applied =
1916 resolveDamageEffect(source, target, effect, params, rng, qualityLevel, hitPool, result);
1917 } else if (kind == "condition") {
1918 applied = resolveConditionEffect(source, target, effect, params, rng, qualityLevel, result);
1919 } else if (kind == "heal") {
1920 applied = resolveHealEffect(source, target, effect, rng, qualityLevel, hitPool, result);
1921 } else if (kind == "temp_hp") {
1922 applied = resolveTempHpEffect(source, target, effect, rng, qualityLevel);
1923 } else if (kind == "resist") {
1924 applied = resolveResistEffect(target, effect);
1925 } else if (kind == "stat_bonus") {
1926 applied = resolveStatBonusEffect(source, target, effect, qualityLevel, result);
1927 } else if (kind == "bonus_die") {
1928 applied = resolveBonusDieEffect(target, effect);
1929 } else if (kind == "options") {
1930 applied = resolveOptionsEffect(source, target, effect, params, rng, qualityLevel, result,
1931 selections);
1932 }
1933
1934 // Recurring (ongoing) effects: after the base effect resolves and
1935 // *applies* (the attack hit / the save failed / it is unconditional),
1936 // remember what to re-apply at the declared phase of the target's turn.
1937 // The stored record is the `ongoing.effect` when given (the recurring
1938 // part may differ from the initial one — an arrow deals 4d4 now and 2d4
1939 // at the end of its next turn), else the base effect with `ongoing`
1940 // stripped, so a re-resolution never re-registers.
1941 if (applied && effect.contains("ongoing") && effect.at("ongoing").is_object()) {
1942 const Json &ongoingJson = effect.at("ongoing");
1943 OngoingEffect ongoing;
1944 if (ongoingJson.contains("effect")) {
1945 ongoing.effect = ongoingJson.at("effect");
1946 } else {
1947 ongoing.effect = effect;
1948 ongoing.effect.erase("ongoing");
1949 }
1950 ongoing.phase = ongoingJson.value("at", "end_of_turn");
1951 ongoing.remaining = ongoingJson.value("duration", 1);
1952 ongoing.source = "effect";
1953 target.effects().addOngoing(ongoing);
1954 }
1955 }
1956 return result;
1957 }

1966 template <RandomNumberGenerator Rng>
1967 bool resolveDamageEffect(DynamicEntity &source, DynamicEntity &target, const Json &effect,
1968 const CheckParams &params, Rng &rng, int32_t qualityLevel,
1969 const std::string &hitPool, EffectsResult &result) {
1970 if (!effect.contains("dice")) {
1971 return false;
1972 }
1973 // A dice expression is rolled explicitly so the individual dice can be
1974 // reported on the result (a fight transcript narrates them). This is
1975 // observation-only: roll() consumes exactly the RNG stream rollSum()
1976 // would, so a call that records dice and one that only sums stay
1977 // byte-identical (pinned by the combat tests). Plain numbers and range
1978 // objects keep their generic readVariantValue path (no dice to record).
1979 int32_t final = 0;
1980 const Json &diceJson = effect.at("dice");
1981 if (diceJson.is_string()) {
1982 const DiceExpression diceExpr(diceJson.get<std::string>());
1983 const std::vector<int> rolled = diceExpr.roll(rng);
1984 final = diceExpr.constant();
1985 const auto &groups = diceExpr.dice();
1986 for (std::size_t g = 0, i = 0; g < groups.size(); ++g) {
1987 for (int j = 0; j < groups[g].count; ++j, ++i) {
1988 final += groups[g].sign * rolled[i];
1989 result.damageDice.push_back(rolled[i]);
1990 }
1991 }
1992 } else {
1993 final = static_cast<int32_t>(readVariantValue(diceJson, Variance::Random, rng));
1994 }
1995 if (effect.contains("add")) {
1996 final += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1997 }
1998 bool applied = true;
1999 if (effect.contains("save")) {
2000 if (resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
2001 ++result.savesPassed;
2002 if (effect.at("save").value("on_success", "none") == "half") {
2003 final /= 2;
2004 } else {
2005 final = 0;
2006 applied = false;
2007 }
2008 }
2009 } else if (effect.contains("attack")) {
2010 const Json &attack = effect.at("attack");
2011 if (!resolveAttack(source, target, attack, params, rng)) {
2012 if (attack.value("on_miss", "none") == "half") {
2013 final /= 2;
2014 } else {
2015 final = 0;
2016 applied = false;
2017 }
2018 }
2019 }
2020 if (final > 0 && !hitPool.empty()) {
2021 // applyDamage reports the negative pool delta; negate for "dealt".
2022 result.damageDealt += -applyDamage(source, target, hitPool, final);
2023 }
2024 return applied;
2025 }
