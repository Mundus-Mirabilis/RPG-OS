215 [[nodiscard]] static bool scopeMatches(std::string_view pattern,
216 std::string_view scope) noexcept {
217 if (pattern == "all" || pattern == scope) {
218 return true;
219 }
220 if (!pattern.empty() && pattern.back() == '*') {
221 return scope.starts_with(pattern.substr(0, pattern.size() - 1));
222 }
223 std::size_t begin = 0;
224 while (begin <= scope.size()) {
225 const std::size_t end = scope.find('_', begin);
226 const std::string_view segment =
227 scope.substr(begin, end == std::string_view::npos ? std::string_view::npos : end - begin);
228 if (segment == pattern) {
229 return true;
230 }
231 if (end == std::string_view::npos) {
232 break;
233 }
234 begin = end + 1;
235 }
236 return false;
237 }

254 [[nodiscard]] CheckParams withConditionModifiers(const DynamicEntity &actor,
255 const DynamicEntity *target,
256 std::string_view scope,
257 CheckParams params) const {
258 int32_t bonus = 0;
259 int advantage = 0;
260 int disadvantage = 0;
261 bool autoFail = false;
262 std::vector<std::string> bonusDice;
263 accumulateCheckModifiers(actor, scope, "actor", bonus, advantage, disadvantage, autoFail,
264 bonusDice);
265 if (target != nullptr) {
266 accumulateCheckModifiers(*target, scope, "target", bonus, advantage, disadvantage, autoFail,
267 bonusDice);
268 }
269 // Temporary bonus dice from effects (the `bonus_die` effect kind) apply to
270 // the carrier's own checks of the matching scope.
271 for (const BonusDie &die : actor.effects().bonusDice()) {
272 if (scopeMatches(die.scope, scope)) {
273 bonusDice.push_back(die.dice);
274 }
275 }
276 params.situationalModifier += bonus;
277 if (advantage > 0 && disadvantage > 0) {
278 params.advantage = AdvantageMode::None;
279 } else if (advantage > 0) {
280 params.advantage = AdvantageMode::Advantage;
281 } else if (disadvantage > 0) {
282 params.advantage = AdvantageMode::Disadvantage;
283 }
284 params.autoFail = autoFail;
285 for (const std::string &dice : bonusDice) {
286 params.bonusDice.push_back(DiceExpression(dice));
287 }
288 return params;
289 }

295 void accumulateCheckModifiers(const DynamicEntity &entity, std::string_view scope,
296 std::string_view wantedSide, int32_t &bonus, int &advantage,
297 int &disadvantage, bool &autoFail,
298 std::vector<std::string> &bonusDice) const {
299 for (const auto &[conditionId, stacks] : entity.conditions()) {
300 const Json *condition = findCondition(conditionId);
301 if (condition == nullptr || !condition->contains("check_modifiers")) {
302 continue;
303 }
304 const Json &modifiers = condition->at("check_modifiers");
305 if (!modifiers.is_array()) {
306 continue;
307 }
308 for (const Json &mod : modifiers) {
309 accumulateModifierEntry(mod, scope, wantedSide, stacks, bonus, advantage, disadvantage,
310 autoFail, bonusDice);
311 }
312 }
313 for (const std::string &traitId : entity.traits()) {
314 const Json *trait = findDataRecord("traits", traitId);
315 if (trait == nullptr || !trait->contains("check_modifiers")) {
316 continue;
317 }
318 const Json &modifiers = trait->at("check_modifiers");
319 if (!modifiers.is_array()) {
320 continue;
321 }
322 for (const Json &mod : modifiers) {
323 accumulateModifierEntry(mod, scope, wantedSide, 1, bonus, advantage, disadvantage, autoFail,
324 bonusDice);
325 }
326 }
327 }
328
332 void accumulateModifierEntry(const Json &mod, std::string_view scope, std::string_view wantedSide,
333 int32_t stacks, int32_t &bonus, int &advantage, int &disadvantage,
334 bool &autoFail, std::vector<std::string> &bonusDice) const {
335 const std::string side = mod.value("side", "actor");
336 if (side != wantedSide) {
337 return;
338 }
339 if (!scopeMatches(mod.value("scope", "all"), scope)) {
340 return;
341 }
342 const std::string mode = mod.value("mode", "none");
343 if (mode == "advantage") {
344 ++advantage;
345 } else if (mode == "disadvantage") {
346 ++disadvantage;
347 } else if (mode == "auto_fail") {
348 autoFail = true;
349 }
350 if (mod.contains("bonus")) {
351 int32_t modBonus = mod.at("bonus").get<int32_t>();
352 if (mod.value("per_stack", false)) {
353 modBonus *= stacks;
354 }
355 bonus += modBonus;
356 }
357 if (mod.contains("bonus_dice")) {
358 bonusDice.push_back(mod.at("bonus_dice").get<std::string>());
359 }
360 }

444 template <RandomNumberGenerator Rng>
445 [[nodiscard]] CheckResult executeSkillCheck(std::string_view skillId, const DynamicEntity &actor,
446 const CheckParams &params, Rng &rng) const {
447 const SkillDef *skill = m_ruleset.findSkill(skillId);
448 if (skill == nullptr) {
449 throw std::invalid_argument("unknown skill '" + std::string(skillId) + "'");
450 }
451 if (skill->attributes.size() != 3) {
452 throw std::invalid_argument("skill '" + std::string(skillId) +
453 "' is not a three-attribute skill check");
454 }
455 CheckRecipe recipe = makePoolRecipe();
456 recipe.numPoolAttributes = 3;
457 recipe.poolAttributes = {skill->attributes[0], skill->attributes[1], skill->attributes[2]};
458 recipe.poolStat = std::string(skillId);
459 const CheckParams adjusted = withConditionModifiers(actor, nullptr, "skill", params);
460 announceCheckStart(skillId, actor);
461 const CheckResult result = resolveCheck(actor, NullStatProvider{}, recipe, adjusted, rng);
462 announceCheckResult(skillId, actor, result);
463 return result;
464 }

486 int32_t applyDamage(DynamicEntity &actor, DynamicEntity &target, std::string_view resourceId,
487 int32_t rawDamage, const Json &env = {}) {
488 EventData data;
489 data.payload = {{"damage", rawDamage}, {"raw_damage", rawDamage},
490 {"attacker_id", actor.id()}, {"attacker_instance_id", actor.entityId().value},
491 {"target_id", target.id()}, {"target_instance_id", target.entityId().value}};
492 fireEvent(EventType::OnDamageCalculated, data, actor, &target, env);
493 int32_t finalDamage = data.getInt("damage", rawDamage);
494 if (finalDamage < 0) {
495 finalDamage = 0;
496 }
497 // Temporary Hit Points are a buffer: they absorb damage before the real
498 // hit-point pool ("Lose Temporary Hit Points First").
499 if (target.temporaryHitPoints() > 0 && finalDamage > 0) {
500 const int32_t absorbed = std::min(target.temporaryHitPoints(), finalDamage);
501 target.addTemporaryHitPoints(-absorbed);
502 finalDamage -= absorbed;
503 data.payload["temporary_hp_absorbed"] = absorbed;
504 }
505 const int32_t applied = target.modifyResource(resourceId, -finalDamage);
506 data.payload["applied_damage"] = -applied;
507 fireEvent(EventType::OnDamageTaken, data, actor, &target, env);
508 return applied;
509 }

593 template <RandomNumberGenerator Rng>
594 void applySpellEffects(DynamicEntity &actor, DynamicEntity *target, const Json *spell,
595 SpellResult &result, const CheckParams &params, Rng &rng,
596 int32_t qualityLevel,
597 const std::unordered_map<std::string, int32_t> *selections) {
598 if (!result.cast || target == nullptr) {
599 return;
600 }
601 if (spell->contains("effects") && spell->at("effects").is_array()) {
602 const EffectsResult effects = resolveEffects(actor, *target, spell->at("effects"), params,
603 rng, qualityLevel, selections);
604 result.appliedDamage = effects.damageDealt;
605 result.choicesRequired = effects.choicesRequired;
606 } else if (spell->contains("damage")) {
607 // `applyDamage` reports the (negative) pool delta, so negate it into the
608 // positive "damage dealt" the caller expects.
609 const int32_t damage = readVariantValue(spell->at("damage"), Variance::Random, rng);
610 const std::string hitPool = resolveHitPointPoolId();
611 if (!hitPool.empty()) {
612 result.appliedDamage = -applyDamage(actor, *target, hitPool, damage);
613 }
614 }
615 }

633 template <RandomNumberGenerator Rng>
634 [[nodiscard]] SpellResult
635 castSpell(std::string_view spellId, DynamicEntity &actor, DynamicEntity *target,
636 std::string_view resourceId, const CheckParams &params, Rng &rng,
637 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
638 SpellResult result;
639 const Json *spell = findSpell(spellId);
640 if (spell == nullptr) {
641 throw std::invalid_argument("unknown spell '" + std::string(spellId) + "'");
642 }
643 // The actor's restrictions are enforced: a creature that cannot take
644 // actions (incapacitated, stunned, ...) or cannot cast is refused before
645 // any resource is spent, with the reason recorded on the result.
646 if (!actionAllowed(actor, "action") || !actionAllowed(actor, "cast")) {
647 result.cast = false;
648 result.denied = actor.hasRestriction("no_cast") ? "no_cast" : "no_action";
649 return result;
650 }
651 // Cost: `cost` or `ae_cost`, else `level` (one point per level).
652 result.cost = rpg_os::spellCost(*spell);
653 result.resourceId = std::string(resourceId);
654 if (result.cost > 0 && !resourceId.empty()) {
655 if (actor.resource(resourceId) < result.cost) {
656 result.cast = false; // cannot afford the spell
657 return result;
658 }
659 (void)actor.modifyResource(resourceId, -result.cost);
660 }
661
662 // Casting check (named check type, or an "A/B/C" attribute list).
663 if (spell->contains("check") && spell->at("check").is_string()) {
664 const std::string checkText = spell->at("check").get<std::string>();
665 if (m_ruleset.findCheckType(checkText) != nullptr) {
666 const NullStatProvider noTarget;
667 result.check =
668 target != nullptr
669 ? CheckResolver::resolve(m_ruleset, actor, *target, checkText, params, rng)
670 : CheckResolver::resolve(m_ruleset, actor, noTarget, checkText, params, rng);
671 } else {
672 result.check = resolveSpellCheck(actor, checkText, params, rng);
673 }
674 result.cast = result.check.isSuccess;
675 } else {
676 result.cast = true;
677 }
678
679 // Structured effects take precedence over the bare `damage` field: they
680 // are the extracted, machine-readable form of the spell's prose (saves,
681 // half-on-save damage, conditions, healing, resistances). A spell with no
682 // `effects` falls back to the simple damage field. Both paths require the
683 // casting check to have passed: a failed check fizzles — the action and
684 // the resource are spent, but the spell has no effect (and no dice are
685 // rolled for its damage), so the combat log can trust that a fizzle
686 // changes no stat.
687 if (result.cast) {
688 // The casting check's quality level (The Dark Eye's QL) is threaded into
689 // the effect formulas so QL-scaled spells ("2D6 + QLx2") resolve from
690 // data alone.
691 applySpellEffects(actor, target, spell, result, params, rng, result.check.qualityLevel,
692 selections);
693 }
694
695 // Announce the cast attempt (cast = true when it went off). Refused casts
696 // (not allowed to act / not enough resource) return before reaching here
697 // and are not announced; a failed casting check still announces with
698 // cast = false so UI can narrate the fizzle.
699 EventData data;
700 data.payload = {{"spell", std::string(spellId)},
701 {"actor_id", actor.id()},
702 {"instance_id", actor.entityId().value},
703 {"cast", result.cast}};
704 fireEvent(EventType::OnSpellCast, data, actor, target, Json{});
705 return result;
706 }

733 template <RandomNumberGenerator Rng>
734 [[nodiscard]] AfflictionResult applyAffliction(std::string_view section, std::string_view id,
735 DynamicEntity &victim, const CheckParams &params,
736 Rng &rng) {
737 AfflictionResult result;
738 const Json *affliction = findDataRecord(section, id);
739 if (affliction == nullptr) {
740 throw std::invalid_argument("unknown " + std::string(section) + " '" + std::string(id) + "'");
741 }
742 if (affliction->contains("save") && affliction->at("save").is_string()) {
743 const std::string save = affliction->at("save").get<std::string>();
744 if (m_ruleset.hasStat(save)) {
745 result.saveRolled = true;
746 CheckRecipe recipe;
747 recipe.resolution = Resolution::Threshold;
748 recipe.dice = "1d20"_dice;
749 recipe.comparison = Comparison::LessEqual;
750 recipe.thresholdSource = ThresholdSource::ActorStat;
751 recipe.thresholdStat = save;
752 recipe.difficultyMode = DifficultyMode::ToStat;
753 const CheckResult saveResult =
754 resolveCheck(victim, NullStatProvider{}, recipe, params, rng);
755 if (saveResult.isSuccess) {
756 result.resisted = true;
757 return result;
758 }
759 }
760 }
761 if (affliction->contains("effects") && affliction->at("effects").is_array()) {
762 std::vector<std::string> appliedConditions;
763 for (const Json &effect : affliction->at("effects")) {
764 const std::string stat = effect.value("stat", "");
765 if (stat.empty()) {
766 continue;
767 }
768 int32_t amount = 1;
769 if (effect.contains("amount")) {
770 amount = readVariantValue(effect.at("amount"), Variance::Random, rng);
771 }
772 if (m_ruleset.isResourcePool(stat)) {
773 (void)victim.modifyResource(stat, -amount);
774 ++result.effectsApplied;
775 } else if (effect.contains("condition") && effect.at("condition").is_string()) {
776 const std::string conditionId = effect.at("condition").get<std::string>();
777 victim.addCondition(conditionId, amount);
778 appliedConditions.push_back(conditionId);
779 ++result.effectsApplied;
780 } else if (m_ruleset.isCondition(stat)) {
781 victim.addCondition(stat, amount);
782 appliedConditions.push_back(stat);
783 ++result.effectsApplied;
784 } else if (m_ruleset.findAttribute(stat) != nullptr) {
785 victim.setBaseAttribute(stat, victim.baseAttribute(stat) - amount);
786 ++result.effectsApplied;
787 }
788 }
789 // Remember what was applied so an application can cure it later.
790 if (result.effectsApplied > 0) {
791 victim.addAffliction(section, id, appliedConditions);
792 EventData data;
793 data.payload = {{"section", std::string(section)},
794 {"affliction", std::string(id)},
795 {"victim_id", victim.id()},
796 {"effects_applied", result.effectsApplied}};
797 fireEvent(EventType::OnAfflictionApplied, data, victim, nullptr, Json{});
798 }
799 }
800 return result;
801 }

1464 template <RandomNumberGenerator Rng>
1465 [[nodiscard]] EffectsResult
1466 resolveEffects(DynamicEntity &source, DynamicEntity &target, const Json &effects,
1467 const CheckParams &params, Rng &rng, int32_t qualityLevel = 0,
1468 const std::unordered_map<std::string, int32_t> *selections = nullptr) {
1469 EffectsResult result;
1470 if (!effects.is_array()) {
1471 return result;
1472 }
1473 const std::string hitPool = resolveHitPointPoolId();
1474 for (const Json &effect : effects) {
1475 const std::string kind = effect.value("kind", "");
1476 bool applied = true; // whether the base effect actually took effect
1477 if (kind == "damage") {
1478 applied =
1479 resolveDamageEffect(source, target, effect, params, rng, qualityLevel, hitPool, result);
1480 } else if (kind == "condition") {
1481 applied = resolveConditionEffect(source, target, effect, params, rng, qualityLevel, result);
1482 } else if (kind == "heal") {
1483 applied = resolveHealEffect(source, target, effect, rng, qualityLevel, hitPool, result);
1484 } else if (kind == "temp_hp") {
1485 applied = resolveTempHpEffect(source, target, effect, rng, qualityLevel);
1486 } else if (kind == "resist") {
1487 applied = resolveResistEffect(target, effect);
1488 } else if (kind == "stat_bonus") {
1489 applied = resolveStatBonusEffect(source, target, effect, qualityLevel, result);
1490 } else if (kind == "bonus_die") {
1491 applied = resolveBonusDieEffect(target, effect);
1492 } else if (kind == "options") {
1493 applied = resolveOptionsEffect(source, target, effect, params, rng, qualityLevel, result,
1494 selections);
1495 }
1496
1497 // Recurring (ongoing) effects: after the base effect resolves and
1498 // *applies* (the attack hit / the save failed / it is unconditional),
1499 // remember what to re-apply at the declared phase of the target's turn.
1500 // The stored record is the `ongoing.effect` when given (the recurring
1501 // part may differ from the initial one — an arrow deals 4d4 now and 2d4
1502 // at the end of its next turn), else the base effect with `ongoing`
1503 // stripped, so a re-resolution never re-registers.
1504 if (applied && effect.contains("ongoing") && effect.at("ongoing").is_object()) {
1505 const Json &ongoingJson = effect.at("ongoing");
1506 OngoingEffect ongoing;
1507 if (ongoingJson.contains("effect")) {
1508 ongoing.effect = ongoingJson.at("effect");
1509 } else {
1510 ongoing.effect = effect;
1511 ongoing.effect.erase("ongoing");
1512 }
1513 ongoing.phase = ongoingJson.value("at", "end_of_turn");
1514 ongoing.remaining = ongoingJson.value("duration", 1);
1515 ongoing.source = "effect";
1516 target.effects().addOngoing(ongoing);
1517 }
1518 }
1519 return result;
1520 }

1529 template <RandomNumberGenerator Rng>
1530 bool resolveDamageEffect(DynamicEntity &source, DynamicEntity &target, const Json &effect,
1531 const CheckParams &params, Rng &rng, int32_t qualityLevel,
1532 const std::string &hitPool, EffectsResult &result) {
1533 if (!effect.contains("dice")) {
1534 return false;
1535 }
1536 int32_t final =
1537 static_cast<int32_t>(readVariantValue(effect.at("dice"), Variance::Random, rng));
1538 if (effect.contains("add")) {
1539 final += evaluateEffectAdd(source, &target, effect.at("add"), qualityLevel);
1540 }
1541 bool applied = true;
1542 if (effect.contains("save")) {
1543 if (resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
1544 ++result.savesPassed;
1545 if (effect.at("save").value("on_success", "none") == "half") {
1546 final /= 2;
1547 } else {
1548 final = 0;
1549 applied = false;
1550 }
1551 }
1552 } else if (effect.contains("attack")) {
1553 const Json &attack = effect.at("attack");
1554 if (!resolveAttack(source, target, attack, params, rng)) {
1555 if (attack.value("on_miss", "none") == "half") {
1556 final /= 2;
1557 } else {
1558 final = 0;
1559 applied = false;
1560 }
1561 }
1562 }
1563 if (final > 0 && !hitPool.empty()) {
1564 // applyDamage reports the negative pool delta; negate for "dealt".
1565 result.damageDealt += -applyDamage(source, target, hitPool, final);
1566 }
1567 return applied;
1568 }

1573 template <RandomNumberGenerator Rng>
1574 bool resolveConditionEffect(DynamicEntity &source, DynamicEntity &target, const Json &effect,
1575 const CheckParams &params, Rng &rng, int32_t qualityLevel,
1576 EffectsResult &result) {
1577 const std::string conditionId = effect.value("condition", "");
1578 if (conditionId.empty()) {
1579 return false;
1580 }
1581 if (effect.contains("save") &&
1582 resolveSave(source, target, effect.at("save"), params, rng, qualityLevel)) {
1583 ++result.savesPassed;
1584 return false;
1585 }
1586 if (effect.contains("attack") &&
1587 !resolveAttack(source, target, effect.at("attack"), params, rng)) {

1986 parsePriceString(std::string_view text) const {
1987 std::string trimmed = trimWhitespace(text);
1988 if (trimmed.empty() || trimmed == "-" || trimmed == "—" || trimmed == "Varies" ||
1989 trimmed == "varies") {
1990 return Money{0};
1991 }
1992 std::size_t pos = 0;
1993 bool sawDigit = false;
1994 while (pos < trimmed.size()) {
1995 const char c = trimmed[pos];
1996 if (c == ',' || c == '.' || c == ' ' || (c >= '0' && c <= '9')) {
1997 if (c >= '0' && c <= '9') {
1998 sawDigit = true;
1999 }
2000 ++pos;
2001 continue;
2002 }
2003 break;
2004 }
2005 if (!sawDigit) {
2006 return Money{0};
2007 }
2008 std::string number = trimmed.substr(0, pos);
2009 std::string symbol = trimWhitespace(trimmed.substr(pos));
2010 number.erase(std::remove(number.begin(), number.end(), ','), number.end());
2011 number.erase(std::remove(number.begin(), number.end(), ' '), number.end());
2012 const int64_t amount = static_cast<int64_t>(std::llround(std::strtod(number.c_str(), nullptr)));

2014 return Money{amount}; // a bare number is base units
2015 }
2016 for (const Denomination &d : m_ruleset.currencySystem.denominations) {
2017 if (d.symbol == symbol || d.id == symbol) {

2019 }
