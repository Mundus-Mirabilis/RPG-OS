38void runTdeExample(const std::string &root) {
39 std::cout << "=== The Dark Eye 5e (universal mode) ===\n";
40 rpg_os::RulesetEngine engine;
41 if (!engine.loadRulesetFromFile(rulesetPath(root, "tde5e_core.json"))) {
42 std::cerr << "failed to load tde5e_core.json: " << engine.lastError() << '\n';
43 return;
44 }
45 auto geron = engine.createEntity("geron");
46 if (!geron) {
47 std::cerr << "archetype 'geron' not found\n";
48 return;
49 }
50 std::cout << "Geron: COU " << engine.calculateStat(*geron, "COU") << ", Life Points "
51 << geron->resource("LP") << "/" << engine.calculateStat(*geron, "LifePoints_Max")
52 << ", Attack " << engine.calculateStat(*geron, "Attack") << ", Dodge "
53 << engine.calculateStat(*geron, "Dodge") << '\n';
54
55 // A climbing skill check (linked attributes COU/AGI/STR; rating is the pool).
56 const rpg_os::CheckResult climb =
57 engine.executeSkillCheck("climbing", *geron, rpg_os::CheckParams{});
58 std::cout << "Climbing check: " << (climb.isSuccess ? "success" : "failure") << " (SP left "
59 << climb.remainingPool << ", QL " << climb.qualityLevel << ")\n";
60}

64void runDndExample(const std::string &root) {
65 std::cout << "=== D&D 5e SRD (universal mode) ===\n";
66 rpg_os::RulesetEngine engine;
67 if (!engine.loadRulesetFromFile(rulesetPath(root, "dnd5e_srd.json"))) {
68 std::cerr << "failed to load dnd5e_srd.json: " << engine.lastError() << '\n';
69 return;
70 }
71 auto fighter = engine.createEntity("fighter_lvl1");
72 if (!fighter) {
73 std::cerr << "archetype 'fighter_lvl1' not found\n";
74 return;
75 }
76 std::cout << "Fighter: STR " << engine.calculateStat(*fighter, "STR") << " (mod "
77 << engine.calculateStat(*fighter, "STR_mod") << "), AC "
78 << engine.calculateStat(*fighter, "AC") << ", HP " << fighter->resource("HP") << '\n';
79
80 // A melee attack roll: 1d20 + STR mod + proficiency vs the target's AC.
81 // Attacking itself with no target entity demonstrates using a dynamic
82 // entity as both actor and target — the AC read from the same character.
83 const rpg_os::CheckResult attack =
84 engine.executeCheck("dnd5e_attack_melee", *fighter, fighter.get(), rpg_os::CheckParams{});
85 std::cout << "Melee attack vs AC " << engine.calculateStat(*fighter, "AC") << ": "
86 << (attack.isSuccess ? "hit" : "miss") << " (roll "
87 << (attack.rawDiceRolls.empty() ? 0 : attack.rawDiceRolls.front()) << ")\n";
88}
